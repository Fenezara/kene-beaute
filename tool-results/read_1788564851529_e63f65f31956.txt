     1→"use client";
     2→// Kènè Cliente — Boutique : catalogue, fiche produit, panier, checkout Wave/Orange/Wallet simulé
     3→// + « Mes commandes » : historique des commandes enregistrées (consultation par la cliente).
     4→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     5→import { AnimatePresence, motion } from "framer-motion";
     6→import { BadgeCheck, History, Loader2, Minus, Plus, Search, ShoppingBag, Tag, Trash2, X } from "lucide-react";
     7→import { toast } from "sonner";
     8→import { apiGet, apiPost } from "@/lib/kene/api";
     9→import { xof, CASHBACK_RATE, formatDate, formatTime } from "@/lib/kene/format";
    10→import { HAPTIC, haptic } from "@/lib/kene/ux";
    11→import { MOMO_OPERATORS } from "@/lib/kene/rfm";
    12→import { Sheet, SheetContent, SheetHeader, SheetTitle } from "@/components/ui/sheet";
    13→import { Skeleton } from "@/components/ui/skeleton";
    14→import { useKene } from "@/store/kene";
    15→import { KenteWeaveCard } from "@/components/kene/weave/KenteWeaveCard";
    16→import { categoryThread } from "@/components/kene/weave/threads";
    17→import type { ApiOrder, ApiPayment, ApiProduct, ApiWallet } from "./types";
    18→import { SHOP_CATEGORIES } from "./types";
    19→import { EmptyBlock, Stars, SuccessBurst } from "./bits";
    20→
    21→type PayMethod = "wave" | "orange" | "wallet";
    22→
    23→/* Coupon validé au checkout (aperçu local — la consommation a lieu à la commande) */
    24→interface AppliedPromo {
    25→  code: string;
    26→  label: string | null;
    27→  discount: number;
    28→  subtotal: number; // panier au moment de la validation (invalidation si changement)
    29→}
    30→
    31→export function ShopScreen() {
    32→  const user = useKene((s) => s.user)!;
    33→  const cart = useKene((s) => s.cart);
    34→  const addToCart = useKene((s) => s.addToCart);
    35→  const setCartQty = useKene((s) => s.setCartQty);
    36→  const clearCart = useKene((s) => s.clearCart);
    37→
    38→  const [products, setProducts] = useState<ApiProduct[] | null>(null);
    39→  const [cat, setCat] = useState("");
    40→  const [q, setQ] = useState("");
    41→  const [detail, setDetail] = useState<ApiProduct | null>(null);
    42→  const [qty, setQty] = useState(1);
    43→  const [checkout, setCheckout] = useState(false);
    44→  const [wallet, setWallet] = useState<ApiWallet | null>(null);
    45→  const [payState, setPayState] = useState<{ phase: "processing" | "success"; method: PayMethod; amount: number } | null>(null);
    46→  const [paying, setPaying] = useState(false);
    47→
    48→  // ─── Double-tap « ajout rapide » (TikTok Shop / Instagram) ───
    49→  // 1er tap = ouvre la fiche (avec un délai court annulable) ; 2e tap < 320 ms
    50→  // = ajoute directement au panier + burst animé sur la carte.
    51→  const lastTap = useRef<{ id: string; t: number; timer: number | null }>({ id: "", t: 0, timer: null });
    52→  const [burst, setBurst] = useState<{ id: string; x: number; y: number } | null>(null);
    53→
    54→  const onCardTap = (p: ApiProduct, e: React.MouseEvent<HTMLButtonElement>) => {
    55→    const now = Date.now();
    56→    const s = lastTap.current;
    57→    if (s.id === p.id && now - s.t < 320 && s.timer) {
    58→      window.clearTimeout(s.timer);
    59→      lastTap.current = { id: "", t: 0, timer: null };
    60→      addToCart({ productId: p.id, name: p.name, price: p.price, qty: 1, image: p.image });
    61→      haptic(HAPTIC.light);
    62→      const rect = e.currentTarget.getBoundingClientRect();
    63→      setBurst({ id: p.id, x: e.clientX - rect.left, y: e.clientY - rect.top });
    64→      window.setTimeout(() => setBurst(null), 700);
    65→      toast.success(`${p.name} ajouté au panier`, { description: "Astuce : double-tape une carte pour l'ajouter en 1 geste" });
    66→    } else {
    67→      if (s.timer) window.clearTimeout(s.timer);
    68→      const timer = window.setTimeout(() => {
    69→        setDetail(p);
    70→        setQty(1);
    71→      }, 240);
    72→      lastTap.current = { id: p.id, t: now, timer };
    73→    }
    74→  };
    75→  const [promoInput, setPromoInput] = useState("");
    76→  const [promoChecking, setPromoChecking] = useState(false);
    77→  const [promo, setPromo] = useState<AppliedPromo | null>(null);
    78→  const inputRef = useRef<HTMLInputElement>(null);
    79→
    80→  /* « Mes commandes » — historique consultable (chargé au montage, rafraîchi
    81→   * après chaque commande réussie : la cliente suit ses données enregistrées). */
    82→  const [view, setView] = useState<"catalogue" | "commandes">("catalogue");
    83→  const [orders, setOrders] = useState<ApiOrder[] | null>(null);
    84→  const refreshOrders = useCallback(() => {
    85→    apiGet<{ orders: ApiOrder[] }>(`/api/orders?userId=${user.id}`)
    86→      .then((r) => setOrders(r.orders ?? []))
    87→      .catch(() => setOrders((prev) => prev ?? []));
    88→  }, [user.id]);
    89→
    90→  const subtotal = cart.reduce((s, l) => s + l.price * l.qty, 0);
    91→  /* Coupon invalidé si le panier a changé depuis la validation (remise recalculée
    92→   * sur un autre sous-total) — la cliente re-valide, jamais de surprise serveur. */
    93→  const livePromo = promo && promo.subtotal === subtotal ? promo : null;
    94→  const discount = livePromo?.discount ?? 0;
    95→  const total = subtotal - discount;
    96→  /* taux de cashback réellement appliqué (wallet de la cliente, sinon défaut) */
    97→  const cashbackRate = wallet?.cashbackRate ?? CASHBACK_RATE;
    98→
    99→  useEffect(() => {
   100→    apiGet<{ products: ApiProduct[] }>("/api/shop/products")
   101→      .then((r) => setProducts(r.products ?? []))
   102→      .catch(() => setProducts([]));
   103→    apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).then((r) => setWallet(r.wallet)).catch(() => {});
   104→    refreshOrders();
   105→  }, [user.id, refreshOrders]);
   106→
   107→  const filtered = useMemo(() => {
   108→    const nq = q.trim().toLowerCase();
   109→    return (products ?? []).filter((p) => (!cat || p.category === cat) && (!nq || `${p.name} ${p.botanicals} ${p.description}`.toLowerCase().includes(nq)));
   110→  }, [products, cat, q]);
   111→
   112→  /* le fil de la catégorie — la navette l'illumine dans la bande tissée */
   113→  const weaveCaption =
   114→    products === null
   115→      ? "La navette monte le métier…"
   116→      : `${filtered.length} soin${filtered.length > 1 ? "s" : ""}${
   117→          cat ? ` · ${SHOP_CATEGORIES.find((c) => c.id === cat)?.label.toLowerCase() ?? cat}` : " au catalogue"
   118→        }`;
   119→
   120→  /* Applique un code promo : aperçu de remise sans consommer le coupon
   121→   * (la consommation a lieu à la commande — toutes les gardes côté serveur). */
   122→  async function applyPromo() {
   123→    const code = promoInput.trim().toUpperCase();
   124→    if (!code || promoChecking) return;
   125→    setPromoChecking(true);
   126→    try {
   127→      const r = await apiPost<{ ok: true; coupon: { code: string; label: string | null }; discount: number; total: number }>(
   128→        "/api/coupons/validate",
   129→        { code, userId: user.id, subtotal }
   130→      );
   131→      setPromo({ code: r.coupon.code, label: r.coupon.label, discount: r.discount, subtotal });
   132→      setPromoInput("");
   133→      toast.success(`Code ${r.coupon.code} appliqué`, { description: `Remise de ${xof(r.discount)} déduite du total.` });
   134→    } catch (e) {
   135→      setPromo(null);
   136→      toast.error(e instanceof Error ? e.message : "Code promo invalide");
   137→    } finally {
   138→      setPromoChecking(false);
   139→    }
   140→  }
   141→
   142→  async function pay(method: PayMethod) {
   143→    setPaying(true);
   144→    try {
   145→      const items = cart.map((l) => ({ productId: l.productId, qty: l.qty }));
   146→      const r = await apiPost<{ order: ApiOrder; payment: ApiPayment | null; paid: boolean }>("/api/orders", {
   147→        userId: user.id,
   148→        items,
   149→        paymentMethod: method,
   150→        ...(livePromo ? { couponCode: livePromo.code } : {}),
   151→      });
   152→      const amount = r.order.total;
   153→      if (method !== "wallet" && r.payment) {
   154→        setCheckout(false);
   155→        setPayState({ phase: "processing", method, amount });
   156→        await new Promise((res) => setTimeout(res, 3000));
   157→        const c = await apiPost<{ payment: ApiPayment; order?: ApiOrder; wallet?: ApiWallet }>("/api/payments/confirm", { paymentId: r.payment.id });
   158→        if (c.wallet) setWallet(c.wallet);
   159→      } else {
   160→        setCheckout(false);
   161→        setPayState({ phase: "processing", method: "wallet", amount });
   162→        await new Promise((res) => setTimeout(res, 1200));
   163→      }
   164→      setPayState({ phase: "success", method, amount });
   165→      haptic(HAPTIC.success);
   166→      clearCart();
   167→      setPromo(null); // le coupon est consommé : remise à zéro pour la prochaine commande
   168→      void refreshOrders(); // la nouvelle commande apparaît dans « Mes commandes »
   169→      const cb = r.order.cashback;
   170→      toast.success(cb > 0 ? `Commande confirmée — cashback ${xof(cb)} crédité sur ton wallet` : "Commande confirmée et payée");
   171→      if (method === "wallet") {
   172→        const w = await apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).catch(() => null);
   173→        if (w) setWallet(w.wallet);
   174→      }
   175→    } catch (e) {
   176→      setPayState(null);
   177→      toast.error(e instanceof Error ? e.message : "Paiement impossible");
   178→    } finally {
   179→      setPaying(false);
   180→    }
   181→  }
   182→
   183→  const op = (m: PayMethod) => MOMO_OPERATORS.find((o) => o.code === m);
   184→
   185→  return (
   186→    <div className="pt-4 pb-2">
   187→      <header className="flex items-center justify-between">
   188→        <h1 className="font-heading font-black text-xl">Boutique Kènè</h1>
   189→        <span className="rounded-full bg-primary/10 text-primary text-[10px] font-bold px-2.5 py-1 uppercase tracking-wide">Cashback {Math.round(cashbackRate * 100)} %</span>
   190→      </header>
   191→
   192→      {/* Bascule Catalogue ↔ Mes commandes — la cliente consulte ses données */}
   193→      <div className="mt-3 grid grid-cols-2 gap-1 rounded-2xl border border-border bg-card p-1" role="tablist" aria-label="Vues boutique">
   194→        <button
   195→          role="tab"
   196→          aria-selected={view === "catalogue"}
   197→          onClick={() => setView("catalogue")}
   198→          className={`h-10 rounded-xl flex items-center justify-center gap-1.5 text-xs font-bold transition-all active:scale-[0.98] focus-visible:outline-2 focus-visible:outline-primary ${view === "catalogue" ? "bg-primary text-primary-foreground shadow-sm" : "text-muted-foreground hover:text-foreground"}`}
   199→        >
   200→          <ShoppingBag size={14} aria-hidden="true" /> Catalogue
   201→        </button>
   202→        <button
   203→          role="tab"
   204→          aria-selected={view === "commandes"}
   205→          onClick={() => setView("commandes")}
   206→          className={`h-10 rounded-xl flex items-center justify-center gap-1.5 text-xs font-bold transition-all active:scale-[0.98] focus-visible:outline-2 focus-visible:outline-primary ${view === "commandes" ? "bg-primary text-primary-foreground shadow-sm" : "text-muted-foreground hover:text-foreground"}`}
   207→        >
   208→          <History size={14} aria-hidden="true" /> Mes commandes{orders !== null && orders.length > 0 ? ` (${orders.length})` : ""}
   209→        </button>
   210→      </div>
   211→
   212→      {view === "commandes" ? (
   213→        <OrdersView orders={orders} onRefresh={refreshOrders} onShop={() => setView("catalogue")} />
   214→      ) : (
   215→        <>
   216→      {/* Le Fil de Kente — hero tissé, le fil de la catégorie s'illumine */}
   217→      <div className="mt-4">
   218→        <KenteWeaveCard highlightIndex={cat ? categoryThread(cat) : -1} caption={weaveCaption} />
   219→      </div>
   220→
   221→      {/* Recherche */}
   222→      <div className="mt-3 relative">
   223→        <Search size={16} className="absolute left-3.5 top-1/2 -translate-y-1/2 text-muted-foreground" aria-hidden="true" />
   224→        <input
   225→          ref={inputRef}
   226→          value={q}
   227→          onChange={(e) => setQ(e.target.value)}
   228→          placeholder="Rechercher un soin, un botanique…"
   229→          aria-label="Rechercher un produit"
   230→          className="h-12 w-full rounded-2xl border border-border bg-card pl-10 pr-4 text-sm focus-visible:outline-2 focus-visible:outline-primary"
   231→        />
   232→      </div>
   233→
   234→      {/* Catégories */}
   235→      <div className="flex gap-2 overflow-x-auto py-3 scrollbar-thin -mx-1 px-1" role="tablist" aria-label="Catégories">
   236→        {SHOP_CATEGORIES.map((c) => (
   237→          <button
   238→            key={c.id}
   239→            role="tab"
   240→            aria-selected={cat === c.id}
   241→            onClick={() => setCat(c.id)}
   242→            className={`shrink-0 rounded-full px-3.5 min-h-10 text-xs font-semibold transition-all active:scale-95 focus-visible:outline-2 focus-visible:outline-primary ${cat === c.id ? "bg-primary text-primary-foreground shadow" : "border border-border bg-card text-foreground/80"}`}
   243→          >
   244→            {c.label}
   245→          </button>
   246→        ))}
   247→      </div>
   248→
   249→      {/* Grille produits */}
   250→      {products === null ? (
   251→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   252→          {[0, 1, 2, 3].map((i) => (
   253→            <div key={i} className="space-y-2">
   254→              <Skeleton className="aspect-square rounded-2xl" />
   255→              <Skeleton className="h-3.5 w-4/5" />
   256→              <Skeleton className="h-3 w-2/3" />
   257→            </div>
   258→          ))}
   259→        </div>
   260→      ) : filtered.length === 0 ? (
   261→        <EmptyBlock icon={<Search size={22} />} title="Aucun produit trouvé" text="Essaie un autre mot-clé ou une autre catégorie." />
   262→      ) : (
   263→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   264→          {filtered.map((p) => (
   265→            <button
   266→              key={p.id}
   267→              onClick={(e) => onCardTap(p, e)}
   268→              className="relative text-left rounded-2xl border border-border bg-card overflow-hidden shadow-sm active:scale-[0.98] transition-transform hover:border-primary/40 touch-manipulation focus-visible:outline-2 focus-visible:outline-primary"
   269→              aria-label={`${p.name}, ${xof(p.price)} — appuie une fois pour la fiche, deux fois pour l'ajouter au panier`}
   270→            >
   271→              {/* Burst double-tap — panier kente qui jaillit sous le doigt */}
   272→              {burst?.id === p.id && (
   273→                <motion.span
   274→                  aria-hidden="true"
   275→                  initial={{ scale: 0.3, opacity: 0.95 }}
   276→                  animate={{ scale: 1.7, opacity: 0 }}
   277→                  transition={{ duration: 0.65, ease: "easeOut" }}
   278→                  className="pointer-events-none absolute z-20"
   279→                  style={{ left: burst.x, top: burst.y }}
   280→                >
   281→                  <span className="grid place-items-center h-20 w-20 -ml-10 -mt-10 rounded-full bg-[#FFF9EC]/30 backdrop-blur-[2px] shadow-xl">
   282→                    <span className="grid place-items-center h-12 w-12 rounded-full bg-gradient-to-br from-[#A0522D] to-[#8B1A3B] shadow-lg">
   283→                      <ShoppingBag size={22} className="text-[#FFF9EC]" />
   284→                    </span>
   285→                  </span>
   286→                </motion.span>
   287→              )}
   288→              <img src={p.image} alt={p.name} loading="lazy" className="aspect-square w-full object-cover" />
   289→              <div className="p-2.5">
   290→                <p className="text-[13px] font-semibold leading-tight line-clamp-2 min-h-9">{p.name}</p>
   291→                <p className="text-[10px] text-terre mt-1 truncate">{p.botanicals}</p>
   292→                <div className="flex items-center justify-between mt-1.5">
   293→                  <span className="font-mono text-[13px] font-bold">{xof(p.price)}</span>
   294→                  <Stars rating={p.rating} size={9} />
   295→                </div>
   296→              </div>
   297→            </button>
   298→          ))}
   299→        </div>
   300→      )}
   301→        </>
   302→      )}
   303→
   304→      {/* Barre panier sticky au-dessus de la nav */}
   305→      <AnimatePresence>
   306→        {cart.length > 0 && !payState && (
   307→          <motion.div initial={{ y: 80, opacity: 0 }} animate={{ y: 0, opacity: 1 }} exit={{ y: 80, opacity: 0 }} className="sticky bottom-[84px] z-20 mt-4">
   308→            <button
   309→              onClick={() => { setCheckout(true); haptic(HAPTIC.tap); }}
   310→              className="w-full h-14 rounded-2xl bg-gradient-to-r from-[#A0522D] to-[#8B1A3B] text-[#FFF9EC] shadow-xl flex items-center justify-between px-4 active:scale-[0.99] transition-transform focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary"
   311→              aria-label={`Panier ${cart.length} articles, total ${xof(subtotal)} — commander`}
   312→            >
   313→              <span className="flex items-center gap-2 text-sm font-semibold">
   314→                <span className="relative" aria-hidden="true">
   315→                  <ShoppingBag size={20} />
   316→                  <span className="absolute -top-1.5 -right-2 h-4 min-w-4 px-0.5 rounded-full bg-[#FFF9EC] text-[#A0520F] text-[9px] font-black grid place-items-center">{cart.reduce((s, l) => s + l.qty, 0)}</span>
   317→                </span>
   318→                <span aria-hidden="true">{cart.length} article{cart.length > 1 ? "s" : ""}</span>
   319→              </span>
   320→              <span className="font-mono font-black text-base" aria-hidden="true">{xof(subtotal)}</span>
   321→            </button>
   322→          </motion.div>
   323→        )}
   324→      </AnimatePresence>
   325→
   326→      {/* Sheet fiche produit */}
   327→      <Sheet open={!!detail} onOpenChange={(o) => !o && setDetail(null)}>
   328→        <SheetContent side="bottom" className="max-h-[88vh] rounded-t-3xl px-0 overflow-y-auto max-w-[560px] mx-auto">
   329→          {detail && (
   330→            <>
   331→              <div className="mx-auto sticky top-0 z-10 bg-card/95 backdrop-blur pt-2 pb-1">
   332→                <div className="h-1 w-10 rounded-full bg-muted mx-auto" aria-hidden="true" />
   333→              </div>
   334→              <img src={detail.image} alt={detail.name} className="aspect-square w-full object-cover px-0" />
   335→              <SheetHeader className="px-5 pt-4 text-left">
   336→                <SheetTitle className="font-heading font-black text-lg leading-tight">{detail.name}</SheetTitle>
   337→                <div className="flex items-center gap-2">
   338→                  <Stars rating={detail.rating} />
   339→                  <span className="text-[11px] text-muted-foreground">{detail.rating.toFixed(1)} ({detail.reviewCount} avis)</span>
   340→                </div>
   341→              </SheetHeader>
   342→              <div className="px-5 pb-6 space-y-4">
   343→                <p className="text-xs text-muted-foreground leading-relaxed">{detail.description}</p>
   344→                <div className="flex flex-wrap gap-1.5">
   345→                  {detail.botanicals.split(/[,;]/).map((b, i) => (
   346→                    <span key={i} className="rounded-full bg-karite border border-[#C8951E]/30 px-2.5 py-1 text-[10px] font-semibold text-terre">{b.trim()}</span>
   347→                  ))}
   348→                </div>
   349→                <div className="flex items-center justify-between">
   350→                  <div>
   351→                    <p className="font-mono text-xl font-black text-primary">{xof(detail.price)}</p>
   352→                    {detail.compareAt && detail.compareAt > detail.price && <p className="text-[11px] text-muted-foreground line-through">{xof(detail.compareAt)}</p>}
   353→                  </div>
   354→                  <div className="flex items-center gap-3 rounded-full border border-border px-2 py-1.5">
   355→                    <button onClick={() => setQty((n) => Math.max(1, n - 1))} className="h-9 w-9 grid place-items-center rounded-full hover:bg-muted active:scale-90 transition-all" aria-label="Diminuer la quantité"><Minus size={16} /></button>
   356→                    <span className="font-mono font-bold w-5 text-center" aria-live="polite">{qty}</span>
   357→                    <button onClick={() => setQty((n) => Math.min(detail.stock, n + 1))} className="h-9 w-9 grid place-items-center rounded-full hover:bg-muted active:scale-90 transition-all" aria-label="Augmenter la quantité"><Plus size={16} /></button>
   358→                  </div>
   359→                </div>
   360→                <button
   361→                  onClick={() => { addToCart({ productId: detail.id, name: detail.name, price: detail.price, qty, image: detail.image }); toast.success(`${qty} × ${detail.name} ajouté au panier`); setDetail(null); }}
   362→                  className="h-12 w-full rounded-xl bg-primary text-primary-foreground font-semibold text-sm shadow-md active:scale-[0.98] transition-all flex items-center justify-center gap-2 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary"
   363→                >
   364→                  <Plus size={17} /> Ajouter au panier — {xof(detail.price * qty)}
   365→                </button>
   366→              </div>
   367→            </>
   368→          )}
   369→        </SheetContent>
   370→      </Sheet>
   371→
   372→      {/* Sheet checkout */}
   373→      <Sheet open={checkout} onOpenChange={(o) => !o && setCheckout(false)}>
   374→        <SheetContent side="bottom" className="max-h-[90vh] rounded-t-3xl overflow-y-auto max-w-[560px] mx-auto">
   375→          <SheetHeader className="text-left">
   376→            <SheetTitle className="font-heading font-black">Ma commande</SheetTitle>
   377→          </SheetHeader>
   378→          <div className="px-4 pb-6 space-y-4">
   379→            <div className="space-y-2">
   380→              {cart.map((l) => (
   381→                <div key={l.productId} className="flex items-center gap-3 rounded-xl border border-border bg-card p-2">
   382→                  <img src={l.image} alt={l.name} className="h-12 w-12 rounded-lg object-cover" />
