   179→    } catch (e) {

   237→        ownerName: ownerName.trim() || undefined,

   248→        // Lecture de secours : si le compte est déjà pro (et qu'on retrouve
   249→        // le tenant), on entre directement dans l'espace entreprise — sinon
   250→        // on affiche l'erreur renvoyée par l'API.

   262→                : "Tu gères déjà ton espace entreprise — on y retourne",
   263→            );
   264→          }
   265→        } catch {
   266→          /* session injoignable → message du 409 ci-dessous */
   267→        }
   268→        if (!entered) toast.error(e.message || "Un espace entreprise existe déjà pour ce numéro");
   269→      } else {
   270→        toast.error(e instanceof Error ? e.message : "Création impossible");
   271→      }
   272→    } finally {
   273→      setSavingPro(false);
   274→    }
