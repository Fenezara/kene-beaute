   275→    swipeStart.current = null;
   276→    if (!s || !s.ok || refreshing) return;

   278→    const dx = t.clientX - s.x;
   279→    const dy = t.clientY - s.y;

   285→      setNavDir(dx < 0 ? 1 : -1);
   286→      haptic(HAPTIC.tap);
   287→      setClientTab(next);
   288→    }
