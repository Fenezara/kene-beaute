   186→    // Résultat acquis (POST direct ou récupéré après un 502) : même animation,
   187→    // durée minimale ~3,4 s, minuteurs suivis et nettoyés dans tous les cas.
   188→    function showResult(d: { id: string; result: DiagnosisResult; imageData: string; createdAt: string }, recovered = false) {
   189→      const wait = Math.max(0, 3400 - (Date.now() - t0));
   190→      const to1 = setTimeout(() => {
   191→        setProgress(100);
   192→        setCheckedSteps(ANALYSIS_STEPS.length);
   193→        const to2 = setTimeout(() => {
   194→          setDiag(d);
   195→          setAnalyzing(false);
   196→          setNetNotice(null);
   197→          setStep(3);
   198→          clearTimers();

   200→            toast.success("Analyse retrouvée — voici tes résultats", {
   201→              description: "Le réseau a été instable une seconde, mais ton diagnostic était bien enregistré.",
   202→            });
   203→          }
   204→        }, 450);
   205→        timersRef.current.timeouts.push(to2);
   206→      }, wait);
   207→      timersRef.current.timeouts.push(to1);
   208→    }
   209→
   210→    // Récupération après échec réseau : le diagnostic a très bien pu être
   211→    // ENREGISTRÉ avant le redémarrage du serveur (le back crée la ligne
   212→    // « pending » puis la passe à « done » avant de répondre). On va le
   213→    // chercher : done récent (< 4 min) → résultats ; pending → re-poll 3 s
   214→    // × 10 max. Retourne true si les résultats sont affichés.
   215→    async function tryRecover(): Promise<boolean> {
   216→      const RECENT_MS = 4 * 60 * 1000;
   217→      const findRecent = (list: ApiDiagnosis[] | undefined): ApiDiagnosis | undefined =>
   218→        (list ?? [])
   219→          .filter(
   220→            (d) =>
   221→              d.zone === zone &&
   222→              (d.status === "done" || d.status === "pending") &&
   223→              Date.now() - new Date(d.createdAt).getTime() < RECENT_MS,
   224→          )
   225→          .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())[0];
   226→      try {
   227→        const first = findRecent((await apiGet<{ diagnoses: ApiDiagnosis[] }>(`/api/diagnoses?userId=${user.id}`)).diagnoses);
   228→        let found = first;
   229→        let polls = 0;
   230→        while (found?.status === "pending" && polls < 10) {
   231→          polls += 1;
   232→          setNetNotice("Récupération de ton analyse…");
   233→          await new Promise((res) => setTimeout(res, 3000));
   234→          const again = await apiGet<{ diagnoses: ApiDiagnosis[] }>(`/api/diagnoses?userId=${user.id}`);
   235→          found = findRecent(again.diagnoses);
   236→        }
   237→        if (!found || found.status !== "done") return false;
   238→        const result = parseDiagnosis(found.resultJson);
   239→        if (!result) return false;
   240→        showResult(
   241→          { id: found.id, result, imageData: diagImgSrc(found.imageData), createdAt: found.createdAt },
   242→          true,
   243→        );
   244→        return true;
   245→      } catch {
   246→        // Le GET lui-même est injoignable (serveur toujours au redémarrage) :
   247→        // échec final honnête, sans crash.
   248→        return false;
   249→      }
   250→    }
   251→
   252→    try {
   253→      // POST /api/diagnoses : 3 tentatives quand la gateway renvoie 502/503/504
   254→      // (fenêtre transitoire de redémarrage du serveur Next) — backoff 1,5 s
   255→      // puis 4 s. Les autres erreurs (400/404/429…) ne sont JAMAIS rejouées.
   256→      let r: { diagnosis: ApiDiagnosis } | undefined;
   257→      let fatal: unknown = new Error("Analyse impossible");
   258→      const backoffs = [1500, 4000];
   259→      for (let attempt = 0; attempt < 3; attempt += 1) {
   260→        try {
   261→          r = await apiPost<{ diagnosis: ApiDiagnosis }>("/api/diagnoses", {
   262→            userId: user.id,
   263→            zone,
   264→            image,
   265→            fitzpatrick: user.fitzpatrick ?? undefined,
   266→            allergies: user.allergies ?? undefined,
   267→          });
   268→          break;
   269→        } catch (e) {
   270→          fatal = e;
   271→          const retryable = e instanceof ApiError && [502, 503, 504].includes(e.status);
   272→          if (!retryable || attempt === 2) break;
   273→          setNetNotice("Rétablissement de la connexion…");
   274→          await new Promise((res) => setTimeout(res, backoffs[attempt]));
   275→          setNetNotice(null);
   276→        }
   277→      }
   278→      if (!r) throw fatal;
   279→      const result = parseDiagnosis(r.diagnosis.resultJson);
   280→      if (!result) throw new Error("Résultat IA illisible");
   281→      showResult({ id: r.diagnosis.id, result, imageData: image, createdAt: r.diagnosis.createdAt });
   282→    } catch (e) {
   283→      // 502/503/504 après 3 tentatives : tenter la récupération AVANT l'erreur.
   284→      const gatewayish = e instanceof ApiError && [502, 503, 504].includes(e.status);
   285→      if (gatewayish && (await tryRecover())) return;
   286→      clearTimers();
   287→      setAnalyzing(false);
   288→      setNetNotice(null);
   289→      toast.error(
   290→        gatewayish
   291→          ? "Moteur d'analyse momentanément injoignable — tes photos restent prêtes, retente dans un instant."
   292→          : e instanceof Error
   293→            ? e.message
   294→            : "Analyse impossible",
   295→      );
   296→      setStep(1); // retour capture : la photo est conservée
   297→    }
   298→  }

   343→          <ArrowLeft size={15} /> Changer de zone

   346→        <h2 className="font-heading font-black text-xl mt-3">Prends ta photo</h2>

   348→        <ul className="mt-4 space-y-2">

   373→            >
   374→              <span className="flex flex-col items-center gap-3 text-primary">

   411→          <img src={image} alt="Photo en cours d'analyse" className="aspect-square w-full object-cover" />
   412→          <motion.div
   413→            className="absolute left-0 right-0 h-16 bg-gradient-to-b from-transparent via-[#C8951E]/50 to-transparent border-y-2 border-[#C8951E]"
   414→            animate={{ top: ["8%", "78%", "8%"] }}
   415→            transition={{ duration: 2.6, repeat: Infinity, ease: "easeInOut" }}
   416→            aria-hidden="true"
   417→          />
   418→        </RevealItem>
   419→        <RevealItem className="mt-6 w-full max-w-[320px]">
   420→          <GlassCard hero>
   421→            <div className="divide-y divide-border/60">
   422→              {ANALYSIS_STEPS.map((label, i) => {
   423→                const done = i < checkedSteps;
   424→                const active = i === checkedSteps;
   425→                return (
   426→                  <div key={i} className={`flex items-center gap-3 p-3 transition-colors ${done ? "bg-success/5" : active ? "bg-primary/5" : "opacity-50"}`}>
   427→                    <span className={`grid place-items-center h-6 w-6 rounded-full shrink-0 ${done ? "bg-success text-white" : active ? "bg-primary/20 text-primary" : "bg-muted text-muted-foreground"}`}>

   825→/* ══════════════ Historique + comparaison ══════════════ */
   826→function HistoryView({
   827→  userId, history, compareMode, setCompareMode, compareSel, setCompareSel, onBack, onOpen,
   828→}: {
   829→  userId: string;
   830→  history: ApiDiagnosis[] | null;
   831→  compareMode: boolean;
   832→  setCompareMode: (v: boolean) => void;
   833→  compareSel: string[];
   834→  setCompareSel: (v: string[]) => void;
   835→  onBack: () => void;
   836→  onOpen: (d: ApiDiagnosis) => void;
   837→}) {
   838→  const list = useMemo(
   839→    () => [...(history ?? [])].filter((d) => d.status === "done").sort((a, b) => +new Date(b.createdAt) - +new Date(a.createdAt)),
   840→    [history]
   841→  );
   842→  const sameZoneSel = compareSel.length === 2 && list.find((d) => d.id === compareSel[0])?.zone === list.find((d) => d.id === compareSel[1])?.zone;
   843→  const a = list.find((d) => d.id === compareSel[0]);
   844→  const b = list.find((d) => d.id === compareSel[1]);
   845→  const ra = a ? parseDiagnosis(a.resultJson) : null;
   846→  const rb = b ? parseDiagnosis(b.resultJson) : null;
   847→
   848→  const evolutions =
   849→    ra && rb
   850→      ? rb.indicateurs
   851→          .map((ib) => {
   852→            const ia = ra.indicateurs.find((x) => x.nom === ib.nom);
   853→            return ia ? { nom: ib.nom, delta: ib.pourcentage - ia.pourcentage } : null;
   854→          })
   855→          .filter((x): x is { nom: string; delta: number } => x !== null)
   856→          .sort((x, y) => Math.abs(y.delta) - Math.abs(x.delta))
   857→          .slice(0, 3)
   858→      : [];
   859→
   860→  return (
   861→    <div className="pt-4 pb-2">
   862→      <button onClick={onBack} className="inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground mb-4 focus-visible:outline-2 focus-visible:outline-primary rounded" aria-label="Retour">
   863→        <ArrowLeft size={15} /> Nouveau diagnostic
   864→      </button>
   865→      <div className="flex items-center justify-between gap-2">
   866→        <h2 className="font-heading font-black text-xl">Mon historique</h2>
   867→        <Chip selected={compareMode} onClick={() => { setCompareMode(!compareMode); setCompareSel([]); }}>
   868→          <GitCompareArrows size={14} /> Comparer
   869→        </Chip>
   870→      </div>
   871→
   872→      {history === null ? (
   873→        <div className="space-y-3 mt-5">
   874→          {[0, 1, 2].map((i) => (
   875→            <Shimmer key={i} className="h-20 rounded-[18px]" />
   876→          ))}
   877→        </div>
   878→      ) : list.length === 0 ? (
   879→        <div className="mt-6">
   880→          <EmptyBlock
   881→            icon={<NeaOnnimIcon size={30} />}
   882→            title="Aucun diagnostic pour l'instant"
   883→            text="Ton premier scan débloquera le suivi d'évolution."
   884→          />
   885→        </div>
   886→      ) : (
   887→        <>
   888→          {/* Le Fil du Temps — courbes d'évolution */}
   889→          <EvolutionCard userId={userId} className="mt-4" />
   890→
   891→          <Reveal className="space-y-2.5 mt-4" stagger={0.07}>
   892→            {list.map((d) => {
   893→              const sel = compareSel.includes(d.id);
   894→              return (
   895→                <RevealItem key={d.id}>
   896→                <div className={`k-card flex items-center gap-3 rounded-[18px] p-2.5 ${sel ? "ring-2 ring-primary/60" : ""}`}>
   897→                  <button
   898→                    onClick={() => {
   899→                      if (compareMode) {
   900→                        setCompareSel(sel ? compareSel.filter((x) => x !== d.id) : compareSel.length < 2 ? [...compareSel, d.id] : [compareSel[1], d.id]);
   901→                      } else onOpen(d);
   902→                    }}
   903→                    className="flex items-center gap-3 flex-1 min-w-0 text-left focus-visible:outline-2 focus-visible:outline-primary rounded-xl"
   904→                    aria-label={`Diagnostic ${d.zone} du ${formatDate(d.createdAt)}, score ${d.scoreGlobal}`}
   905→                  >
   906→                    <img src={diagImgSrc(d.imageData)} alt={`Diagnostic ${d.zone}`} loading="lazy" className="h-14 w-14 rounded-xl object-cover shrink-0" />
   907→                    <div className="min-w-0">
   908→                      <p className="text-sm font-semibold">{BODY_ZONES.find((z) => z.id === d.zone)?.label ?? d.zone}</p>
   909→                      <p className="text-[11px] text-muted-foreground">{formatDate(d.createdAt)}</p>
   910→                    </div>
   911→                    <div className="ml-auto"><ScoreChip score={d.scoreGlobal} /></div>
   912→                  </button>
   913→                  {compareMode && (
   914→                    <button
   915→                      onClick={() => setCompareSel(sel ? compareSel.filter((x) => x !== d.id) : compareSel.length < 2 ? [...compareSel, d.id] : [compareSel[1], d.id])}
   916→                      aria-pressed={sel}
   917→                      aria-label="Sélectionner pour comparaison"
   918→                      className={`h-9 w-9 grid place-items-center rounded-full border-2 shrink-0 transition-colors ${sel ? "bg-primary border-primary text-primary-foreground" : "border-border"}`}
   919→                    >
   920→                      {sel && <Check size={14} />}
   921→                    </button>
   922→                  )}
   923→                </div>
   924→                </RevealItem>
   925→              );
   926→            })}
   927→          </Reveal>
   928→
   929→          {compareMode && (
   930→            <AnimatePresence>
   931→              {compareSel.length === 2 && sameZoneSel && ra && rb && a && b ? (
   932→                <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: "auto" }} exit={{ opacity: 0, height: 0 }} className="mt-5 overflow-hidden">
   933→                  <div className="k-card rounded-[24px] p-4">
   934→                    <p className="font-heading font-bold text-sm mb-3 flex items-center gap-2">
   935→                      <GitCompareArrows size={15} className="text-primary" /> Avant / Après — {BODY_ZONES.find((z) => z.id === a.zone)?.label}
   936→                    </p>
   937→                    <div className="grid grid-cols-2 gap-3">
   938→                      {[{ d: a, r: ra, tag: "Avant" }, { d: b, r: rb, tag: "Après" }].map(({ d, r, tag }) => (
   939→                        <div key={tag}>
   940→                          <img src={diagImgSrc(d.imageData)} alt={`${tag} — ${d.zone}`} loading="lazy" className="aspect-square w-full rounded-2xl object-cover border border-border" />
   941→                          <div className="flex items-center justify-between mt-1.5">
   942→                            <span className="text-[10px] uppercase tracking-wide font-bold text-muted-foreground">{tag} · {formatDate(d.createdAt, { day: "numeric", month: "short" })}</span>
   943→                            <ScoreChip score={r.score_global} />
   944→                          </div>
   945→                        </div>
   946→                      ))}
   947→                    </div>
   948→                    <div className="mt-3 rounded-xl bg-muted/60 p-3 flex items-center justify-between">
   949→                      <span className="text-xs font-semibold">Évolution du score</span>
   950→                      <span className="font-mono text-lg font-black tabular-nums" style={{ color: rb.score_global >= ra.score_global ? "#346834" : "#8B1A3B" }}>
   951→                        {rb.score_global >= ra.score_global ? "+" : ""}{rb.score_global - ra.score_global} pts
   952→                      </span>
   953→                    </div>
   954→                    {evolutions.length > 0 && (
   955→                      <div className="mt-3 space-y-1.5">
   956→                        {evolutions.map((e) => (
   957→                          <div key={e.nom} className="flex items-center justify-between text-[11px]">
   958→                            <span className="text-muted-foreground">{e.nom}</span>
   959→                            <span className="font-mono font-bold tabular-nums" style={{ color: e.delta >= 0 ? "#346834" : "#8B1A3B" }}>
   960→                              {e.delta >= 0 ? "+" : ""}{e.delta} %
   961→                            </span>
   962→                          </div>
   963→                        ))}
   964→                      </div>
   965→                    )}
   966→                  </div>
   967→                </motion.div>
   968→              ) : (
   969→                <p className="mt-4 text-center text-[11px] text-muted-foreground">Sélectionne 2 diagnostics de la même zone pour comparer l&apos;évolution.</p>
   970→              )}
   971→            </AnimatePresence>
   972→          )}
   973→        </>
   974→      )}
   975→      <div className="h-2" />
   976→    </div>
   977→  );
   978→}
