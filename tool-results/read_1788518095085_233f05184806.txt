   233→  { n: 2, label: "Questionnaire" },

   238→const ANALYSIS_STEPS = [

   245→function DiagWizard({
   246→  tenantId,
   247→  preselectClientId,
   248→  onClose,
   249→  onSaved,
   250→}: {
   251→  tenantId: string;
   252→  preselectClientId?: string | null;
   253→  onClose: () => void;
   254→  onSaved: () => void;
   255→}) {
   256→  const [step, setStep] = useState(1);
   257→  const [zone, setZone] = useState<BodyZone>("visage");
   258→  const [selected, setSelected] = useState<WizardClient | null>(null);
   259→  const [answers, setAnswers] = useState<QAnswers>(defaultAnswers);
   260→  const [photo, setPhoto] = useState<string | null>(null);
   261→  const [practitioner, setPractitioner] = useState("Fatou Koné");
   262→  const [submitting, setSubmitting] = useState(false);
   263→  const [result, setResult] = useState<ProDiagnosisResult | null>(null);
   264→  const [submitError, setSubmitError] = useState<string | null>(null);
   265→  const [analysisMsg, setAnalysisMsg] = useState(ANALYSIS_STEPS[0]);
   266→
   267→  // Message d'attente rotatif pendant l'analyse VLM (10-30 s possibles)
   268→  const msgIdx = useRef(0);
   269→  useEffect(() => {
   270→    if (!submitting) return;
   271→    msgIdx.current = 0;
   272→    setAnalysisMsg(ANALYSIS_STEPS[0]);
   273→    const t = window.setInterval(() => {
   274→      msgIdx.current = (msgIdx.current + 1) % ANALYSIS_STEPS.length;
   275→      setAnalysisMsg(ANALYSIS_STEPS[msgIdx.current]);
   276→    }, 4_000);
   277→    return () => window.clearInterval(t);
   278→  }, [submitting]);
   279→
   280→  const progress = useMemo(() => questionnaireProgress(answers), [answers]);
   281→  const zoneLabel = BODY_ZONES.find((z) => z.id === zone)?.label ?? zone;
   282→
   283→  const reset = () => {
   284→    setStep(1);
   285→    setZone("visage");
   286→    setSelected(null);
   287→    setAnswers(defaultAnswers());
   288→    setPhoto(null);
   289→    setResult(null);
   290→    setSubmitError(null);
   291→  };
   292→
   293→  const submit = async () => {
   294→    setSubmitting(true);
   295→    setSubmitError(null);
   296→    setResult(null);
   297→    setStep(4);
   298→    try {
   299→      const res = await apiPost<{ diagnosis: ProDiagnosisItem; result: ProDiagnosisResult; reusedClient: boolean }>(
   300→        "/api/pro/diagnoses",
   301→        {
   302→          tenantId,
   303→          zone,
   304→          answers,
   305→          clientProfileId: selected?.id,
   306→          photo: photo ?? undefined,
   307→          practitioner: practitioner.trim() || undefined,
   308→        }
   309→      );
   310→      setResult(res.result);
   311→      toast.success(`Diagnostic enregistré — ${res.diagnosis.clientName} · ${res.result.score_global}/100`, {
   312→        description: res.result.questionnaire.flags.length > 0 ? `${res.result.questionnaire.flags.length} point(s) de vigilance noté(s) — visible dans la fiche CRM.` : "Résultat fusionné disponible dans le CRM.",
   313→        duration: 7_000,
   314→      });
   315→      if (res.reusedClient) toast.info("Fiche CRM existante réutilisée (téléphone déjà enregistré).");
   316→      onSaved();
   317→    } catch (e) {
   318→      setSubmitError(e instanceof Error ? e.message : "Analyse impossible");
   319→      toastError(e, "Analyse impossible");
   320→    } finally {
   321→      setSubmitting(false);
   322→    }
   323→  };
   324→
   325→  return (
   326→    <Sheet open onOpenChange={(o) => !o && onClose()}>
   327→      <SheetContent side="right" aria-describedby={undefined} className="w-full sm:max-w-xl overflow-y-auto pretty-scroll p-0 [&>button]:z-30">
   328→        <SheetHeader className="p-4 pb-3 border-b border-border bg-muted/40 sticky top-0 z-10">
   329→          <SheetTitle className="font-heading text-lg leading-tight">
   330→            Diagnostic en cabine {result ? "— résultat" : `— étape ${step}/3`}
   331→          </SheetTitle>
   332→          <SheetDescription className="text-xs">
   333→            {result
   334→              ? `${selected?.name ?? ""} · ${zoneLabel.toLowerCase()} · ${practitioner || "praticienne"}`.replace(/^ · /, "")
   335→              : "Questionnaire dermatologique ± photo — fusionnée en un score unique."}
   336→          </SheetDescription>
   337→          {/* Stepper */}
   338→          <div className="mt-2 flex items-center gap-1.5" role="group" aria-label="Étapes du diagnostic">
   339→            {STEPS.map((s, i) => {
   340→              const state = result ? 4 : step;
   341→              const active = s.n === state;
   342→              const done = s.n < state;
   343→              return (
   344→                <div key={s.n} className="flex items-center gap-1.5 min-w-0">
   345→                  <span
   346→                    className={cn(
   347→                      "inline-flex items-center gap-1.5 rounded-full px-2.5 py-1 text-[11px] font-medium transition-colors",
   348→                      active ? "bg-primary text-primary-foreground" : done ? "bg-success/15 text-success" : "bg-muted text-muted-foreground"
   349→                    )}
   350→                    aria-current={active ? "step" : undefined}
   351→                  >
   352→                    <span className="font-mono">{done ? "✓" : s.n}</span>
   353→                    <span className="hidden xs:inline sm:inline">{s.label}</span>
   354→                  </span>
   355→                  {i < STEPS.length - 1 && <span className="h-px w-3 bg-border shrink-0" aria-hidden="true" />}
   356→                </div>
   357→              );
   358→            })}
   359→          </div>
   360→        </SheetHeader>
   361→
   362→        <div className="p-4 space-y-4">
   363→          {step === 1 && (
   364→            <ClientStep
   365→              tenantId={tenantId}
   366→              preselectClientId={preselectClientId}
   367→              selected={selected}
   368→              onSelect={setSelected}
   369→              zone={zone}
   370→              onZone={setZone}
   371→              practitioner={practitioner}
   372→              onPractitioner={setPractitioner}
   373→              onNext={() => setStep(2)}
   374→            />
   375→          )}
   376→          {step === 2 && (
   377→            <QuestionnaireStep
   378→              answers={answers}
   379→              onChange={setAnswers}
   380→              progress={progress}
   381→              onBack={() => setStep(1)}
   382→              onNext={() => setStep(3)}
   383→            />
   384→          )}
   385→          {step === 3 && (
   386→            <PhotoStep
   387→              photo={photo}
   388→              onPhoto={setPhoto}
   389→              zoneLabel={zoneLabel}
   390→              onBack={() => setStep(2)}
   391→              onSubmit={submit}
   392→            />
   393→          )}
   394→          {step === 4 && (
   395→            <ResultStep
   396→              submitting={submitting}
   397→              analysisMsg={analysisMsg}
   398→              error={submitError}
   399→              result={result}
   400→              clientName={selected?.name}
   401→              onRetryBack={() => setStep(3)}
   402→              onNew={reset}
   403→              onDone={onClose}
   404→            />
   405→          )}
   406→        </div>
   407→      </SheetContent>
   408→    </Sheet>
   409→  );
   410→}

   413→function ClientStep({
   414→  tenantId,
   415→  preselectClientId,
   416→  selected,
   417→  onSelect,
   418→  zone,
   419→  onZone,
   420→  practitioner,
   421→  onPractitioner,
   422→  onNext,
   423→}: {
   424→  tenantId: string;
   425→  preselectClientId?: string | null;
   426→  selected: WizardClient | null;
   427→  onSelect: (c: WizardClient) => void;
   428→  zone: BodyZone;
   429→  onZone: (z: BodyZone) => void;
   430→  practitioner: string;
   431→  onPractitioner: (v: string) => void;
   432→  onNext: () => void;
   433→}) {
   434→  const [query, setQuery] = useState("");
   435→  const [debounced, setDebounced] = useState("");
   436→  const [express, setExpress] = useState({ name: "", phone: "" });
   437→  const [creating, setCreating] = useState(false);
   438→
   439→  useEffect(() => {
   440→    const t = window.setTimeout(() => setDebounced(query.trim()), 300);
   441→    return () => window.clearTimeout(t);
   442→  }, [query]);
   443→
   444→  // Préselection CRM : charge la fiche dès l'ouverture
   445→  const preselected = useApi<WizardClient | null>(
   446→    () =>
   447→      preselectClientId
   448→        ? apiGet<{ client: { id: string; name: string; phone: string } }>(`/api/pro/clients/${preselectClientId}`).then((r) => r.client)
   449→        : Promise.resolve(null),
   450→    [preselectClientId]
   451→  );
   452→  useEffect(() => {
   453→    if (preselected.data && !selected) onSelect(preselected.data);
   454→  }, [preselected.data, selected, onSelect]);
   455→
   456→  const results = useApi<ProClient[]>(
   457→    () =>
   458→      tenantId && debounced.length >= 2
   459→        ? apiGet<{ clients: ProClient[] }>(`/api/pro/clients?tenantId=${tenantId}&q=${encodeURIComponent(debounced)}`).then((r) => r.clients ?? [])
   460→        : Promise.resolve([]),
   461→    [tenantId, debounced]
   462→  );
   463→
   464→  const createExpress = async () => {
   465→    if (express.name.trim().length < 2) return toast.error("Nom trop court");
   466→    if (express.phone.replace(/\D/g, "").length < 8) return toast.error("Téléphone invalide (8 chiffres min.)");
   467→    setCreating(true);
   468→    try {
   469→      const res = await apiPost<{ client: ProClient; reused: boolean }>("/api/pro/clients", {
   470→        tenantId,
   471→        name: express.name,
   472→        phone: express.phone,
   473→      });
   474→      onSelect({ id: res.client.id, name: res.client.name, phone: res.client.phone });
   475→      toast.success(res.reused ? `Fiche existante retrouvée : ${res.client.name}` : `Cliente créée : ${res.client.name}`);
   476→    } catch (e) {
   477→      toastError(e, "Création impossible");
   478→    } finally {
   479→      setCreating(false);
   480→    }
   481→  };
   482→
   483→  return (
   484→    <div className="space-y-5">
   485→      {/* Cliente sélectionnée */}
   486→      {selected ? (
   487→        <Card className="border-success/40 bg-success/5">
   488→          <CardContent className="p-3.5 flex items-center gap-3">
