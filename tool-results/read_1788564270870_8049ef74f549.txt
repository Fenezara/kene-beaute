     4→// vocale TTS, CTA scan, Route de l'Or, prochain RDV, wallet, recommandations

     6→
     7→import { useEffect, useMemo, useState } from "react";
     8→import { motion } from "framer-motion";
     9→import {
    10→  BellRing,
    11→  Brush,
    12→  CalendarClock,
    13→  CheckCircle2,
    14→  ChevronRight,
    15→  CircleDot,
    16→  Hand,

    25→} from "lucide-react";
    26→import { apiGet } from "@/lib/kene/api";
    27→import { formatDate, formatTime, scoreColor, readableTextColor, xof, CASHBACK_RATE } from "@/lib/kene/format";
    28→import { nextClientStep } from "@/lib/kene/followups";
    29→import { channelLabel, humanWhen } from "@/lib/kene/reminders";
    30→import { BODY_ZONES, type BodyZone } from "@/lib/kene/types";
    31→import { NeaOnnimIcon, SankofaIcon } from "@/components/kene/icons";
    32→import { RitualJourney } from "@/components/kene/route/RitualJourney";
    33→import { Skeleton } from "@/components/ui/skeleton";
    34→import { useKene } from "@/store/kene";
    35→import { VoiceNarration } from "./VoiceNarration";
    36→import type { ApiAppointment, ApiDiagnosis, ApiProduct, ApiReminderFeed, ApiWallet } from "./types";
    37→import { parseDiagnosis } from "./types";
    38→import { ScoreGauge, ScrollFadeRow, SectionTitle, Stars, WalletPill } from "./bits";
    39→
    40→interface HomeData {
    41→  diagnoses: ApiDiagnosis[];
    42→  appointments: ApiAppointment[];
    43→  wallet: ApiWallet | null;
    44→  products: ApiProduct[];
    45→  reminders: ApiReminderFeed | null;
    46→}
    47→
    48→/** Icône par zone de scan (stories du feed) */
    49→const ZONE_ICON: Record<BodyZone, React.ComponentType<{ className?: string }>> = {
    50→  visage: ScanFace,
    51→  dos: PersonStanding,
    52→  cuir_chevelu: Waves,
    53→  mains: Hand,
    54→  barbe: Brush,
    55→  naevi: CircleDot,
    56→};
    57→
    58→/** Libellé court des stories (2 lignes autorisées, façon Instagram) */
    59→const STORY_LABEL: Record<BodyZone, string> = {
    60→  visage: "Visage",
    61→  dos: "Dos",
    62→  cuir_chevelu: "Cuir chevelu",
    63→  mains: "Mains",
    64→  barbe: "Barbe",
    65→  naevi: "Grains de beauté",
    66→};
    67→
    68→export function HomeScreen({
    69→  onScanZone,
    70→  refreshKey = 0,
    71→  onRefreshed,
    72→}: {
    73→  onScanZone: (z: BodyZone) => void;
    74→  /** Incrémenté par le pull-to-refresh du shell — déclenche un rechargement */
    75→  refreshKey?: number;
    76→  /** Appelé à la fin du chargement (le shell ferme l'indicateur de tirage) */
    77→  onRefreshed?: () => void;
    78→}) {
    79→  const user = useKene((s) => s.user)!;
    80→  const setClientTab = useKene((s) => s.setClientTab);
    81→  const [data, setData] = useState<HomeData | null>(null);
    82→  const [err, setErr] = useState<string | null>(null);
    83→  const [ritualOpen, setRitualOpen] = useState(false);
    84→
    85→  useEffect(() => {
    86→    let alive = true;
    87→    (async () => {
    88→      try {
    89→        const [d, a, w, p, r] = await Promise.all([
    90→          apiGet<{ diagnoses: ApiDiagnosis[] }>(`/api/diagnoses?userId=${user.id}`),
    91→          apiGet<{ appointments: ApiAppointment[] }>(`/api/appointments?userId=${user.id}`),
    92→          apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).catch(() => null),
    93→          apiGet<{ products: ApiProduct[] }>("/api/shop/products"),
    94→          apiGet<ApiReminderFeed>(`/api/notifications?userId=${user.id}`).catch(() => null),
    95→        ]);
    96→        if (alive) {
    97→          setData({
    98→            diagnoses: d.diagnoses ?? [],
    99→            appointments: a.appointments ?? [],
   100→            wallet: w?.wallet ?? null,
   101→            products: p.products ?? [],
   102→            reminders: r ?? null,
   103→          });
   104→          if (alive) onRefreshed?.();
   105→        }
   106→      } catch (e) {
   107→        if (alive) {
   108→          setErr(e instanceof Error ? e.message : "Chargement impossible");
   109→          onRefreshed?.();
   110→        }
   111→      }
   112→    })();
   113→    return () => {
   114→      alive = false;
   115→    };
   116→  }, [user.id, refreshKey, onRefreshed]);
   117→
   118→  // Score multi-zones pondéré (PRD §8.8) — dernier diagnostic par zone
   119→  const multi = useMemo(() => {
   120→    const byZone = new Map<BodyZone, ApiDiagnosis>();
   121→    [...(data?.diagnoses ?? [])]
   122→      .filter((d) => d.status === "done")
   123→      .sort((x, y) => +new Date(y.createdAt) - +new Date(x.createdAt))
   124→      .forEach((d) => {
   125→        if (!byZone.has(d.zone)) byZone.set(d.zone, d);
   126→      });
   127→    const covered = BODY_ZONES.filter((z) => byZone.has(z.id));
   128→    const weightSum = covered.reduce((s, z) => s + z.weight, 0);
   129→    const score =
   130→      covered.length && weightSum > 0
   131→        ? Math.round(covered.reduce((s, z) => s + (byZone.get(z.id)?.scoreGlobal ?? 0) * z.weight, 0) / weightSum)
   132→        : null;
   133→    const missing = BODY_ZONES.filter((z) => !byZone.has(z.id));
   134→    const last = covered.length ? byZone.get(covered[0].id) : null;
   135→    return { byZone, covered, score, missing, last };
   136→  }, [data?.diagnoses]);
   137→
   138→  const nextAppt = useMemo(() => {
   139→    const now = Date.now();
   140→    return (
   141→      [...(data?.appointments ?? [])]
   142→        .filter((a) => new Date(a.startAt).getTime() >= now && a.status !== "cancelled")
   143→        .sort((x, y) => +new Date(x.startAt) - +new Date(y.startAt))[0] ?? null
   144→    );
   145→  }, [data?.appointments]);
   146→
   147→  const reco = useMemo(() => {
   148→    const all = data?.products ?? [];
   149→    const byCat = (cats: string[]) => all.filter((p) => cats.includes(p.category));
   150→    const pick =
   151→      user.skinType === "grasse"
   152→        ? byCat(["serum", "savon"])
   153→        : user.skinType === "seche"
   154→          ? byCat(["creme", "huile"])
   155→          : byCat(["serum", "creme"]);
   156→    return (pick.length >= 3 ? pick : [...pick, ...all.filter((p) => !pick.includes(p))]).slice(0, 3);
   157→  }, [data?.products, user.skinType]);
   158→
   159→  // Route de l'Or : dernier diagnostic parsable → rituel tissable depuis l'accueil
   160→  const lastResult = useMemo(() => (multi.last ? parseDiagnosis(multi.last.resultJson) : null), [multi.last]);
   161→
   162→  // Le Fil du Retour : prochaine étape dérivée de l'activité (contrôle protocole / soin de suite)
   163→  const nextStep = useMemo(
   164→    () => (data ? nextClientStep(new Date(), data.diagnoses, data.appointments) : null),
   165→    [data],
   166→  );
   167→
   168→  const first = user.name.split(" ")[0];
   169→
   170→  return (
   171→    <div className="flex flex-col gap-6 pt-1">
   172→      {/* ───── Salutation ───── */}
   173→      <header className="flex items-end justify-between gap-3">
   174→        <div className="min-w-0">
   175→          <p className="text-[11px] uppercase tracking-[0.16em] text-primary font-semibold">Bonjour</p>
   176→          <h2 className="font-black text-[26px] leading-tight tracking-tight truncate"><span className="kente-text">{first}</span> ✨</h2>
   177→          <p className="flex items-center gap-1 text-[11px] text-muted-foreground mt-0.5">
   178→            <MapPin size={11} /> {user.city || "Abidjan"} · {user.fitzpatrick ? `Fitzpatrick ${user.fitzpatrick}` : "Phototype à définir"}
   179→          </p>
   180→        </div>
   181→        {data?.wallet && <WalletPill balance={data.wallet.balance} onClick={() => setClientTab("profil")} />}
   182→      </header>
   183→
   184→      {/* ───── Stories : scan rapide + zones avec score ───── */}
   185→      <section aria-label="Scan rapide par zone" className="-mx-3 sm:-mx-5 px-3 sm:px-5">
   186→        <ScrollFadeRow label="Stories des zones — fais défiler horizontalement" className="flex gap-3.5 overflow-x-auto no-scrollbar py-1.5 pr-2">
   187→          {/* Story Scanner */}
   188→          <button
   189→            onClick={() => setClientTab("diagnostic")}
   190→            aria-label="Scanner ma peau — nouveau diagnostic"
   191→            className="shrink-0 w-[68px] flex flex-col items-center gap-1.5 focus-visible:outline-2 focus-visible:-outline-offset-2 focus-visible:outline-primary rounded-xl"
   192→          >
   193→            <span className="grid place-items-center h-[66px] w-[66px] rounded-full p-[3px] bg-gradient-to-br from-[#C8951E] via-[#A0522D] to-[#8B1A3B] shadow-md active:scale-95 transition-transform">
   194→              <span className="grid place-items-center h-full w-full rounded-full bg-background">
   195→                <NeaOnnimIcon size={26} className="text-primary" />
   196→              </span>
   197→            </span>
   198→            <span className="text-[10px] font-bold text-primary">Scanner</span>
   199→          </button>
   200→
   201→          {/* Stories zones */}
   202→          {BODY_ZONES.map((z) => {
   203→            const d = multi.byZone.get(z.id);
   204→            const covered = !!d;
   205→            const Icon = ZONE_ICON[z.id];
   206→            const color = covered ? scoreColor(d!.scoreGlobal) : "var(--border)";
   207→            return (
   208→              <button
   209→                key={z.id}
   210→                onClick={() => onScanZone(z.id)}
   211→                aria-label={`${covered ? `Re-scanner ${z.label} — dernier score ${d!.scoreGlobal}` : `Scanner ${z.label} pour compléter ton score`} · pondération ${Math.round(z.weight * 100)} %`}
   212→                className="shrink-0 w-[68px] flex flex-col items-center gap-1.5 focus-visible:outline-2 focus-visible:-outline-offset-2 focus-visible:outline-primary rounded-xl"
   213→              >
   214→                <span
   215→                  className="relative grid place-items-center h-[66px] w-[66px] rounded-full p-[3px] active:scale-95 transition-transform shadow-md"
   216→                  style={
   217→                    covered
   218→                      ? { background: `conic-gradient(from 210deg, ${color} 0%, ${color} 45%, #C8951E 68%, ${color} 100%)` }
   219→                      : undefined
   220→                  }
   221→                >
   222→                  <span
   223→                    className={`grid place-items-center h-full w-full rounded-full bg-background ${covered ? "" : "border-[2.5px] border-dashed border-border"}`}
   224→                  >
   225→                    <Icon className={covered ? "text-foreground/80" : "text-muted-foreground"} />
   226→                  </span>
   227→                  {!covered && (
   228→                    <span className="absolute -bottom-0.5 -right-0.5 grid place-items-center h-6 w-6 rounded-full bg-primary text-primary-foreground border-2 border-background">
   229→                      <Plus size={13} strokeWidth={2.5} />
   230→                    </span>
   231→                  )}
   232→                  {covered && (
   233→                    <span
   234→                      className="absolute -bottom-1 left-1/2 -translate-x-1/2 rounded-full px-1.5 py-px font-mono text-[10px] font-bold tabular-nums shadow-sm"
   235→                      style={{ backgroundColor: color, color: readableTextColor(color) }}
   236→                    >
   237→                      {d!.scoreGlobal}
   238→                    </span>
   239→                  )}
   240→                </span>
   241→                <span className={`text-[11px] w-full text-center leading-tight line-clamp-2 break-words min-h-[26px] ${covered ? "font-semibold" : "text-muted-foreground font-medium"}`}>{STORY_LABEL[z.id]}</span>
   242→              </button>
   243→            );
   244→          })}
   245→        </ScrollFadeRow>
   246→      </section>
   247→
   248→      {err && <p className="rounded-xl bg-destructive/10 text-destructive text-xs p-3">{err}</p>}
   249→
   250→      {/* ───── Carte score multi-zones (avec lecture vocale) ───── */}
   251→      <section aria-labelledby="sc-t" className="rounded-[26px] border border-border/70 overflow-hidden shadow-[0_18px_44px_-18px_rgba(28,17,9,0.28)] bg-gradient-to-br from-card via-card to-primary/[0.08]">
   252→        <div className="kente-band h-[3px] w-full" aria-hidden="true" />
   253→        <div id="sc-t" className="p-6">
   254→          {!data ? (
   255→            <div className="flex items-center gap-4">
   256→              <Skeleton className="h-[130px] w-[130px] rounded-full" />
   257→              <div className="flex-1 space-y-2">
   258→                <Skeleton className="h-4 w-3/4" />
   259→                <Skeleton className="h-3 w-full" />
   260→                <Skeleton className="h-3 w-2/3" />
   261→              </div>
   262→            </div>
   263→          ) : multi.score !== null ? (
   264→            <div className="flex flex-wrap items-center gap-4">
   265→              <ScoreGauge score={multi.score} label="Multi-zones" />
   266→              <div className="min-w-0 flex-1">
   267→                <h3 className="font-heading font-bold text-base">Santé de ta peau</h3>
   268→                <p className="text-xs text-muted-foreground mt-1 leading-relaxed">
   269→                  {multi.covered.length} zone{multi.covered.length > 1 ? "s" : ""} analysée{multi.covered.length > 1 ? "s" : ""} · pondération PRD
   270→                  {multi.last && <span className="block mt-1">Dernier scan : {formatDate(multi.last.createdAt)}</span>}
   271→                </p>
   272→                <div className="flex flex-wrap gap-1.5 mt-2">
   273→                  {multi.covered.map((z) => (
   274→                    <span key={z.id} className="rounded-full bg-muted px-2 py-0.5 text-[10px] font-medium">
   275→                      {z.label} {multi.byZone.get(z.id)?.scoreGlobal}
   276→                    </span>
   277→                  ))}
   278→                  {multi.missing.map((z) => (
   279→                    <button
   280→                      key={z.id}
   281→                      onClick={() => onScanZone(z.id)}
   282→                      className="rounded-full border border-primary/40 bg-primary/10 px-3 min-h-10 inline-flex items-center text-[11px] font-semibold text-primary active:scale-95 transition-transform focus-visible:outline-2 focus-visible:outline-primary"
   283→                    >
   284→                      + {z.label}
   285→                    </button>
   286→                  ))}
   287→                </div>
   288→                {/* Lecture vocale du dernier diagnostic (accès non-lectrices) */}
   289→                {lastResult && (
   290→                  <div className="mt-5 pt-4 border-t border-border/60">
   291→                    <VoiceNarration result={lastResult} userName={user.name} />
   292→                  </div>
   293→                )}
   294→              </div>
   295→            </div>
   296→          ) : (
   297→            <div className="text-center py-2">
   298→              <NeaOnnimIcon size={40} className="mx-auto text-primary" />
   299→              <p className="font-heading font-bold mt-2">Ton premier diagnostic t&apos;attend</p>
   300→              <p className="text-xs text-muted-foreground mt-1">Analyse IA de 6 zones — commence par le visage.</p>
   301→            </div>
   302→          )}
   303→          {data && multi.missing.length > 0 && (
   304→            <div className="mt-4 pt-4 border-t border-dashed border-border">
   305→              <p className="text-[11px] font-semibold text-muted-foreground mb-2 flex items-center gap-1.5">
   306→                <Sparkles size={12} className="text-primary" /> Zones à scanner pour compléter ton score
   307→              </p>
   308→              <ScrollFadeRow label="Zones manquantes — fais défiler" className="flex gap-2 overflow-x-auto no-scrollbar pb-1 pr-1">
   309→                {multi.missing.map((z) => (
   310→                  <button
   311→                    key={z.id}
   312→                    onClick={() => onScanZone(z.id)}
   313→                    className="shrink-0 rounded-full border border-primary/40 bg-primary/10 px-3 min-h-10 inline-flex items-center text-[11px] font-semibold text-primary active:scale-95 transition-transform focus-visible:outline-2 focus-visible:outline-primary"
   314→                  >
   315→                    + {z.label} · {Math.round(z.weight * 100)} %
   316→                  </button>
   317→                ))}
   318→              </ScrollFadeRow>
   319→            </div>
   320→          )}
   321→        </div>
   322→      </section>
   323→
   324→      {/* ───── CTA Scanner ───── */}
   325→      <motion.button
   326→        initial={{ scale: 0.97, opacity: 0 }}
   327→        animate={{ scale: 1, opacity: 1 }}
   328→        whileTap={{ scale: 0.98 }}
   329→        onClick={() => setClientTab("diagnostic")}
   330→        className="relative h-24 rounded-3xl bg-gradient-to-br from-[#A0522D] via-[#8B1A3B] to-[#6B2416] text-[#FFF9EC] shadow-lg overflow-hidden focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary"
   331→        aria-label="Scanner ma peau maintenant"
   332→      >
   333→        <div aria-hidden="true" className="absolute inset-0 bogolan-dots opacity-30" />
   334→        <div className="relative h-full flex items-center justify-center gap-3 px-4">
   335→          <span className="grid place-items-center h-14 w-14 rounded-2xl bg-[#FFF9EC]/15 backdrop-blur border border-[#FFF9EC]/30">
   336→            <NeaOnnimIcon size={30} />
   337→          </span>
   338→          <span className="text-left">
   339→            <span className="block font-black text-lg tracking-tight">Scanner ma peau</span>
   340→            <span className="block text-[11px] opacity-90">Analyse IA VISIA-like · 6 zones · 30 s</span>
   341→          </span>
   342→          <ChevronRight size={22} className="ml-auto opacity-80" aria-hidden="true" />
   343→        </div>
   344→      </motion.button>
   345→
   346→      {/* ───── Route de l'Or — rituel tissé depuis le dernier scan ───── */}
   347→      {data && lastResult && multi.last && (
   348→        <motion.button
   349→          initial={{ opacity: 0, y: 10 }}
   350→          animate={{ opacity: 1, y: 0 }}
   351→          transition={{ delay: 0.1 }}
   352→          whileTap={{ scale: 0.98 }}
   353→          onClick={() => setRitualOpen(true)}
   354→          className="relative w-full rounded-3xl border-2 border-[#C8951E]/50 bg-card shadow-sm overflow-hidden text-left focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary"
   355→          aria-label="Reprendre la Route de l'Or — tisser ma routine depuis mon dernier diagnostic"
   356→        >
   357→          <div aria-hidden="true" className="h-2 w-full" style={{ backgroundImage: "repeating-linear-gradient(90deg,#8B1A3B 0 12px,#3F7D3F 12px 20px,#C8951E 20px 28px,#E07A2B 28px 36px,#A0522D 36px 46px)" }} />
   358→          <div className="flex items-center gap-3 p-4">
   359→            <span className="grid place-items-center h-12 w-12 rounded-2xl bg-gradient-to-br from-[#C8951E] to-[#A0522D] text-[#FFF9EC] shrink-0 shadow">
   360→              <NeaOnnimIcon size={26} />
   361→            </span>
   362→            <div className="min-w-0 flex-1">
   363→              <p className="font-heading font-bold text-sm">La Route de l&apos;Or</p>
   364→              <p className="text-[11px] text-muted-foreground mt-0.5 leading-snug">
   365→                Ta routine se tisse en 4 stations, selon ton scan du {formatDate(multi.last.createdAt, { day: "numeric", month: "short" })}
   366→              </p>
   367→            </div>
   368→            <ChevronRight size={18} className="text-primary shrink-0" />
   369→          </div>
   370→        </motion.button>
   371→      )}
   372→
   373→      {/* ───── Le Fil du Retour — ta prochaine étape ───── */}
   374→      {nextStep && (
   375→        <motion.section
   376→          initial={{ opacity: 0, y: 10 }}
   377→          animate={{ opacity: 1, y: 0 }}
   378→          transition={{ delay: 0.05 }}
   379→          aria-labelledby="ns-t"
   380→          className="rounded-2xl border border-gold/40 bg-gradient-to-br from-gold/10 via-card to-card p-4 shadow-sm"
   381→        >
   382→          <div className="flex items-center gap-2">
   383→            <span className="grid place-items-center h-9 w-9 rounded-xl bg-gold/15 text-gold shrink-0">
   384→              <SankofaIcon size={19} />
   385→            </span>
   386→            <p id="ns-t" className="font-heading font-bold text-sm">Ta prochaine étape</p>
   387→            <span
   388→              className={
   389→                "ml-auto rounded-full px-2 py-0.5 text-[10px] font-semibold tabular-nums " +
   390→                (nextStep.overdue ? "bg-bissap/15 text-destructive" : "bg-gold/15 text-gold-text")
   391→              }
   392→            >
   393→              {nextStep.overdue ? `En retard de ${-nextStep.days} j` : nextStep.days === 0 ? "Aujourd'hui" : `Dans ${nextStep.days} j`}
   394→            </span>
   395→          </div>
   396→          <p className="text-sm font-semibold mt-2">{nextStep.title}</p>
   397→          <p className="text-[11px] text-muted-foreground mt-0.5 leading-snug">{nextStep.detail}</p>
   398→          <button
   399→            onClick={() => setClientTab(nextStep.ctaTab)}
   400→            className="mt-3 h-11 w-full rounded-xl bg-primary text-primary-foreground text-xs font-bold inline-flex items-center justify-center gap-1.5 shadow active:scale-[0.98] transition-transform focus-visible:outline-2 focus-visible:outline-primary"
   401→          >
   402→            {nextStep.ctaLabel}
   403→            <ChevronRight size={15} aria-hidden="true" />
   404→          </button>
   405→        </motion.section>
   406→      )}
   407→
   408→      {/* ───── Prochain RDV ───── */}
   409→      <section aria-labelledby="rdv-t">
   410→        <SectionTitle icon={<SankofaIcon size={17} />}>
   411→          <span id="rdv-t">Prochain rendez-vous</span>
   412→        </SectionTitle>
   413→        {!data ? (
   414→          <Skeleton className="h-24 rounded-2xl" />
   415→        ) : nextAppt ? (
   416→          <button onClick={() => setClientTab("rdv")} className="w-full text-left rounded-2xl border border-border bg-card p-4 shadow-sm active:scale-[0.99] transition-transform focus-visible:outline-2 focus-visible:outline-primary">
