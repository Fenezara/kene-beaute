   110→  // true quand la relecture localStorage est finie (onRehydrateStorage, même
   111→  // en cas d'erreur — la porte ne se verrouille jamais). Typage tolérant : si
   112→  // 63-b était absent, fallback true = comportement d'avant (zéro régression).
   113→  const hydrated = useKene((s) => (s as { _keneHydrated?: boolean })._keneHydrated ?? true);
   114→  // Badge chat honnête : faux par défaut — seul un message ENTRANT (événement
   115→  // « kene:chat:new » dispatché par ChatScreen) l'allume, voir l'effet plus bas

   120→  // (useSyncExternalStore sur localStorage — sans mismatch d'hydratation)

   122→
   123→  // ─── Pull-to-refresh (Instagram / Wave) ───
   124→  const [pull, setPull] = useState(0);          // distance d'étirement (px, résistive)
   125→  const [refreshing, setRefreshing] = useState(false);
   126→  const [refreshKey, setRefreshKey] = useState(0);
   127→  const pullStart = useRef<{ y: number; x: number; atTop: boolean } | null>(null);
   128→  const wasRefreshing = useRef(false);
   129→
   130→  // ─── Swipe horizontal entre onglets (TikTok) ───
   131→  const swipeStart = useRef<{ x: number; y: number; ok: boolean } | null>(null);
   132→
   133→  // ─── Direction de transition (sens de navigation) + connectivité ───
   134→  const [navDir, setNavDir] = useState<1 | -1>(1);
   135→  const [online, setOnline] = useState(true);
   136→

   138→  useEffect(() => {
   139→    const update = () => setOnline(isOnline());
   140→    update();
   141→    window.addEventListener("online", update);
   142→    window.addEventListener("offline", update);
   143→    return () => {
   144→      window.removeEventListener("online", update);
   145→      window.removeEventListener("offline", update);
   146→    };
   147→  }, []);
   148→
   149→  // Relecture du store persisté (kene-store, skipHydration t. 63-b) — UNE fois

   161→  useEffect(() => {

   166→    return () => window.removeEventListener("kene:chat:new", onNewChatMessage);
   167→  }, []);

   169→  // Remonte en haut du flux à chaque changement d'onglet (le tirage est
   170→  // réinitialisé par les gestionnaires tactiles, jamais par cet effet)
   171→  useEffect(() => {

   173→  }, [tab]);

   175→  function goTab(t: ClientTab) {
   176→    if (t !== tab) {
   177→      const from = SWIPE_ORDER.indexOf(tab);
   178→      const to = SWIPE_ORDER.indexOf(t);
   179→      if (from >= 0 && to >= 0) setNavDir(to > from ? 1 : -1);
   180→      haptic(HAPTIC.tap);
   181→    }
   182→    if (t === "chat") setChatUnread(false);
   183→    setClientTab(t);
   184→  }
   185→
   186→  // ─── Gestes tactiles du conteneur de flux ───
   187→  const onTouchStart = (e: React.TouchEvent) => {
   188→    const el = scrollRef.current;
   189→    const t = e.touches[0];
   190→    pullStart.current = { y: t.clientY, x: t.clientX, atTop: !el || el.scrollTop <= 0 };
   191→    // le swipe est ignoré s'il démarre dans une rangée horizontale scrollable
   192→    const target = e.target as HTMLElement;
   193→    const inScrollRow = !!target.closest("[data-scroll-row]");
   194→    swipeStart.current = { x: t.clientX, y: t.clientY, ok: !inScrollRow };
   195→  };
   196→
   197→  const onTouchMove = (e: React.TouchEvent) => {
   198→    if (refreshing) return;
   199→    const s = pullStart.current;
   200→    if (!s || !s.atTop) return;
   201→    const t = e.touches[0];
   202→    const dy = t.clientY - s.y;
   203→    const dx = t.clientX - s.x;
   204→    if (dy <= 0 || Math.abs(dx) > Math.abs(dy)) return; // tirage vertical uniquement
   205→    setPull(Math.min(dy / 2.2, 96)); // résistif : l'icône s'alourdit en fin de course
   206→  };
   207→
   208→  const onTouchEnd = (e: React.TouchEvent) => {
   209→    // 1) fin de tirage → actualisation si seuil franchi
   210→    const dist = pull;
   211→    pullStart.current = null;
   212→    if (dist > 56) {
   213→      haptic(HAPTIC.medium);
   214→      setRefreshing(true);
   215→      setPull(0);
   216→      wasRefreshing.current = true;
   217→      setRefreshKey((k) => k + 1);
   218→    } else {
   219→      setPull(0);
   220→    }
   221→    // 2) swipe horizontal → onglet voisin
   222→    const s = swipeStart.current;
   223→    swipeStart.current = null;
   224→    if (!s || !s.ok || refreshing) return;
   225→    const t = e.changedTouches[0];
   226→    const dx = t.clientX - s.x;
   227→    const dy = t.clientY - s.y;
   228→    if (Math.abs(dx) < 64 || Math.abs(dx) < Math.abs(dy) * 1.8) return;
   229→    const idx = SWIPE_ORDER.indexOf(tab);
   230→    if (idx < 0) return;
   231→    const next = dx < 0 ? SWIPE_ORDER[idx + 1] : SWIPE_ORDER[idx - 1];
   232→    if (next) {
   233→      setNavDir(dx < 0 ? 1 : -1);
   234→      haptic(HAPTIC.tap);
   235→      setClientTab(next);
   236→    }
   237→  };

   240→  const onRefreshed = useCallback(() => {
   241→    if (!wasRefreshing.current) return;
   242→    wasRefreshing.current = false;
   243→    setRefreshing(false);
   244→    haptic(HAPTIC.success);
   245→    toast.success("Fil actualisé", { description: `Mis à jour à ${formatTime(new Date())}` });

   298→                    "flex flex-col items-center gap-1 py-2.5",
