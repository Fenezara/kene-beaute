   255→            );
   256→          }
   257→        } catch {
   258→          /* session injoignable → message du 409 ci-dessous */
   259→        }
   260→        if (!entered) toast.error(e.message || "Un espace entreprise existe déjà pour ce numéro");
   261→      } else {
   262→        toast.error(e instanceof Error ? e.message : "Création impossible");
   263→      }
   264→    } finally {
   265→      setSavingPro(false);
   266→    }
