   103→export function ClientApp() {

   110→  // true quand la relecture localStorage est finie (onRehydrateStorage, même

   112→  // 63-b était absent, fallback true = comportement d'avant (zéro régression).
   113→  const hydrated = useKene((s) => (s as { _keneHydrated?: boolean })._keneHydrated ?? true);
   114→  // Badge chat honnête : faux par défaut — seul un message ENTRANT (événement
   115→  // « kene:chat:new » dispatché par ChatScreen) l'allume, voir l'effet plus bas
   116→  const [chatUnread, setChatUnread] = useState(false);
   117→  const [pendingZone, setPendingZone] = useState<BodyZone | null>(null);
   118→  const scrollRef = useRef<HTMLDivElement | null>(null);
   119→  // Fil de Kente : l'introduction immersive ne se montre qu'une fois
   120→  // (useSyncExternalStore sur localStorage — sans mismatch d'hydratation)
   121→  const introDone = useIntroDone();
   122→
   123→  // ─── Pull-to-refresh (Instagram / Wave) ───
   124→  const [pull, setPull] = useState(0);          // distance d'étirement (px, résistive)
   125→  const [refreshing, setRefreshing] = useState(false);
   126→  const [refreshKey, setRefreshKey] = useState(0);
   127→  const pullStart = useRef<{ y: number; x: number; atTop: boolean } | null>(null);
   128→  const wasRefreshing = useRef(false);
   129→
   130→  // ─── Swipe horizontal entre onglets (TikTok) ───
   131→  const swipeStart = useRef<{ x: number; y: number; ok: boolean } | null>(null);
   132→
   133→  // ─── Direction de transition (sens de navigation) + connectivité ───
   134→  const [navDir, setNavDir] = useState<1 | -1>(1);
   135→  const [online, setOnline] = useState(true);
   136→
   137→  // Bandeau hors-ligne — résilience réseau façon Wave
   138→  useEffect(() => {
   139→    const update = () => setOnline(isOnline());
   140→    update();
   141→    window.addEventListener("online", update);
   142→    window.addEventListener("offline", update);
   143→    return () => {
   144→      window.removeEventListener("online", update);
   145→      window.removeEventListener("offline", update);
   146→    };
   147→  }, []);
   148→
   149→  // Relecture du store persisté (kene-store, skipHydration t. 63-b) — UNE fois
   150→  // au montage, idempotente (le store a aussi son propre filet « load ») :
   151→  // c'est ceci qui restaure l'état persisté (space/user/panier/onglet) après
   152→  // le premier rendu client — le BootSkeleton couvre exactement cette fenêtre.
   153→  useEffect(() => {
   154→    void useKene.persist.rehydrate();
   155→  }, []);
   156→
   157→  // ── Validation silencieuse de la session au boot ── DÉPLACÉE à la racine
   158→  // (t. 66-d) : le check vivait ici, or ClientApp n'est PAS monté quand
   159→  // l'espace persisté est pro/admin → une session invalidée n'était pas
   160→  // purgée au retour dans ces espaces. Voir src/components/kene/SessionKeeper.tsx
   161→  // (monté dans page.tsx, couvre les 3 espaces).
   162→
   163→  // Badge chat honnête : l'événement « kene:chat:new » (CustomEvent, détail
   164→  // { at } — dispatché par ChatScreen, contrat figé t. 63) n'allume le point
   165→  // QUE si tu n'es pas déjà sur l'onglet chat (lecture fraîche via getState,

   167→  useEffect(() => {

   169→      if (useKene.getState().clientTab !== "chat") setChatUnread(true);
   170→    };
   171→    window.addEventListener("kene:chat:new", onNewChatMessage);
   172→    return () => window.removeEventListener("kene:chat:new", onNewChatMessage);
   173→  }, []);
   174→
   175→  // Remonte en haut du flux à chaque changement d'onglet (le tirage est
   176→  // réinitialisé par les gestionnaires tactiles, jamais par cet effet)
   177→  useEffect(() => {
   178→    scrollRef.current?.scrollTo({ top: 0 });
   179→  }, [tab]);
   180→
   181→  function goTab(t: ClientTab) {
   182→    if (t !== tab) {
   183→      const from = SWIPE_ORDER.indexOf(tab);
   184→      const to = SWIPE_ORDER.indexOf(t);
   185→      if (from >= 0 && to >= 0) setNavDir(to > from ? 1 : -1);
   186→      haptic(HAPTIC.tap);
   187→    }
   188→    if (t === "chat") setChatUnread(false);
   189→    setClientTab(t);
   190→  }
   191→
   192→  // ─── Gestes tactiles du conteneur de flux ───
   193→  const onTouchStart = (e: React.TouchEvent) => {
   194→    const el = scrollRef.current;
   195→    const t = e.touches[0];
   196→    pullStart.current = { y: t.clientY, x: t.clientX, atTop: !el || el.scrollTop <= 0 };
   197→    // le swipe est ignoré s'il démarre dans une rangée horizontale scrollable
   198→    const target = e.target as HTMLElement;
   199→    const inScrollRow = !!target.closest("[data-scroll-row]");
   200→    swipeStart.current = { x: t.clientX, y: t.clientY, ok: !inScrollRow };
   201→  };
   202→
   203→  const onTouchMove = (e: React.TouchEvent) => {
   204→    if (refreshing) return;
   205→    const s = pullStart.current;
   206→    if (!s || !s.atTop) return;
   207→    const t = e.touches[0];
   208→    const dy = t.clientY - s.y;
   209→    const dx = t.clientX - s.x;
   210→    if (dy <= 0 || Math.abs(dx) > Math.abs(dy)) return; // tirage vertical uniquement
   211→    setPull(Math.min(dy / 2.2, 96)); // résistif : l'icône s'alourdit en fin de course
   212→  };
   213→
   214→  const onTouchEnd = (e: React.TouchEvent) => {
   215→    // 1) fin de tirage → actualisation si seuil franchi
   216→    const dist = pull;
   217→    pullStart.current = null;
   218→    if (dist > 56) {
   219→      haptic(HAPTIC.medium);
   220→      setRefreshing(true);
   221→      setPull(0);
   222→      wasRefreshing.current = true;
   223→      setRefreshKey((k) => k + 1);
   224→    } else {
   225→      setPull(0);
   226→    }
   227→    // 2) swipe horizontal → onglet voisin
   228→    const s = swipeStart.current;
   229→    swipeStart.current = null;
   230→    if (!s || !s.ok || refreshing) return;
   231→    const t = e.changedTouches[0];
   232→    const dx = t.clientX - s.x;
   233→    const dy = t.clientY - s.y;
   234→    if (Math.abs(dx) < 64 || Math.abs(dx) < Math.abs(dy) * 1.8) return;
   235→    const idx = SWIPE_ORDER.indexOf(tab);
   236→    if (idx < 0) return;
   237→    const next = dx < 0 ? SWIPE_ORDER[idx + 1] : SWIPE_ORDER[idx - 1];
   238→    if (next) {
   239→      setNavDir(dx < 0 ? 1 : -1);
   240→      haptic(HAPTIC.tap);
   241→      setClientTab(next);
   242→    }
   243→  };
   244→
   245→  // Le fil signale la fin de son rechargement (déclenché par refreshKey)
   246→  const onRefreshed = useCallback(() => {
   247→    if (!wasRefreshing.current) return;
   248→    wasRefreshing.current = false;
   249→    setRefreshing(false);
   250→    haptic(HAPTIC.success);
   251→    toast.success("Fil actualisé", { description: `Mis à jour à ${formatTime(new Date())}` });
   252→  }, []);
   253→
   254→  const onScanZone = useCallback(
   255→    (z: BodyZone) => {
   256→      setPendingZone(z);
   257→      setClientTab("diagnostic");
   258→    },
   259→    [setClientTab],
   260→  );
   261→
   262→  const onZoneConsumed = useCallback(() => setPendingZone(null), []);
   263→
   264→  // Store persisté pas encore relu → squelette d'amorçage (évite le flash
   265→  // d'onboarding avant la restauration de la session — t. 63)
   266→  if (!hydrated) return <BootSkeleton />;
   267→
   268→  if (!user) {
   269→    if (!introDone) return <KenteIntro />;
   270→    return <Onboarding />;
   271→  }
   272→
   273→  const first = user.name.split(" ")[0];
   274→
   275→  return (
   276→    <div className="h-dvh flex overflow-hidden">
   277→      {/* Atmosphère ÉCLAT 2026 — lueurs aurora derrière tout le shell (le fond

   287→          <span className="hidden xl:block">
   288→            <KeneLogo size={34} withText />
   289→          </span>
   290→          <span className="xl:hidden mx-auto">
   291→            <KeneLogo size={34} />
   292→          </span>
   293→        </div>
   294→
   295→        <nav className="flex-1 overflow-y-auto pretty-scroll px-2.5 xl:px-4 py-4 flex flex-col gap-1.5">
   296→          {NAV_DESKTOP.map((n) => {

   302→                  key={n.tab}
   303→                  onClick={() => goTab(n.tab)}
   304→                  aria-current={active ? "page" : undefined}
   305→                  className={cn(
   306→                    "k-cta flex flex-col items-center gap-1 py-2.5 rounded-2xl text-[#FFF9EC] focus-visible:outline-2 focus-visible:outline-primary",
   307→                    "xl:flex-row xl:items-center xl:justify-start xl:gap-3 xl:px-3 xl:h-12",
   308→                    active && "outline-2 outline-[#FFF9EC]/80",
   309→                  )}
   310→                >
   311→                  <NeaOnnimIcon size={22} className="xl:hidden" />
   312→                  <NeaOnnimIcon size={19} className="hidden xl:block" />
   313→                  <span className="text-[10px] xl:text-sm font-semibold xl:font-bold tracking-wide">{t(n.labelKey)}</span>
   314→                </button>
   315→              );
   316→            }
   317→            return (
   318→              <button
   319→                key={n.tab}
   320→                onClick={() => goTab(n.tab)}
   321→                aria-current={active ? "page" : undefined}
   322→                aria-label={`${t(n.labelKey)}${n.tab === "chat" && chatUnread ? " — 1 nouveau message" : ""}${n.tab === "boutique" && cartCount > 0 ? ` — ${cartCount} article${cartCount > 1 ? "s" : ""} au panier` : ""}`}
   323→                className={cn(

   325→                  active ? "bg-primary/12 text-primary" : "text-muted-foreground hover:bg-accent/60 hover:text-foreground",

   342→                    >

   348→                {active && <motion.span layoutId="side-indicator" className="k-rail-line hidden xl:block absolute left-0 top-1/2 -translate-y-1/2 h-6 w-[3px] rounded-full" aria-hidden="true" />}
   349→              </button>
   350→            );
   351→          })}
   352→        </nav>
   353→
   354→        {/* Bas de sidebar : micro légal (l'isolation des comptes t. 69-a a
