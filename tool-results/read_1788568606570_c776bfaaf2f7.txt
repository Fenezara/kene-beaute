     4→import { motion } from "framer-motion";
     5→import {

     7→  Pencil, Phone, Plus, ShieldCheck, Smartphone, Sparkles, Wallet as WalletIcon,
     8→} from "lucide-react";
     9→import { toast } from "sonner";
    10→import { apiGet, apiPatch, apiPost } from "@/lib/kene/api";
    11→import { formatDate, xof, CASHBACK_RATE } from "@/lib/kene/format";
    12→import { LANGS, type Lang } from "@/lib/kene/i18n";
    13→import { useT } from "@/lib/kene/use-t";
    14→import { MOMO_OPERATORS } from "@/lib/kene/rfm";
    15→import { Sheet, SheetContent, SheetHeader, SheetTitle } from "@/components/ui/sheet";
    16→import { Skeleton } from "@/components/ui/skeleton";
    17→import { Switch } from "@/components/ui/switch";
    18→import { useKene, type SessionUser } from "@/store/kene";
    19→import { useSecurity } from "@/store/security";
    20→import type { ApiUser, ApiWallet, ApiWalletTx } from "./types";
    21→import { FITZPATRICK_CARDS, SKIN_GOALS, SKIN_TYPES } from "./types";
    22→import { SectionTitle } from "./bits";

    25→
    26→/** SessionUser + goals (string JSON) renvoyé par PATCH profile */
    27→type ClientUser = SessionUser & { goals?: string | null };
    28→
    29→const REASON_LABELS: Record<string, string> = {
    30→  cashback: "Cashback commande",
    31→  referral: "Bonus parrainage",
    32→  topup: "Approvisionnement",
    33→  payment: "Paiement",
    34→  refund: "Remboursement RDV",
    35→};
    36→
    37→export function ProfileScreen() {
    38→  const user = useKene((s) => s.user) as ClientUser;
    39→  const setUser = useKene((s) => s.setUser);
    40→  const setSpace = useKene((s) => s.setSpace);
    41→  const setClientTab = useKene((s) => s.setClientTab);
    42→  const { t, lang, setLang } = useT();
    43→
    44→  const [edit, setEdit] = useState(false);
    45→  const [name, setName] = useState(user.name);
    46→  const [city, setCity] = useState(user.city ?? "");
    47→  const [savingId, setSavingId] = useState(false);
    48→
    49→  const [fitz, setFitz] = useState(user.fitzpatrick ?? "V");
    50→  const [skinType, setSkinType] = useState(user.skinType ?? "mixte");
    51→  const [allergies, setAllergies] = useState(user.allergies ?? "");
    52→  const [goals, setGoals] = useState<string[]>(() => {
    53→    try {
    54→      return (JSON.parse(user.goals ?? "[]") as { id: string }[]).map((g) => g.id);
    55→    } catch {
    56→      return [];
    57→    }
    58→  });
    59→  const [savingSkin, setSavingSkin] = useState(false);
    60→
    61→  const [wallet, setWallet] = useState<ApiWallet | null>(null);
    62→  const [txs, setTxs] = useState<ApiWalletTx[] | null>(null);
    63→  const [topup, setTopup] = useState(false);
    64→  const [amount, setAmount] = useState(5000);
    65→  const [method, setMethod] = useState<"wave" | "orange">("wave");
    66→  const [topupState, setTopupState] = useState<"idle" | "processing" | "done">("idle");
    67→  const [topupBusy, setTopupBusy] = useState(false);
    68→  const [exportBusy, setExportBusy] = useState(false);
    69→
    70→  // Installation PWA — même source d'événement que la bannière d'accueil.
    71→  const { canInstall, promptInstall, isStandalone, isIOS } = useInstallPrompt();
    72→
    73→  // Sécurité renforcée (2FA-lite) : code SMS exigé avant chaque paiement.
    74→  const secureEnabled = useSecurity((s) => s.enabled);
    75→  const setSecureEnabled = useSecurity((s) => s.setEnabled);
    76→
    77→  const loadWallet = useCallback(() => {
    78→    apiGet<{ wallet: ApiWallet; transactions: ApiWalletTx[] }>(`/api/wallet?userId=${user.id}`)
    79→      .then((r) => { setWallet(r.wallet); setTxs(r.transactions ?? []); })
    80→      .catch(() => setTxs([]));
    81→  }, [user.id]);
    82→
    83→  useEffect(() => {
    84→    loadWallet();
    85→  }, [loadWallet]);
    86→
    87→  /** Portabilité RGPD — télécharge « mes données » en JSON via blob */
    88→  async function downloadMyData() {
    89→    setExportBusy(true);
    90→    try {
    91→      const res = await fetch(`/api/profile/export?userId=${encodeURIComponent(user.id)}`);
    92→      if (!res.ok) throw new Error("Export impossible");
    93→      const blob = await res.blob();
    94→      const cd = res.headers.get("Content-Disposition") ?? "";
    95→      const m = /filename="([^"]+)"/.exec(cd);
    96→      const url = URL.createObjectURL(blob);
    97→      const a = document.createElement("a");
    98→      a.href = url;
    99→      a.download = m?.[1] ?? `kene-mes-donnees-${new Date().toISOString().slice(0, 10)}.json`;
   100→      document.body.appendChild(a);
   101→      a.click();
   102→      a.remove();
   103→      setTimeout(() => URL.revokeObjectURL(url), 4000);
   104→      const kb = Math.max(1, Math.round(blob.size / 1024));
   105→      toast.success(`Mes données téléchargées · ${kb} Ko`);
   106→    } catch (e) {
   107→      toast.error(e instanceof Error ? e.message : "Export impossible");
   108→    } finally {
   109→      setExportBusy(false);
   110→    }
   111→  }
   112→
   113→  /** Installer l'app (prompt natif, ou instructions iOS / navigateur). */
   114→  async function installApp() {
   115→    if (canInstall) {
   116→      const accepted = await promptInstall();
   117→      if (accepted) toast.success("Kènè installée sur ton écran d'accueil 💛");
   118→      return;
   119→    }
   120→    if (isIOS) {
   121→      toast.info("Installer sur iPhone", {
   122→        description: "Bouton Partager ⬆️ puis « Sur l'écran d'accueil ».",
   123→        duration: 8000,
   124→      });
   125→      return;
   126→    }
   127→    toast.info("Installation manuelle", {
   128→      description: "Menu du navigateur → « Installer l'application » ou « Ajouter à l'écran d'accueil ».",
   129→      duration: 8000,
   130→    });
   131→  }
   132→
   133→  async function saveIdentity() {
   134→    setSavingId(true);
   135→    try {
   136→      const r = await apiPatch<{ user: ApiUser }>("/api/auth/profile", { userId: user.id, name: name.trim() || undefined, city: city.trim() || undefined });
   137→      setUser({ ...user, ...r.user } as SessionUser);
   138→      setEdit(false);
   139→      toast.success("Profil mis à jour");
   140→    } catch (e) {
   141→      toast.error(e instanceof Error ? e.message : "Mise à jour impossible");
   142→    } finally {
   143→      setSavingId(false);
   144→    }
   145→  }
   146→
   147→  async function saveSkin() {
   148→    setSavingSkin(true);
   149→    try {
   150→      const r = await apiPatch<{ user: ApiUser }>("/api/auth/profile", {
   151→        userId: user.id,
   152→        skinType,
   153→        fitzpatrick: fitz,
   154→        allergies: allergies.trim() || undefined,
   155→        goals: goals.map((id) => ({ id, label: SKIN_GOALS.find((g) => g.id === id)?.label ?? id })),
   156→      });
   157→      setUser({ ...user, ...r.user } as SessionUser);
   158→      toast.success("Profil peau mis à jour");
   159→    } catch (e) {
   160→      toast.error(e instanceof Error ? e.message : "Mise à jour impossible");
   161→    } finally {
   162→      setSavingSkin(false);
   163→    }
   164→  }
   165→
   166→  /** Langue de l'interface (i18n) — indépendante de la langue de lecture vocale. */
   167→  function selectLang(l: Lang) {
   168→    if (l === lang) return;
   169→    setLang(l);
   170→    const label = LANGS.find((x) => x.id === l)?.label ?? "";
   171→    toast.success(`Interface en ${label.toLowerCase()}`);
   172→  }
   173→
   174→  /** Sécurité renforcée : bascule la re-vérification par code avant paiement. */
   175→  function toggleSecure() {
   176→    const next = !secureEnabled;
   177→    setSecureEnabled(next);
   178→    if (next) toast.success("Sécurité renforcée activée");
   179→    else toast("Sécurité renforcée désactivée");
   180→  }
   181→
   182→  async function runTopup() {
   183→    setTopupBusy(true);
   184→    setTopupState("processing");
   185→    try {
   186→      const r = await apiPost<{ payment: { id: string } }>("/api/wallet/topup", { userId: user.id, amount, method });
   187→      await new Promise((res) => setTimeout(res, 2600));
   188→      await apiPost("/api/payments/confirm", { paymentId: r.payment.id });
   189→      setTopupState("done");
   190→      toast.success(`${xof(amount)} crédités sur ton wallet`);
   191→      loadWallet();
   192→    } catch (e) {
   193→      setTopupState("idle");
   194→      toast.error(e instanceof Error ? e.message : "Approvisionnement impossible");
   195→    } finally {
   196→      setTopupBusy(false);
   197→    }
   198→  }
   199→
   200→  return (
   201→    <div className="pt-4 pb-2 flex flex-col gap-6">
   202→      <button onClick={() => setClientTab("accueil")} className="self-start inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground focus-visible:outline-2 focus-visible:outline-primary rounded min-h-10 px-1" aria-label={t("profile.back.aria")}>
   203→        <ArrowLeft size={15} /> {t("tab.home")}
   204→      </button>
   205→
   206→      {/* Identité */}
   207→      <section aria-labelledby="me-t" className="rounded-3xl border border-border bg-card shadow-sm overflow-hidden">
   208→        <div className="kente-band h-1.5 w-full" aria-hidden="true" />
   209→        <div id="me-t" className="p-5">
   210→          <div className="flex items-center gap-4">
   211→            <span className="grid place-items-center h-16 w-16 rounded-full bg-gradient-to-br from-[#C8951E] to-[#A0522D] text-[#FFF9EC] font-heading font-black text-2xl shadow">
   212→              {user.name.charAt(0)}
   213→            </span>
   214→            <div className="min-w-0 flex-1">
   215→              <p className="font-heading font-black text-lg leading-tight truncate">{user.name}</p>
   216→              <p className="text-xs text-muted-foreground flex items-center gap-1.5 mt-1 font-mono"><Phone size={12} /> {user.phone}</p>
   217→              <p className="text-xs text-muted-foreground flex items-center gap-1.5 mt-0.5"><MapPin size={12} /> {user.city || "Ville non renseignée"}</p>
   218→            </div>
   219→            <button onClick={() => setEdit((v) => !v)} aria-label="Modifier mon profil" className="h-10 w-10 grid place-items-center rounded-full border border-border text-muted-foreground hover:text-primary active:scale-90 transition-all focus-visible:outline-2 focus-visible:outline-primary">
   220→              <Pencil size={16} />
   221→            </button>
   222→          </div>
   223→          {edit && (
   224→            <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: "auto" }} className="overflow-hidden">
   225→              <div className="mt-4 space-y-3">
   226→                <div>
   227→                  <label htmlFor="p-name" className="text-[11px] font-semibold text-muted-foreground">Prénom & nom</label>
   228→                  <input id="p-name" value={name} onChange={(e) => setName(e.target.value)} className="mt-1 h-11 w-full rounded-xl border border-border bg-background px-3 text-sm focus-visible:outline-2 focus-visible:outline-primary" />
   229→                </div>
   230→                <div>
   231→                  <label htmlFor="p-city" className="text-[11px] font-semibold text-muted-foreground">Ville</label>
   232→                  <input id="p-city" value={city} onChange={(e) => setCity(e.target.value)} placeholder="Abidjan" className="mt-1 h-11 w-full rounded-xl border border-border bg-background px-3 text-sm focus-visible:outline-2 focus-visible:outline-primary" />
   233→                </div>
   234→                <button onClick={saveIdentity} disabled={savingId} className="h-11 w-full rounded-xl bg-primary text-primary-foreground text-sm font-bold flex items-center justify-center gap-2 active:scale-[0.98] transition-transform disabled:opacity-60 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary">
   235→                  {savingId ? <Loader2 size={16} className="animate-spin" /> : <Check size={16} />} Enregistrer
   236→                </button>
   237→              </div>
   238→            </motion.div>
   239→          )}
   240→        </div>
   241→      </section>
   242→
   243→      {/* Profil peau */}
   244→      <section aria-labelledby="skin-t">
   245→        <SectionTitle icon={<Sparkles size={16} />}><span id="skin-t">Mon profil peau</span></SectionTitle>
   246→        <div className="rounded-2xl border border-border bg-card p-4 space-y-4">
   247→          <div>
   248→            <p className="text-[11px] font-semibold text-muted-foreground mb-2">Phototype Fitzpatrick</p>
   249→            <div className="grid grid-cols-3 gap-2">
   250→              {FITZPATRICK_CARDS.map((f) => (
   251→                <button key={f.id} onClick={() => setFitz(f.id)} aria-pressed={fitz === f.id} className={`rounded-xl p-1.5 border-2 text-left active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${fitz === f.id ? "border-primary" : "border-transparent"}`}>
   252→                  <div className="h-8 rounded-lg mb-1" style={{ background: f.gradient }} aria-hidden="true" />
   253→                  <p className="text-[11px] font-bold">{f.id}</p>
   254→                </button>
   255→              ))}
   256→            </div>
   257→          </div>
   258→          <div>
   259→            <p className="text-[11px] font-semibold text-muted-foreground mb-2">Type de peau</p>
   260→            <div className="flex flex-wrap gap-1.5">
   261→              {SKIN_TYPES.map((t) => (
   262→                <button key={t.id} onClick={() => setSkinType(t.id)} aria-pressed={skinType === t.id} className={`rounded-full px-3 min-h-10 text-xs font-medium active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${skinType === t.id ? "bg-primary text-primary-foreground" : "border border-border"}`}>{t.label}</button>
   263→              ))}
   264→            </div>
   265→          </div>
   266→          <div>
   267→            <p className="text-[11px] font-semibold text-muted-foreground mb-2">Objectifs</p>
   268→            <div className="flex flex-wrap gap-1.5">
   269→              {SKIN_GOALS.map((g) => (
   270→                <button key={g.id} onClick={() => setGoals((s) => (s.includes(g.id) ? s.filter((x) => x !== g.id) : [...s, g.id]))} aria-pressed={goals.includes(g.id)} className={`rounded-full px-3 min-h-10 text-xs font-medium active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${goals.includes(g.id) ? "bg-primary text-primary-foreground" : "border border-border"}`}>{g.label}</button>
   271→              ))}
   272→            </div>
   273→          </div>
   274→          <div>
   275→            <label htmlFor="p-all" className="text-[11px] font-semibold text-muted-foreground">Allergies</label>
   276→            <textarea id="p-all" value={allergies} onChange={(e) => setAllergies(e.target.value)} rows={2} placeholder="Ex. huile de coco…" className="mt-1 w-full rounded-xl border border-border bg-background px-3 py-2 text-sm focus-visible:outline-2 focus-visible:outline-primary" />
   277→          </div>
   278→          <button onClick={saveSkin} disabled={savingSkin} className="h-11 w-full rounded-xl border border-primary/60 text-primary text-sm font-bold flex items-center justify-center gap-2 active:scale-[0.98] transition-transform disabled:opacity-60 focus-visible:outline-2 focus-visible:outline-primary">
   279→            {savingSkin ? <Loader2 size={15} className="animate-spin" /> : <Check size={15} />} Mettre à jour mon profil peau
   280→          </button>
   281→        </div>
   282→      </section>
   283→
   284→      {/* Langue de l'interface — i18n UI, indépendante de la lecture vocale
   285→          (la langue TTS se règle dans les pilules du résumé vocal, accueil) */}
   286→      <section aria-labelledby="lang-t" className="rounded-2xl border border-border bg-card p-4">
   287→        <div className="flex items-center gap-3">
   288→          <span className="grid place-items-center h-10 w-10 rounded-xl bg-primary/15 text-primary shrink-0">
   289→            <Languages size={19} />
   290→          </span>
   291→          <div className="flex-1 min-w-0">
   292→            <p id="lang-t" className="text-xs font-bold">{t("lang.selector.label")}</p>
   293→            <p className="text-[11px] text-muted-foreground mt-0.5 leading-snug">{t("lang.selector.note")}</p>
   294→          </div>
   295→        </div>
   296→        <div className="mt-3 grid grid-cols-2 gap-2">
   297→          {LANGS.map((l) => {
   298→            const active = lang === l.id;
   299→            return (
   300→              <button
   301→                key={l.id}
   302→                onClick={() => selectLang(l.id)}
   303→                aria-pressed={active}
   304→                className={`rounded-xl border p-3 min-h-12 text-left active:scale-[0.98] transition-all focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary ${
   305→                  active ? "border-primary bg-primary/10" : "border-border hover:border-primary/40"
   306→                }`}
   307→              >
   308→                <span className="flex items-center justify-between gap-1.5">
   309→                  <span className={`text-xs font-bold ${active ? "text-primary" : "text-foreground"}`}>{l.label}</span>
   310→                  {active ? (
   311→                    <Check size={13} className="text-primary shrink-0" aria-hidden="true" />
   312→                  ) : (
   313→                    <span className="text-[9px] font-mono font-bold text-muted-foreground/70" aria-hidden="true">{l.flag}</span>
   314→                  )}
   315→                </span>
   316→                <span className="block text-[10px] text-muted-foreground mt-1">{l.note}</span>
   317→              </button>
   318→            );
   319→          })}
   320→        </div>
   321→      </section>
   322→
   323→      {/* Wallet */}
   324→      <section aria-labelledby="wa-t">
   325→        <SectionTitle icon={<WalletIcon size={16} />}><span id="wa-t">Mon wallet Kènè</span></SectionTitle>
   326→        <div className="rounded-3xl bg-[#1A1410] text-[#F8F1E4] p-5 shadow-md relative overflow-hidden">
   327→          <div aria-hidden="true" className="absolute inset-0 bogolan-dots opacity-20" />
   328→          <div className="relative">
   329→            <p className="text-[10px] uppercase tracking-[0.18em] opacity-70">Solde disponible</p>
   330→            <p className="font-mono font-black text-3xl mt-1">{wallet ? xof(wallet.balance) : "···"}</p>
   331→            <div className="flex items-center gap-2 mt-2 text-[11px] opacity-80">
   332→              <BadgeCheck size={13} className="text-[#C8951E]" /> Cashback {Math.round((wallet?.cashbackRate ?? CASHBACK_RATE) * 100)} % sur chaque commande
   333→            </div>
   334→            <button onClick={() => { setTopup(true); setTopupState("idle"); }} className="mt-4 h-11 w-full rounded-xl bg-[#C8951E] text-[#1A1410] font-bold text-sm flex items-center justify-center gap-2 active:scale-[0.98] transition-transform focus-visible:outline-2 focus-visible:outline-[#C8951E]">
   335→              <Plus size={16} /> Approvisionner
   336→            </button>
   337→          </div>
   338→        </div>
   339→        <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wide mt-4 mb-2">Dernières transactions</p>
   340→        {txs === null ? (
   341→          <div className="space-y-2">{[0, 1, 2].map((i) => <Skeleton key={i} className="h-12 rounded-xl" />)}</div>
   342→        ) : txs.length === 0 ? (
   343→          <p className="text-xs text-muted-foreground text-center py-3">Aucune transaction pour l&apos;instant.</p>
   344→        ) : (
   345→          <div className="rounded-2xl border border-border bg-card divide-y divide-border max-h-72 overflow-y-auto scrollbar-thin">
   346→            {txs.map((t) => {
   347→              const credit = t.type === "credit";
   348→              return (
   349→                <div key={t.id} className="flex items-center gap-3 px-4 py-3">
   350→                  <span className={`grid place-items-center h-8 w-8 rounded-full shrink-0 ${credit ? "bg-success/15 text-success" : "bg-destructive/10 text-destructive"}`}>
   351→                    {credit ? <ArrowDownLeft size={15} /> : <ArrowUpRight size={15} />}
   352→                  </span>
   353→                  <div className="min-w-0 flex-1">
   354→                    <p className="text-xs font-semibold">{REASON_LABELS[t.reason] ?? t.reason}</p>
   355→                    <p className="text-[10px] text-muted-foreground">{formatDate(t.createdAt, { day: "numeric", month: "short", year: "2-digit" })}</p>
   356→                  </div>
   357→                  <span className={`font-mono text-sm font-bold ${credit ? "text-success" : "text-destructive"}`}>
   358→                    {credit ? "+" : "−"}{xof(t.amount)}
   359→                  </span>
   360→                </div>
   361→              );
   362→            })}
   363→          </div>
   364→        )}
   365→      </section>
   366→
   367→      {/* Parrainage — le fil qui relie les amies */}
   368→      <ParrainageCard userId={user.id} userName={user.name} onRedeemed={loadWallet} />
   369→
   370→      {/* Consentement */}
   371→      <section aria-labelledby="cons-t" className="rounded-2xl border border-border bg-card p-4 flex items-center gap-3">
   372→        <ShieldCheck size={20} className={user.consentHealth ? "text-success" : "text-destructive"} />
   373→        <div className="flex-1">
   374→          <p className="text-xs font-semibold">Consentement données santé</p>
   375→          <p className="text-[11px] text-muted-foreground mt-0.5">{user.consentHealth ? "Accordé — photos et diagnostics utilisés uniquement pour tes analyses." : "Non accordé — requis pour le diagnostic IA."}</p>
   376→        </div>
   377→        <span className={`rounded-full px-2.5 py-1 text-[10px] font-bold ${user.consentHealth ? "bg-success/15 text-success" : "bg-destructive/10 text-destructive"}`}>
   378→          {user.consentHealth ? "Actif" : "Inactif"}
   379→        </span>
   380→      </section>
   381→
   382→      {/* Mes données — portabilité RGPD (art. 20) */}
