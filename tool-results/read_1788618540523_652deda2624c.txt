   119→  // « kene:chat:new » dispatché par ChatScreen) l'allume, voir l'effet plus bas

   130→  const [refreshKey, setRefreshKey] = useState(0);
   131→  const pullStart = useRef<{ y: number; x: number; atTop: boolean } | null>(null);
   132→  const wasRefreshing = useRef(false);
   133→
   134→  // ─── Swipe horizontal entre onglets (TikTok) ───
   135→  const swipeStart = useRef<{ x: number; y: number; ok: boolean } | null>(null);
   136→
   137→  // ─── Direction de transition (sens de navigation) + connectivité ───
   138→  const [navDir, setNavDir] = useState<1 | -1>(1);

   148→      window.removeEventListener("online", update);

   185→  function goTab(t: ClientTab) {
   186→    if (t !== tab) {
   187→      const from = SWIPE_ORDER.indexOf(tab);
   188→      const to = SWIPE_ORDER.indexOf(t);
   189→      if (from >= 0 && to >= 0) setNavDir(to > from ? 1 : -1);
   190→      haptic(HAPTIC.tap);
   191→    }
   192→    if (t === "chat") setChatUnread(false);
   193→    setClientTab(t);
   194→  }
   195→
   196→  // ─── Gestes tactiles du conteneur de flux ───
   197→  const onTouchStart = (e: React.TouchEvent) => {
   198→    const el = scrollRef.current;
   199→    const t = e.touches[0];
   200→    pullStart.current = { y: t.clientY, x: t.clientX, atTop: !el || el.scrollTop <= 0 };
   201→    // le swipe est ignoré s'il démarre dans une rangée horizontale scrollable
   202→    const target = e.target as HTMLElement;
   203→    const inScrollRow = !!target.closest("[data-scroll-row]");
   204→    swipeStart.current = { x: t.clientX, y: t.clientY, ok: !inScrollRow };
   205→  };
   206→
   207→  const onTouchMove = (e: React.TouchEvent) => {
   208→    if (refreshing) return;
   209→    const s = pullStart.current;
   210→    if (!s || !s.atTop) return;
   211→    const t = e.touches[0];
   212→    const dy = t.clientY - s.y;
   213→    const dx = t.clientX - s.x;
   214→    if (dy <= 0 || Math.abs(dx) > Math.abs(dy)) return; // tirage vertical uniquement
   215→    setPull(Math.min(dy / 2.2, 96)); // résistif : l'icône s'alourdit en fin de course
   216→  };
   217→
   218→  const onTouchEnd = (e: React.TouchEvent) => {
   219→    // 1) fin de tirage → actualisation si seuil franchi
   220→    const dist = pull;
   221→    pullStart.current = null;
   222→    if (dist > 56) {
   223→      haptic(HAPTIC.medium);
   224→      setRefreshing(true);
   225→      setPull(0);
   226→      wasRefreshing.current = true;
   227→      setRefreshKey((k) => k + 1);
   228→    } else {
   229→      setPull(0);
   230→    }
   231→    // 2) swipe horizontal → onglet voisin
   232→    const s = swipeStart.current;
   233→    swipeStart.current = null;
   234→    if (!s || !s.ok || refreshing) return;
   235→    const t = e.changedTouches[0];
   236→    const dx = t.clientX - s.x;
   237→    const dy = t.clientY - s.y;
   238→    if (Math.abs(dx) < 64 || Math.abs(dx) < Math.abs(dy) * 1.8) return;
   239→    const idx = SWIPE_ORDER.indexOf(tab);
   240→    if (idx < 0) return;
   241→    const next = dx < 0 ? SWIPE_ORDER[idx + 1] : SWIPE_ORDER[idx - 1];
   242→    if (next) {
   243→      setNavDir(dx < 0 ? 1 : -1);
   244→      haptic(HAPTIC.tap);
   245→      setClientTab(next);
   246→    }
   247→  };
   248→
   249→  // Le fil signale la fin de son rechargement (déclenché par refreshKey)
   250→  const onRefreshed = useCallback(() => {
   251→    if (!wasRefreshing.current) return;
   252→    wasRefreshing.current = false;
   253→    setRefreshing(false);
   254→    haptic(HAPTIC.success);
   255→    toast.success("Fil actualisé", { description: `Mis à jour à ${formatTime(new Date())}` });
   256→  }, []);

   260→      setPendingZone(z);
   261→      setClientTab("diagnostic");

   263→    [setClientTab],
   264→  );

   266→  const onZoneConsumed = useCallback(() => setPendingZone(null), []);
   267→
   268→  // Store persisté pas encore relu → squelette d'amorçage (évite le flash
   269→  // d'onboarding avant la restauration de la session — t. 63)
   270→  if (!hydrated) return <BootSkeleton />;
   271→

   273→    if (!introDone) return <KenteIntro />;
   274→    return <Onboarding />;
   275→  }

   292→            <KeneLogo size={34} withText />
   293→          </span>
   294→          <span className="xl:hidden mx-auto">
   295→            <KeneLogo size={34} />
   296→          </span>
   297→        </div>

   311→                    "xl:flex-row xl:items-center xl:justify-start xl:gap-3 xl:px-3 xl:h-12",
