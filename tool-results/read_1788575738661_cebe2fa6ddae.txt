    92→    92→  const inputRef = useRef<HTMLInputElement>(null);
    93→    93→

   107→   107→  const livePromo = promo && promo.subtotal === subtotal ? promo : null;
   108→   108→  const discount = livePromo?.discount ?? 0;
   109→   109→  const total = subtotal - discount;
   110→   110→  /* taux de cashback réellement appliqué (wallet de la cliente, sinon défaut) */
   111→   111→  const cashbackRate = wallet?.cashbackRate ?? CASHBACK_RATE;
   112→   112→
   113→   113→  /* Solde wallet — échec explicite (walletError) plutôt que silence : le

   116→   116→    apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`)
   117→   117→      .then((r) => { setWallet(r.wallet); setWalletError(false); })

   138→   138→
   139→   139→  /* le fil de la catégorie — la navette l'illumine dans la bande tissée */

   141→   141→    products === null
   142→   142→      ? "La navette monte le métier…"

   144→   144→          cat ? ` · ${SHOP_CATEGORIES.find((c) => c.id === cat)?.label.toLowerCase() ?? cat}` : " au catalogue"
   145→   145→        }${favOnly ? " · favoris" : ""}`;
   146→   146→
   147→   147→  /* Applique un code promo : aperçu de remise sans consommer le coupon
   148→   148→   * (la consommation a lieu à la commande — toutes les gardes côté serveur). */
   149→   149→  async function applyPromo() {
   150→   150→    const code = promoInput.trim().toUpperCase();
   151→   151→    if (!code || promoChecking) return;
   152→   152→    setPromoChecking(true);
   153→   153→    try {
   154→   154→      const r = await apiPost<{ ok: true; coupon: { code: string; label: string | null }; discount: number; total: number }>(
   155→   155→        "/api/coupons/validate",
   156→   156→        { code, userId: user.id, subtotal }
   157→   157→      );
   158→   158→      setPromo({ code: r.coupon.code, label: r.coupon.label, discount: r.discount, subtotal });
   159→   159→      setPromoInput("");
   160→   160→      toast.success(`Code ${r.coupon.code} appliqué`, { description: `Remise de ${xof(r.discount)} déduite du total.` });
   161→   161→    } catch (e) {
   162→   162→      setPromo(null);
   163→   163→      toast.error(e instanceof Error ? e.message : "Code promo invalide");
   164→   164→    } finally {
   165→   165→      setPromoChecking(false);
   166→   166→    }
   167→   167→  }
   168→   168→
   169→   169→  /** Passerelle paiement : vérification d'identité par code si la sécurité
   170→   170→   *  renforcée est active — le paiement initialement prévu (pay) n'est lancé
   171→   171→   *  qu'une fois le code confirmé ; annulé sinon, rien n'est engagé. */
   172→   172→  function startPay(method: PayMethod) {
   173→   173→    if (securityEnabled) {
   174→   174→      haptic(HAPTIC.tap);
   175→   175→      setPendingPay(method);
   176→   176→      return;
   177→   177→    }
   178→   178→    void pay(method);
   179→   179→  }
   180→   180→
   181→   181→  async function pay(method: PayMethod) {
   182→   182→    setPaying(true);
   183→   183→    try {
   184→   184→      const items = cart.map((l) => ({ productId: l.productId, qty: l.qty }));
   185→   185→      const r = await apiPost<{ order: ApiOrder; payment: ApiPayment | null; paid: boolean }>("/api/orders", {
   186→   186→        userId: user.id,
   187→   187→        items,
   188→   188→        paymentMethod: method,
   189→   189→        ...(livePromo ? { couponCode: livePromo.code } : {}),
   190→   190→      });
   191→   191→      const amount = r.order.total;
   192→   192→      if (method !== "wallet" && r.payment) {
   193→   193→        // Contrat confirmToken (63-b/63-c) : un paiement mobile money en attente
   194→   194→        // porte son jeton — absent, on n'appelle JAMAIS confirm et on rollback
   195→   195→        // l'état UI comme un échec de paiement (panier intact, aucun overlay).
   196→   196→        const confirmToken = r.payment.confirmToken;
   197→   197→        if (!confirmToken) {
   198→   198→          setCheckout(false);
   199→   199→          toast.error("Paiement impossible — réessaie dans quelques instants");
   200→   200→          return;
   201→   201→        }
   202→   202→        setCheckout(false);
   203→   203→        setPayState({ phase: "processing", method, amount });
   204→   204→        await new Promise((res) => setTimeout(res, 3000));
   205→   205→        const c = await apiPost<{ payment: ApiPayment; order?: ApiOrder; wallet?: ApiWallet }>("/api/payments/confirm", { paymentId: r.payment.id, confirmToken });
   206→   206→        if (c.wallet) { setWallet(c.wallet); setWalletError(false); }
   207→   207→      } else {
   208→   208→        setCheckout(false);
   209→   209→        setPayState({ phase: "processing", method: "wallet", amount });
   210→   210→        await new Promise((res) => setTimeout(res, 1200));
   211→   211→      }
   212→   212→      setPayState({ phase: "success", method, amount });
   213→   213→      haptic(HAPTIC.success);
   214→   214→      clearCart();
   215→   215→      setPromo(null); // le coupon est consommé : remise à zéro pour la prochaine commande
   216→   216→      void refreshOrders(); // la nouvelle commande apparaît dans « Mes commandes »
   217→   217→      const cb = r.order.cashback;
   218→   218→      toast.success(cb > 0 ? `Commande confirmée — cashback ${xof(cb)} crédité sur ton wallet` : "Commande confirmée et payée");
   219→   219→      if (method === "wallet") {
   220→   220→        const w = await apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).catch(() => null);
   221→   221→        if (w) setWallet(w.wallet);
   222→   222→      }
   223→   223→    } catch (e) {
   224→   224→      setPayState(null);
   225→   225→      toast.error(e instanceof Error ? e.message : "Paiement impossible");
   226→   226→    } finally {
   227→   227→      setPaying(false);
   228→   228→    }
   229→   229→  }
   230→   230→
   231→   231→  const op = (m: PayMethod) => MOMO_OPERATORS.find((o) => o.code === m);
   232→   232→
   233→   233→  return (
   234→   234→    <div className="pt-4 pb-2">
   235→   235→      <header className="flex items-center justify-between">
   236→   236→        <h1 className="font-heading font-black text-xl">Boutique Kènè</h1>
   237→   237→        <span className="rounded-full bg-primary/10 text-primary text-[10px] font-bold px-2.5 py-1 uppercase tracking-wide">Cashback {Math.round(cashbackRate * 100)} %</span>

   240→   240→      {/* Bascule Catalogue ↔ Mes commandes — la cliente consulte ses données */}

   242→   242→        <button
   243→   243→          role="tab"
   244→   244→          aria-selected={view === "catalogue"}
   245→   245→          onClick={() => setView("catalogue")}
   246→   246→          className={`h-10 rounded-xl flex items-center justify-center gap-1.5 text-xs font-bold transition-all active:scale-[0.98] focus-visible:outline-2 focus-visible:outline-primary ${view === "catalogue" ? "bg-primary text-primary-foreground shadow-sm" : "text-muted-foreground hover:text-foreground"}`}
   247→   247→        >
   248→   248→          <ShoppingBag size={14} aria-hidden="true" /> Catalogue
   249→   249→        </button>
   250→   250→        <button
   251→   251→          role="tab"
   252→   252→          aria-selected={view === "commandes"}
   253→   253→          onClick={() => setView("commandes")}
   254→   254→          className={`h-10 rounded-xl flex items-center justify-center gap-1.5 text-xs font-bold transition-all active:scale-[0.98] focus-visible:outline-2 focus-visible:outline-primary ${view === "commandes" ? "bg-primary text-primary-foreground shadow-sm" : "text-muted-foreground hover:text-foreground"}`}
   255→   255→        >
   256→   256→          <History size={14} aria-hidden="true" /> Mes commandes{orders !== null && orders.length > 0 ? ` (${orders.length})` : ""}
   257→   257→        </button>
   258→   258→      </div>
   259→   259→
   260→   260→      {view === "commandes" ? (
   261→   261→        <OrdersView orders={orders} onRefresh={refreshOrders} onShop={() => setView("catalogue")} />
   262→   262→      ) : (
   263→   263→        <>
   264→   264→      {/* Le Fil de Kente — hero tissé, le fil de la catégorie s'illumine */}
   265→   265→      <div className="mt-4">
   266→   266→        <KenteWeaveCard highlightIndex={cat ? categoryThread(cat) : -1} caption={weaveCaption} />
   267→   267→      </div>
   268→   268→
   269→   269→      {/* Recherche */}

   272→   272→        <input
   273→   273→          ref={inputRef}
   274→   274→          value={q}
   275→   275→          onChange={(e) => setQ(e.target.value)}
   276→   276→          placeholder="Rechercher un soin, un botanique…"
   277→   277→          aria-label="Rechercher un produit"
   278→   278→          className="h-12 w-full rounded-2xl border border-border bg-card pl-10 pr-4 text-sm focus-visible:outline-2 focus-visible:outline-primary"
   279→   279→        />
   280→   280→      </div>
   281→   281→
   282→   282→      {/* Filtres : catégories + favoris (bascules aria-pressed, cumulables) */}

   284→   284→        {SHOP_CATEGORIES.map((c) => (
   285→   285→          <button
   286→   286→            key={c.id}
   287→   287→            aria-pressed={cat === c.id}
   288→   288→            onClick={() => setCat(c.id)}
   289→   289→            className={`shrink-0 rounded-full px-3.5 min-h-10 text-xs font-semibold transition-all active:scale-95 focus-visible:outline-2 focus-visible:outline-primary ${cat === c.id ? "bg-primary text-primary-foreground shadow" : "border border-border bg-card text-foreground/80"}`}
   290→   290→          >
   291→   291→            {c.label}
   292→   292→          </button>
   293→   293→        ))}
   294→   294→        {/* ♥ Favoris — filtre cumulable, badge count si ≥ 1 */}
   295→   295→        <button
   296→   296→          aria-pressed={favOnly}
   297→   297→          onClick={() => { setFavOnly((v) => !v); haptic(HAPTIC.tap); }}
   298→   298→          className={`shrink-0 inline-flex items-center gap-1.5 rounded-full px-3.5 min-h-10 text-xs font-semibold transition-all active:scale-95 focus-visible:outline-2 focus-visible:outline-primary ${favOnly ? "bg-primary text-primary-foreground shadow" : "border border-border bg-card text-foreground/80"}`}
   299→   299→        >
   300→   300→          <Heart size={13} fill={favOnly ? "currentColor" : "none"} aria-hidden="true" />
   301→   301→          Favoris{favs.length > 0 ? ` · ${favs.length}` : ""}
   302→   302→        </button>
   303→   303→      </div>
   304→   304→
   305→   305→      {/* Grille produits */}
   306→   306→      {products === null ? (
   307→   307→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   308→   308→          {[0, 1, 2, 3].map((i) => (
   309→   309→            <div key={i} className="space-y-2">
   310→   310→              <Skeleton className="aspect-square rounded-2xl" />
   311→   311→              <Skeleton className="h-3.5 w-4/5" />
   312→   312→              <Skeleton className="h-3 w-2/3" />
   313→   313→            </div>
   314→   314→          ))}
   315→   315→        </div>
   316→   316→      ) : filtered.length === 0 ? (
   317→   317→        favOnly && favs.length === 0 ? (
   318→   318→          <EmptyBlock
   319→   319→            icon={<Heart size={22} />}
   320→   320→            title="Aucun favori pour l'instant"
   321→   321→            text="Touche le cœur sur un soin pour le retrouver ici."
   322→   322→          />
   323→   323→        ) : favOnly ? (
   324→   324→          <EmptyBlock
   325→   325→            icon={<Heart size={22} />}
   326→   326→            title="Aucun favori dans cette sélection"
   327→   327→            text="Tes favoris ne passent pas ce filtre — essaie une autre catégorie ou efface la recherche."
   328→   328→            cta={
   329→   329→              <button
   330→   330→                onClick={() => { setCat(""); setQ(""); }}
   331→   331→                className="h-11 px-6 rounded-xl bg-primary text-primary-foreground text-sm font-bold active:scale-95 transition-transform focus-visible:outline-2 focus-visible:outline-primary"
   332→   332→              >
   333→   333→                Voir tous mes favoris
   334→   334→              </button>
   335→   335→            }
   336→   336→          />
   337→   337→        ) : (
   338→   338→          <EmptyBlock icon={<Search size={22} />} title="Aucun produit trouvé" text="Essaie un autre mot-clé ou une autre catégorie." />
   339→   339→        )
   340→   340→      ) : (
   341→   341→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   342→   342→          {filtered.map((p) => (
   343→   343→            /* Wrapper relatif : le cœur est un FRÈRE de la carte (jamais de <button>
   344→   344→               imbriqué — HTML valide, focus/a11y propres), posé sur l'image en absolu. */
   345→   345→            <div key={p.id} className="relative">
   346→   346→              <button
   347→   347→                onClick={(e) => onCardTap(p, e)}
   348→   348→                className="relative block w-full text-left rounded-2xl border border-border bg-card overflow-hidden shadow-sm active:scale-[0.98] transition-transform hover:border-primary/40 touch-manipulation focus-visible:outline-2 focus-visible:outline-primary"
