   216→    // le swipe est ignoré s'il démarre dans une rangée horizontale scrollable
   217→    const target = e.target as HTMLElement;
   218→    const inScrollRow = !!target.closest("[data-scroll-row]");
   219→    swipeStart.current = { x: t.clientX, y: t.clientY, ok: !inScrollRow };
   220→  };

   222→  const onTouchMove = (e: React.TouchEvent) => {
   223→    if (refreshing) return;
   224→    const s = pullStart.current;
   225→    if (!s || !s.atTop) return;
   226→    const t = e.touches[0];
   227→    const dy = t.clientY - s.y;
   228→    const dx = t.clientX - s.x;

   231→  };

   235→    const dist = pull;
   236→    pullStart.current = null;

   238→      haptic(HAPTIC.medium);
   239→      setRefreshing(true);
   240→      setPull(0);
   241→      wasRefreshing.current = true;
   242→      setRefreshKey((k) => k + 1);
   243→    } else {
   244→      setPull(0);
   245→    }

   247→    const s = swipeStart.current;
   248→    swipeStart.current = null;
   249→    if (!s || !s.ok || refreshing) return;
   250→    const t = e.changedTouches[0];
   251→    const dx = t.clientX - s.x;
   252→    const dy = t.clientY - s.y;
   253→    if (Math.abs(dx) < 64 || Math.abs(dx) < Math.abs(dy) * 1.8) return;

   255→    if (idx < 0) return;
   256→    const next = dx < 0 ? SWIPE_ORDER[idx + 1] : SWIPE_ORDER[idx - 1];
   257→    if (next) {
   258→      setNavDir(dx < 0 ? 1 : -1);
   259→      haptic(HAPTIC.tap);
   260→      setClientTab(next);
   261→    }
   262→  };
