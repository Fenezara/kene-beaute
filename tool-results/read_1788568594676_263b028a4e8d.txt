   290→            aria-hidden="true"
   291→          />

   294→          {ANALYSIS_STEPS.map((label, i) => {

   296→            const active = i === checkedSteps;
   297→            return (
   298→              <div key={i} className={`flex items-center gap-3 rounded-xl border p-3 transition-colors ${done ? "border-[#3F7D3F]/40 bg-[#3F7D3F]/5" : active ? "border-primary/50 bg-primary/5" : "border-border bg-card opacity-50"}`}>

   300→                  {done ? <Check size={13} /> : active ? <Loader2 size={13} className="animate-spin" /> : <span className="font-mono text-[10px] font-bold">{i + 1}</span>}
   301→                </span>
   302→                <span className={`text-xs ${done ? "text-[#3F7D3F] font-semibold" : "text-muted-foreground"}`}>{label}</span>
   303→              </div>
   304→            );
   305→          })}
   306→        </div>
   307→        <div className="w-full max-w-[320px] mt-5">
   308→          <Progress value={progress} className="h-2" aria-label="Progression de l'analyse" />
   309→          <p className="text-center text-[11px] text-muted-foreground mt-2">Vision par ordinateur spécialisée mélanoderme — 5 à 30 s</p>
   310→        </div>
   311→      </div>
   312→    );
   313→  }
   314→
   315→  /* ─────────── Étape 4 — Historique ─────────── */
   316→  if (step === 4) return <HistoryView userId={user.id} history={history} compareMode={compareMode} setCompareMode={setCompareMode} compareSel={compareSel} setCompareSel={setCompareSel} onBack={() => setStep(0)} onOpen={(d) => { const r = parseDiagnosis(d.resultJson); if (r) { setDiag({ id: d.id, result: r, imageData: diagImgSrc(d.imageData), createdAt: d.createdAt }); setStep(3); } }} />;
   317→
   318→  /* ─────────── Étape 3 — Résultats ─────────── */
   319→  if (step === 3 && diag) {
   320→    return <ResultView diag={diag} products={products} onNewZone={() => { setStep(0); setImage(""); setDiag(null); }} onHistory={goHistory} />;
   321→  }
   322→
   323→  return null;
   324→}

   327→function ResultView({ diag, products, onNewZone, onHistory }: { diag: { id: string; result: DiagnosisResult; imageData: string; createdAt: string }; products: ApiProduct[]; onNewZone: () => void; onHistory: () => void }) {
   328→  const user = useKene((s) => s.user)!;
   329→  const setClientTab = useKene((s) => s.setClientTab);
   330→  const addToCart = useKene((s) => s.addToCart);
   331→  const [view, setView] = useState<string>("standard");
   332→  const [ritualOpen, setRitualOpen] = useState(false);
   333→  const [glossary, setGlossary] = useState<GlossaryEntry | null>(null);
   334→  const r = diag.result;
   335→  const weakest = useMemo(() => [...r.indicateurs].sort((a, b) => a.pourcentage - b.pourcentage).slice(0, 8), [r.indicateurs]);
   336→  const viewDef = SPECTRAL_VIEWS.find((v) => v.id === view) ?? SPECTRAL_VIEWS[0];
   337→  const heatmapCls = view === "pigment" ? "heatmap-pigment" : view === "inflammation" ? "heatmap-inflammation" : view === "acne" ? "heatmap-acne" : "";
   338→
   339→  const botanicIcon = (name: string): React.ReactNode => {
   340→    const n = norm(name);
   341→    if (n.includes("moringa")) return <MoringaIcon size={13} />;
   342→    if (n.includes("karite")) return <KariteIcon size={13} />;
   343→    if (n.includes("baobab")) return <BaobabIcon size={13} />;
   344→    return <Sparkles size={13} />;
   345→  };
   346→
   347→  function askGlossary(term: string) {
   348→    const entry = glossaryFor(term);
   349→    if (entry) setGlossary(entry);
   350→  }
   351→
   352→  return (
   353→    <div className="pt-4 pb-2">
   354→      {/* Header score */}
   355→      <div className="rounded-3xl border border-border bg-card shadow-sm overflow-hidden">
   356→        <div className="kente-band h-1.5 w-full" aria-hidden="true" />
   357→        <div className="p-5 flex items-center gap-4">
   358→          <ScoreGauge score={r.score_global} label="Zone" size={110} />
   359→          <div className="min-w-0">
   360→            <div className="flex items-center gap-2 flex-wrap">
   361→              <h2 className="font-heading font-black text-lg">{BODY_ZONES.find((z) => z.id === r.zone)?.label ?? r.zone}</h2>
   362→              {r.fitzpatrick_estime && <span className="rounded-full bg-melanine text-[#F8F1E4] px-2 py-0.5 text-[10px] font-bold">Fitz {r.fitzpatrick_estime}</span>}
   363→            </div>
   364→            <p className="text-[11px] text-muted-foreground mt-1">{formatDate(diag.createdAt)} · {r.source === "vlm" ? "Analyse VLM" : "Analyse heuristique"} · {r.indicateurs.length} indicateurs</p>
   365→            <div className="flex gap-1.5 mt-2">
   366→              <span className="rounded-full px-2 py-0.5 text-[10px] font-bold" style={{ backgroundColor: scoreColor(r.score_global), color: readableTextColor(scoreColor(r.score_global)) }}>
   367→                {r.score_global >= 80 ? "Excellente santé" : r.score_global >= 60 ? "Bon équilibre" : r.score_global >= 40 ? "Points à surveiller" : "Besoin de soin"}
   368→              </span>
   369→            </div>
   370→          </div>
   371→        </div>
   372→      </div>

   374→      {/* Lecture vocale — accès non-lectrices & confort audio (TTS) */}
   375→      <VoiceNarration result={r} userName={user.name} />
   376→
   377→      {/* Résumé en pictos — tuiles tapables lues à voix haute (non-lectrices) */}

   379→        <PictoSummary result={r} zoneLabel={BODY_ZONES.find((z) => z.id === r.zone)?.label ?? r.zone} />
   380→      </div>

   382→      {/* Alerte orientation dermato */}
   383→      {r.orientation_dermato && (

   386→            <TriangleAlert size={17} /> Consultation dermatologique conseillée
   387→          </p>
   388→          <p className="text-xs mt-1.5 leading-relaxed">{r.raison_orientation ?? "Une lésion présente des signes justifiant un avis médical."}</p>
   389→          {r.abcde && r.abcde.length > 0 && (
   390→            <div className="flex gap-1.5 flex-wrap mt-3">

   626→function IndicatorBar({ ind, onAsk }: { ind: Indicator; onAsk?: (term: string) => void }) {

   659→function HistoryView({
   660→  userId, history, compareMode, setCompareMode, compareSel, setCompareSel, onBack, onOpen,
   661→}: {
   662→  userId: string;
   663→  history: ApiDiagnosis[] | null;

   665→  setCompareMode: (v: boolean) => void;
   666→  compareSel: string[];

   668→  onBack: () => void;
   669→  onOpen: (d: ApiDiagnosis) => void;
   670→}) {
   671→  const list = useMemo(
   672→    () => [...(history ?? [])].filter((d) => d.status === "done").sort((a, b) => +new Date(b.createdAt) - +new Date(a.createdAt)),
   673→    [history]
   674→  );
   675→  const sameZoneSel = compareSel.length === 2 && list.find((d) => d.id === compareSel[0])?.zone === list.find((d) => d.id === compareSel[1])?.zone;
   676→  const a = list.find((d) => d.id === compareSel[0]);
   677→  const b = list.find((d) => d.id === compareSel[1]);
   678→  const ra = a ? parseDiagnosis(a.resultJson) : null;
   679→  const rb = b ? parseDiagnosis(b.resultJson) : null;
   680→
   681→  const evolutions =
   682→    ra && rb
   683→      ? rb.indicateurs
   684→          .map((ib) => {
   685→            const ia = ra.indicateurs.find((x) => x.nom === ib.nom);
   686→            return ia ? { nom: ib.nom, delta: ib.pourcentage - ia.pourcentage } : null;
   687→          })
   688→          .filter((x): x is { nom: string; delta: number } => x !== null)
   689→          .sort((x, y) => Math.abs(y.delta) - Math.abs(x.delta))
   690→          .slice(0, 3)
   691→      : [];
   692→
   693→  return (
   694→    <div className="pt-4 pb-2">
   695→      <button onClick={onBack} className="inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground mb-4 focus-visible:outline-2 focus-visible:outline-primary rounded" aria-label="Retour">

   698→      <div className="flex items-center justify-between gap-2">
   699→        <h2 className="font-heading font-black text-xl">Mon historique</h2>
   700→        <button
   701→          onClick={() => { setCompareMode(!compareMode); setCompareSel([]); }}
   702→          aria-pressed={compareMode}
   703→          className={`h-10 px-3 rounded-full text-[11px] font-bold flex items-center gap-1.5 active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${compareMode ? "bg-primary text-primary-foreground" : "border border-border bg-card"}`}
   704→        >
   705→          <GitCompareArrows size={14} /> Comparer
   706→        </button>
   707→      </div>
   708→
   709→      {history === null ? (
   710→        <div className="space-y-3 mt-5">
   711→          {[0, 1, 2].map((i) => (
   712→            <Skeleton key={i} className="h-20 rounded-2xl" />
   713→          ))}
   714→        </div>
   715→      ) : list.length === 0 ? (
   716→        <div className="mt-6 rounded-2xl border border-dashed border-border bg-card/60 p-8 text-center">
   717→          <NeaOnnimIcon size={38} className="mx-auto text-primary" />
   718→          <p className="font-heading font-bold text-sm mt-3">Aucun diagnostic pour l&apos;instant</p>
   719→          <p className="text-xs text-muted-foreground mt-1">Ton premier scan débloquera le suivi d&apos;évolution.</p>
   720→        </div>

   722→        <>
   723→          {/* Le Fil du Temps — courbes d'évolution */}
   724→          <EvolutionCard userId={userId} className="mt-4" />
   725→
   726→          <div className="space-y-2.5 mt-4">
   727→            {list.map((d) => {
   728→              const sel = compareSel.includes(d.id);
   729→              return (
   730→                <div key={d.id} className={`flex items-center gap-3 rounded-2xl border p-2.5 transition-colors ${sel ? "border-primary bg-primary/5" : "border-border bg-card"}`}>
   731→                  <button

   733→                      if (compareMode) {
   734→                        setCompareSel(sel ? compareSel.filter((x) => x !== d.id) : compareSel.length < 2 ? [...compareSel, d.id] : [compareSel[1], d.id]);
   735→                      } else onOpen(d);
   736→                    }}
   737→                    className="flex items-center gap-3 flex-1 min-w-0 text-left focus-visible:outline-2 focus-visible:outline-primary rounded-xl"
   738→                    aria-label={`Diagnostic ${d.zone} du ${formatDate(d.createdAt)}, score ${d.scoreGlobal}`}

   741→                    <div className="min-w-0">

   750→                      aria-pressed={sel}
