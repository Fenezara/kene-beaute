    23→
    24→type PayMethod = "wave" | "wallet";
    25→
    26→function isoDate(d: Date): string {
    27→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
    28→}
    29→
    30→export function BookingScreen() {
    31→  const user = useKene((s) => s.user)!;
    32→  const [tab, setTab] = useState<"book" | "mine">("book");
    33→
    34→  // ── Réserver ──
    35→  const [institutes, setInstitutes] = useState<ApiInstitute[] | null>(null);
    36→  const [inst, setInst] = useState<ApiInstitute | null>(null);
    37→  const [services, setServices] = useState<ApiService[]>([]);
    38→  const [resources, setResources] = useState<ApiResource[]>([]);
    39→  const [reviews, setReviews] = useState<ApiReview[]>([]);
    40→  const [detailLoading, setDetailLoading] = useState(false);
    41→  const [service, setService] = useState<ApiService | null>(null);
    42→  const [day, setDay] = useState<Date>(new Date());
    43→  const [slots, setSlots] = useState<ApiSlot[] | null>(null);
    44→  const [slot, setSlot] = useState<ApiSlot | null>(null);
    45→  const [wallet, setWallet] = useState<ApiWallet | null>(null);
    46→  const [payMethod, setPayMethod] = useState<PayMethod>("wave");
    47→  const [paying, setPaying] = useState(false);
    48→  const [payOverlay, setPayOverlay] = useState<{ phase: "processing" | "done"; amount: number } | null>(null);
    49→  const [confirmed, setConfirmed] = useState<ApiAppointment | null>(null);
    50→
    51→  // Sécurité renforcée (2FA-lite) : si activée ET acompte payant, la cliente
    52→  // re-vérifie son code AVANT la confirmation du RDV (voir startBook + SecureVerify).
    53→  const securityEnabled = useSecurity((s) => s.enabled);
    54→  const [pendingBook, setPendingBook] = useState(false);
    55→
    56→  // ── Mes RDV ──
    57→  const [mine, setMine] = useState<ApiAppointment[] | null>(null);
    58→  const [reviewFor, setReviewFor] = useState<ApiAppointment | null>(null);
    59→  const [rating, setRating] = useState(5);
    60→  const [comment, setComment] = useState("");
    61→  const [cancelFor, setCancelFor] = useState<ApiAppointment | null>(null);
    62→
    63→  const days = useMemo(() => Array.from({ length: 7 }, (_, i) => addDays(new Date(), i)), []);
    64→  const deposit = service ? Math.round(service.price * DEPOSIT_RATE) : 0;
    65→  const practitioner = useMemo(() => (slot?.resourceIds?.length ? resources.find((r) => r.id === slot.resourceIds![0]) ?? null : null), [slot, resources]);
    66→
    67→  const loadInstitutes = useCallback(async () => {
    68→    setInstitutes(null);
    69→    try {
    70→      const r = await apiGet<{ institutes: ApiInstitute[] }>("/api/institutes");
    71→      setInstitutes(r.institutes ?? []);
    72→    } catch {
    73→      setInstitutes([]);
    74→    }
    75→  }, []);
    76→
    77→  const loadMine = useCallback(async () => {
    78→    setMine(null);
    79→    try {
    80→      const r = await apiGet<{ appointments: ApiAppointment[] }>(`/api/appointments?userId=${user.id}`);
    81→      setMine(r.appointments ?? []);
    82→    } catch {
    83→      setMine([]);
    84→    }
    85→  }, [user.id]);
    86→
    87→  useEffect(() => {
    88→    loadInstitutes();
    89→    apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).then((r) => setWallet(r.wallet)).catch(() => {});
    90→  }, [loadInstitutes, user.id]);
    91→
    92→  useEffect(() => {
    93→    if (tab === "mine") loadMine();
    94→  }, [tab, loadMine]);
    95→
    96→  async function openInstitute(i: ApiInstitute) {
    97→    setInst(i);
    98→    setService(null);
    99→    setSlot(null);
   100→    setSlots(null);
   101→    setDetailLoading(true);
   102→    try {
   103→      const r = await apiGet<{ institute: ApiInstitute; services: ApiService[]; resources: ApiResource[]; reviews: ApiReview[] }>(`/api/institutes/${i.id}`);
   104→      setServices(r.services ?? []);
   105→      setResources(r.resources ?? []);
   106→      setReviews(r.reviews ?? []);
   107→    } catch {
   108→      toast.error("Institut indisponible");
   109→      setInst(null);
   110→    } finally {
   111→      setDetailLoading(false);
   112→    }
   113→  }
   114→
   115→  async function pickDay(d: Date) {
   116→    setDay(d);
   117→    setSlot(null);
   118→    setSlots(null);
   119→    if (!service || !inst) return;
   120→    try {
   121→      const r = await apiGet<{ slots: ApiSlot[] }>(`/api/institutes/${inst.id}/availability?date=${isoDate(d)}&serviceId=${service.id}`);
   122→      setSlots(r.slots ?? []);
   123→    } catch {
   124→      setSlots([]);
   125→    }
   126→  }
   127→
   128→  async function chooseService(s: ApiService) {
   129→    setService(s);
   130→    setSlot(null);
   131→    setSlots(null);
   132→    if (!inst) return;
   133→    try {
   134→      const r = await apiGet<{ slots: ApiSlot[] }>(`/api/institutes/${inst.id}/availability?date=${isoDate(day)}&serviceId=${s.id}`);
   135→      setSlots(r.slots ?? []);
   136→    } catch {
   137→      setSlots([]);
   138→    }
   139→  }
   140→
   141→  /** Passerelle confirmation : vérification d'identité par code si la
   142→   *  sécurité renforcée est active et qu'un acompte est réglé — la
   143→   *  réservation (book) ne part qu'une fois le code confirmé. */
   144→  function startBook() {
   145→    if (securityEnabled && deposit > 0) {
   146→      haptic(HAPTIC.tap);
   147→      setPendingBook(true);
   148→      return;
   149→    }
   150→    void book();
   151→  }
   152→
   153→  async function book() {
   154→    if (!inst || !service || !slot) return;
   155→    setPaying(true);
   156→    try {
   157→      const [h, m] = slot.time.split(":").map(Number);
   158→      const startAt = new Date(day);
   159→      startAt.setHours(h, m, 0, 0);
   160→      const r = await apiPost<{ appointment: ApiAppointment; payment?: ApiPayment }>("/api/appointments", {
   161→        tenantId: inst.id,
   162→        serviceId: service.id,
   163→        resourceId: slot.resourceIds?.[0] ?? resources[0]?.id,
   164→        startAt: startAt.toISOString(),
   165→        clientName: user.name,
   166→        clientPhone: user.phone,
   167→        userId: user.id,
   168→        depositAmount: deposit,
   169→        paymentMethod: payMethod,
   170→      });
   171→      if (payMethod === "wave" && r.payment) {
   172→        setPayOverlay({ phase: "processing", amount: deposit });
   173→        await new Promise((res) => setTimeout(res, 3000));
   174→        await apiPost("/api/payments/confirm", { paymentId: r.payment.id });
   175→      } else {
   176→        setPayOverlay({ phase: "processing", amount: deposit });
   177→        await new Promise((res) => setTimeout(res, 1200));
   178→      }
   179→      setPayOverlay({ phase: "done", amount: deposit });
   180→      haptic(HAPTIC.success);
   181→      setTimeout(() => setPayOverlay(null), 1400);
   182→      setConfirmed(r.appointment);
   183→      const w = await apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).catch(() => null);
   184→      if (w) setWallet(w.wallet);
   185→    } catch (e) {
   186→      setPayOverlay(null);
   187→      toast.error(e instanceof Error ? e.message : "Réservation impossible");
   188→    } finally {
   189→      setPaying(false);
   190→    }
   191→  }
   192→
   193→  async function submitReview() {
   194→    if (!reviewFor) return;
   195→    try {
   196→      await apiPost(`/api/appointments/${reviewFor.id}/review`, { userId: user.id, rating, comment: comment.trim() || undefined });
   197→      toast.success("Merci pour ton avis");
   198→      setReviewFor(null);
   199→      setComment("");
   200→      setRating(5);
   201→      loadMine();
   202→    } catch (e) {
   203→      toast.error(e instanceof Error ? e.message : "Envoi impossible");
   204→    }
   205→  }
   206→
   207→  async function doCancel() {
   208→    if (!cancelFor) return;
   209→    try {
   210→      const r = await apiPost<{ appointment: ApiAppointment; refund: { amount: number; label: string } }>(`/api/appointments/${cancelFor.id}/cancel`, { userId: user.id });
   211→      toast.success(r.refund.amount > 0 ? `RDV annulé — ${xof(r.refund.amount)} remboursés sur ton wallet` : "RDV annulé");
   212→      setCancelFor(null);
   213→      loadMine();
   214→    } catch (e) {
   215→      toast.error(e instanceof Error ? e.message : "Annulation impossible");
   216→    }
   217→  }
   218→
   219→  const now = Date.now();
   220→  const upcoming = (mine ?? []).filter((a) => new Date(a.startAt).getTime() >= now && a.status !== "cancelled").sort((x, y) => +new Date(x.startAt) - +new Date(y.startAt));
   221→  const past = (mine ?? []).filter((a) => new Date(a.startAt).getTime() < now || a.status === "cancelled" || a.status === "completed").sort((x, y) => +new Date(y.startAt) - +new Date(x.startAt));
   222→
   223→  return (
   224→    <div className="pt-4 pb-2">
   225→      <Tabs value={tab} onValueChange={(v) => setTab(v as "book" | "mine")}>
   226→        <TabsList className="w-full grid grid-cols-2 bg-muted/60 rounded-2xl p-1">
   227→          <TabsTrigger value="book" className="rounded-xl data-[state=active]:bg-card data-[state=active]:text-primary font-semibold text-sm py-2.5">Réserver</TabsTrigger>
   228→          <TabsTrigger value="mine" className="rounded-xl data-[state=active]:bg-card data-[state=active]:text-primary font-semibold text-sm py-2.5">Mes RDV</TabsTrigger>
   229→        </TabsList>
   230→      </Tabs>
   231→
   232→      {/* ═══════ RÉSERVER ═══════ */}
   233→      {tab === "book" && (
   234→        <AnimatePresence mode="wait">
   235→          {confirmed ? (
   236→            /* — Confirmation — */
   237→            <motion.div key="ok" initial={{ opacity: 0, scale: 0.96 }} animate={{ opacity: 1, scale: 1 }} className="pt-8 flex flex-col items-center text-center">
   238→              <span className="grid place-items-center h-20 w-20 rounded-full bg-[#3F7D3F]/15 text-[#3F7D3D]">
   239→                <SankofaIcon size={38} />
   240→              </span>
   241→              <h2 className="font-heading font-black text-xl mt-4">Rendez-vous confirmé</h2>
   242→              <div className="mt-4 w-full rounded-2xl border border-border bg-card p-4 text-left space-y-1.5">
   243→                <p className="font-semibold text-sm">{confirmed.service?.name ?? "Soin"}</p>
   244→                <p className="text-xs text-muted-foreground">{confirmed.tenant?.name}</p>
   245→                <p className="text-xs text-muted-foreground flex items-center gap-1.5"><CalendarDays size={13} /> {formatDate(confirmed.startAt)} · <Clock size={13} /> {formatTime(confirmed.startAt)}</p>
   246→                {confirmed.resource?.name && <p className="text-xs text-muted-foreground flex items-center gap-1.5"><Users size={13} /> {confirmed.resource.name}</p>}
   247→                {confirmed.depositAmount > 0 && <p className="text-xs font-mono text-primary font-bold pt-1 border-t border-dashed border-border">Acompte réglé : {xof(confirmed.depositAmount)}</p>}
   248→              </div>
   249→              <p className="mt-4 inline-flex items-center gap-1.5 rounded-full bg-[#3F7D3F]/10 text-[#3F7D3D] px-3 py-1.5 text-[11px] font-semibold">
   250→                <MessageSquareQuote size={13} /> Rappel SMS J-1 programmé (simulé)
   251→              </p>
   252→              <div className="mt-6 grid grid-cols-2 gap-3 w-full">
   253→                <button onClick={() => { setConfirmed(null); setInst(null); setService(null); loadInstitutes(); }} className="h-12 rounded-xl border border-primary/60 text-primary text-sm font-bold active:scale-[0.98] transition-transform focus-visible:outline-2 focus-visible:outline-primary">
   254→                  Autre RDV
   255→                </button>
   256→                <button onClick={() => setTab("mine")} className="h-12 rounded-xl bg-primary text-primary-foreground text-sm font-bold active:scale-[0.98] transition-transform focus-visible:outline-2 focus-visible:outline-primary">
   257→                  Mes RDV
   258→                </button>
   259→              </div>
   260→            </motion.div>
   261→          ) : !inst ? (
   262→            /* — Liste instituts — */
   263→            <motion.div key="list" initial={{ opacity: 0, x: 30 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -30 }}>
   264→              <SectionTitle icon={<MapPin size={16} />}>Instituts partenaires</SectionTitle>
   265→              {institutes === null ? (
   266→                <div className="space-y-3">
   267→                  {[0, 1].map((i) => <Skeleton key={i} className="h-36 rounded-2xl" />)}
   268→                </div>
   269→              ) : institutes.length === 0 ? (
   270→                <EmptyBlock icon={<MapPin size={22} />} title="Aucun institut pour l'instant" text="De nouveaux instituts Kènè arrivent à Abidjan et Dakar." />
   271→              ) : (
   272→                <div className="space-y-3">
   273→                  {institutes.map((i) => (
   274→                    <button key={i.id} onClick={() => openInstitute(i)} className="w-full text-left rounded-2xl border border-border bg-card overflow-hidden shadow-sm active:scale-[0.99] transition-transform hover:border-primary/40 focus-visible:outline-2 focus-visible:outline-primary">
   275→                      <div className="relative">
   276→                        <img src={i.image} alt={i.name} loading="lazy" className="h-32 w-full object-cover" />
   277→                        <div className="absolute inset-0 bg-gradient-to-t from-[#1A1410]/70 to-transparent" />
   278→                        <div className="absolute bottom-2 left-3 right-3 flex items-end justify-between gap-2 text-[#F8F1E4]">
   279→                          <div>
   280→                            <p className="font-heading font-bold text-sm leading-tight">{i.name}</p>
   281→                            <p className="text-[10px] opacity-85 flex items-center gap-1"><MapPin size={10} /> {i.city}, {i.country}</p>
   282→                          </div>
   283→                          <span className="flex items-center gap-1 rounded-full bg-[#1A1410]/60 backdrop-blur px-2 py-0.5 text-[10px] font-bold">
   284→                            <Star size={10} className="fill-[#C8951E] text-[#C8951E]" /> {i.rating.toFixed(1)} ({i.reviewCount})
   285→                          </span>
   286→                        </div>
   287→                      </div>
   288→                      <div className="px-3 py-2.5 flex items-center justify-between text-[11px] text-muted-foreground">
   289→                        <span>{i._count?.services ?? 0} soins · {i._count?.reviews ?? 0} avis</span>
   290→                        <span className="flex items-center gap-0.5 text-primary font-semibold">Réserver <ChevronRight size={13} /></span>
   291→                      </div>
   292→                    </button>
   293→                  ))}
   294→                </div>
   295→              )}
   296→            </motion.div>
   297→          ) : (
   298→            /* — Détail institut — */
   299→            <motion.div key="detail" initial={{ opacity: 0, x: 30 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -30 }}>
   300→              <button onClick={() => setInst(null)} className="mt-4 inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground focus-visible:outline-2 focus-visible:outline-primary rounded" aria-label="Retour aux instituts">
   301→                <ArrowLeft size={15} /> Tous les instituts
   302→              </button>
   303→              <div className="relative mt-3 rounded-2xl overflow-hidden">
   304→                <img src={inst.image} alt={inst.name} className="h-44 w-full object-cover" />
   305→                <div className="absolute inset-0 bg-gradient-to-t from-[#1A1410]/80 to-transparent" />
   306→                <div className="absolute bottom-3 left-4 right-4 text-[#F8F1E4]">
   307→                  <h2 className="font-heading font-black text-lg">{inst.name}</h2>
   308→                  <p className="text-[11px] opacity-90 flex items-center gap-2">
   309→                    <span className="flex items-center gap-1"><Star size={11} className="fill-[#C8951E] text-[#C8951E]" /> {inst.rating.toFixed(1)}</span>
   310→                    <span className="flex items-center gap-1"><MapPin size={11} /> {inst.city}</span>
   311→                    <span className="flex items-center gap-1"><Clock size={11} /> {inst.openingHour}h–{inst.closingHour}h</span>
   312→                  </p>
   313→                </div>
   314→              </div>
   315→              {inst.description && <p className="text-xs text-muted-foreground leading-relaxed mt-3">{inst.description}</p>}
   316→
   317→              {detailLoading ? (
   318→                <div className="space-y-2 mt-4">{[0, 1, 2].map((i) => <Skeleton key={i} className="h-16 rounded-xl" />)}</div>
   319→              ) : (
   320→                <>
   321→                  {/* Avis */}
   322→                  {reviews.length > 0 && (
   323→                    <div className="mt-4">
   324→                      <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wide mb-2">Ce que disent les clientes</p>
   325→                      <div className="flex gap-2.5 overflow-x-auto pb-1 scrollbar-thin">
   326→                        {reviews.slice(0, 5).map((rv) => (
   327→                          <div key={rv.id} className="shrink-0 w-56 rounded-2xl border border-border bg-card p-3">
   328→                            <div className="flex items-center justify-between">
   329→                              <p className="text-xs font-semibold">{rv.user?.name ?? "Cliente"}</p>
   330→                              <Stars rating={rv.rating} size={9} />
   331→                            </div>
   332→                            <p className="text-[11px] text-muted-foreground mt-1.5 leading-snug line-clamp-3">{rv.comment ?? "Expérience agréable."}</p>
   333→                          </div>
   334→                        ))}
   335→                      </div>
   336→                    </div>
   337→                  )}
   338→
   339→                  {/* Services */}
   340→                  <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wide mt-5 mb-2">Soins & tarifs</p>
   341→                  <div className="space-y-2">
   342→                    {services.map((s) => (
   343→                      <div key={s.id} className={`rounded-2xl border p-3.5 transition-colors ${service?.id === s.id ? "border-primary bg-primary/5" : "border-border bg-card"}`}>
   344→                        <div className="flex items-start justify-between gap-2">
   345→                          <div className="min-w-0">
   346→                            <p className="text-sm font-semibold">{s.name}</p>
   347→                            <p className="text-[11px] text-muted-foreground mt-0.5 flex items-center gap-2">
   348→                              <span className="flex items-center gap-1"><Clock size={11} /> {s.durationMin} min</span>
   349→                              {s.botanicals && <span className="text-terre truncate">{s.botanicals}</span>}
   350→                            </p>
   351→                          </div>
   352→                          <p className="font-mono text-sm font-bold shrink-0">{xof(s.price)}</p>
   353→                        </div>
   354→                        <button onClick={() => chooseService(s)} className={`mt-2.5 h-10 w-full rounded-xl text-xs font-bold active:scale-[0.98] transition-all focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary ${service?.id === s.id ? "bg-primary text-primary-foreground" : "border border-primary/50 text-primary"}`}>
   355→                          {service?.id === s.id ? "Choisi" : "Choisir"}
   356→                        </button>
   357→                      </div>
   358→                    ))}
   359→                  </div>
   360→
   361→                  {/* Jours + créneaux */}
   362→                  {service && (
   363→                    <motion.div initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} className="mt-6">
   364→                      <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wide mb-2">Choisis ton jour</p>
   365→                      <div className="flex gap-2 overflow-x-auto pb-1 scrollbar-thin">
   366→                        {days.map((d) => {

   369→                            <button key={d.toISOString()} onClick={() => pickDay(d)} aria-pressed={sel} className={`shrink-0 rounded-2xl border px-3.5 py-2 text-center transition-all active:scale-95 focus-visible:outline-2 focus-visible:outline-primary ${sel ? "border-primary bg-primary text-primary-foreground" : "border-border bg-card"}`}>
   370→                              <span className="block text-[9px] uppercase font-bold opacity-70">{formatDate(d, { weekday: "short" })}</span>
   371→                              <span className="block font-mono text-sm font-bold mt-0.5">{d.getDate()}</span>
   372→                            </button>
   373→                          );
   374→                        })}
   375→                      </div>
   376→
   377→                      <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wide mt-5 mb-2">Créneaux — {formatDate(day, { weekday: "long", day: "numeric", month: "long" })}</p>
   378→                      {slots === null ? (
   379→                        <div className="flex justify-center py-6"><Loader2 size={22} className="animate-spin text-primary" /></div>
   380→                      ) : slots.length === 0 ? (
   381→                        <p className="text-xs text-muted-foreground text-center py-4">Aucun créneau affiché — choisis un autre jour.</p>
   382→                      ) : (
   383→                        <div className="grid grid-cols-4 gap-2">
   384→                          {slots.map((s) => (
   385→                            <button
