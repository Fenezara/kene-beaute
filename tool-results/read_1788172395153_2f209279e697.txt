   276→          <img src={image} alt="Photo en cours d'analyse" className="aspect-square w-full object-cover" />
   277→          <motion.div

   280→            transition={{ duration: 2.6, repeat: Infinity, ease: "easeInOut" }}
   281→            aria-hidden="true"
   282→          />
   283→        </div>
   284→        <div className="w-full max-w-[320px] mt-6 space-y-3">
   285→          {ANALYSIS_STEPS.map((label, i) => {
   286→            const done = i < checkedSteps;
   287→            const active = i === checkedSteps;
   288→            return (
   289→              <div key={i} className={`flex items-center gap-3 rounded-xl border p-3 transition-colors ${done ? "border-[#3F7D3F]/40 bg-[#3F7D3F]/5" : active ? "border-primary/50 bg-primary/5" : "border-border bg-card opacity-50"}`}>
   290→                <span className={`grid place-items-center h-6 w-6 rounded-full shrink-0 ${done ? "bg-[#3F7D3F] text-white" : active ? "bg-primary/20 text-primary" : "bg-muted text-muted-foreground"}`}>
   291→                  {done ? <Check size={13} /> : active ? <Loader2 size={13} className="animate-spin" /> : <span className="font-mono text-[10px] font-bold">{i + 1}</span>}
   292→                </span>
   293→                <span className={`text-xs ${done ? "text-[#3F7D3F] font-semibold" : "text-muted-foreground"}`}>{label}</span>
   294→              </div>
   295→            );
   296→          })}
   297→        </div>
   298→        <div className="w-full max-w-[320px] mt-5">
   299→          <Progress value={progress} className="h-2" aria-label="Progression de l'analyse" />
   300→          <p className="text-center text-[11px] text-muted-foreground mt-2">Vision par ordinateur spécialisée mélanoderme — 5 à 30 s</p>
   301→        </div>
   302→      </div>
   303→    );
   304→  }
   305→
   306→  /* ─────────── Étape 4 — Historique ─────────── */
   307→  if (step === 4) return <HistoryView history={history} compareMode={compareMode} setCompareMode={setCompareMode} compareSel={compareSel} setCompareSel={setCompareSel} onBack={() => setStep(0)} onOpen={(d) => { const r = parseDiagnosis(d.resultJson); if (r) { setDiag({ id: d.id, result: r, imageData: diagImgSrc(d.imageData), createdAt: d.createdAt }); setStep(3); } }} />;
   308→
   309→  /* ─────────── Étape 3 — Résultats ─────────── */
   310→  if (step === 3 && diag) {
   311→    return <ResultView diag={diag} products={products} onNewZone={() => { setStep(0); setImage(""); setDiag(null); }} onHistory={goHistory} />;
   312→  }
   313→
   314→  return null;
   315→}

   318→function ResultView({ diag, products, onNewZone, onHistory }: { diag: { id: string; result: DiagnosisResult; imageData: string; createdAt: string }; products: ApiProduct[]; onNewZone: () => void; onHistory: () => void }) {

   617→function HistoryView({
   618→  history, compareMode, setCompareMode, compareSel, setCompareSel, onBack, onOpen,
   619→}: {
   620→  history: ApiDiagnosis[] | null;
   621→  compareMode: boolean;
   622→  setCompareMode: (v: boolean) => void;
   623→  compareSel: string[];
   624→  setCompareSel: (v: string[]) => void;
   625→  onBack: () => void;
   626→  onOpen: (d: ApiDiagnosis) => void;
   627→}) {
   628→  const list = useMemo(
   629→    () => [...(history ?? [])].filter((d) => d.status === "done").sort((a, b) => +new Date(b.createdAt) - +new Date(a.createdAt)),
   630→    [history]
   631→  );
   632→  const sameZoneSel = compareSel.length === 2 && list.find((d) => d.id === compareSel[0])?.zone === list.find((d) => d.id === compareSel[1])?.zone;
   633→  const a = list.find((d) => d.id === compareSel[0]);
   634→  const b = list.find((d) => d.id === compareSel[1]);
   635→  const ra = a ? parseDiagnosis(a.resultJson) : null;
   636→  const rb = b ? parseDiagnosis(b.resultJson) : null;
   637→
   638→  const evolutions =
   639→    ra && rb
   640→      ? rb.indicateurs
   641→          .map((ib) => {
   642→            const ia = ra.indicateurs.find((x) => x.nom === ib.nom);
   643→            return ia ? { nom: ib.nom, delta: ib.pourcentage - ia.pourcentage } : null;
   644→          })
   645→          .filter((x): x is { nom: string; delta: number } => x !== null)
   646→          .sort((x, y) => Math.abs(y.delta) - Math.abs(x.delta))
   647→          .slice(0, 3)
   648→      : [];
   649→
   650→  return (
   651→    <div className="pt-4 pb-2">
   652→      <button onClick={onBack} className="inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground mb-4 focus-visible:outline-2 focus-visible:outline-primary rounded" aria-label="Retour">
   653→        <ArrowLeft size={15} /> Nouveau diagnostic
   654→      </button>
   655→      <div className="flex items-center justify-between gap-2">
   656→        <h1 className="font-heading font-black text-xl">Mon historique</h1>
   657→        <button
   658→          onClick={() => { setCompareMode(!compareMode); setCompareSel([]); }}
   659→          aria-pressed={compareMode}
   660→          className={`h-9 px-3 rounded-full text-[11px] font-bold flex items-center gap-1.5 active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${compareMode ? "bg-primary text-primary-foreground" : "border border-border bg-card"}`}
   661→        >
   662→          <GitCompareArrows size={14} /> Comparer
   663→        </button>
   664→      </div>
   665→
   666→      {history === null ? (
   667→        <div className="space-y-3 mt-5">
   668→          {[0, 1, 2].map((i) => (
   669→            <Skeleton key={i} className="h-20 rounded-2xl" />
   670→          ))}
   671→        </div>
   672→      ) : list.length === 0 ? (
   673→        <div className="mt-6 rounded-2xl border border-dashed border-border bg-card/60 p-8 text-center">
   674→          <NeaOnnimIcon size={38} className="mx-auto text-primary" />
   675→          <p className="font-heading font-bold text-sm mt-3">Aucun diagnostic pour l&apos;instant</p>
   676→          <p className="text-xs text-muted-foreground mt-1">Ton premier scan débloquera le suivi d&apos;évolution.</p>
   677→        </div>
   678→      ) : (
   679→        <>
   680→          <div className="space-y-2.5 mt-4">
   681→            {list.map((d) => {
   682→              const sel = compareSel.includes(d.id);
   683→              return (
   684→                <div key={d.id} className={`flex items-center gap-3 rounded-2xl border p-2.5 transition-colors ${sel ? "border-primary bg-primary/5" : "border-border bg-card"}`}>
   685→                  <button
   686→                    onClick={() => {
   687→                      if (compareMode) {
   688→                        setCompareSel(sel ? compareSel.filter((x) => x !== d.id) : compareSel.length < 2 ? [...compareSel, d.id] : [compareSel[1], d.id]);
   689→                      } else onOpen(d);
   690→                    }}
   691→                    className="flex items-center gap-3 flex-1 min-w-0 text-left focus-visible:outline-2 focus-visible:outline-primary rounded-xl"
   692→                    aria-label={`Diagnostic ${d.zone} du ${formatDate(d.createdAt)}, score ${d.scoreGlobal}`}
   693→                  >
   694→                    <img src={diagImgSrc(d.imageData)} alt={`Diagnostic ${d.zone}`} loading="lazy" className="h-14 w-14 rounded-xl object-cover shrink-0" />
   695→                    <div className="min-w-0">
   696→                      <p className="text-sm font-semibold">{BODY_ZONES.find((z) => z.id === d.zone)?.label ?? d.zone}</p>
   697→                      <p className="text-[11px] text-muted-foreground">{formatDate(d.createdAt)}</p>
   698→                    </div>
   699→                    <div className="ml-auto"><ScoreChip score={d.scoreGlobal} /></div>
   700→                  </button>
   701→                  {compareMode && (
   702→                    <button
   703→                      onClick={() => setCompareSel(sel ? compareSel.filter((x) => x !== d.id) : compareSel.length < 2 ? [...compareSel, d.id] : [compareSel[1], d.id])}
   704→                      aria-pressed={sel}
   705→                      aria-label="Sélectionner pour comparaison"
   706→                      className={`h-8 w-8 grid place-items-center rounded-full border-2 shrink-0 transition-colors ${sel ? "bg-primary border-primary text-primary-foreground" : "border-border"}`}
   707→                    >
   708→                      {sel && <Check size={14} />}
   709→                    </button>

   713→            })}
   714→          </div>
   715→
   716→          {compareMode && (
   717→            <AnimatePresence>
   718→              {compareSel.length === 2 && sameZoneSel && ra && rb && a && b ? (
   719→                <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: "auto" }} exit={{ opacity: 0, height: 0 }} className="mt-5 overflow-hidden">
   720→                  <div className="rounded-3xl border border-primary/40 bg-card p-4 shadow-md">

   722→                      <GitCompareArrows size={15} className="text-primary" /> Avant / Après — {BODY_ZONES.find((z) => z.id === a.zone)?.label}
   723→                    </p>
   724→                    <div className="grid grid-cols-2 gap-3">
   725→                      {[{ d: a, r: ra, tag: "Avant" }, { d: b, r: rb, tag: "Après" }].map(({ d, r, tag }) => (

   728→                          <div className="flex items-center justify-between mt-1.5">
   729→                            <span className="text-[10px] uppercase tracking-wide font-bold text-muted-foreground">{tag} · {formatDate(d.createdAt, { day: "numeric", month: "short" })}</span>
   730→                            <ScoreChip score={r.score_global} />
   731→                          </div>
   732→                        </div>
   733→                      ))}
   734→                    </div>
   735→                    <div className="mt-3 rounded-xl bg-muted/60 p-3 flex items-center justify-between">
   736→                      <span className="text-xs font-semibold">Évolution du score</span>
   737→                      <span className="font-mono text-lg font-black" style={{ color: rb.score_global >= ra.score_global ? "#3F7D3F" : "#8B1A3B" }}>
   738→                        {rb.score_global >= ra.score_global ? "+" : ""}{rb.score_global - ra.score_global} pts
   739→                      </span>
   740→                    </div>
   741→                    {evolutions.length > 0 && (
   742→                      <div className="mt-3 space-y-1.5">
   743→                        {evolutions.map((e) => (
   744→                          <div key={e.nom} className="flex items-center justify-between text-[11px]">
   745→                            <span className="text-muted-foreground">{e.nom}</span>
