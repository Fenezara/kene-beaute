   387→/** Une question est-elle renseignée ? (texte = jamais requis) */
   388→export function isAnswered(q: Question, answers: QAnswers): boolean {
   389→  const v = answers[q.id];
   390→  if (q.type === "multi") return Array.isArray(v) && v.length > 0;
   391→  if (q.type === "single") return typeof v === "string" && v.length > 0;
   392→  return true;
   393→}
   394→
   395→/** Progression 0-1 sur les questions requises. */
   396→export function questionnaireProgress(answers: QAnswers): number {
   397→  const done = REQUIRED_QUESTIONS.filter((q) => isAnswered(q, answers)).length;
   398→  return REQUIRED_QUESTIONS.length === 0 ? 1 : done / REQUIRED_QUESTIONS.length;
   399→}
   400→
   401→/** Validation serveur : ids des questions requises manquantes. */
   402→export function missingRequired(answers: QAnswers): string[] {
   403→  return REQUIRED_QUESTIONS.filter((q) => !isAnswered(q, answers)).map((q) => q.id);
   404→}

   406→// ─────────────────────────── Moteur de scoring ───────────────────────────
   407→
   408→const norm = (s: string) =>
   409→  s
   410→    .normalize("NFD")
   411→    .replace(/[\u0300-\u036f]/g, "")
   412→    .toLowerCase();
   413→
   414→const BASE_SCORE = 78;
   415→
   416→/** Applique un delta à l'indicateur dont le nom (normalisé) contient la clé. */
   417→function applyEffects(scores: Map<string, number>, effects: Effects | undefined, zoneIndicators: string[]) {
   418→  if (!effects) return;
   419→  for (const [key, delta] of Object.entries(effects)) {
   420→    const k = norm(key);
   421→    for (const name of zoneIndicators) {
   422→      if (norm(name).includes(k)) scores.set(name, (scores.get(name) ?? BASE_SCORE) + delta);
   423→    }
   424→  }
   425→}
   426→
   427→const clampScore = (n: number) => Math.max(5, Math.min(98, Math.round(n)));
   428→
   429→/** Construit le jeu de recommandations ciblées à partir des indicateurs faibles + drapeaux. */

   431→  // Indicateurs les plus faibles d'abord (hors zone nævi — pas de conseil questionnaire)

   436→

   447→    if (has("hydrat", "secheresse")) {

   463→      produits.push("Sérum Éclat Moringa", "SPF 50 teinté minéral");

   538→/** Scoring complet du questionnaire pour une zone. */
   539→export function scoreQuestionnaire(answers: QAnswers, zone: BodyZone): QuestionnaireResult {
   540→  const zoneIndicators = zone === "naevi" ? ZONE_INDICATORS.visage : ZONE_INDICATORS[zone];
   541→  const scores = new Map<string, number>(zoneIndicators.map((n) => [n, BASE_SCORE]));
   542→  const flags: QFlag[] = [];
   543→
   544→  let answered = 0;
   545→  for (const q of QUESTIONS) {
   546→    const v = answers[q.id];
   547→    if (q.type === "single" && typeof v === "string" && v.length > 0) {
   548→      answered++;
   549→      const opt = q.options.find((o) => o.value === v);
   550→      if (opt) {
   551→        applyEffects(scores, opt.effects, zoneIndicators);
   552→        if (opt.flag) flags.push({ level: opt.flag.level ?? "warn", label: opt.flag.label, detail: opt.flag.detail });
   553→      }
   554→    } else if (q.type === "multi" && Array.isArray(v) && v.length > 0) {
   555→      answered++;
   556→      for (const sel of v) {
   557→        const opt = q.options.find((o) => o.value === sel);
   558→        if (opt) applyEffects(scores, opt.effects, zoneIndicators);
   559→      }
   560→    }
   561→  }
   562→  const total = REQUIRED_QUESTIONS.length;
   563→
   564→  // Allergies déclarées → drapeau info
   565→  const allergies = typeof answers.allergies === "string" ? answers.allergies.trim() : "";
   566→  if (allergies) {
   567→    flags.unshift({ level: "info", label: "Allergies déclarées", detail: allergies.slice(0, 200) });
   568→  }
   569→
   570→  // Nævi : le questionnaire ne peut pas évaluer ABCDE — honnêteté clinique
   571→  if (zone === "naevi") {
   572→    flags.push({
   573→      level: "info",
   574→      label: "Nævi : photo indispensable",
   575→      detail: "L'évaluation ABCDE repose sur l'observation — joindre une photo rapprochée pour l'analyse IA (le questionnaire seul ne note pas ces critères).",
   576→    });
   577→  }
   578→
   579→  const indicateurs: Indicator[] = zoneIndicators.map((nom) => {
   580→    const pct = clampScore(scores.get(nom) ?? BASE_SCORE);
   581→    return { nom, pourcentage: pct, severite: severityFromPercent(pct) };
   582→  });
   583→  const scoreGlobal = Math.round(indicateurs.reduce((s, i) => s + i.pourcentage, 0) / Math.max(1, indicateurs.length));
   584→
   585→  const recommandations = buildRecommendations(indicateurs, flags, zone);
   586→  const verdict =
   587→    scoreGlobal >= 80
   588→      ? "la peau déclarée est équilibrée — consolider les bonnes habitudes"
   589→      : scoreGlobal >= 65
   590→        ? "l'équilibre est correct avec des zones d'attention ciblées"
   591→        : scoreGlobal >= 50
   592→          ? "des déséquilibres identifiés méritent un protocole structuré"
   593→          : "un protocole de fond sur plusieurs semaines est recommandé";
   594→  const worst = indicateurs.slice().sort((a, b) => a.pourcentage - b.pourcentage)[0];
   595→  recommandations.resume = `Entretien mené en institut : ${verdict}${worst ? ` — priorité n°1 : ${worst.nom.toLowerCase()} (${worst.pourcentage}/100)` : ""}.`;
   596→
   597→  return { scoreGlobal, indicateurs, flags, recommandations, answered, total };
   598→}

   602→const PHOTO_WEIGHT = 0.62; // l'observation (photo) pèse plus que le déclaratif
   603→const Q_WEIGHT = 0.38;
   604→
   605→/** Fusionne le résultat questionnaire avec l'analyse VLM de la photo (si fournie). */
   606→export function mergeResults(qr: QuestionnaireResult, vlm: DiagnosisResult | null, zone: BodyZone, photoUsed: boolean): ProDiagnosisResult {
   607→  if (!vlm) {
   608→    return {
   609→      score_global: qr.scoreGlobal,
   610→      zone,
   611→      indicateurs: qr.indicateurs,
   612→      zones_marquages: [],
   613→      recommandations: qr.recommandations,
   614→      orientation_dermato: qr.flags.some((f) => f.level === "danger"),
   615→      raison_orientation: qr.flags.find((f) => f.level === "danger")?.label,
   616→      avertissement:
   617→        "Évaluation issue de l'entretien (questionnaire) réalisée en institut — ne constitue pas un diagnostic médical. Joindre une photo à un prochain passage pour une analyse visuelle.",
   618→      source: "questionnaire",
   619→      questionnaire: { score: qr.scoreGlobal, flags: qr.flags, answered: qr.answered, total: qr.total, photoUsed },
   620→    };
   621→  }
   622→
   623→  // Fusion par indicateur de la zone VLM (les noms font autorité côté photo)
   624→  const merged: Indicator[] = vlm.indicateurs.map((ind) => {
   625→    const q = qr.indicateurs.find((qi) => norm(qi.nom).slice(0, 8) === norm(ind.nom).slice(0, 8));
   626→    const pct = q ? Math.round(ind.pourcentage * PHOTO_WEIGHT + q.pourcentage * Q_WEIGHT) : ind.pourcentage;
   627→    return { ...ind, pourcentage: pct, severite: severityFromPercent(pct) };
   628→  });
   629→  const scoreGlobal = Math.round(vlm.score_global * PHOTO_WEIGHT + qr.scoreGlobal * Q_WEIGHT);
   630→
   631→  // Recommandations : base observation VLM + vigilances questionnaire
   632→  const rec: RecommendationSet = {
   633→    ...vlm.recommandations,
   634→    conseils_hygiene_vie: Array.from(
   635→      new Set([...(vlm.recommandations?.conseils_hygiene_vie ?? []), ...qr.recommandations.conseils_hygiene_vie])
   636→    ).slice(0, 6),
   637→    soins_conseilles: Array.from(
   638→      new Set([...(vlm.recommandations?.soins_conseilles ?? []), ...qr.recommandations.soins_conseilles])
   639→    ).slice(0, 6),
   640→  };
   641→
   642→  return {
   643→    score_global: scoreGlobal,
   644→    fitzpatrick_estime: vlm.fitzpatrick_estime,
   645→    zone,
   646→    indicateurs: merged,
   647→    zones_marquages: vlm.zones_marquages ?? [],
   648→    recommandations: rec,
   649→    orientation_dermato: vlm.orientation_dermato,
   650→    raison_orientation: vlm.raison_orientation,
   651→    abcde: vlm.abcde,
   652→    avertissement:
   653→      "Diagnostic réalisé en institut (entretien + analyse IA photo) — estimation éducative, ne constitue pas un diagnostic médical. En cas de lésion évolutive, consultez un dermatologue.",
   654→    source: "vlm+questionnaire",
   655→    questionnaire: { score: qr.scoreGlobal, flags: qr.flags, answered: qr.answered, total: qr.total, photoUsed },
   656→  };
   657→}
