   286→            const active = i === checkedSteps;
   287→            return (
   288→              <div key={i} className={`flex items-center gap-3 rounded-xl border p-3 transition-colors ${done ? "border-[#3F7D3F]/40 bg-[#3F7D3F]/5" : active ? "border-primary/50 bg-primary/5" : "border-border bg-card opacity-50"}`}>
   289→                <span className={`grid place-items-center h-6 w-6 rounded-full shrink-0 ${done ? "bg-[#3F7D3F] text-white" : active ? "bg-primary/20 text-primary" : "bg-muted text-muted-foreground"}`}>

   291→                </span>
   292→                <span className={`text-xs ${done ? "text-[#3F7D3F] font-semibold" : "text-muted-foreground"}`}>{label}</span>
   293→              </div>
   294→            );
   295→          })}
   296→        </div>
   297→        <div className="w-full max-w-[320px] mt-5">
   298→          <Progress value={progress} className="h-2" aria-label="Progression de l'analyse" />
   299→          <p className="text-center text-[11px] text-muted-foreground mt-2">Vision par ordinateur spécialisée mélanoderme — 5 à 30 s</p>
   300→        </div>
   301→      </div>
   302→    );
   303→  }
   304→
   305→  /* ─────────── Étape 4 — Historique ─────────── */

   307→
   308→  /* ─────────── Étape 3 — Résultats ─────────── */
   309→  if (step === 3 && diag) {
   310→    return <ResultView diag={diag} products={products} onNewZone={() => { setStep(0); setImage(""); setDiag(null); }} onHistory={goHistory} />;
   311→  }
   312→
   313→  return null;
   314→}

   617→function HistoryView({
   618→  userId, history, compareMode, setCompareMode, compareSel, setCompareSel, onBack, onOpen,
   619→}: {
   620→  userId: string;
   621→  history: ApiDiagnosis[] | null;
   622→  compareMode: boolean;
   623→  setCompareMode: (v: boolean) => void;
   624→  compareSel: string[];
   625→  setCompareSel: (v: string[]) => void;
   626→  onBack: () => void;
   627→  onOpen: (d: ApiDiagnosis) => void;
   628→}) {
   629→  const list = useMemo(
   630→    () => [...(history ?? [])].filter((d) => d.status === "done").sort((a, b) => +new Date(b.createdAt) - +new Date(a.createdAt)),
   631→    [history]
   632→  );
   633→  const sameZoneSel = compareSel.length === 2 && list.find((d) => d.id === compareSel[0])?.zone === list.find((d) => d.id === compareSel[1])?.zone;
   634→  const a = list.find((d) => d.id === compareSel[0]);
   635→  const b = list.find((d) => d.id === compareSel[1]);
   636→  const ra = a ? parseDiagnosis(a.resultJson) : null;
   637→  const rb = b ? parseDiagnosis(b.resultJson) : null;
   638→
   639→  const evolutions =
   640→    ra && rb
   641→      ? rb.indicateurs
   642→          .map((ib) => {
   643→            const ia = ra.indicateurs.find((x) => x.nom === ib.nom);
   644→            return ia ? { nom: ib.nom, delta: ib.pourcentage - ia.pourcentage } : null;
   645→          })
   646→          .filter((x): x is { nom: string; delta: number } => x !== null)
   647→          .sort((x, y) => Math.abs(y.delta) - Math.abs(x.delta))
   648→          .slice(0, 3)
   649→      : [];
   650→
   651→  return (
   652→    <div className="pt-4 pb-2">
   653→      <button onClick={onBack} className="inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground mb-4 focus-visible:outline-2 focus-visible:outline-primary rounded" aria-label="Retour">
   654→        <ArrowLeft size={15} /> Nouveau diagnostic
   655→      </button>
   656→      <div className="flex items-center justify-between gap-2">
   657→        <h1 className="font-heading font-black text-xl">Mon historique</h1>
   658→        <button
   659→          onClick={() => { setCompareMode(!compareMode); setCompareSel([]); }}
   660→          aria-pressed={compareMode}
   661→          className={`h-9 px-3 rounded-full text-[11px] font-bold flex items-center gap-1.5 active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${compareMode ? "bg-primary text-primary-foreground" : "border border-border bg-card"}`}
   662→        >
   663→          <GitCompareArrows size={14} /> Comparer
   664→        </button>
   665→      </div>
   666→
   667→      {history === null ? (
   668→        <div className="space-y-3 mt-5">
   669→          {[0, 1, 2].map((i) => (
   670→            <Skeleton key={i} className="h-20 rounded-2xl" />
   671→          ))}
   672→        </div>
   673→      ) : list.length === 0 ? (
   674→        <div className="mt-6 rounded-2xl border border-dashed border-border bg-card/60 p-8 text-center">
   675→          <NeaOnnimIcon size={38} className="mx-auto text-primary" />
   676→          <p className="font-heading font-bold text-sm mt-3">Aucun diagnostic pour l&apos;instant</p>
   677→          <p className="text-xs text-muted-foreground mt-1">Ton premier scan débloquera le suivi d&apos;évolution.</p>
   678→        </div>
   679→      ) : (
   680→        <>
   681→          {/* Le Fil du Temps — courbes d'évolution */}
   682→          <EvolutionCard userId={userId} className="mt-4" />
   683→
   684→          <div className="space-y-2.5 mt-4">
   685→            {list.map((d) => {
   686→              const sel = compareSel.includes(d.id);
   687→              return (
   688→                <div key={d.id} className={`flex items-center gap-3 rounded-2xl border p-2.5 transition-colors ${sel ? "border-primary bg-primary/5" : "border-border bg-card"}`}>
   689→                  <button
   690→                    onClick={() => {
   691→                      if (compareMode) {
   692→                        setCompareSel(sel ? compareSel.filter((x) => x !== d.id) : compareSel.length < 2 ? [...compareSel, d.id] : [compareSel[1], d.id]);
   693→                      } else onOpen(d);
   694→                    }}
   695→                    className="flex items-center gap-3 flex-1 min-w-0 text-left focus-visible:outline-2 focus-visible:outline-primary rounded-xl"
   696→                    aria-label={`Diagnostic ${d.zone} du ${formatDate(d.createdAt)}, score ${d.scoreGlobal}`}
   697→                  >
   698→                    <img src={diagImgSrc(d.imageData)} alt={`Diagnostic ${d.zone}`} loading="lazy" className="h-14 w-14 rounded-xl object-cover shrink-0" />
   699→                    <div className="min-w-0">
   700→                      <p className="text-sm font-semibold">{BODY_ZONES.find((z) => z.id === d.zone)?.label ?? d.zone}</p>
   701→                      <p className="text-[11px] text-muted-foreground">{formatDate(d.createdAt)}</p>

   706→                    <button

   708→                      aria-pressed={sel}
   709→                      aria-label="Sélectionner pour comparaison"
   710→                      className={`h-8 w-8 grid place-items-center rounded-full border-2 shrink-0 transition-colors ${sel ? "bg-primary border-primary text-primary-foreground" : "border-border"}`}

   720→          {compareMode && (

   727→                    </p>
   728→                    <div className="grid grid-cols-2 gap-3">
   729→                      {[{ d: a, r: ra, tag: "Avant" }, { d: b, r: rb, tag: "Après" }].map(({ d, r, tag }) => (
   730→                        <div key={tag}>

   732→                          <div className="flex items-center justify-between mt-1.5">
   733→                            <span className="text-[10px] uppercase tracking-wide font-bold text-muted-foreground">{tag} · {formatDate(d.createdAt, { day: "numeric", month: "short" })}</span>
   734→                            <ScoreChip score={r.score_global} />
   735→                          </div>
   736→                        </div>
   737→                      ))}
   738→                    </div>
   739→                    <div className="mt-3 rounded-xl bg-muted/60 p-3 flex items-center justify-between">
   740→                      <span className="text-xs font-semibold">Évolution du score</span>

   742→                        {rb.score_global >= ra.score_global ? "+" : ""}{rb.score_global - ra.score_global} pts
   743→                      </span>
   744→                    </div>
   745→                    {evolutions.length > 0 && (
   746→                      <div className="mt-3 space-y-1.5">
   747→                        {evolutions.map((e) => (
   748→                          <div key={e.nom} className="flex items-center justify-between text-[11px]">
