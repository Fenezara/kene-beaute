   244→  // Suivi d'un diagnostic serveur: progression honnête pilotée par le temps
   245→  // écoulé (le serveur n'expose pas d'avancement réel) + poll
   246→  // GET /api/diagnoses/{id} toutes les 1,8 s. Timeout global 100 s → échec
   247→  // honnête + Réessayer (retour capture, photo conservée — comportement 66-b).
   248→  // Échec réseau d'un tick → on continue de poller (garde-fou: timeout).
   249→  const trackDiagnosis = useCallback(
   250→    (diagId: string, opts: { recovered?: boolean; startedAt?: number } = {}) => {
   251→      clearTimers(); // filet: aucun reliquat d'un suivi précédent
   252→      const gen = trackGenRef.current;
   253→      const t0 = opts.startedAt ?? Date.now();
   254→      const recovered = opts.recovered ?? false;
   255→      if (recovered) setNetNotice("Récupération de ton analyse…");
   256→
   257→      function stopIntervals() {
   258→        if (timersRef.current.timer) { clearInterval(timersRef.current.timer); timersRef.current.timer = undefined; }
   259→        if (timersRef.current.poll) { clearInterval(timersRef.current.poll); timersRef.current.poll = undefined; }
   260→      }
   261→      function failAnalysis(msg: string) {
   262→        if (trackGenRef.current !== gen) return;
   263→        stopIntervals();
   264→        setAnalyzing(false);
   265→        setNetNotice(null);
   266→        toast.error(msg);
   267→        setStep(1); // retour capture: la photo est conservée → Réessayer
   268→      }
   269→
   270→      // Progression: courbe exponentielle vers 96 % (jamais 100 avant la fin
   271→      // réelle) — la reprise au montage repart de createdAt (startedAt).
   272→      const timer = setInterval(() => {
   273→        const elapsed = Date.now() - t0;
   274→        setProgress(Math.min(96, Math.round(96 * (1 - Math.exp(-elapsed / 15000)))));
   275→        setCheckedSteps(elapsed > 16000 ? 2 : 1);
   276→      }, 420);
   277→      timersRef.current.timer = timer;
   278→
   279→      const check = (d: ApiDiagnosis | undefined | null) => {
   280→        if (trackGenRef.current !== gen) return;
   281→        if (!d) { failAnalysis("Diagnostic introuvable — retente dans un instant"); return; }
   282→        if (d.status === "done") {
   283→          const result = parseDiagnosis(d.resultJson);
   284→          stopIntervals();
   285→          if (!result) { failAnalysis("Résultat IA illisible — retente dans un instant"); return; }
   286→          showResult({ id: d.id, result, imageData: diagImgSrc(d.imageData), createdAt: d.createdAt }, t0, recovered);
   287→        } else if (d.status === "error") {
   288→          failAnalysis("Moteur d'analyse momentanément injoignable — tes photos restent prêtes, retente dans un instant.");
   289→        }
   290→        // "pending" → on continue de poller (timeout global en garde-fou)
   291→      };
   292→
   293→      const POLL_MS = 1800;
   294→      const TIMEOUT_MS = 100_000;
   295→      let busy = false; // un fetch lent n'empile pas les ticks suivants
   296→      const poll = setInterval(() => {
   297→        if (Date.now() - t0 > TIMEOUT_MS) {
   298→          failAnalysis("Ton analyse met plus de temps que prévu — ta photo reste prête, retente dans un instant.");
   299→          return;
   300→        }
   301→        if (busy) return;
   302→        busy = true;
   303→        apiGet<{ diagnosis: ApiDiagnosis }>(`/api/diagnoses/${diagId}`)
   304→          .then((r) => { busy = false; check(r.diagnosis); })
   305→          .catch(() => { busy = false; /* réseau: le tick suivant réessaie */ });
   306→      }, POLL_MS);
   307→      timersRef.current.poll = poll;
   308→
   309→      // Premier statut immédiat: si le worker a déjà fini (reprise au
   310→      // montage), les résultats arrivent sans attendre le 1er tick.
   311→      apiGet<{ diagnosis: ApiDiagnosis }>(`/api/diagnoses/${diagId}`)
   312→        .then((r) => check(r.diagnosis))
   313→        .catch(() => undefined);
   314→    },
   315→    [clearTimers, showResult],
   316→  );
   317→
   318→  // Récupération après échec réseau (66-b): le diagnostic a très bien pu
   319→  // être ENREGISTRÉ avant la coupure. Done récent (< 4 min) → résultats;
   320→  // pending → on reprend son suivi (poll du worker). Retourne true si pris
   321→  // en charge.
   322→  async function tryRecover(t0: number): Promise<boolean> {
   323→    const RECENT_MS = 4 * 60 * 1000;
   324→    const findRecent = (list: ApiDiagnosis[] | undefined): ApiDiagnosis | undefined =>
   325→      (list ?? [])
   326→        .filter(
   327→          (d) =>
   328→            d.zone === zone &&
   329→            (d.status === "done" || d.status === "pending") &&
   330→            Date.now() - new Date(d.createdAt).getTime() < RECENT_MS,
   331→        )
   332→        .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())[0];
   333→    try {
   334→      const found = findRecent((await apiGet<{ diagnoses: ApiDiagnosis[] }>(`/api/diagnoses?userId=${user.id}`)).diagnoses);
   335→      if (!found) return false;
   336→      if (found.status === "done") {
   337→        const result = parseDiagnosis(found.resultJson);
   338→        if (!result) return false;
   339→        showResult(
   340→          { id: found.id, result, imageData: diagImgSrc(found.imageData), createdAt: found.createdAt },
   341→          t0,
   342→          true,
   343→        );
   344→        return true;
   345→      }
   346→      // pending: le worker tourne encore côté serveur → on le suit.
   347→      setZone(found.zone);
   348→      setImage(diagImgSrc(found.imageData));
   349→      trackDiagnosis(found.id, { recovered: true, startedAt: new Date(found.createdAt).getTime() });
   350→      return true;
   351→    } catch {
   352→      // Le GET lui-même est injoignable (serveur toujours au redémarrage):
   353→      // échec final honnête, sans crash.
   354→      return false;
   355→    }
   356→  }

   359→  // l'historique est "pending" et récent (< 3 min), on reprend son suivi —
   360→  // la cliente qui quitte et revient ne perd rien. Les setState vivent dans
   361→  // la continuation async (pattern loadProducts — règle set-state-in-effect).
   362→  const resumePending = useCallback(async () => {
   363→    try {
   364→      const r = await apiGet<{ diagnoses: ApiDiagnosis[] }>(`/api/diagnoses?userId=${user.id}`);

   380→    }
   381→  }, [user.id, trackDiagnosis]);

   387→  async function launch() {
   388→    if (!image) return;
   389→    launchedRef.current = true;
   390→    setStep(2);
   391→    setAnalyzing(true);
   392→    setProgress(0);
   393→    setCheckedSteps(0);
   394→    setNetNotice(null);
   395→    setQuotaUpsell(false);
   396→    const t0 = Date.now();
   397→    // Hors-ligne dès le départ: la photo part en file d'attente —
   398→    // envoyée toute seule au retour du réseau (la replay vit dans ClientApp,
   399→    // elle prévient par toast). Jamais d'échec sec pour une photo déjà cadrée.
   400→    if (!isOnline()) {
   401→      const q = enqueueDiag({ userId: user.id, zone, image, fitzpatrick: user.fitzpatrick ?? undefined, allergies: user.allergies ?? undefined });
   402→      haptic(HAPTIC.light);
   403→      setAnalyzing(false);
   404→      setStep(1); // la photo reste affichée: la cliente voit qu'elle est gardée
   405→      toast.success("Diagnostic mis en attente", {
   406→        description: q.ok
   407→          ? "Ta photo partira toute seule dès que le réseau revient — tu peux même quitter l’app."
   408→          : "File indisponible sur cet appareil — retente quand le réseau revient.",
   409→      });
   410→      return;
   411→    }
   412→    try {
   413→      // POST /api/diagnoses: répond 202 en < 1 s avec la ligne "pending"
   414→      // (pipeline asynchrone). 4 tentatives (: 3 → 4, backoff
   415→      // jusqu'à 8 s ≈ ~13,5 s de fenêtre) quand la gateway renvoie
   416→      // 502/503/504 (recompilation/restart du serveur Next en dev) — backoff
   417→      // 1,5 s → 4 s → 8 s. Les autres erreurs (400/404/429…) ne sont JAMAIS
   418→      // rejouées; chaque retry crée une NOUVELLE ligne côté serveur si le
   419→      // POST a échoué AVANT d'atteindre l'app (échec réseau = rien reçu).
   420→      let r: { diagnosis: ApiDiagnosis } | undefined;
   421→      let fatal: unknown = new Error("Analyse impossible");
   422→      const backoffs = [1500, 4000, 8000];
   423→      for (let attempt = 0; attempt < 4; attempt += 1) {
   424→        try {
   425→          r = await apiPost<{ diagnosis: ApiDiagnosis }>("/api/diagnoses", {
   426→            userId: user.id,
   427→            zone,
   428→            image,
   429→            fitzpatrick: user.fitzpatrick ?? undefined,
   430→            allergies: user.allergies ?? undefined,
   431→          });
   432→          break;
   433→        } catch (e) {
   434→          fatal = e;
   435→          const retryable = e instanceof ApiError && [502, 503, 504].includes(e.status);
   436→          if (!retryable || attempt === 3) break;
   437→          setNetNotice("Rétablissement de la connexion…");
   438→          await new Promise((res) => setTimeout(res, backoffs[attempt]));
   439→          setNetNotice(null);
   440→        }
   441→      }
   442→      if (!r) throw fatal;
   443→      // Compat: un back encore synchrone (status "done" direct) est accepté
   444→      // tel quel — même animation de fin.
   445→      if (r.diagnosis.status === "done") {
   446→        const result = parseDiagnosis(r.diagnosis.resultJson);
   447→        if (result) {

  1214→function HistoryView({
  1215→  userId, history, compareMode, setCompareMode, compareSel, setCompareSel, onBack, onOpen,
  1216→}: {
  1217→  userId: string;
  1218→  history: ApiDiagnosis[] | null;
  1219→  compareMode: boolean;
  1220→  setCompareMode: (v: boolean) => void;
  1221→  compareSel: string[];

  1223→  onBack: () => void;
  1224→  onOpen: (d: ApiDiagnosis) => void;
  1225→}) {
  1226→  const list = useMemo(
  1227→    () => [...(history ?? [])].filter((d) => d.status === "done").sort((a, b) => +new Date(b.createdAt) - +new Date(a.createdAt)),
  1228→    [history]
  1229→  );
  1230→  const sameZoneSel = compareSel.length === 2 && list.find((d) => d.id === compareSel[0])?.zone === list.find((d) => d.id === compareSel[1])?.zone;
  1231→  const a = list.find((d) => d.id === compareSel[0]);
  1232→  const b = list.find((d) => d.id === compareSel[1]);
  1233→  const ra = a ? parseDiagnosis(a.resultJson) : null;
  1234→  const rb = b ? parseDiagnosis(b.resultJson) : null;
  1235→
  1236→  const evolutions =
  1237→    ra && rb
  1238→      ? rb.indicateurs
  1239→          .map((ib) => {
  1240→            const ia = ra.indicateurs.find((x) => x.nom === ib.nom);
  1241→            return ia ? { nom: ib.nom, delta: ib.pourcentage - ia.pourcentage } : null;
  1242→          })
  1243→          .filter((x): x is { nom: string; delta: number } => x !== null)
  1244→          .sort((x, y) => Math.abs(y.delta) - Math.abs(x.delta))

  1248→  return (
  1249→    <div className="pt-4 pb-2">
  1250→      <button onClick={onBack} className="inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground mb-4 focus-visible:outline-2 focus-visible:outline-primary rounded" aria-label="Retour">
  1251→        <ArrowLeft size={15} /> Nouveau diagnostic
  1252→      </button>
  1253→      <div className="flex items-center justify-between gap-2">
  1254→        <h2 className="font-heading font-black text-xl">Mon historique</h2>
  1255→        <Chip selected={compareMode} onClick={() => { setCompareMode(!compareMode); setCompareSel([]); }}>
  1256→          <GitCompareArrows size={14} /> Comparer
  1257→        </Chip>
  1258→      </div>

  1261→        <div className="space-y-3 mt-5">
  1262→          {[0, 1, 2].map((i) => (
  1263→            <Shimmer key={i} className="h-20 rounded-[18px]" />
  1264→          ))}
  1265→        </div>
  1266→      ) : list.length === 0 ? (
  1267→        <div className="mt-6">
  1268→          <EmptyBlock
  1269→            icon={<NeaOnnimIcon size={30} />}
  1270→            title="Aucun diagnostic pour l'instant"
  1271→            text="Ton premier scan débloquera le suivi d'évolution."
  1272→          />
  1273→        </div>
  1274→      ) : (
  1275→        <>
  1276→          {/* Le Fil du Temps — courbes d'évolution */}
  1277→          <EvolutionCard userId={userId} className="mt-4" />
  1278→
  1279→          <Reveal className="space-y-2.5 mt-4" stagger={0.07}>
  1280→            {list.map((d) => {
