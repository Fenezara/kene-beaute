    36→export function ProfileScreen() {
    37→  const user = useKene((s) => s.user) as ClientUser;
    38→  const setUser = useKene((s) => s.setUser);
    39→  const setSpace = useKene((s) => s.setSpace);
    40→  const setClientTab = useKene((s) => s.setClientTab);
    41→  const clearCart = useKene((s) => s.clearCart);
    42→  const { t, lang, setLang } = useT();
    43→
    44→  const [edit, setEdit] = useState(false);
    45→  const [name, setName] = useState(user.name);
    46→  const [city, setCity] = useState(user.city ?? "");
    47→  const [savingId, setSavingId] = useState(false);
    48→
    49→  const [fitz, setFitz] = useState(user.fitzpatrick ?? "V");
    50→  const [skinType, setSkinType] = useState(user.skinType ?? "mixte");
    51→  const [allergies, setAllergies] = useState(user.allergies ?? "");
    52→  const [goals, setGoals] = useState<string[]>(() => {
    53→    try {
    54→      return (JSON.parse(user.goals ?? "[]") as { id: string }[]).map((g) => g.id);
    55→    } catch {
    56→      return [];
    57→    }
    58→  });
    59→  const [savingSkin, setSavingSkin] = useState(false);
    60→
    61→  const [wallet, setWallet] = useState<ApiWallet | null>(null);
    62→  const [txs, setTxs] = useState<ApiWalletTx[] | null>(null);
    63→  const [topup, setTopup] = useState(false);
    64→  const [amount, setAmount] = useState(5000);
    65→  const [method, setMethod] = useState<"wave" | "orange">("wave");
    66→  const [topupState, setTopupState] = useState<"idle" | "processing" | "done">("idle");
    67→  const [topupBusy, setTopupBusy] = useState(false);
    68→  const [exportBusy, setExportBusy] = useState(false);
    69→
    70→  // Installation PWA — même source d'événement que la bannière d'accueil.
    71→  const { canInstall, promptInstall, isStandalone, isIOS } = useInstallPrompt();
    72→
    73→  // Sécurité renforcée (2FA-lite) : code SMS exigé avant chaque paiement.
    74→  const secureEnabled = useSecurity((s) => s.enabled);
    75→  const setSecureEnabled = useSecurity((s) => s.setEnabled);
    76→
    77→  const loadWallet = useCallback(() => {
    78→    apiGet<{ wallet: ApiWallet; transactions: ApiWalletTx[] }>(`/api/wallet?userId=${user.id}`)
    79→      .then((r) => { setWallet(r.wallet); setTxs(r.transactions ?? []); })
    80→      .catch(() => setTxs([]));
    81→  }, [user.id]);
    82→
    83→  useEffect(() => {
    84→    loadWallet();
    85→  }, [loadWallet]);
    86→
    87→  /** Portabilité RGPD — télécharge « mes données » en JSON via blob */
    88→  async function downloadMyData() {
    89→    setExportBusy(true);
    90→    try {
    91→      const res = await fetch(`/api/profile/export?userId=${encodeURIComponent(user.id)}`);
    92→      if (!res.ok) throw new Error("Export impossible");
    93→      const blob = await res.blob();
    94→      const cd = res.headers.get("Content-Disposition") ?? "";
    95→      const m = /filename="([^"]+)"/.exec(cd);
    96→      const url = URL.createObjectURL(blob);
    97→      const a = document.createElement("a");
    98→      a.href = url;
    99→      a.download = m?.[1] ?? `kene-mes-donnees-${new Date().toISOString().slice(0, 10)}.json`;
   100→      document.body.appendChild(a);
   101→      a.click();
   102→      a.remove();
   103→      setTimeout(() => URL.revokeObjectURL(url), 4000);
   104→      const kb = Math.max(1, Math.round(blob.size / 1024));
   105→      toast.success(`Mes données téléchargées · ${kb} Ko`);
   106→    } catch (e) {
   107→      toast.error(e instanceof Error ? e.message : "Export impossible");
   108→    } finally {
   109→      setExportBusy(false);
   110→    }
   111→  }
   112→
   113→  /** Installer l'app (prompt natif, ou instructions iOS / navigateur). */
   114→  async function installApp() {
   115→    if (canInstall) {
   116→      const accepted = await promptInstall();
   117→      if (accepted) toast.success("Kènè installée sur ton écran d'accueil 💛");
   118→      return;
   119→    }
   120→    if (isIOS) {
   121→      toast.info("Installer sur iPhone", {
   122→        description: "Bouton Partager ⬆️ puis « Sur l'écran d'accueil ».",
   123→        duration: 8000,
   124→      });
   125→      return;
   126→    }
   127→    toast.info("Installation manuelle", {
   128→      description: "Menu du navigateur → « Installer l'application » ou « Ajouter à l'écran d'accueil ».",
   129→      duration: 8000,
   130→    });
   131→  }
   132→
   133→  async function saveIdentity() {
   134→    setSavingId(true);
   135→    try {
   136→      const r = await apiPatch<{ user: ApiUser }>("/api/auth/profile", { userId: user.id, name: name.trim() || undefined, city: city.trim() || undefined });
   137→      setUser({ ...user, ...r.user } as SessionUser);
   138→      setEdit(false);
   139→      toast.success("Profil mis à jour");
   140→    } catch (e) {
   141→      toast.error(e instanceof Error ? e.message : "Mise à jour impossible");
   142→    } finally {
   143→      setSavingId(false);
   144→    }
   145→  }
   146→
   147→  async function saveSkin() {
   148→    setSavingSkin(true);
   149→    try {
   150→      const r = await apiPatch<{ user: ApiUser }>("/api/auth/profile", {
   151→        userId: user.id,
   152→        skinType,
   153→        fitzpatrick: fitz,
   154→        allergies: allergies.trim() || undefined,
   155→        goals: goals.map((id) => ({ id, label: SKIN_GOALS.find((g) => g.id === id)?.label ?? id })),
   156→      });
   157→      setUser({ ...user, ...r.user } as SessionUser);
   158→      toast.success("Profil peau mis à jour");
   159→    } catch (e) {
   160→      toast.error(e instanceof Error ? e.message : "Mise à jour impossible");
   161→    } finally {
   162→      setSavingSkin(false);
   163→    }
   164→  }
   165→
   166→  /** Langue de l'interface (i18n) — indépendante de la langue de lecture vocale. */
   167→  function selectLang(l: Lang) {
   168→    if (l === lang) return;
   169→    setLang(l);
   170→    const label = LANGS.find((x) => x.id === l)?.label ?? "";
   171→    toast.success(`Interface en ${label.toLowerCase()}`);
   172→  }
   173→
   174→  /** Sécurité renforcée : bascule la re-vérification par code avant paiement. */
   175→  function toggleSecure() {
   176→    const next = !secureEnabled;
   177→    setSecureEnabled(next);
   178→    if (next) toast.success("Sécurité renforcée activée");
   179→    else toast("Sécurité renforcée désactivée");
   180→  }
   181→
   182→  async function runTopup() {
   183→    setTopupBusy(true);
   184→    setTopupState("processing");
   185→    try {
   186→      const r = await apiPost<{ payment: { id: string; confirmToken?: string } }>("/api/wallet/topup", { userId: user.id, amount, method });
   187→      // Contrat confirmToken (63-b/63-c) : un approvisionnement mobile money en
   188→      // attente porte son jeton — absent, on n'appelle JAMAIS confirm et on
   189→      // revient au formulaire comme après un échec (aucun crédit fantôme).
   190→      const confirmToken = r.payment.confirmToken;
   191→      if (!confirmToken) {
   192→        setTopupState("idle");
   193→        toast.error("Paiement impossible — réessaie dans quelques instants");
   194→        return;
   195→      }
   196→      await new Promise((res) => setTimeout(res, 2600));
   197→      await apiPost("/api/payments/confirm", { paymentId: r.payment.id, confirmToken });
   198→      setTopupState("done");
   199→      toast.success(`${xof(amount)} crédités sur ton wallet`);
   200→      loadWallet();
   201→    } catch (e) {
   202→      setTopupState("idle");
   203→      toast.error(e instanceof Error ? e.message : "Approvisionnement impossible");
   204→    } finally {
   205→      setTopupBusy(false);
   206→    }
   207→  }
   208→
   209→  return (
   210→    <>
   211→      <Reveal className="pt-4 pb-2 flex flex-col gap-6" stagger={0.07}>
   212→        <RevealItem className="self-start">
   213→          <button onClick={() => setClientTab("accueil")} className="inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground focus-visible:outline-2 focus-visible:outline-primary rounded min-h-10 px-1" aria-label={t("profile.back.aria")}>
   214→            <ArrowLeft size={15} /> {t("tab.home")}
   215→          </button>
   216→        </RevealItem>
   217→
   218→      {/* Identité */}
   219→      <RevealItem>
   220→        <section aria-labelledby="me-t" className="k-card overflow-hidden rounded-[24px]">
   221→          <div className="kente-band h-1.5 w-full" aria-hidden="true" />
   222→          <div id="me-t" className="p-5">
   223→            <div className="flex items-center gap-4">
   224→              <span className="k-glow-gold grid place-items-center h-16 w-16 rounded-full bg-gradient-to-br from-[#C8951E] to-[#A0522D] text-[#FFF9EC] font-heading font-black text-2xl">
   225→                {user.name.charAt(0)}
   226→              </span>
   227→              <div className="min-w-0 flex-1">
   228→                <p className="font-heading font-black text-[22px] leading-tight truncate">{user.name}</p>
   229→                <p className="text-xs text-muted-foreground flex items-center gap-1.5 mt-1 font-mono"><Phone size={12} /> {user.phone}</p>
   230→                <p className="text-xs text-muted-foreground flex items-center gap-1.5 mt-0.5"><MapPin size={12} /> {user.city || "Ville non renseignée"}</p>
   231→              </div>

   233→                <Pencil size={16} />

   246→                  </div>
   247→                  <button onClick={saveIdentity} disabled={savingId} className="k-btn-gold h-11 w-full rounded-xl text-primary-foreground text-sm font-bold flex items-center justify-center gap-2 disabled:opacity-60 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary">
   248→                    {savingId ? <Loader2 size={16} className="animate-spin" /> : <Check size={16} />} Enregistrer
   249→                  </button>
   250→                </div>
   251→              </motion.div>

   253→          </div>
   254→        </section>

   256→
   257→      {/* Profil peau */}

"border-primary" : "border-transparent"}`}>
   267→                    <div className="h-8 rounded-lg mb-1" style={{ background: f.gradient }} aria-hidden="true" />

   276→                {SKIN_TYPES.map((t) => (

"k-btn-gold text-primary-foreground" : "k-chip"}`}>{t.label}</button>
   278→                ))}
   279→              </div>

   294→              {savingSkin ? <Loader2 size={15} className="animate-spin" /> : <Check size={15} />} Mettre à jour mon profil peau
   295→            </button>
   296→          </div>
   297→        </section>

   300→      {/* Langue de l'interface — i18n UI, indépendante de la lecture vocale

   305→            <IconBadge icon={<Languages size={19} />} tone="gold" />
   306→            <div className="flex-1 min-w-0">
   307→              <p id="lang-t" className="text-xs font-bold">{t("lang.selector.label")}</p>

   310→          </div>

   344→            <div aria-hidden="true" className="absolute inset-0 bogolan-dots opacity-20" />
   345→            <div className="relative">

   351→              <button onClick={() => { setTopup(true); setTopupState("idle"); }} className="k-btn-gold mt-4 h-11 w-full rounded-xl text-primary-foreground font-bold text-sm flex items-center justify-center gap-2 focus-visible:outline-2 focus-visible:outline-[#C8951E]">

   385→      {/* Parrainage — le fil qui relie les amies */}

   388→      {/* Consentement */}

   390→        <section aria-labelledby="cons-t" className="k-card rounded-[24px] p-4 flex items-center gap-3">
