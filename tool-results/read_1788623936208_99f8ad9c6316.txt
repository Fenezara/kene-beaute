   286→            );
   287→          }
   288→        } catch {
   289→          /* session injoignable → message du 409 ci-dessous */
   290→        }
   291→        if (!entered) toast.error(e.message || "Un espace entreprise existe déjà pour ce numéro");
   292→      } else {
   293→        toast.error(e instanceof Error ? e.message : "Création impossible");
   294→      }
   295→    } finally {
   296→      setSavingPro(false);
   297→    }
