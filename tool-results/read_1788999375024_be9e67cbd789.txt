     1→// Kènè — « cerveau » du Dr Kènè : base de connaissances structurée.
     2→// Lib PURE (aucune dépendance React/serveur). Deux usages :
     3→//  1. Les données typées ci-dessous (conditions, dépigmentation, botaniques,
     4→//     actifs, signes rouges, contexte ivoirien) : richesse pour l'app
     5→//     (futurs écrans pédagogiques, croissements questionnaire/glossaire).
     6→//  2. KNOWLEDGE_DIGEST : version CONDENSÉE (≤ ~900 mots) assemblée par code
     7→//     depuis ces données — injectée dans le prompt système du chat LLM
     8→//     (POST /api/dermato/chat).
     9→// Positionnement : le Dr Kènè ÉDUQUE et ORIENTE. Elle ne pose JAMAIS de
    10→// diagnostic formel et ne prescrit JAMAIS de médicament (voir CHARACTER_RULES).
    11→
    12→// ─────────────────────────── Types ───────────────────────────
    13→
    14→/** Niveau d'une condition : « éducatif » (conseils doux possibles) ou
    15→ *  « référer » (orientation professionnelle immédiate). */
    16→export type KnowledgeLevel = "educatif" | "referer";
    17→
    18→export interface KnowledgeCondition {
    19→  id: string;
    20→  name: string;
    21→  level: KnowledgeLevel;
    22→  /** Résumé 2-3 phrases, niveau patiente (mots du quotidien). */
    23→  summary: string;
    24→  /** Signes d'alerte : ce qui doit faire sortir du cadre « éducatif ». */
    25→  redFlags: string[];
    26→  /** Conduite à tenir par Kènè (hygiène, soins doux, prévention). */
    27→  action: string;
    28→  /** Quand référer (dermatologue / médecin / urgence). */
    29→  refer: string;
    30→  /** Ligne ultra-compacte (≈ 15-20 mots) pour le digest LLM. */
    31→  digest: string;
    32→}
    33→
    34→export interface KnowledgeBotanical {
    35→  id: string;
    36→  name: string;
    37→  /** Vertus principales. */
    38→  virtues: string;
    39→  /** Forme d'usage. */
    40→  usage: string;
    41→  /** Précautions d'emploi. */
    42→  caution: string;
    43→  digest: string;
    44→}
    45→
    46→export interface KnowledgeActive {
    47→  id: string;
    48→  name: string;
    49→  /** Rôle sur peau foncée. */
    50→  role: string;
    51→  /** Conseil d'introduction. */
    52→  advice: string;
    53→  digest: string;
    54→}
    55→
    56→export interface KnowledgeRedFlag {
    57→  id: string;
    58→  /** Le signe, décrit simplement. */
    59→  sign: string;
    60→  /** Conduite : qui voir, à quelle vitesse. */
    61→  action: string;
    62→  digest: string;
    63→}
    64→
    65→export interface DepigmentationDanger {
    66→  id: string;
    67→  title: string;
    68→  products: string;
    69→  complications: string[];
    70→  message: string;
    71→  approach: string[];
    72→  refer: string;
    73→  digest: string;
    74→}
