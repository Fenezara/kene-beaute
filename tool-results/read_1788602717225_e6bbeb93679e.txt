   131→  // ─── Swipe horizontal entre onglets (TikTok) ───

   133→

   182→  function goTab(t: ClientTab) {

   184→      const from = SWIPE_ORDER.indexOf(tab);
   185→      const to = SWIPE_ORDER.indexOf(t);

   192→
   193→  // ─── Gestes tactiles du conteneur de flux ───
   194→  const onTouchStart = (e: React.TouchEvent) => {
   195→    const el = scrollRef.current;
   196→    const t = e.touches[0];
   197→    pullStart.current = { y: t.clientY, x: t.clientX, atTop: !el || el.scrollTop <= 0 };
   198→    // le swipe est ignoré s'il démarre dans une rangée horizontale scrollable
   199→    const target = e.target as HTMLElement;
   200→    const inScrollRow = !!target.closest("[data-scroll-row]");
   201→    swipeStart.current = { x: t.clientX, y: t.clientY, ok: !inScrollRow };
   202→  };
   203→
   204→  const onTouchMove = (e: React.TouchEvent) => {
   205→    if (refreshing) return;
   206→    const s = pullStart.current;
   207→    if (!s || !s.atTop) return;
   208→    const t = e.touches[0];
   209→    const dy = t.clientY - s.y;
   210→    const dx = t.clientX - s.x;
   211→    if (dy <= 0 || Math.abs(dx) > Math.abs(dy)) return; // tirage vertical uniquement
   212→    setPull(Math.min(dy / 2.2, 96)); // résistif : l'icône s'alourdit en fin de course
   213→  };

   215→  const onTouchEnd = (e: React.TouchEvent) => {
   216→    // 1) fin de tirage → actualisation si seuil franchi
   217→    const dist = pull;
   218→    pullStart.current = null;

   220→      haptic(HAPTIC.medium);
   221→      setRefreshing(true);
   222→      setPull(0);
   223→      wasRefreshing.current = true;
   224→      setRefreshKey((k) => k + 1);
   225→    } else {
   226→      setPull(0);
   227→    }

   229→    const s = swipeStart.current;
   230→    swipeStart.current = null;
   231→    if (!s || !s.ok || refreshing) return;
   232→    const t = e.changedTouches[0];
   233→    const dx = t.clientX - s.x;
   234→    const dy = t.clientY - s.y;
   235→    if (Math.abs(dx) < 64 || Math.abs(dx) < Math.abs(dy) * 1.8) return;

   237→    if (idx < 0) return;
   238→    const next = dx < 0 ? SWIPE_ORDER[idx + 1] : SWIPE_ORDER[idx - 1];
   239→    if (next) {
   240→      setNavDir(dx < 0 ? 1 : -1);
   241→      haptic(HAPTIC.tap);
   242→      setClientTab(next);
   243→    }
   244→  };
   245→
   246→  // Le fil signale la fin de son rechargement (déclenché par refreshKey)
   247→  const onRefreshed = useCallback(() => {
   248→    if (!wasRefreshing.current) return;
   249→    wasRefreshing.current = false;
   250→    setRefreshing(false);
   251→    haptic(HAPTIC.success);
