     1→"use client";
     2→// Kènè Cliente — Boutique : catalogue, fiche produit, panier, checkout Wave/Orange/Wallet simulé
     3→// + « Mes commandes » : historique des commandes enregistrées (consultation par la cliente).
     4→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     5→import { AnimatePresence, motion } from "framer-motion";
     6→import { BadgeCheck, Check, History, Loader2, Minus, Plus, Search, ShoppingBag, Tag, Trash2, X } from "lucide-react";
     7→import { toast } from "sonner";
     8→import { apiGet, apiPost } from "@/lib/kene/api";
     9→import { xof, CASHBACK_RATE, formatDate, formatTime } from "@/lib/kene/format";
    10→import { MOMO_OPERATORS } from "@/lib/kene/rfm";
    11→import { Sheet, SheetContent, SheetHeader, SheetTitle } from "@/components/ui/sheet";
    12→import { Skeleton } from "@/components/ui/skeleton";
    13→import { useKene } from "@/store/kene";
    14→import { KenteWeaveCard } from "@/components/kene/weave/KenteWeaveCard";
    15→import { categoryThread } from "@/components/kene/weave/threads";
    16→import type { ApiOrder, ApiPayment, ApiProduct, ApiWallet } from "./types";
    17→import { SHOP_CATEGORIES } from "./types";
    18→import { EmptyBlock, Stars } from "./bits";
    19→
    20→type PayMethod = "wave" | "orange" | "wallet";
    21→
    22→/* Coupon validé au checkout (aperçu local — la consommation a lieu à la commande) */
    23→interface AppliedPromo {
    24→  code: string;
    25→  label: string | null;
    26→  discount: number;
    27→  subtotal: number; // panier au moment de la validation (invalidation si changement)
    28→}
    29→
    30→export function ShopScreen() {
    31→  const user = useKene((s) => s.user)!;
    32→  const cart = useKene((s) => s.cart);
    33→  const addToCart = useKene((s) => s.addToCart);
    34→  const setCartQty = useKene((s) => s.setCartQty);
    35→  const clearCart = useKene((s) => s.clearCart);
    36→
    37→  const [products, setProducts] = useState<ApiProduct[] | null>(null);
    38→  const [cat, setCat] = useState("");
    39→  const [q, setQ] = useState("");
    40→  const [detail, setDetail] = useState<ApiProduct | null>(null);
    41→  const [qty, setQty] = useState(1);
    42→  const [checkout, setCheckout] = useState(false);
    43→  const [wallet, setWallet] = useState<ApiWallet | null>(null);
    44→  const [payState, setPayState] = useState<{ phase: "processing" | "success"; method: PayMethod; amount: number } | null>(null);
    45→  const [paying, setPaying] = useState(false);
    46→  const [promoInput, setPromoInput] = useState("");
    47→  const [promoChecking, setPromoChecking] = useState(false);
    48→  const [promo, setPromo] = useState<AppliedPromo | null>(null);
    49→  const inputRef = useRef<HTMLInputElement>(null);
    50→
    51→  /* « Mes commandes » — historique consultable (chargé au montage, rafraîchi
    52→   * après chaque commande réussie : la cliente suit ses données enregistrées). */
    53→  const [view, setView] = useState<"catalogue" | "commandes">("catalogue");
    54→  const [orders, setOrders] = useState<ApiOrder[] | null>(null);
    55→  const refreshOrders = useCallback(() => {
    56→    apiGet<{ orders: ApiOrder[] }>(`/api/orders?userId=${user.id}`)
    57→      .then((r) => setOrders(r.orders ?? []))
    58→      .catch(() => setOrders((prev) => prev ?? []));
    59→  }, [user.id]);
    60→
    61→  const subtotal = cart.reduce((s, l) => s + l.price * l.qty, 0);
    62→  /* Coupon invalidé si le panier a changé depuis la validation (remise recalculée
    63→   * sur un autre sous-total) — la cliente re-valide, jamais de surprise serveur. */
    64→  const livePromo = promo && promo.subtotal === subtotal ? promo : null;
    65→  const discount = livePromo?.discount ?? 0;
    66→  const total = subtotal - discount;
    67→  /* taux de cashback réellement appliqué (wallet de la cliente, sinon défaut) */
    68→  const cashbackRate = wallet?.cashbackRate ?? CASHBACK_RATE;
    69→
    70→  useEffect(() => {
    71→    apiGet<{ products: ApiProduct[] }>("/api/shop/products")
    72→      .then((r) => setProducts(r.products ?? []))
    73→      .catch(() => setProducts([]));
    74→    apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).then((r) => setWallet(r.wallet)).catch(() => {});
    75→    refreshOrders();
    76→  }, [user.id, refreshOrders]);
    77→
    78→  const filtered = useMemo(() => {
    79→    const nq = q.trim().toLowerCase();
    80→    return (products ?? []).filter((p) => (!cat || p.category === cat) && (!nq || `${p.name} ${p.botanicals} ${p.description}`.toLowerCase().includes(nq)));
    81→  }, [products, cat, q]);
    82→
    83→  /* le fil de la catégorie — la navette l'illumine dans la bande tissée */
    84→  const weaveCaption =
    85→    products === null
    86→      ? "La navette monte le métier…"
    87→      : `${filtered.length} soin${filtered.length > 1 ? "s" : ""}${
    88→          cat ? ` · ${SHOP_CATEGORIES.find((c) => c.id === cat)?.label.toLowerCase() ?? cat}` : " au catalogue"
    89→        }`;
    90→
    91→  /* Applique un code promo : aperçu de remise sans consommer le coupon
    92→   * (la consommation a lieu à la commande — toutes les gardes côté serveur). */
    93→  async function applyPromo() {
    94→    const code = promoInput.trim().toUpperCase();
    95→    if (!code || promoChecking) return;
    96→    setPromoChecking(true);
    97→    try {
    98→      const r = await apiPost<{ ok: true; coupon: { code: string; label: string | null }; discount: number; total: number }>(
    99→        "/api/coupons/validate",
   100→        { code, userId: user.id, subtotal }
   101→      );
   102→      setPromo({ code: r.coupon.code, label: r.coupon.label, discount: r.discount, subtotal });
   103→      setPromoInput("");
   104→      toast.success(`Code ${r.coupon.code} appliqué`, { description: `Remise de ${xof(r.discount)} déduite du total.` });
   105→    } catch (e) {
   106→      setPromo(null);
   107→      toast.error(e instanceof Error ? e.message : "Code promo invalide");
   108→    } finally {
   109→      setPromoChecking(false);
   110→    }
   111→  }
   112→
   113→  async function pay(method: PayMethod) {
   114→    setPaying(true);
   115→    try {
   116→      const items = cart.map((l) => ({ productId: l.productId, qty: l.qty }));
   117→      const r = await apiPost<{ order: ApiOrder; payment: ApiPayment | null; paid: boolean }>("/api/orders", {
   118→        userId: user.id,
   119→        items,
   120→        paymentMethod: method,
   121→        ...(livePromo ? { couponCode: livePromo.code } : {}),
   122→      });
   123→      const amount = r.order.total;
   124→      if (method !== "wallet" && r.payment) {
   125→        setCheckout(false);
   126→        setPayState({ phase: "processing", method, amount });
   127→        await new Promise((res) => setTimeout(res, 3000));
   128→        const c = await apiPost<{ payment: ApiPayment; order?: ApiOrder; wallet?: ApiWallet }>("/api/payments/confirm", { paymentId: r.payment.id });
   129→        if (c.wallet) setWallet(c.wallet);
   130→      } else {
   131→        setCheckout(false);
   132→        setPayState({ phase: "processing", method: "wallet", amount });
   133→        await new Promise((res) => setTimeout(res, 1200));
   134→      }
   135→      setPayState({ phase: "success", method, amount });
   136→      clearCart();
   137→      setPromo(null); // le coupon est consommé : remise à zéro pour la prochaine commande
   138→      void refreshOrders(); // la nouvelle commande apparaît dans « Mes commandes »
   139→      const cb = r.order.cashback;
   140→      toast.success(cb > 0 ? `Commande confirmée — cashback ${xof(cb)} crédité sur ton wallet` : "Commande confirmée et payée");
   141→      if (method === "wallet") {
   142→        const w = await apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).catch(() => null);
   143→        if (w) setWallet(w.wallet);
   144→      }
   145→    } catch (e) {
   146→      setPayState(null);
   147→      toast.error(e instanceof Error ? e.message : "Paiement impossible");
   148→    } finally {
   149→      setPaying(false);
   150→    }
   151→  }
   152→
   153→  const op = (m: PayMethod) => MOMO_OPERATORS.find((o) => o.code === m);
   154→
   155→  return (
   156→    <div className="pt-4 pb-2">
   157→      <header className="flex items-center justify-between">
   158→        <h1 className="font-heading font-black text-xl">Boutique Kènè</h1>
   159→        <span className="rounded-full bg-primary/10 text-primary text-[10px] font-bold px-2.5 py-1 uppercase tracking-wide">Cashback {Math.round(cashbackRate * 100)} %</span>
   160→      </header>
   161→
   162→      {/* Bascule Catalogue ↔ Mes commandes — la cliente consulte ses données */}
   163→      <div className="mt-3 grid grid-cols-2 gap-1 rounded-2xl border border-border bg-card p-1" role="tablist" aria-label="Vues boutique">
   164→        <button
   165→          role="tab"
   166→          aria-selected={view === "catalogue"}
   167→          onClick={() => setView("catalogue")}
   168→          className={`h-10 rounded-xl flex items-center justify-center gap-1.5 text-xs font-bold transition-all active:scale-[0.98] focus-visible:outline-2 focus-visible:outline-primary ${view === "catalogue" ? "bg-primary text-primary-foreground shadow-sm" : "text-muted-foreground hover:text-foreground"}`}
   169→        >
   170→          <ShoppingBag size={14} aria-hidden="true" /> Catalogue
   171→        </button>
   172→        <button
   173→          role="tab"
   174→          aria-selected={view === "commandes"}
   175→          onClick={() => setView("commandes")}
   176→          className={`h-10 rounded-xl flex items-center justify-center gap-1.5 text-xs font-bold transition-all active:scale-[0.98] focus-visible:outline-2 focus-visible:outline-primary ${view === "commandes" ? "bg-primary text-primary-foreground shadow-sm" : "text-muted-foreground hover:text-foreground"}`}
   177→        >
   178→          <History size={14} aria-hidden="true" /> Mes commandes{orders !== null && orders.length > 0 ? ` (${orders.length})` : ""}
   179→        </button>
   180→      </div>
   181→
   182→      {view === "commandes" ? (
   183→        <OrdersView orders={orders} onRefresh={refreshOrders} onShop={() => setView("catalogue")} />
   184→      ) : (
   185→        <>
   186→      {/* Le Fil de Kente — hero tissé, le fil de la catégorie s'illumine */}
   187→      <div className="mt-4">
   188→        <KenteWeaveCard highlightIndex={cat ? categoryThread(cat) : -1} caption={weaveCaption} />
   189→      </div>
   190→
   191→      {/* Recherche */}
   192→      <div className="mt-3 relative">
   193→        <Search size={16} className="absolute left-3.5 top-1/2 -translate-y-1/2 text-muted-foreground" aria-hidden="true" />
   194→        <input
   195→          ref={inputRef}
   196→          value={q}
   197→          onChange={(e) => setQ(e.target.value)}
   198→          placeholder="Rechercher un soin, un botanique…"
   199→          aria-label="Rechercher un produit"
   200→          className="h-12 w-full rounded-2xl border border-border bg-card pl-10 pr-4 text-sm focus-visible:outline-2 focus-visible:outline-primary"
   201→        />
   202→      </div>
   203→
   204→      {/* Catégories */}
   205→      <div className="flex gap-2 overflow-x-auto py-3 scrollbar-thin -mx-1 px-1" role="tablist" aria-label="Catégories">
   206→        {SHOP_CATEGORIES.map((c) => (
   207→          <button
   208→            key={c.id}
   209→            role="tab"
   210→            aria-selected={cat === c.id}
   211→            onClick={() => setCat(c.id)}
   212→            className={`shrink-0 rounded-full px-3.5 py-2 text-xs font-semibold transition-all active:scale-95 focus-visible:outline-2 focus-visible:outline-primary ${cat === c.id ? "bg-primary text-primary-foreground shadow" : "border border-border bg-card text-foreground/80"}`}
   213→          >
   214→            {c.label}
   215→          </button>
   216→        ))}
   217→      </div>
   218→
   219→      {/* Grille produits */}
   220→      {products === null ? (
   221→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   222→          {[0, 1, 2, 3].map((i) => (
   223→            <div key={i} className="space-y-2">
   224→              <Skeleton className="aspect-square rounded-2xl" />
   225→              <Skeleton className="h-3.5 w-4/5" />
   226→              <Skeleton className="h-3 w-2/3" />
   227→            </div>
   228→          ))}
   229→        </div>
   230→      ) : filtered.length === 0 ? (
   231→        <EmptyBlock icon={<Search size={22} />} title="Aucun produit trouvé" text="Essaie un autre mot-clé ou une autre catégorie." />
   232→      ) : (
   233→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   234→          {filtered.map((p) => (
   235→            <button
   236→              key={p.id}
   237→              onClick={() => { setDetail(p); setQty(1); }}
   238→              className="text-left rounded-2xl border border-border bg-card overflow-hidden shadow-sm active:scale-[0.98] transition-transform hover:border-primary/40 focus-visible:outline-2 focus-visible:outline-primary"
   239→              aria-label={`${p.name}, ${xof(p.price)}`}
   240→            >
   241→              <img src={p.image} alt={p.name} loading="lazy" className="aspect-square w-full object-cover" />
   242→              <div className="p-2.5">
   243→                <p className="text-[13px] font-semibold leading-tight line-clamp-2 min-h-9">{p.name}</p>
   244→                <p className="text-[10px] text-terre mt-1 truncate">{p.botanicals}</p>
   245→                <div className="flex items-center justify-between mt-1.5">
   246→                  <span className="font-mono text-[13px] font-bold">{xof(p.price)}</span>
   247→                  <Stars rating={p.rating} size={9} />
   248→                </div>
   249→              </div>
   250→            </button>
   251→          ))}
   252→        </div>
   253→      )}
   254→        </>
   255→      )}
   256→
   257→      {/* Barre panier sticky au-dessus de la nav */}
   258→      <AnimatePresence>
   259→        {cart.length > 0 && !payState && (
   260→          <motion.div initial={{ y: 80, opacity: 0 }} animate={{ y: 0, opacity: 1 }} exit={{ y: 80, opacity: 0 }} className="sticky bottom-[84px] z-20 mt-4">
   261→            <button
   262→              onClick={() => setCheckout(true)}
   263→              className="w-full h-14 rounded-2xl bg-gradient-to-r from-[#C8951E] to-[#A0522D] text-[#FFF9EC] shadow-xl flex items-center justify-between px-4 active:scale-[0.99] transition-transform focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary"
   264→              aria-label={`Panier ${cart.length} articles, total ${xof(subtotal)} — commander`}
   265→            >
   266→              <span className="flex items-center gap-2 text-sm font-semibold">
   267→                <span className="relative">
   268→                  <ShoppingBag size={20} />
   269→                  <span className="absolute -top-1.5 -right-2 h-4 min-w-4 px-0.5 rounded-full bg-[#FFF9EC] text-[#A0520F] text-[9px] font-black grid place-items-center">{cart.reduce((s, l) => s + l.qty, 0)}</span>
   270→                </span>
   271→                {cart.length} article{cart.length > 1 ? "s" : ""}
   272→              </span>
   273→              <span className="font-mono font-black text-base">{xof(subtotal)}</span>
   274→            </button>
   275→          </motion.div>
   276→        )}
   277→      </AnimatePresence>
   278→
   279→      {/* Sheet fiche produit */}
   280→      <Sheet open={!!detail} onOpenChange={(o) => !o && setDetail(null)}>
   281→        <SheetContent side="bottom" className="max-h-[88vh] rounded-t-3xl px-0 overflow-y-auto max-w-[560px] mx-auto">
   282→          {detail && (
   283→            <>
   284→              <div className="mx-auto sticky top-0 z-10 bg-card/95 backdrop-blur pt-2 pb-1">
   285→                <div className="h-1 w-10 rounded-full bg-muted mx-auto" aria-hidden="true" />
   286→              </div>
   287→              <img src={detail.image} alt={detail.name} className="aspect-square w-full object-cover px-0" />
   288→              <SheetHeader className="px-5 pt-4 text-left">
   289→                <SheetTitle className="font-heading font-black text-lg leading-tight">{detail.name}</SheetTitle>
   290→                <div className="flex items-center gap-2">
   291→                  <Stars rating={detail.rating} />
   292→                  <span className="text-[11px] text-muted-foreground">{detail.rating.toFixed(1)} ({detail.reviewCount} avis)</span>
   293→                </div>
   294→              </SheetHeader>
   295→              <div className="px-5 pb-6 space-y-4">
   296→                <p className="text-xs text-muted-foreground leading-relaxed">{detail.description}</p>
   297→                <div className="flex flex-wrap gap-1.5">
   298→                  {detail.botanicals.split(/[,;]/).map((b, i) => (
   299→                    <span key={i} className="rounded-full bg-karite border border-[#C8951E]/30 px-2.5 py-1 text-[10px] font-semibold text-terre">{b.trim()}</span>
   300→                  ))}
   301→                </div>
   302→                <div className="flex items-center justify-between">
   303→                  <div>
   304→                    <p className="font-mono text-xl font-black text-primary">{xof(detail.price)}</p>
   305→                    {detail.compareAt && detail.compareAt > detail.price && <p className="text-[11px] text-muted-foreground line-through">{xof(detail.compareAt)}</p>}
   306→                  </div>
   307→                  <div className="flex items-center gap-3 rounded-full border border-border px-2 py-1.5">
   308→                    <button onClick={() => setQty((n) => Math.max(1, n - 1))} className="h-9 w-9 grid place-items-center rounded-full hover:bg-muted active:scale-90 transition-all" aria-label="Diminuer la quantité"><Minus size={16} /></button>
   309→                    <span className="font-mono font-bold w-5 text-center" aria-live="polite">{qty}</span>
   310→                    <button onClick={() => setQty((n) => Math.min(detail.stock, n + 1))} className="h-9 w-9 grid place-items-center rounded-full hover:bg-muted active:scale-90 transition-all" aria-label="Augmenter la quantité"><Plus size={16} /></button>
   311→                  </div>
   312→                </div>
   313→                <button
   314→                  onClick={() => { addToCart({ productId: detail.id, name: detail.name, price: detail.price, qty, image: detail.image }); toast.success(`${qty} × ${detail.name} ajouté au panier`); setDetail(null); }}
   315→                  className="h-12 w-full rounded-xl bg-primary text-primary-foreground font-semibold text-sm shadow-md active:scale-[0.98] transition-all flex items-center justify-center gap-2 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary"
   316→                >
   317→                  <Plus size={17} /> Ajouter au panier — {xof(detail.price * qty)}
   318→                </button>
   319→              </div>
   320→            </>
   321→          )}
   322→        </SheetContent>
   323→      </Sheet>
   324→
   325→      {/* Sheet checkout */}
   326→      <Sheet open={checkout} onOpenChange={(o) => !o && setCheckout(false)}>
   327→        <SheetContent side="bottom" className="max-h-[90vh] rounded-t-3xl overflow-y-auto max-w-[560px] mx-auto">
   328→          <SheetHeader className="text-left">
   329→            <SheetTitle className="font-heading font-black">Ma commande</SheetTitle>
   330→          </SheetHeader>
   331→          <div className="px-4 pb-6 space-y-4">
   332→            <div className="space-y-2">
   333→              {cart.map((l) => (
   334→                <div key={l.productId} className="flex items-center gap-3 rounded-xl border border-border bg-card p-2">
   335→                  <img src={l.image} alt={l.name} className="h-12 w-12 rounded-lg object-cover" />
   336→                  <div className="min-w-0 flex-1">
   337→                    <p className="text-xs font-semibold leading-tight line-clamp-2">{l.name}</p>
   338→                    <p className="font-mono text-[11px] text-muted-foreground mt-0.5">{xof(l.price)} × {l.qty}</p>
   339→                  </div>
   340→                  <div className="flex items-center gap-1.5">
   341→                    <button onClick={() => setCartQty(l.productId, l.qty - 1)} className="h-8 w-8 grid place-items-center rounded-full border border-border active:scale-90 transition-transform" aria-label={`Retirer un ${l.name}`}><Minus size={13} /></button>
   342→                    <span className="font-mono text-xs font-bold w-4 text-center">{l.qty}</span>
   343→                    <button onClick={() => setCartQty(l.productId, l.qty + 1)} className="h-8 w-8 grid place-items-center rounded-full border border-border active:scale-90 transition-transform" aria-label={`Ajouter un ${l.name}`}><Plus size={13} /></button>
   344→                    <button onClick={() => setCartQty(l.productId, 0)} className="h-8 w-8 grid place-items-center rounded-full text-destructive hover:bg-destructive/10 active:scale-90 transition-all" aria-label={`Supprimer ${l.name}`}><Trash2 size={14} /></button>
   345→                  </div>
   346→                </div>
   347→              ))}
   348→            </div>
   349→
   350→            {/* Code promo — l'aperçu vit côté serveur (lib/kene/coupons) */}
   351→            {livePromo ? (
   352→              <div className="rounded-2xl border border-[#3F7D3F]/30 bg-[#3F7D3F]/[0.06] p-3.5 flex items-center gap-3" aria-live="polite">
   353→                <span className="grid place-items-center h-9 w-9 rounded-full bg-[#3F7D3F]/15 text-[#3F7D3F] shrink-0">
   354→                  <Tag size={16} aria-hidden="true" />
   355→                </span>
   356→                <div className="min-w-0 flex-1">
   357→                  <p className="text-xs font-bold font-mono tracking-wide">{livePromo.code}</p>
   358→                  <p className="text-[11px] text-[#3F7D3F] leading-tight">
   359→                    {livePromo.label ? `${livePromo.label} · ` : ""}−{xof(livePromo.discount)}
   360→                  </p>
   361→                </div>
   362→                <button
   363→                  onClick={() => setPromo(null)}
   364→                  aria-label={`Retirer le code ${livePromo.code}`}
   365→                  className="h-8 w-8 grid place-items-center rounded-full text-muted-foreground hover:bg-muted active:scale-90 transition-all"
   366→                >
   367→                  <X size={15} />
   368→                </button>
   369→              </div>
   370→            ) : (
   371→              <div className="flex gap-2">
   372→                <div className="relative flex-1">
   373→                  <Tag size={15} className="absolute left-3.5 top-1/2 -translate-y-1/2 text-muted-foreground" aria-hidden="true" />
   374→                  <input
   375→                    value={promoInput}
   376→                    onChange={(e) => setPromoInput(e.target.value.toUpperCase())}
