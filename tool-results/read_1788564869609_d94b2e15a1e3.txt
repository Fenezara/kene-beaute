   100→   100→      window.removeEventListener("offline", update);

   168→   168→      setNavDir(dx < 0 ? 1 : -1);
   169→   169→      haptic(HAPTIC.tap);
   170→   170→      setClientTab(next);
   171→   171→    }
   172→   172→  };

   183→   183→  const onScanZone = useCallback(

   189→   189→  );
   190→   190→
