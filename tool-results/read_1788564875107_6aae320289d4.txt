     6→
     7→import { useEffect, useMemo, useState } from "react";
     8→import { motion } from "framer-motion";
     9→import {

    11→  Brush,
    12→  CalendarClock,
    13→  CheckCircle2,

    15→  CircleDot,

    25→} from "lucide-react";
    26→import { apiGet } from "@/lib/kene/api";
    27→import { formatDate, formatTime, scoreColor, readableTextColor, xof, CASHBACK_RATE } from "@/lib/kene/format";
    28→import { nextClientStep } from "@/lib/kene/followups";
    29→import { channelLabel, humanWhen } from "@/lib/kene/reminders";
    30→import { BODY_ZONES, type BodyZone } from "@/lib/kene/types";
    31→import { NeaOnnimIcon, SankofaIcon } from "@/components/kene/icons";
    32→import { InstallBanner } from "@/components/kene/pwa/InstallBanner";
    33→import { RitualJourney } from "@/components/kene/route/RitualJourney";
    34→import { Skeleton } from "@/components/ui/skeleton";
    35→import { useKene } from "@/store/kene";
    36→import { VoiceNarration } from "./VoiceNarration";
    37→import type { ApiAppointment, ApiDiagnosis, ApiProduct, ApiReminderFeed, ApiWallet } from "./types";

    39→import { ScoreGauge, ScrollFadeRow, SectionTitle, Stars, WalletPill } from "./bits";
    40→
    41→interface HomeData {
    42→  diagnoses: ApiDiagnosis[];
    43→  appointments: ApiAppointment[];
    44→  wallet: ApiWallet | null;
    45→  products: ApiProduct[];
    46→  reminders: ApiReminderFeed | null;
    47→}
    48→

    50→const ZONE_ICON: Record<BodyZone, React.ComponentType<{ className?: string }>> = {
    51→  visage: ScanFace,
    52→  dos: PersonStanding,
    53→  cuir_chevelu: Waves,
    54→  mains: Hand,
    55→  barbe: Brush,
    56→  naevi: CircleDot,
    57→};
    58→
    59→/** Libellé court des stories (2 lignes autorisées, façon Instagram) */
    60→const STORY_LABEL: Record<BodyZone, string> = {
    61→  visage: "Visage",
    62→  dos: "Dos",
    63→  cuir_chevelu: "Cuir chevelu",
    64→  mains: "Mains",
    65→  barbe: "Barbe",
    66→  naevi: "Grains de beauté",
    67→};
    68→
    69→export function HomeScreen({
    70→  onScanZone,
    71→  refreshKey = 0,
    72→  onRefreshed,
    73→}: {
    74→  onScanZone: (z: BodyZone) => void;
    75→  /** Incrémenté par le pull-to-refresh du shell — déclenche un rechargement */
    76→  refreshKey?: number;
    77→  /** Appelé à la fin du chargement (le shell ferme l'indicateur de tirage) */
    78→  onRefreshed?: () => void;
    79→}) {
    80→  const user = useKene((s) => s.user)!;
    81→  const setClientTab = useKene((s) => s.setClientTab);
    82→  const [data, setData] = useState<HomeData | null>(null);
    83→  const [err, setErr] = useState<string | null>(null);
    84→  const [ritualOpen, setRitualOpen] = useState(false);
    85→
    86→  useEffect(() => {
    87→    let alive = true;
    88→    (async () => {
    89→      try {
    90→        const [d, a, w, p, r] = await Promise.all([
    91→          apiGet<{ diagnoses: ApiDiagnosis[] }>(`/api/diagnoses?userId=${user.id}`),
    92→          apiGet<{ appointments: ApiAppointment[] }>(`/api/appointments?userId=${user.id}`),
    93→          apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).catch(() => null),
    94→          apiGet<{ products: ApiProduct[] }>("/api/shop/products"),
    95→          apiGet<ApiReminderFeed>(`/api/notifications?userId=${user.id}`).catch(() => null),
    96→        ]);
    97→        if (alive) {
    98→          setData({
    99→            diagnoses: d.diagnoses ?? [],
   100→            appointments: a.appointments ?? [],
   101→            wallet: w?.wallet ?? null,
   102→            products: p.products ?? [],
   103→            reminders: r ?? null,
   104→          });
   105→          if (alive) onRefreshed?.();
   106→        }
   107→      } catch (e) {
   108→        if (alive) {
   109→          setErr(e instanceof Error ? e.message : "Chargement impossible");
   110→          onRefreshed?.();
   111→        }
   112→      }
   113→    })();
   114→    return () => {
   115→      alive = false;
   116→    };
   117→  }, [user.id, refreshKey, onRefreshed]);
   118→
   119→  // Score multi-zones pondéré (PRD §8.8) — dernier diagnostic par zone
   120→  const multi = useMemo(() => {
   121→    const byZone = new Map<BodyZone, ApiDiagnosis>();
   122→    [...(data?.diagnoses ?? [])]
   123→      .filter((d) => d.status === "done")
   124→      .sort((x, y) => +new Date(y.createdAt) - +new Date(x.createdAt))
   125→      .forEach((d) => {
   126→        if (!byZone.has(d.zone)) byZone.set(d.zone, d);
   127→      });
   128→    const covered = BODY_ZONES.filter((z) => byZone.has(z.id));
   129→    const weightSum = covered.reduce((s, z) => s + z.weight, 0);
   130→    const score =
   131→      covered.length && weightSum > 0
   132→        ? Math.round(covered.reduce((s, z) => s + (byZone.get(z.id)?.scoreGlobal ?? 0) * z.weight, 0) / weightSum)
   133→        : null;
   134→    const missing = BODY_ZONES.filter((z) => !byZone.has(z.id));
   135→    const last = covered.length ? byZone.get(covered[0].id) : null;
   136→    return { byZone, covered, score, missing, last };
   137→  }, [data?.diagnoses]);
   138→
   139→  const nextAppt = useMemo(() => {
   140→    const now = Date.now();
   141→    return (
   142→      [...(data?.appointments ?? [])]
   143→        .filter((a) => new Date(a.startAt).getTime() >= now && a.status !== "cancelled")
   144→        .sort((x, y) => +new Date(x.startAt) - +new Date(y.startAt))[0] ?? null
   145→    );
   146→  }, [data?.appointments]);
   147→
   148→  const reco = useMemo(() => {
   149→    const all = data?.products ?? [];
   150→    const byCat = (cats: string[]) => all.filter((p) => cats.includes(p.category));
   151→    const pick =
   152→      user.skinType === "grasse"
   153→        ? byCat(["serum", "savon"])
   154→        : user.skinType === "seche"
   155→          ? byCat(["creme", "huile"])
   156→          : byCat(["serum", "creme"]);
   157→    return (pick.length >= 3 ? pick : [...pick, ...all.filter((p) => !pick.includes(p))]).slice(0, 3);
   158→  }, [data?.products, user.skinType]);
   159→
   160→  // Route de l'Or : dernier diagnostic parsable → rituel tissable depuis l'accueil
   161→  const lastResult = useMemo(() => (multi.last ? parseDiagnosis(multi.last.resultJson) : null), [multi.last]);
   162→
   163→  // Le Fil du Retour : prochaine étape dérivée de l'activité (contrôle protocole / soin de suite)
   164→  const nextStep = useMemo(
   165→    () => (data ? nextClientStep(new Date(), data.diagnoses, data.appointments) : null),
   166→    [data],
   167→  );
   168→
   169→  const first = user.name.split(" ")[0];
   170→
   171→  return (
   172→    <div className="flex flex-col gap-6 pt-1">
   173→      {/* ───── Salutation ───── */}
   174→      <header className="flex items-end justify-between gap-3">
   175→        <div className="min-w-0">
   176→          <p className="text-[11px] uppercase tracking-[0.16em] text-primary font-semibold">Bonjour</p>
   177→          <h2 className="font-black text-[26px] leading-tight tracking-tight truncate"><span className="kente-text">{first}</span> ✨</h2>
   178→          <p className="flex items-center gap-1 text-[11px] text-muted-foreground mt-0.5">
   179→            <MapPin size={11} /> {user.city || "Abidjan"} · {user.fitzpatrick ? `Fitzpatrick ${user.fitzpatrick}` : "Phototype à définir"}
   180→          </p>
   181→        </div>
   182→        {data?.wallet && <WalletPill balance={data.wallet.balance} onClick={() => setClientTab("profil")} />}
   183→      </header>
   184→
   185→      {/* ───── Bannière d'installation PWA (auto-masquée : installée / fermée / standalone) ───── */}
   186→      <InstallBanner />
   187→
   188→      {/* ───── Stories : scan rapide + zones avec score ───── */}
   189→      <section aria-label="Scan rapide par zone" className="-mx-3 sm:-mx-5 px-3 sm:px-5">
   190→        <ScrollFadeRow label="Stories des zones — fais défiler horizontalement" className="flex gap-3.5 overflow-x-auto no-scrollbar py-1.5 pr-2">
   191→          {/* Story Scanner */}
   192→          <button
   193→            onClick={() => setClientTab("diagnostic")}
   194→            aria-label="Scanner ma peau — nouveau diagnostic"
   195→            className="shrink-0 w-[68px] flex flex-col items-center gap-1.5 focus-visible:outline-2 focus-visible:-outline-offset-2 focus-visible:outline-primary rounded-xl"
   196→          >
   197→            <span className="grid place-items-center h-[66px] w-[66px] rounded-full p-[3px] bg-gradient-to-br from-[#C8951E] via-[#A0522D] to-[#8B1A3B] shadow-md active:scale-95 transition-transform">
   198→              <span className="grid place-items-center h-full w-full rounded-full bg-background">
   199→                <NeaOnnimIcon size={26} className="text-primary" />
   200→              </span>
   201→            </span>
   202→            <span className="text-[10px] font-bold text-primary">Scanner</span>
   203→          </button>
   204→
   205→          {/* Stories zones */}
   206→          {BODY_ZONES.map((z) => {
   207→            const d = multi.byZone.get(z.id);
   208→            const covered = !!d;
   209→            const Icon = ZONE_ICON[z.id];
   210→            const color = covered ? scoreColor(d!.scoreGlobal) : "var(--border)";
   211→            return (
   212→              <button
   213→                key={z.id}
   214→                onClick={() => onScanZone(z.id)}
   215→                aria-label={`${covered ? `Re-scanner ${z.label} — dernier score ${d!.scoreGlobal}` : `Scanner ${z.label} pour compléter ton score`} · pondération ${Math.round(z.weight * 100)} %`}
   216→                className="shrink-0 w-[68px] flex flex-col items-center gap-1.5 focus-visible:outline-2 focus-visible:-outline-offset-2 focus-visible:outline-primary rounded-xl"
   217→              >
   218→                <span
   219→                  className="relative grid place-items-center h-[66px] w-[66px] rounded-full p-[3px] active:scale-95 transition-transform shadow-md"
   220→                  style={
   221→                    covered
   222→                      ? { background: `conic-gradient(from 210deg, ${color} 0%, ${color} 45%, #C8951E 68%, ${color} 100%)` }
   223→                      : undefined
   224→                  }
   225→                >
   226→                  <span
   227→                    className={`grid place-items-center h-full w-full rounded-full bg-background ${covered ? "" : "border-[2.5px] border-dashed border-border"}`}
   228→                  >
   229→                    <Icon className={covered ? "text-foreground/80" : "text-muted-foreground"} />
   230→                  </span>
   231→                  {!covered && (
   232→                    <span className="absolute -bottom-0.5 -right-0.5 grid place-items-center h-6 w-6 rounded-full bg-primary text-primary-foreground border-2 border-background">
   233→                      <Plus size={13} strokeWidth={2.5} />
   234→                    </span>
   235→                  )}
   236→                  {covered && (
   237→                    <span
   238→                      className="absolute -bottom-1 left-1/2 -translate-x-1/2 rounded-full px-1.5 py-px font-mono text-[10px] font-bold tabular-nums shadow-sm"
   239→                      style={{ backgroundColor: color, color: readableTextColor(color) }}
   240→                    >
   241→                      {d!.scoreGlobal}
   242→                    </span>
   243→                  )}
   244→                </span>
   245→                <span className={`text-[11px] w-full text-center leading-tight line-clamp-2 break-words min-h-[26px] ${covered ? "font-semibold" : "text-muted-foreground font-medium"}`}>{STORY_LABEL[z.id]}</span>
   246→              </button>
   247→            );
   248→          })}
   249→        </ScrollFadeRow>
   250→      </section>
   251→
   252→      {err && <p className="rounded-xl bg-destructive/10 text-destructive text-xs p-3">{err}</p>}
   253→
   254→      {/* ───── Carte score multi-zones (avec lecture vocale) ───── */}
   255→      <section aria-labelledby="sc-t" className="rounded-[26px] border border-border/70 overflow-hidden shadow-[0_18px_44px_-18px_rgba(28,17,9,0.28)] bg-gradient-to-br from-card via-card to-primary/[0.08]">
   256→        <div className="kente-band h-[3px] w-full" aria-hidden="true" />
   257→        <div id="sc-t" className="p-6">
   258→          {!data ? (
   259→            <div className="flex items-center gap-4">
   260→              <Skeleton className="h-[130px] w-[130px] rounded-full" />
   261→              <div className="flex-1 space-y-2">
   262→                <Skeleton className="h-4 w-3/4" />
   263→                <Skeleton className="h-3 w-full" />
   264→                <Skeleton className="h-3 w-2/3" />
   265→              </div>
   266→            </div>
   267→          ) : multi.score !== null ? (
   268→            <div className="flex flex-wrap items-center gap-4">
   269→              <ScoreGauge score={multi.score} label="Multi-zones" />
   270→              <div className="min-w-0 flex-1">
   271→                <h3 className="font-heading font-bold text-base">Santé de ta peau</h3>
   272→                <p className="text-xs text-muted-foreground mt-1 leading-relaxed">
   273→                  {multi.covered.length} zone{multi.covered.length > 1 ? "s" : ""} analysée{multi.covered.length > 1 ? "s" : ""} · pondération PRD
   274→                  {multi.last && <span className="block mt-1">Dernier scan : {formatDate(multi.last.createdAt)}</span>}
   275→                </p>
   276→                <div className="flex flex-wrap gap-1.5 mt-2">
   277→                  {multi.covered.map((z) => (
   278→                    <span key={z.id} className="rounded-full bg-muted px-2 py-0.5 text-[10px] font-medium">
   279→                      {z.label} {multi.byZone.get(z.id)?.scoreGlobal}
   280→                    </span>
   281→                  ))}
   282→                  {multi.missing.map((z) => (
   283→                    <button
   284→                      key={z.id}
   285→                      onClick={() => onScanZone(z.id)}
   286→                      className="rounded-full border border-primary/40 bg-primary/10 px-3 min-h-10 inline-flex items-center text-[11px] font-semibold text-primary active:scale-95 transition-transform focus-visible:outline-2 focus-visible:outline-primary"
   287→                    >
   288→                      + {z.label}
   289→                    </button>
   290→                  ))}
   291→                </div>
   292→                {/* Lecture vocale du dernier diagnostic (accès non-lectrices) */}
   293→                {lastResult && (
   294→                  <div className="mt-5 pt-4 border-t border-border/60">
   295→                    <VoiceNarration result={lastResult} userName={user.name} />
   296→                  </div>
   297→                )}
   298→              </div>
   299→            </div>
   300→          ) : (
   301→            <div className="text-center py-2">
   302→              <NeaOnnimIcon size={40} className="mx-auto text-primary" />
   303→              <p className="font-heading font-bold mt-2">Ton premier diagnostic t&apos;attend</p>
   304→              <p className="text-xs text-muted-foreground mt-1">Analyse IA de 6 zones — commence par le visage.</p>
   305→            </div>
   306→          )}
   307→          {data && multi.missing.length > 0 && (
   308→            <div className="mt-4 pt-4 border-t border-dashed border-border">
   309→              <p className="text-[11px] font-semibold text-muted-foreground mb-2 flex items-center gap-1.5">
   310→                <Sparkles size={12} className="text-primary" /> Zones à scanner pour compléter ton score
   311→              </p>
   312→              <ScrollFadeRow label="Zones manquantes — fais défiler" className="flex gap-2 overflow-x-auto no-scrollbar pb-1 pr-1">
   313→                {multi.missing.map((z) => (
   314→                  <button
   315→                    key={z.id}
   316→                    onClick={() => onScanZone(z.id)}
   317→                    className="shrink-0 rounded-full border border-primary/40 bg-primary/10 px-3 min-h-10 inline-flex items-center text-[11px] font-semibold text-primary active:scale-95 transition-transform focus-visible:outline-2 focus-visible:outline-primary"
   318→                  >
   319→                    + {z.label} · {Math.round(z.weight * 100)} %
   320→                  </button>
   321→                ))}
   322→              </ScrollFadeRow>
   323→            </div>
   324→          )}
   325→        </div>
   326→      </section>
   327→
   328→      {/* ───── CTA Scanner ───── */}
   329→      <motion.button
   330→        initial={{ scale: 0.97, opacity: 0 }}
   331→        animate={{ scale: 1, opacity: 1 }}
   332→        whileTap={{ scale: 0.98 }}
   333→        onClick={() => setClientTab("diagnostic")}
   334→        className="relative h-24 rounded-3xl bg-gradient-to-br from-[#A0522D] via-[#8B1A3B] to-[#6B2416] text-[#FFF9EC] shadow-lg overflow-hidden focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary"
   335→        aria-label="Scanner ma peau maintenant"
   336→      >
   337→        <div aria-hidden="true" className="absolute inset-0 bogolan-dots opacity-30" />
   338→        <div className="relative h-full flex items-center justify-center gap-3 px-4">
   339→          <span className="grid place-items-center h-14 w-14 rounded-2xl bg-[#FFF9EC]/15 backdrop-blur border border-[#FFF9EC]/30">
   340→            <NeaOnnimIcon size={30} />
   341→          </span>
   342→          <span className="text-left">
   343→            <span className="block font-black text-lg tracking-tight">Scanner ma peau</span>
   344→            <span className="block text-[11px] opacity-90">Analyse IA VISIA-like · 6 zones · 30 s</span>
   345→          </span>
   346→          <ChevronRight size={22} className="ml-auto opacity-80" aria-hidden="true" />
   347→        </div>
   348→      </motion.button>
   349→
   350→      {/* ───── Route de l'Or — rituel tissé depuis le dernier scan ───── */}
   351→      {data && lastResult && multi.last && (
   352→        <motion.button
   353→          initial={{ opacity: 0, y: 10 }}
   354→          animate={{ opacity: 1, y: 0 }}
   355→          transition={{ delay: 0.1 }}
   356→          whileTap={{ scale: 0.98 }}
   357→          onClick={() => setRitualOpen(true)}
   358→          className="relative w-full rounded-3xl border-2 border-[#C8951E]/50 bg-card shadow-sm overflow-hidden text-left focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary"
   359→          aria-label="Reprendre la Route de l'Or — tisser ma routine depuis mon dernier diagnostic"
   360→        >
   361→          <div aria-hidden="true" className="h-2 w-full" style={{ backgroundImage: "repeating-linear-gradient(90deg,#8B1A3B 0 12px,#3F7D3F 12px 20px,#C8951E 20px 28px,#E07A2B 28px 36px,#A0522D 36px 46px)" }} />
   362→          <div className="flex items-center gap-3 p-4">
   363→            <span className="grid place-items-center h-12 w-12 rounded-2xl bg-gradient-to-br from-[#C8951E] to-[#A0522D] text-[#FFF9EC] shrink-0 shadow">
   364→              <NeaOnnimIcon size={26} />
   365→            </span>
   366→            <div className="min-w-0 flex-1">
   367→              <p className="font-heading font-bold text-sm">La Route de l&apos;Or</p>
   368→              <p className="text-[11px] text-muted-foreground mt-0.5 leading-snug">
   369→                Ta routine se tisse en 4 stations, selon ton scan du {formatDate(multi.last.createdAt, { day: "numeric", month: "short" })}
   370→              </p>
   371→            </div>
   372→            <ChevronRight size={18} className="text-primary shrink-0" />
   373→          </div>
   374→        </motion.button>
   375→      )}
   376→
   377→      {/* ───── Le Fil du Retour — ta prochaine étape ───── */}
   378→      {nextStep && (
   379→        <motion.section
   380→          initial={{ opacity: 0, y: 10 }}
   381→          animate={{ opacity: 1, y: 0 }}
   382→          transition={{ delay: 0.05 }}
   383→          aria-labelledby="ns-t"
   384→          className="rounded-2xl border border-gold/40 bg-gradient-to-br from-gold/10 via-card to-card p-4 shadow-sm"
   385→        >
   386→          <div className="flex items-center gap-2">
   387→            <span className="grid place-items-center h-9 w-9 rounded-xl bg-gold/15 text-gold shrink-0">
   388→              <SankofaIcon size={19} />
   389→            </span>
   390→            <p id="ns-t" className="font-heading font-bold text-sm">Ta prochaine étape</p>
   391→            <span
   392→              className={
   393→                "ml-auto rounded-full px-2 py-0.5 text-[10px] font-semibold tabular-nums " +
   394→                (nextStep.overdue ? "bg-bissap/15 text-destructive" : "bg-gold/15 text-gold-text")
   395→              }
   396→            >
   397→              {nextStep.overdue ? `En retard de ${-nextStep.days} j` : nextStep.days === 0 ? "Aujourd'hui" : `Dans ${nextStep.days} j`}
   398→            </span>
   399→          </div>
   400→          <p className="text-sm font-semibold mt-2">{nextStep.title}</p>
   401→          <p className="text-[11px] text-muted-foreground mt-0.5 leading-snug">{nextStep.detail}</p>
   402→          <button
   403→            onClick={() => setClientTab(nextStep.ctaTab)}
   404→            className="mt-3 h-11 w-full rounded-xl bg-primary text-primary-foreground text-xs font-bold inline-flex items-center justify-center gap-1.5 shadow active:scale-[0.98] transition-transform focus-visible:outline-2 focus-visible:outline-primary"
   405→          >
   406→            {nextStep.ctaLabel}
   407→            <ChevronRight size={15} aria-hidden="true" />
   408→          </button>
   409→        </motion.section>
   410→      )}
   411→
   412→      {/* ───── Prochain RDV ───── */}
   413→      <section aria-labelledby="rdv-t">
   414→        <SectionTitle icon={<SankofaIcon size={17} />}>
   415→          <span id="rdv-t">Prochain rendez-vous</span>
   416→        </SectionTitle>
   417→        {!data ? (
