     1→"use client";
     2→// Kènè Cliente — Profil : identité, profil peau rééditable, wallet complet, consentement, espace pro
     3→import { useCallback, useEffect, useState } from "react";
     4→import { motion } from "framer-motion";
     5→import {
     6→  ArrowDownLeft, ArrowLeft, ArrowUpRight, BadgeCheck, Building2, Check, Download, Languages, Loader2, LogOut, MapPin,
     7→  Pencil, Phone, Plus, ShieldCheck, Smartphone, Sparkles, Wallet as WalletIcon,
     8→} from "lucide-react";
     9→import { toast } from "sonner";
    10→import { apiGet, apiPatch, apiPost } from "@/lib/kene/api";
    11→import { formatDate, xof, CASHBACK_RATE } from "@/lib/kene/format";
    12→import { LANGS, type Lang } from "@/lib/kene/i18n";
    13→import { useT } from "@/lib/kene/use-t";
    14→import { MOMO_OPERATORS } from "@/lib/kene/rfm";
    15→import { Sheet, SheetContent, SheetHeader, SheetTitle } from "@/components/ui/sheet";
    16→import { Skeleton } from "@/components/ui/skeleton";
    17→import { useKene, type SessionUser } from "@/store/kene";
    18→import type { ApiUser, ApiWallet, ApiWalletTx } from "./types";
    19→import { FITZPATRICK_CARDS, SKIN_GOALS, SKIN_TYPES } from "./types";
    20→import { SectionTitle } from "./bits";
    21→import { ParrainageCard } from "./ParrainageCard";
    22→import { useInstallPrompt } from "@/components/kene/pwa/use-install";
    23→
    24→/** SessionUser + goals (string JSON) renvoyé par PATCH profile */
    25→type ClientUser = SessionUser & { goals?: string | null };
    26→
    27→const REASON_LABELS: Record<string, string> = {
    28→  cashback: "Cashback commande",
    29→  referral: "Bonus parrainage",
    30→  topup: "Approvisionnement",
    31→  payment: "Paiement",
    32→  refund: "Remboursement RDV",
    33→};
    34→
    35→export function ProfileScreen() {
    36→  const user = useKene((s) => s.user) as ClientUser;
    37→  const setUser = useKene((s) => s.setUser);
    38→  const setSpace = useKene((s) => s.setSpace);
    39→  const setClientTab = useKene((s) => s.setClientTab);
    40→  const { t, lang, setLang } = useT();
    41→
    42→  const [edit, setEdit] = useState(false);
    43→  const [name, setName] = useState(user.name);
    44→  const [city, setCity] = useState(user.city ?? "");
    45→  const [savingId, setSavingId] = useState(false);
    46→
    47→  const [fitz, setFitz] = useState(user.fitzpatrick ?? "V");
    48→  const [skinType, setSkinType] = useState(user.skinType ?? "mixte");
    49→  const [allergies, setAllergies] = useState(user.allergies ?? "");
    50→  const [goals, setGoals] = useState<string[]>(() => {
    51→    try {
    52→      return (JSON.parse(user.goals ?? "[]") as { id: string }[]).map((g) => g.id);
    53→    } catch {
    54→      return [];
    55→    }
    56→  });
    57→  const [savingSkin, setSavingSkin] = useState(false);
    58→
    59→  const [wallet, setWallet] = useState<ApiWallet | null>(null);
    60→  const [txs, setTxs] = useState<ApiWalletTx[] | null>(null);
    61→  const [topup, setTopup] = useState(false);
    62→  const [amount, setAmount] = useState(5000);
    63→  const [method, setMethod] = useState<"wave" | "orange">("wave");
    64→  const [topupState, setTopupState] = useState<"idle" | "processing" | "done">("idle");
    65→  const [topupBusy, setTopupBusy] = useState(false);
    66→  const [exportBusy, setExportBusy] = useState(false);
    67→
    68→  // Installation PWA — même source d'événement que la bannière d'accueil.
    69→  const { canInstall, promptInstall, isStandalone, isIOS } = useInstallPrompt();
    70→
    71→  const loadWallet = useCallback(() => {
    72→    apiGet<{ wallet: ApiWallet; transactions: ApiWalletTx[] }>(`/api/wallet?userId=${user.id}`)
    73→      .then((r) => { setWallet(r.wallet); setTxs(r.transactions ?? []); })
    74→      .catch(() => setTxs([]));
    75→  }, [user.id]);
    76→
    77→  useEffect(() => {
    78→    loadWallet();
    79→  }, [loadWallet]);
    80→
    81→  /** Portabilité RGPD — télécharge « mes données » en JSON via blob */
    82→  async function downloadMyData() {
    83→    setExportBusy(true);
    84→    try {
    85→      const res = await fetch(`/api/profile/export?userId=${encodeURIComponent(user.id)}`);
    86→      if (!res.ok) throw new Error("Export impossible");
    87→      const blob = await res.blob();
    88→      const cd = res.headers.get("Content-Disposition") ?? "";
    89→      const m = /filename="([^"]+)"/.exec(cd);
    90→      const url = URL.createObjectURL(blob);
    91→      const a = document.createElement("a");
    92→      a.href = url;
    93→      a.download = m?.[1] ?? `kene-mes-donnees-${new Date().toISOString().slice(0, 10)}.json`;
    94→      document.body.appendChild(a);
    95→      a.click();
    96→      a.remove();
    97→      setTimeout(() => URL.revokeObjectURL(url), 4000);
    98→      const kb = Math.max(1, Math.round(blob.size / 1024));
    99→      toast.success(`Mes données téléchargées · ${kb} Ko`);
   100→    } catch (e) {
   101→      toast.error(e instanceof Error ? e.message : "Export impossible");
   102→    } finally {
   103→      setExportBusy(false);
   104→    }
   105→  }
   106→
   107→  /** Installer l'app (prompt natif, ou instructions iOS / navigateur). */
   108→  async function installApp() {
   109→    if (canInstall) {
   110→      const accepted = await promptInstall();
   111→      if (accepted) toast.success("Kènè installée sur ton écran d'accueil 💛");
   112→      return;
   113→    }
   114→    if (isIOS) {
   115→      toast.info("Installer sur iPhone", {
   116→        description: "Bouton Partager ⬆️ puis « Sur l'écran d'accueil ».",
   117→        duration: 8000,
   118→      });
   119→      return;
   120→    }
   121→    toast.info("Installation manuelle", {
   122→      description: "Menu du navigateur → « Installer l'application » ou « Ajouter à l'écran d'accueil ».",
   123→      duration: 8000,
   124→    });
   125→  }
   126→
   127→  async function saveIdentity() {
   128→    setSavingId(true);
   129→    try {
   130→      const r = await apiPatch<{ user: ApiUser }>("/api/auth/profile", { userId: user.id, name: name.trim() || undefined, city: city.trim() || undefined });
   131→      setUser({ ...user, ...r.user } as SessionUser);
   132→      setEdit(false);
   133→      toast.success("Profil mis à jour");
   134→    } catch (e) {
   135→      toast.error(e instanceof Error ? e.message : "Mise à jour impossible");
   136→    } finally {
   137→      setSavingId(false);
   138→    }
   139→  }
   140→
   141→  async function saveSkin() {
   142→    setSavingSkin(true);
   143→    try {
   144→      const r = await apiPatch<{ user: ApiUser }>("/api/auth/profile", {
   145→        userId: user.id,
   146→        skinType,
   147→        fitzpatrick: fitz,
   148→        allergies: allergies.trim() || undefined,
   149→        goals: goals.map((id) => ({ id, label: SKIN_GOALS.find((g) => g.id === id)?.label ?? id })),
   150→      });
   151→      setUser({ ...user, ...r.user } as SessionUser);
   152→      toast.success("Profil peau mis à jour");
   153→    } catch (e) {
   154→      toast.error(e instanceof Error ? e.message : "Mise à jour impossible");
   155→    } finally {
   156→      setSavingSkin(false);
   157→    }
   158→  }
   159→
   160→  /** Langue de l'interface (i18n) — indépendante de la langue de lecture vocale. */
   161→  function selectLang(l: Lang) {
   162→    if (l === lang) return;
   163→    setLang(l);
   164→    const label = LANGS.find((x) => x.id === l)?.label ?? "";
   165→    toast.success(`Interface en ${label.toLowerCase()}`);
   166→  }
   167→
   168→  async function runTopup() {
   169→    setTopupBusy(true);
   170→    setTopupState("processing");
   171→    try {
   172→      const r = await apiPost<{ payment: { id: string } }>("/api/wallet/topup", { userId: user.id, amount, method });
   173→      await new Promise((res) => setTimeout(res, 2600));
   174→      await apiPost("/api/payments/confirm", { paymentId: r.payment.id });
   175→      setTopupState("done");
   176→      toast.success(`${xof(amount)} crédités sur ton wallet`);
   177→      loadWallet();
   178→    } catch (e) {
   179→      setTopupState("idle");
   180→      toast.error(e instanceof Error ? e.message : "Approvisionnement impossible");
   181→    } finally {
   182→      setTopupBusy(false);
   183→    }
   184→  }
   185→
   186→  return (
   187→    <div className="pt-4 pb-2 flex flex-col gap-6">
   188→      <button onClick={() => setClientTab("accueil")} className="self-start inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground focus-visible:outline-2 focus-visible:outline-primary rounded min-h-10 px-1" aria-label={t("profile.back.aria")}>
   189→        <ArrowLeft size={15} /> {t("tab.home")}
   190→      </button>
   191→
   192→      {/* Identité */}
   193→      <section aria-labelledby="me-t" className="rounded-3xl border border-border bg-card shadow-sm overflow-hidden">
   194→        <div className="kente-band h-1.5 w-full" aria-hidden="true" />
   195→        <div id="me-t" className="p-5">
   196→          <div className="flex items-center gap-4">
   197→            <span className="grid place-items-center h-16 w-16 rounded-full bg-gradient-to-br from-[#C8951E] to-[#A0522D] text-[#FFF9EC] font-heading font-black text-2xl shadow">
   198→              {user.name.charAt(0)}
   199→            </span>
   200→            <div className="min-w-0 flex-1">
   201→              <p className="font-heading font-black text-lg leading-tight truncate">{user.name}</p>
   202→              <p className="text-xs text-muted-foreground flex items-center gap-1.5 mt-1 font-mono"><Phone size={12} /> {user.phone}</p>
   203→              <p className="text-xs text-muted-foreground flex items-center gap-1.5 mt-0.5"><MapPin size={12} /> {user.city || "Ville non renseignée"}</p>
   204→            </div>
   205→            <button onClick={() => setEdit((v) => !v)} aria-label="Modifier mon profil" className="h-10 w-10 grid place-items-center rounded-full border border-border text-muted-foreground hover:text-primary active:scale-90 transition-all focus-visible:outline-2 focus-visible:outline-primary">
   206→              <Pencil size={16} />
   207→            </button>
   208→          </div>
   209→          {edit && (
   210→            <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: "auto" }} className="overflow-hidden">
   211→              <div className="mt-4 space-y-3">
   212→                <div>
   213→                  <label htmlFor="p-name" className="text-[11px] font-semibold text-muted-foreground">Prénom & nom</label>
   214→                  <input id="p-name" value={name} onChange={(e) => setName(e.target.value)} className="mt-1 h-11 w-full rounded-xl border border-border bg-background px-3 text-sm focus-visible:outline-2 focus-visible:outline-primary" />
   215→                </div>
   216→                <div>
   217→                  <label htmlFor="p-city" className="text-[11px] font-semibold text-muted-foreground">Ville</label>
   218→                  <input id="p-city" value={city} onChange={(e) => setCity(e.target.value)} placeholder="Abidjan" className="mt-1 h-11 w-full rounded-xl border border-border bg-background px-3 text-sm focus-visible:outline-2 focus-visible:outline-primary" />
   219→                </div>
   220→                <button onClick={saveIdentity} disabled={savingId} className="h-11 w-full rounded-xl bg-primary text-primary-foreground text-sm font-bold flex items-center justify-center gap-2 active:scale-[0.98] transition-transform disabled:opacity-60 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary">
   221→                  {savingId ? <Loader2 size={16} className="animate-spin" /> : <Check size={16} />} Enregistrer
   222→                </button>
   223→              </div>
   224→            </motion.div>
   225→          )}
   226→        </div>
   227→      </section>
   228→
   229→      {/* Profil peau */}
   230→      <section aria-labelledby="skin-t">
   231→        <SectionTitle icon={<Sparkles size={16} />}><span id="skin-t">Mon profil peau</span></SectionTitle>
   232→        <div className="rounded-2xl border border-border bg-card p-4 space-y-4">
   233→          <div>
   234→            <p className="text-[11px] font-semibold text-muted-foreground mb-2">Phototype Fitzpatrick</p>
   235→            <div className="grid grid-cols-3 gap-2">
   236→              {FITZPATRICK_CARDS.map((f) => (
   237→                <button key={f.id} onClick={() => setFitz(f.id)} aria-pressed={fitz === f.id} className={`rounded-xl p-1.5 border-2 text-left active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${fitz === f.id ? "border-primary" : "border-transparent"}`}>
   238→                  <div className="h-8 rounded-lg mb-1" style={{ background: f.gradient }} aria-hidden="true" />
   239→                  <p className="text-[11px] font-bold">{f.id}</p>
   240→                </button>
   241→              ))}
   242→            </div>
   243→          </div>
   244→          <div>
   245→            <p className="text-[11px] font-semibold text-muted-foreground mb-2">Type de peau</p>
   246→            <div className="flex flex-wrap gap-1.5">
   247→              {SKIN_TYPES.map((t) => (
   248→                <button key={t.id} onClick={() => setSkinType(t.id)} aria-pressed={skinType === t.id} className={`rounded-full px-3 min-h-10 text-xs font-medium active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${skinType === t.id ? "bg-primary text-primary-foreground" : "border border-border"}`}>{t.label}</button>
   249→              ))}
   250→            </div>
   251→          </div>
   252→          <div>
   253→            <p className="text-[11px] font-semibold text-muted-foreground mb-2">Objectifs</p>
   254→            <div className="flex flex-wrap gap-1.5">
   255→              {SKIN_GOALS.map((g) => (
   256→                <button key={g.id} onClick={() => setGoals((s) => (s.includes(g.id) ? s.filter((x) => x !== g.id) : [...s, g.id]))} aria-pressed={goals.includes(g.id)} className={`rounded-full px-3 min-h-10 text-xs font-medium active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${goals.includes(g.id) ? "bg-primary text-primary-foreground" : "border border-border"}`}>{g.label}</button>
   257→              ))}
   258→            </div>
   259→          </div>
   260→          <div>
   261→            <label htmlFor="p-all" className="text-[11px] font-semibold text-muted-foreground">Allergies</label>
   262→            <textarea id="p-all" value={allergies} onChange={(e) => setAllergies(e.target.value)} rows={2} placeholder="Ex. huile de coco…" className="mt-1 w-full rounded-xl border border-border bg-background px-3 py-2 text-sm focus-visible:outline-2 focus-visible:outline-primary" />
   263→          </div>
   264→          <button onClick={saveSkin} disabled={savingSkin} className="h-11 w-full rounded-xl border border-primary/60 text-primary text-sm font-bold flex items-center justify-center gap-2 active:scale-[0.98] transition-transform disabled:opacity-60 focus-visible:outline-2 focus-visible:outline-primary">
   265→            {savingSkin ? <Loader2 size={15} className="animate-spin" /> : <Check size={15} />} Mettre à jour mon profil peau
   266→          </button>
   267→        </div>
   268→      </section>
   269→
   270→      {/* Langue de l'interface — i18n UI, indépendante de la lecture vocale
   271→          (la langue TTS se règle dans les pilules du résumé vocal, accueil) */}
   272→      <section aria-labelledby="lang-t" className="rounded-2xl border border-border bg-card p-4">
   273→        <div className="flex items-center gap-3">
   274→          <span className="grid place-items-center h-10 w-10 rounded-xl bg-primary/15 text-primary shrink-0">
   275→            <Languages size={19} />
   276→          </span>
   277→          <div className="flex-1 min-w-0">
   278→            <p id="lang-t" className="text-xs font-bold">{t("lang.selector.label")}</p>
   279→            <p className="text-[11px] text-muted-foreground mt-0.5 leading-snug">{t("lang.selector.note")}</p>
   280→          </div>
   281→        </div>
   282→        <div className="mt-3 grid grid-cols-2 gap-2">
   283→          {LANGS.map((l) => {
   284→            const active = lang === l.id;
   285→            return (
   286→              <button
   287→                key={l.id}
   288→                onClick={() => selectLang(l.id)}
   289→                aria-pressed={active}
   290→                className={`rounded-xl border p-3 min-h-12 text-left active:scale-[0.98] transition-all focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary ${
   291→                  active ? "border-primary bg-primary/10" : "border-border hover:border-primary/40"
   292→                }`}
   293→              >
   294→                <span className="flex items-center justify-between gap-1.5">
   295→                  <span className={`text-xs font-bold ${active ? "text-primary" : "text-foreground"}`}>{l.label}</span>
   296→                  {active ? (
   297→                    <Check size={13} className="text-primary shrink-0" aria-hidden="true" />
   298→                  ) : (
   299→                    <span className="text-[9px] font-mono font-bold text-muted-foreground/70" aria-hidden="true">{l.flag}</span>
   300→                  )}
   301→                </span>
   302→                <span className="block text-[10px] text-muted-foreground mt-1">{l.note}</span>
   303→              </button>
   304→            );
   305→          })}
   306→        </div>
   307→      </section>
   308→
   309→      {/* Wallet */}
   310→      <section aria-labelledby="wa-t">
   311→        <SectionTitle icon={<WalletIcon size={16} />}><span id="wa-t">Mon wallet Kènè</span></SectionTitle>
   312→        <div className="rounded-3xl bg-[#1A1410] text-[#F8F1E4] p-5 shadow-md relative overflow-hidden">
   313→          <div aria-hidden="true" className="absolute inset-0 bogolan-dots opacity-20" />
   314→          <div className="relative">
   315→            <p className="text-[10px] uppercase tracking-[0.18em] opacity-70">Solde disponible</p>
   316→            <p className="font-mono font-black text-3xl mt-1">{wallet ? xof(wallet.balance) : "···"}</p>
   317→            <div className="flex items-center gap-2 mt-2 text-[11px] opacity-80">
   318→              <BadgeCheck size={13} className="text-[#C8951E]" /> Cashback {Math.round((wallet?.cashbackRate ?? CASHBACK_RATE) * 100)} % sur chaque commande
   319→            </div>
   320→            <button onClick={() => { setTopup(true); setTopupState("idle"); }} className="mt-4 h-11 w-full rounded-xl bg-[#C8951E] text-[#1A1410] font-bold text-sm flex items-center justify-center gap-2 active:scale-[0.98] transition-transform focus-visible:outline-2 focus-visible:outline-[#C8951E]">
   321→              <Plus size={16} /> Approvisionner
   322→            </button>
   323→          </div>
   324→        </div>
   325→        <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wide mt-4 mb-2">Dernières transactions</p>
   326→        {txs === null ? (
   327→          <div className="space-y-2">{[0, 1, 2].map((i) => <Skeleton key={i} className="h-12 rounded-xl" />)}</div>
   328→        ) : txs.length === 0 ? (
   329→          <p className="text-xs text-muted-foreground text-center py-3">Aucune transaction pour l&apos;instant.</p>
   330→        ) : (
   331→          <div className="rounded-2xl border border-border bg-card divide-y divide-border max-h-72 overflow-y-auto scrollbar-thin">
   332→            {txs.map((t) => {
   333→              const credit = t.type === "credit";
   334→              return (
   335→                <div key={t.id} className="flex items-center gap-3 px-4 py-3">
   336→                  <span className={`grid place-items-center h-8 w-8 rounded-full shrink-0 ${credit ? "bg-success/15 text-success" : "bg-destructive/10 text-destructive"}`}>
   337→                    {credit ? <ArrowDownLeft size={15} /> : <ArrowUpRight size={15} />}
   338→                  </span>
   339→                  <div className="min-w-0 flex-1">
   340→                    <p className="text-xs font-semibold">{REASON_LABELS[t.reason] ?? t.reason}</p>
   341→                    <p className="text-[10px] text-muted-foreground">{formatDate(t.createdAt, { day: "numeric", month: "short", year: "2-digit" })}</p>
   342→                  </div>
   343→                  <span className={`font-mono text-sm font-bold ${credit ? "text-success" : "text-destructive"}`}>
   344→                    {credit ? "+" : "−"}{xof(t.amount)}
   345→                  </span>
   346→                </div>
   347→              );
   348→            })}
   349→          </div>
   350→        )}
   351→      </section>
   352→
   353→      {/* Parrainage — le fil qui relie les amies */}
   354→      <ParrainageCard userId={user.id} userName={user.name} onRedeemed={loadWallet} />
   355→
   356→      {/* Consentement */}
   357→      <section aria-labelledby="cons-t" className="rounded-2xl border border-border bg-card p-4 flex items-center gap-3">
   358→        <ShieldCheck size={20} className={user.consentHealth ? "text-success" : "text-destructive"} />
   359→        <div className="flex-1">
   360→          <p className="text-xs font-semibold">Consentement données santé</p>
   361→          <p className="text-[11px] text-muted-foreground mt-0.5">{user.consentHealth ? "Accordé — photos et diagnostics utilisés uniquement pour tes analyses." : "Non accordé — requis pour le diagnostic IA."}</p>
   362→        </div>
   363→        <span className={`rounded-full px-2.5 py-1 text-[10px] font-bold ${user.consentHealth ? "bg-success/15 text-success" : "bg-destructive/10 text-destructive"}`}>
   364→          {user.consentHealth ? "Actif" : "Inactif"}
   365→        </span>
   366→      </section>
   367→
   368→      {/* Mes données — portabilité RGPD (art. 20) */}
   369→      <section aria-labelledby="rgpd-t" className="rounded-2xl border border-border bg-card p-4">
   370→        <div className="flex items-center gap-3">
   371→          <span className="grid place-items-center h-10 w-10 rounded-xl bg-terre/15 text-terre shrink-0">
   372→            <Download size={19} />
   373→          </span>
   374→          <div className="flex-1 min-w-0">
   375→            <p id="rgpd-t" className="text-xs font-bold flex items-center gap-1.5">Mes données <span className="rounded-full bg-muted px-1.5 py-px text-[9px] font-semibold text-muted-foreground">RGPD</span></p>
   376→            <p className="text-[11px] text-muted-foreground mt-0.5 leading-snug">Ton dossier complet en un fichier : diagnostics, rendez-vous, commandes, wallet, parrainage, notifications.</p>
   377→          </div>
