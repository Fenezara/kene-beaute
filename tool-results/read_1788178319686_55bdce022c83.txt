    51→function parseDetails(ps: ProPayslip): PayslipDetails {
    52→  try {
    53→    const d = JSON.parse(ps.detailsJson) as PayslipDetails;
    54→    return { ...d, lines: Array.isArray(d.lines) ? d.lines : [] };
    55→  } catch {
    56→    return { lines: [] as PayrollLine[] };
    57→  }
    58→}

   366→                    </div>

   369→              );
   370→            })}
   371→          </div>

   375→      {/* ── Dialog nouveau salarié ── */}

   380→    </div>

   384→function latestCountry(slip: ProPayslip): string {
   385→  // le régime est dérivable du détails (CNPS=CI / IPRES=SN)
   386→  return parseDetails(slip).regime?.includes("IPRES") || parseDetails(slip).regime?.includes("IPM") ? "SN" : "CI";
   387→}

   389→// ═════════════ Embauche ═════════════
   390→function HireDialog({
   391→  open,
   392→  onOpenChange,
   393→  tenantId,
   394→  defaultCountry,
   395→  onCreated,
   396→}: {
   397→  open: boolean;
   398→  onOpenChange: (o: boolean) => void;
   399→  tenantId: string;
   400→  defaultCountry: string;
   401→  onCreated: () => Promise<void>;
   402→}) {
   403→  const [name, setName] = useState("");
   404→  const [role, setRole] = useState("estheticienne");
   405→  const [contractType, setContractType] = useState("CDI");
   406→  const [country, setCountry] = useState(defaultCountry || "CI");
   407→  const [baseSalary, setBaseSalary] = useState("");
   408→  const [transport, setTransport] = useState("");
   409→  const [housing, setHousing] = useState("");
   410→  const [cadres, setCadres] = useState(false);
   411→  const [busy, setBusy] = useState(false);
   412→
   413→  async function submit() {
   414→    const salary = Number(baseSalary);
   415→    if (!name.trim() || !salary || salary <= 0) {
   416→      toast.error("Nom et salaire de base valides requis");
   417→      return;
   418→    }
   419→    setBusy(true);
   420→    try {
   421→      await apiPost("/api/pro/employees", {
   422→        tenantId,
   423→        name: name.trim(),
   424→        role,
   425→        contractType,
   426→        country,
   427→        baseSalary: salary,
   428→        transport: Number(transport) || 0,
   429→        housing: Number(housing) || 0,
   430→        cadres: country === "SN" ? cadres : undefined,
   431→      });
   432→      toast.success(`${name.trim()} ajoutée à l'équipe`);
   433→      setName("");
   434→      setBaseSalary("");
   435→      setTransport("");
   436→      setHousing("");
   437→      setCadres(false);
   438→      onOpenChange(false);
   439→      await onCreated();
   440→    } catch (e) {
