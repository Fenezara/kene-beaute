    40→function norm(s: string): string {
    41→  return s
    42→    .toLowerCase()
    43→    .normalize("NFD")
    44→    .replace(/[\u0300-\u036f]/g, "")
    45→    .replace(/[^a-z0-9 ]/g, " ");
    46→}

    48→/** Match une recommandation libre (« Sérum Moringa éclat ») avec un produit boutique */
    49→function matchProduct(rec: string, products: ApiProduct[]): ApiProduct | null {
    50→  const words = norm(rec).split(" ").filter((w) => w.length >= 5);
    51→  let best: { p: ApiProduct; score: number } | null = null;
    52→  for (const p of products) {
    53→    const target = `${norm(p.name)} ${norm(p.botanicals)} ${norm(p.description)}`;
    54→    let score = 0;
    55→    for (const w of words) if (target.includes(w)) score += 1;
    56→    if (score > 0 && (!best || score > best.score)) best = { p, score };
    57→  }
    58→  return best?.p ?? null;
    59→}

   602→/* ══════════════ Historique + comparaison ══════════════ */
   603→function HistoryView({
   604→  history, compareMode, setCompareMode, compareSel, setCompareSel, onBack, onOpen,
   605→}: {
   606→  history: ApiDiagnosis[] | null;
   607→  compareMode: boolean;
   608→  setCompareMode: (v: boolean) => void;
   609→  compareSel: string[];
   610→  setCompareSel: (v: string[]) => void;
   611→  onBack: () => void;
   612→  onOpen: (d: ApiDiagnosis) => void;
   613→}) {
   614→  const list = useMemo(
   615→    () => [...(history ?? [])].filter((d) => d.status === "done").sort((a, b) => +new Date(b.createdAt) - +new Date(a.createdAt)),
   616→    [history]
   617→  );
   618→  const sameZoneSel = compareSel.length === 2 && list.find((d) => d.id === compareSel[0])?.zone === list.find((d) => d.id === compareSel[1])?.zone;
   619→  const a = list.find((d) => d.id === compareSel[0]);
   620→  const b = list.find((d) => d.id === compareSel[1]);
   621→  const ra = a ? parseDiagnosis(a.resultJson) : null;
   622→  const rb = b ? parseDiagnosis(b.resultJson) : null;
   623→
   624→  const evolutions =
   625→    ra && rb
   626→      ? rb.indicateurs
   627→          .map((ib) => {
   628→            const ia = ra.indicateurs.find((x) => x.nom === ib.nom);
   629→            return ia ? { nom: ib.nom, delta: ib.pourcentage - ia.pourcentage } : null;
   630→          })
   631→          .filter((x): x is { nom: string; delta: number } => x !== null)
   632→          .sort((x, y) => Math.abs(y.delta) - Math.abs(x.delta))
   633→          .slice(0, 3)
   634→      : [];
   635→
   636→  return (
   637→    <div className="pt-4 pb-2">
   638→      <button onClick={onBack} className="inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground mb-4 focus-visible:outline-2 focus-visible:outline-primary rounded" aria-label="Retour">
   639→        <ArrowLeft size={15} /> Nouveau diagnostic
   640→      </button>
   641→      <div className="flex items-center justify-between gap-2">
   642→        <h1 className="font-heading font-black text-xl">Mon historique</h1>
   643→        <button
   644→          onClick={() => { setCompareMode(!compareMode); setCompareSel([]); }}
   645→          aria-pressed={compareMode}
   646→          className={`h-9 px-3 rounded-full text-[11px] font-bold flex items-center gap-1.5 active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${compareMode ? "bg-primary text-primary-foreground" : "border border-border bg-card"}`}
   647→        >
   648→          <GitCompareArrows size={14} /> Comparer
   649→        </button>
   650→      </div>
   651→
   652→      {history === null ? (
   653→        <div className="space-y-3 mt-5">
   654→          {[0, 1, 2].map((i) => (
   655→            <Skeleton key={i} className="h-20 rounded-2xl" />
   656→          ))}
   657→        </div>
   658→      ) : list.length === 0 ? (
   659→        <div className="mt-6 rounded-2xl border border-dashed border-border bg-card/60 p-8 text-center">
   660→          <NeaOnnimIcon size={38} className="mx-auto text-primary" />
   661→          <p className="font-heading font-bold text-sm mt-3">Aucun diagnostic pour l&apos;instant</p>
   662→          <p className="text-xs text-muted-foreground mt-1">Ton premier scan débloquera le suivi d&apos;évolution.</p>
   663→        </div>
   664→      ) : (
   665→        <>
   666→          <div className="space-y-2.5 mt-4">
   667→            {list.map((d) => {
   668→              const sel = compareSel.includes(d.id);
   669→              return (
   670→                <div key={d.id} className={`flex items-center gap-3 rounded-2xl border p-2.5 transition-colors ${sel ? "border-primary bg-primary/5" : "border-border bg-card"}`}>
   671→                  <button
   672→                    onClick={() => {
   673→                      if (compareMode) {
   674→                        setCompareSel(sel ? compareSel.filter((x) => x !== d.id) : compareSel.length < 2 ? [...compareSel, d.id] : [compareSel[1], d.id]);
   675→                      } else onOpen(d);
   676→                    }}
   677→                    className="flex items-center gap-3 flex-1 min-w-0 text-left focus-visible:outline-2 focus-visible:outline-primary rounded-xl"
   678→                    aria-label={`Diagnostic ${d.zone} du ${formatDate(d.createdAt)}, score ${d.scoreGlobal}`}
   679→                  >
   680→                    <img src={diagImgSrc(d.imageData)} alt={`Diagnostic ${d.zone}`} loading="lazy" className="h-14 w-14 rounded-xl object-cover shrink-0" />
   681→                    <div className="min-w-0">
   682→                      <p className="text-sm font-semibold">{BODY_ZONES.find((z) => z.id === d.zone)?.label ?? d.zone}</p>
   683→                      <p className="text-[11px] text-muted-foreground">{formatDate(d.createdAt)}</p>

   690→                      aria-pressed={sel}
