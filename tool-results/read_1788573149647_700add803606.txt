   312→              </div>
   313→            );
   314→          })}

   316→        <div className="w-full max-w-[320px] mt-5">
   317→          <Progress value={progress} className="h-2" aria-label="Progression de l'analyse" />
   318→          <p className="text-center text-[11px] text-muted-foreground mt-2">Vision par ordinateur spécialisée mélanoderme — 5 à 30 s</p>

   326→
   327→  /* ─────────── Étape 3 — Résultats ─────────── */
   328→  if (step === 3 && diag) {

   330→  }
   331→
   332→  return null;
   333→}

   336→function ResultView({ diag, products, productsError, onRetryProducts, onNewZone, onHistory }: { diag: { id: string; result: DiagnosisResult; imageData: string; createdAt: string }; products: ApiProduct[]; productsError: boolean; onRetryProducts: () => void; onNewZone: () => void; onHistory: () => void }) {
   337→  const user = useKene((s) => s.user)!;
   338→  const setClientTab = useKene((s) => s.setClientTab);
   339→  const addToCart = useKene((s) => s.addToCart);
   340→  const [view, setView] = useState<string>("standard");
   341→  const [ritualOpen, setRitualOpen] = useState(false);
   342→  const [glossary, setGlossary] = useState<GlossaryEntry | null>(null);
   343→  const r = diag.result;
   344→  const weakest = useMemo(() => [...r.indicateurs].sort((a, b) => a.pourcentage - b.pourcentage).slice(0, 8), [r.indicateurs]);
   345→  const viewDef = SPECTRAL_VIEWS.find((v) => v.id === view) ?? SPECTRAL_VIEWS[0];
   346→  const heatmapCls = view === "pigment" ? "heatmap-pigment" : view === "inflammation" ? "heatmap-inflammation" : view === "acne" ? "heatmap-acne" : "";
   347→
   348→  const botanicIcon = (name: string): React.ReactNode => {
   349→    const n = norm(name);
   350→    if (n.includes("moringa")) return <MoringaIcon size={13} />;
   351→    if (n.includes("karite")) return <KariteIcon size={13} />;
   352→    if (n.includes("baobab")) return <BaobabIcon size={13} />;
   353→    return <Sparkles size={13} />;
   354→  };
   355→
   356→  function askGlossary(term: string) {
   357→    const entry = glossaryFor(term);
   358→    if (entry) setGlossary(entry);
   359→  }
   360→
   361→  return (
   362→    <div className="pt-4 pb-2">
   363→      {/* Header score */}
   364→      <div className="rounded-3xl border border-border bg-card shadow-sm overflow-hidden">
   365→        <div className="kente-band h-1.5 w-full" aria-hidden="true" />
   366→        <div className="p-5 flex items-center gap-4">
   367→          <ScoreGauge score={r.score_global} label="Zone" size={110} />
   368→          <div className="min-w-0">
   369→            <div className="flex items-center gap-2 flex-wrap">
   370→              <h2 className="font-heading font-black text-lg">{BODY_ZONES.find((z) => z.id === r.zone)?.label ?? r.zone}</h2>
   371→              {r.fitzpatrick_estime && <span className="rounded-full bg-melanine text-[#F8F1E4] px-2 py-0.5 text-[10px] font-bold">Fitz {r.fitzpatrick_estime}</span>}
   372→            </div>
   373→            <p className="text-[11px] text-muted-foreground mt-1">{formatDate(diag.createdAt)} · {r.source === "vlm" ? "Analyse VLM" : "Analyse heuristique"} · {r.indicateurs.length} indicateurs</p>
   374→            <div className="flex gap-1.5 mt-2">
   375→              <span className="rounded-full px-2 py-0.5 text-[10px] font-bold" style={{ backgroundColor: scoreColor(r.score_global), color: readableTextColor(scoreColor(r.score_global)) }}>
   376→                {r.score_global >= 80 ? "Excellente santé" : r.score_global >= 60 ? "Bon équilibre" : r.score_global >= 40 ? "Points à surveiller" : "Besoin de soin"}
   377→              </span>
   378→            </div>
   379→          </div>
   380→        </div>
   381→      </div>
   382→
   383→      {/* Lecture vocale — accès non-lectrices & confort audio (TTS) */}
   384→      <VoiceNarration result={r} userName={user.name} />
   385→
   386→      {/* Résumé en pictos — tuiles tapables lues à voix haute (non-lectrices) */}
   387→      <div className="mt-4">
   388→        <PictoSummary result={r} zoneLabel={BODY_ZONES.find((z) => z.id === r.zone)?.label ?? r.zone} />
   389→      </div>
   390→
   391→      {/* Alerte orientation dermato */}
   392→      {r.orientation_dermato && (
   393→        <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="mt-4 rounded-2xl border-2 border-[#8B1A3B]/60 bg-[#8B1A3B]/5 p-4" role="alert">
   394→          <p className="flex items-center gap-2 font-heading font-bold text-sm text-[#8B1A3B]">

   676→/* ══════════════ Historique + comparaison ══════════════ */
   677→function HistoryView({
   678→  userId, history, compareMode, setCompareMode, compareSel, setCompareSel, onBack, onOpen,
   679→}: {
   680→  userId: string;
   681→  history: ApiDiagnosis[] | null;
   682→  compareMode: boolean;
   683→  setCompareMode: (v: boolean) => void;
   684→  compareSel: string[];
   685→  setCompareSel: (v: string[]) => void;
   686→  onBack: () => void;
   687→  onOpen: (d: ApiDiagnosis) => void;
   688→}) {
   689→  const list = useMemo(
   690→    () => [...(history ?? [])].filter((d) => d.status === "done").sort((a, b) => +new Date(b.createdAt) - +new Date(a.createdAt)),
   691→    [history]
   692→  );
   693→  const sameZoneSel = compareSel.length === 2 && list.find((d) => d.id === compareSel[0])?.zone === list.find((d) => d.id === compareSel[1])?.zone;
   694→  const a = list.find((d) => d.id === compareSel[0]);
   695→  const b = list.find((d) => d.id === compareSel[1]);
   696→  const ra = a ? parseDiagnosis(a.resultJson) : null;
   697→  const rb = b ? parseDiagnosis(b.resultJson) : null;
   698→
   699→  const evolutions =
   700→    ra && rb
   701→      ? rb.indicateurs
   702→          .map((ib) => {
   703→            const ia = ra.indicateurs.find((x) => x.nom === ib.nom);
   704→            return ia ? { nom: ib.nom, delta: ib.pourcentage - ia.pourcentage } : null;
   705→          })
   706→          .filter((x): x is { nom: string; delta: number } => x !== null)
   707→          .sort((x, y) => Math.abs(y.delta) - Math.abs(x.delta))
   708→          .slice(0, 3)
   709→      : [];
   710→
   711→  return (
   712→    <div className="pt-4 pb-2">
   713→      <button onClick={onBack} className="inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground mb-4 focus-visible:outline-2 focus-visible:outline-primary rounded" aria-label="Retour">
   714→        <ArrowLeft size={15} /> Nouveau diagnostic
   715→      </button>
   716→      <div className="flex items-center justify-between gap-2">
   717→        <h2 className="font-heading font-black text-xl">Mon historique</h2>
   718→        <button
   719→          onClick={() => { setCompareMode(!compareMode); setCompareSel([]); }}
   720→          aria-pressed={compareMode}
   721→          className={`h-10 px-3 rounded-full text-[11px] font-bold flex items-center gap-1.5 active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${compareMode ? "bg-primary text-primary-foreground" : "border border-border bg-card"}`}
   722→        >
   723→          <GitCompareArrows size={14} /> Comparer
   724→        </button>
   725→      </div>
   726→
   727→      {history === null ? (
   728→        <div className="space-y-3 mt-5">
   729→          {[0, 1, 2].map((i) => (
   730→            <Skeleton key={i} className="h-20 rounded-2xl" />
   731→          ))}
   732→        </div>
   733→      ) : list.length === 0 ? (
   734→        <div className="mt-6 rounded-2xl border border-dashed border-border bg-card/60 p-8 text-center">
   735→          <NeaOnnimIcon size={38} className="mx-auto text-primary" />
   736→          <p className="font-heading font-bold text-sm mt-3">Aucun diagnostic pour l&apos;instant</p>
   737→          <p className="text-xs text-muted-foreground mt-1">Ton premier scan débloquera le suivi d&apos;évolution.</p>
   738→        </div>
   739→      ) : (
   740→        <>
   741→          {/* Le Fil du Temps — courbes d'évolution */}
   742→          <EvolutionCard userId={userId} className="mt-4" />
   743→
   744→          <div className="space-y-2.5 mt-4">
   745→            {list.map((d) => {
   746→              const sel = compareSel.includes(d.id);
   747→              return (
   748→                <div key={d.id} className={`flex items-center gap-3 rounded-2xl border p-2.5 transition-colors ${sel ? "border-primary bg-primary/5" : "border-border bg-card"}`}>
   749→                  <button
   750→                    onClick={() => {
   751→                      if (compareMode) {
   752→                        setCompareSel(sel ? compareSel.filter((x) => x !== d.id) : compareSel.length < 2 ? [...compareSel, d.id] : [compareSel[1], d.id]);
   753→                      } else onOpen(d);
   754→                    }}
   755→                    className="flex items-center gap-3 flex-1 min-w-0 text-left focus-visible:outline-2 focus-visible:outline-primary rounded-xl"
   756→                    aria-label={`Diagnostic ${d.zone} du ${formatDate(d.createdAt)}, score ${d.scoreGlobal}`}
   757→                  >
   758→                    <img src={diagImgSrc(d.imageData)} alt={`Diagnostic ${d.zone}`} loading="lazy" className="h-14 w-14 rounded-xl object-cover shrink-0" />
