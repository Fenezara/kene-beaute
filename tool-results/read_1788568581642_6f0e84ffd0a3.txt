     1→"use client";
     2→// Kènè Cliente — Boutique : catalogue, fiche produit, panier, checkout Wave/Orange/Wallet simulé
     3→// + « Mes commandes » : historique des commandes enregistrées (consultation par la cliente).
     4→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     5→import { AnimatePresence, motion } from "framer-motion";
     6→import { BadgeCheck, Heart, History, Loader2, Lock, Minus, Plus, Search, ShoppingBag, Tag, Trash2, X } from "lucide-react";
     7→import { toast } from "sonner";
     8→import { apiGet, apiPost } from "@/lib/kene/api";
     9→import { xof, CASHBACK_RATE, formatDate, formatTime } from "@/lib/kene/format";
    10→import { HAPTIC, haptic } from "@/lib/kene/ux";
    11→import { MOMO_OPERATORS } from "@/lib/kene/rfm";
    12→import { Sheet, SheetContent, SheetHeader, SheetTitle } from "@/components/ui/sheet";
    13→import { Skeleton } from "@/components/ui/skeleton";
    14→import { useKene } from "@/store/kene";
    15→import { useFavorites } from "@/store/favorites";
    16→import { useSecurity } from "@/store/security";
    17→import { KenteWeaveCard } from "@/components/kene/weave/KenteWeaveCard";
    18→import { categoryThread } from "@/components/kene/weave/threads";
    19→import type { ApiOrder, ApiPayment, ApiProduct, ApiWallet } from "./types";
    20→import { SHOP_CATEGORIES } from "./types";
    21→import { EmptyBlock, Stars, SuccessBurst } from "./bits";
    22→import { FavButton } from "./FavButton";
    23→import { SecureVerify } from "./SecureVerify";
    24→
    25→type PayMethod = "wave" | "orange" | "wallet";
    26→
    27→/* Coupon validé au checkout (aperçu local — la consommation a lieu à la commande) */
    28→interface AppliedPromo {
    29→  code: string;
    30→  label: string | null;
    31→  discount: number;
    32→  subtotal: number; // panier au moment de la validation (invalidation si changement)
    33→}
    34→
    35→export function ShopScreen() {
    36→  const user = useKene((s) => s.user)!;
    37→  const cart = useKene((s) => s.cart);
    38→  const addToCart = useKene((s) => s.addToCart);
    39→  const setCartQty = useKene((s) => s.setCartQty);
    40→  const clearCart = useKene((s) => s.clearCart);
    41→
    42→  const [products, setProducts] = useState<ApiProduct[] | null>(null);
    43→  const [cat, setCat] = useState("");
    44→  const [q, setQ] = useState("");
    45→  const [favOnly, setFavOnly] = useState(false); // filtre « ♥ Favoris » (cumulable avec catégorie + recherche)
    46→  const favs = useFavorites((s) => s.favs);
    47→  const [detail, setDetail] = useState<ApiProduct | null>(null);
    48→  const [qty, setQty] = useState(1);
    49→  const [checkout, setCheckout] = useState(false);
    50→  const [wallet, setWallet] = useState<ApiWallet | null>(null);
    51→  const [payState, setPayState] = useState<{ phase: "processing" | "success"; method: PayMethod; amount: number } | null>(null);
    52→  const [paying, setPaying] = useState(false);
    53→
    54→  // Sécurité renforcée (2FA-lite) : si activée, la cliente re-vérifie son code
    55→  // AVANT que le moindre appel de paiement ne parte (voir startPay + SecureVerify).
    56→  const securityEnabled = useSecurity((s) => s.enabled);
    57→  const [pendingPay, setPendingPay] = useState<PayMethod | null>(null);
    58→
    59→  // ─── Double-tap « ajout rapide » (TikTok Shop / Instagram) ───
    60→  // 1er tap = ouvre la fiche (avec un délai court annulable) ; 2e tap < 320 ms
    61→  // = ajoute directement au panier + burst animé sur la carte.
    62→  const lastTap = useRef<{ id: string; t: number; timer: number | null }>({ id: "", t: 0, timer: null });
    63→  const [burst, setBurst] = useState<{ id: string; x: number; y: number } | null>(null);
    64→
    65→  const onCardTap = (p: ApiProduct, e: React.MouseEvent<HTMLButtonElement>) => {
    66→    const now = Date.now();
    67→    const s = lastTap.current;
    68→    if (s.id === p.id && now - s.t < 320 && s.timer) {
    69→      window.clearTimeout(s.timer);
    70→      lastTap.current = { id: "", t: 0, timer: null };
    71→      addToCart({ productId: p.id, name: p.name, price: p.price, qty: 1, image: p.image });
    72→      haptic(HAPTIC.light);
    73→      const rect = e.currentTarget.getBoundingClientRect();
    74→      setBurst({ id: p.id, x: e.clientX - rect.left, y: e.clientY - rect.top });
    75→      window.setTimeout(() => setBurst(null), 700);
    76→      toast.success(`${p.name} ajouté au panier`, { description: "Astuce : double-tape une carte pour l'ajouter en 1 geste" });
    77→    } else {
    78→      if (s.timer) window.clearTimeout(s.timer);
    79→      const timer = window.setTimeout(() => {
    80→        setDetail(p);
    81→        setQty(1);
    82→      }, 240);
    83→      lastTap.current = { id: p.id, t: now, timer };
    84→    }
    85→  };
    86→  const [promoInput, setPromoInput] = useState("");
    87→  const [promoChecking, setPromoChecking] = useState(false);
    88→  const [promo, setPromo] = useState<AppliedPromo | null>(null);
    89→  const inputRef = useRef<HTMLInputElement>(null);
    90→
    91→  /* « Mes commandes » — historique consultable (chargé au montage, rafraîchi
    92→   * après chaque commande réussie : la cliente suit ses données enregistrées). */
    93→  const [view, setView] = useState<"catalogue" | "commandes">("catalogue");
    94→  const [orders, setOrders] = useState<ApiOrder[] | null>(null);
    95→  const refreshOrders = useCallback(() => {
    96→    apiGet<{ orders: ApiOrder[] }>(`/api/orders?userId=${user.id}`)
    97→      .then((r) => setOrders(r.orders ?? []))
    98→      .catch(() => setOrders((prev) => prev ?? []));
    99→  }, [user.id]);
   100→
   101→  const subtotal = cart.reduce((s, l) => s + l.price * l.qty, 0);
   102→  /* Coupon invalidé si le panier a changé depuis la validation (remise recalculée
   103→   * sur un autre sous-total) — la cliente re-valide, jamais de surprise serveur. */
   104→  const livePromo = promo && promo.subtotal === subtotal ? promo : null;
   105→  const discount = livePromo?.discount ?? 0;
   106→  const total = subtotal - discount;
   107→  /* taux de cashback réellement appliqué (wallet de la cliente, sinon défaut) */
   108→  const cashbackRate = wallet?.cashbackRate ?? CASHBACK_RATE;
   109→
   110→  useEffect(() => {
   111→    apiGet<{ products: ApiProduct[] }>("/api/shop/products")
   112→      .then((r) => setProducts(r.products ?? []))
   113→      .catch(() => setProducts([]));
   114→    apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).then((r) => setWallet(r.wallet)).catch(() => {});
   115→    refreshOrders();
   116→  }, [user.id, refreshOrders]);
   117→
   118→  const filtered = useMemo(() => {
   119→    const nq = q.trim().toLowerCase();
   120→    return (products ?? []).filter(
   121→      (p) =>
   122→        (!cat || p.category === cat) &&
   123→        (!favOnly || favs.includes(p.id)) &&
   124→        (!nq || `${p.name} ${p.botanicals} ${p.description}`.toLowerCase().includes(nq))
   125→    );
   126→  }, [products, cat, q, favOnly, favs]);
   127→
   128→  /* le fil de la catégorie — la navette l'illumine dans la bande tissée */
   129→  const weaveCaption =
   130→    products === null
   131→      ? "La navette monte le métier…"
   132→      : `${filtered.length} soin${filtered.length > 1 ? "s" : ""}${
   133→          cat ? ` · ${SHOP_CATEGORIES.find((c) => c.id === cat)?.label.toLowerCase() ?? cat}` : " au catalogue"
   134→        }${favOnly ? " · favoris" : ""}`;
   135→
   136→  /* Applique un code promo : aperçu de remise sans consommer le coupon
   137→   * (la consommation a lieu à la commande — toutes les gardes côté serveur). */
   138→  async function applyPromo() {
   139→    const code = promoInput.trim().toUpperCase();
   140→    if (!code || promoChecking) return;
   141→    setPromoChecking(true);
   142→    try {
   143→      const r = await apiPost<{ ok: true; coupon: { code: string; label: string | null }; discount: number; total: number }>(
   144→        "/api/coupons/validate",
   145→        { code, userId: user.id, subtotal }
   146→      );
   147→      setPromo({ code: r.coupon.code, label: r.coupon.label, discount: r.discount, subtotal });
   148→      setPromoInput("");
   149→      toast.success(`Code ${r.coupon.code} appliqué`, { description: `Remise de ${xof(r.discount)} déduite du total.` });
   150→    } catch (e) {
   151→      setPromo(null);
   152→      toast.error(e instanceof Error ? e.message : "Code promo invalide");
   153→    } finally {
   154→      setPromoChecking(false);
   155→    }
   156→  }
   157→
   158→  /** Passerelle paiement : vérification d'identité par code si la sécurité
   159→   *  renforcée est active — le paiement initialement prévu (pay) n'est lancé
   160→   *  qu'une fois le code confirmé ; annulé sinon, rien n'est engagé. */
   161→  function startPay(method: PayMethod) {
   162→    if (securityEnabled) {
   163→      haptic(HAPTIC.tap);
   164→      setPendingPay(method);
   165→      return;
   166→    }
   167→    void pay(method);
   168→  }
   169→
   170→  async function pay(method: PayMethod) {
   171→    setPaying(true);
   172→    try {
   173→      const items = cart.map((l) => ({ productId: l.productId, qty: l.qty }));
   174→      const r = await apiPost<{ order: ApiOrder; payment: ApiPayment | null; paid: boolean }>("/api/orders", {
   175→        userId: user.id,
   176→        items,
   177→        paymentMethod: method,
   178→        ...(livePromo ? { couponCode: livePromo.code } : {}),
   179→      });
   180→      const amount = r.order.total;
   181→      if (method !== "wallet" && r.payment) {
   182→        setCheckout(false);
   183→        setPayState({ phase: "processing", method, amount });
   184→        await new Promise((res) => setTimeout(res, 3000));
   185→        const c = await apiPost<{ payment: ApiPayment; order?: ApiOrder; wallet?: ApiWallet }>("/api/payments/confirm", { paymentId: r.payment.id });
   186→        if (c.wallet) setWallet(c.wallet);
   187→      } else {
   188→        setCheckout(false);
   189→        setPayState({ phase: "processing", method: "wallet", amount });
   190→        await new Promise((res) => setTimeout(res, 1200));
   191→      }
   192→      setPayState({ phase: "success", method, amount });
   193→      haptic(HAPTIC.success);
   194→      clearCart();
   195→      setPromo(null); // le coupon est consommé : remise à zéro pour la prochaine commande
   196→      void refreshOrders(); // la nouvelle commande apparaît dans « Mes commandes »
   197→      const cb = r.order.cashback;
   198→      toast.success(cb > 0 ? `Commande confirmée — cashback ${xof(cb)} crédité sur ton wallet` : "Commande confirmée et payée");
   199→      if (method === "wallet") {
   200→        const w = await apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).catch(() => null);
   201→        if (w) setWallet(w.wallet);
   202→      }
   203→    } catch (e) {
   204→      setPayState(null);
   205→      toast.error(e instanceof Error ? e.message : "Paiement impossible");
   206→    } finally {
   207→      setPaying(false);
   208→    }
   209→  }
   210→
   211→  const op = (m: PayMethod) => MOMO_OPERATORS.find((o) => o.code === m);
   212→
   213→  return (
   214→    <div className="pt-4 pb-2">
   215→      <header className="flex items-center justify-between">
   216→        <h1 className="font-heading font-black text-xl">Boutique Kènè</h1>
   217→        <span className="rounded-full bg-primary/10 text-primary text-[10px] font-bold px-2.5 py-1 uppercase tracking-wide">Cashback {Math.round(cashbackRate * 100)} %</span>
   218→      </header>
   219→
   220→      {/* Bascule Catalogue ↔ Mes commandes — la cliente consulte ses données */}
   221→      <div className="mt-3 grid grid-cols-2 gap-1 rounded-2xl border border-border bg-card p-1" role="tablist" aria-label="Vues boutique">
   222→        <button
   223→          role="tab"
   224→          aria-selected={view === "catalogue"}
   225→          onClick={() => setView("catalogue")}
   226→          className={`h-10 rounded-xl flex items-center justify-center gap-1.5 text-xs font-bold transition-all active:scale-[0.98] focus-visible:outline-2 focus-visible:outline-primary ${view === "catalogue" ? "bg-primary text-primary-foreground shadow-sm" : "text-muted-foreground hover:text-foreground"}`}
   227→        >
   228→          <ShoppingBag size={14} aria-hidden="true" /> Catalogue
   229→        </button>
   230→        <button
   231→          role="tab"
   232→          aria-selected={view === "commandes"}
   233→          onClick={() => setView("commandes")}
   234→          className={`h-10 rounded-xl flex items-center justify-center gap-1.5 text-xs font-bold transition-all active:scale-[0.98] focus-visible:outline-2 focus-visible:outline-primary ${view === "commandes" ? "bg-primary text-primary-foreground shadow-sm" : "text-muted-foreground hover:text-foreground"}`}
   235→        >
   236→          <History size={14} aria-hidden="true" /> Mes commandes{orders !== null && orders.length > 0 ? ` (${orders.length})` : ""}
   237→        </button>
   238→      </div>
   239→
   240→      {view === "commandes" ? (
   241→        <OrdersView orders={orders} onRefresh={refreshOrders} onShop={() => setView("catalogue")} />
   242→      ) : (
   243→        <>
   244→      {/* Le Fil de Kente — hero tissé, le fil de la catégorie s'illumine */}
   245→      <div className="mt-4">
   246→        <KenteWeaveCard highlightIndex={cat ? categoryThread(cat) : -1} caption={weaveCaption} />
   247→      </div>
   248→
   249→      {/* Recherche */}
   250→      <div className="mt-3 relative">
   251→        <Search size={16} className="absolute left-3.5 top-1/2 -translate-y-1/2 text-muted-foreground" aria-hidden="true" />
   252→        <input
   253→          ref={inputRef}
   254→          value={q}
   255→          onChange={(e) => setQ(e.target.value)}
   256→          placeholder="Rechercher un soin, un botanique…"
   257→          aria-label="Rechercher un produit"
   258→          className="h-12 w-full rounded-2xl border border-border bg-card pl-10 pr-4 text-sm focus-visible:outline-2 focus-visible:outline-primary"
   259→        />
   260→      </div>
   261→
   262→      {/* Filtres : catégories + favoris (bascules aria-pressed, cumulables) */}
   263→      <div className="flex gap-2 overflow-x-auto py-3 scrollbar-thin -mx-1 px-1" role="group" aria-label="Filtres de la boutique">
   264→        {SHOP_CATEGORIES.map((c) => (
   265→          <button
   266→            key={c.id}
   267→            aria-pressed={cat === c.id}
   268→            onClick={() => setCat(c.id)}
   269→            className={`shrink-0 rounded-full px-3.5 min-h-10 text-xs font-semibold transition-all active:scale-95 focus-visible:outline-2 focus-visible:outline-primary ${cat === c.id ? "bg-primary text-primary-foreground shadow" : "border border-border bg-card text-foreground/80"}`}
   270→          >
   271→            {c.label}
   272→          </button>
   273→        ))}
   274→        {/* ♥ Favoris — filtre cumulable, badge count si ≥ 1 */}
   275→        <button
   276→          aria-pressed={favOnly}
   277→          onClick={() => { setFavOnly((v) => !v); haptic(HAPTIC.tap); }}
   278→          className={`shrink-0 inline-flex items-center gap-1.5 rounded-full px-3.5 min-h-10 text-xs font-semibold transition-all active:scale-95 focus-visible:outline-2 focus-visible:outline-primary ${favOnly ? "bg-primary text-primary-foreground shadow" : "border border-border bg-card text-foreground/80"}`}
   279→        >
   280→          <Heart size={13} fill={favOnly ? "currentColor" : "none"} aria-hidden="true" />
   281→          Favoris{favs.length > 0 ? ` · ${favs.length}` : ""}
   282→        </button>
   283→      </div>
   284→
   285→      {/* Grille produits */}
   286→      {products === null ? (
   287→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   288→          {[0, 1, 2, 3].map((i) => (
   289→            <div key={i} className="space-y-2">
   290→              <Skeleton className="aspect-square rounded-2xl" />
   291→              <Skeleton className="h-3.5 w-4/5" />
   292→              <Skeleton className="h-3 w-2/3" />
   293→            </div>
   294→          ))}
   295→        </div>
   296→      ) : filtered.length === 0 ? (
   297→        favOnly && favs.length === 0 ? (
   298→          <EmptyBlock
   299→            icon={<Heart size={22} />}
   300→            title="Aucun favori pour l'instant"
   301→            text="Touche le cœur sur un soin pour le retrouver ici."
   302→          />
   303→        ) : favOnly ? (
   304→          <EmptyBlock
   305→            icon={<Heart size={22} />}
   306→            title="Aucun favori dans cette sélection"
   307→            text="Tes favoris ne passent pas ce filtre — essaie une autre catégorie ou efface la recherche."
   308→            cta={
   309→              <button
   310→                onClick={() => { setCat(""); setQ(""); }}
   311→                className="h-11 px-6 rounded-xl bg-primary text-primary-foreground text-sm font-bold active:scale-95 transition-transform focus-visible:outline-2 focus-visible:outline-primary"
   312→              >
   313→                Voir tous mes favoris
   314→              </button>
   315→            }
   316→          />
   317→        ) : (
   318→          <EmptyBlock icon={<Search size={22} />} title="Aucun produit trouvé" text="Essaie un autre mot-clé ou une autre catégorie." />
   319→        )
   320→      ) : (
   321→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   322→          {filtered.map((p) => (
   323→            /* Wrapper relatif : le cœur est un FRÈRE de la carte (jamais de <button>
   324→               imbriqué — HTML valide, focus/a11y propres), posé sur l'image en absolu. */
   325→            <div key={p.id} className="relative">
   326→              <button
   327→                onClick={(e) => onCardTap(p, e)}
   328→                className="relative block w-full text-left rounded-2xl border border-border bg-card overflow-hidden shadow-sm active:scale-[0.98] transition-transform hover:border-primary/40 touch-manipulation focus-visible:outline-2 focus-visible:outline-primary"
   329→                aria-label={`${p.name}, ${xof(p.price)} — appuie une fois pour la fiche, deux fois pour l'ajouter au panier`}
   330→              >
   331→                {/* Burst double-tap — panier kente qui jaillit sous le doigt */}
   332→                {burst?.id === p.id && (
   333→                  <motion.span
   334→                    aria-hidden="true"
   335→                    initial={{ scale: 0.3, opacity: 0.95 }}
   336→                    animate={{ scale: 1.7, opacity: 0 }}
   337→                    transition={{ duration: 0.65, ease: "easeOut" }}
   338→                    className="pointer-events-none absolute z-20"
   339→                    style={{ left: burst.x, top: burst.y }}
   340→                  >
   341→                    <span className="grid place-items-center h-20 w-20 -ml-10 -mt-10 rounded-full bg-[#FFF9EC]/30 backdrop-blur-[2px] shadow-xl">
   342→                      <span className="grid place-items-center h-12 w-12 rounded-full bg-gradient-to-br from-[#A0522D] to-[#8B1A3B] shadow-lg">
   343→                        <ShoppingBag size={22} className="text-[#FFF9EC]" />
   344→                      </span>
   345→                    </span>
   346→                  </motion.span>
   347→                )}
   348→                <img src={p.image} alt={p.name} loading="lazy" className="aspect-square w-full object-cover" />
   349→                <div className="p-2.5">
   350→                  <p className="text-[13px] font-semibold leading-tight line-clamp-2 min-h-9">{p.name}</p>
   351→                  <p className="text-[10px] text-terre mt-1 truncate">{p.botanicals}</p>
   352→                  <div className="flex items-center justify-between mt-1.5">
   353→                    <span className="font-mono text-[13px] font-bold">{xof(p.price)}</span>
   354→                    <Stars rating={p.rating} size={9} />
   355→                  </div>
   356→                </div>
   357→              </button>
   358→              <FavButton variant="card" productId={p.id} productName={p.name} />
   359→            </div>
   360→          ))}
   361→        </div>
   362→      )}
   363→        </>
   364→      )}
   365→
   366→      {/* Barre panier sticky au-dessus de la nav */}
   367→      <AnimatePresence>
   368→        {cart.length > 0 && !payState && (
   369→          <motion.div initial={{ y: 80, opacity: 0 }} animate={{ y: 0, opacity: 1 }} exit={{ y: 80, opacity: 0 }} className="sticky bottom-[84px] z-20 mt-4">
   370→            <button
   371→              onClick={() => { setCheckout(true); haptic(HAPTIC.tap); }}
   372→              className="w-full h-14 rounded-2xl bg-gradient-to-r from-[#A0522D] to-[#8B1A3B] text-[#FFF9EC] shadow-xl flex items-center justify-between px-4 active:scale-[0.99] transition-transform focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary"
   373→              aria-label={`Panier ${cart.length} articles, total ${xof(subtotal)} — commander`}
   374→            >
   375→              <span className="flex items-center gap-2 text-sm font-semibold">
   376→                <span className="relative" aria-hidden="true">
   377→                  <ShoppingBag size={20} />
   378→                  <span className="absolute -top-1.5 -right-2 h-4 min-w-4 px-0.5 rounded-full bg-[#FFF9EC] text-[#A0520F] text-[9px] font-black grid place-items-center">{cart.reduce((s, l) => s + l.qty, 0)}</span>
   379→                </span>
   380→                <span aria-hidden="true">{cart.length} article{cart.length > 1 ? "s" : ""}</span>
   381→              </span>
   382→              <span className="font-mono font-black text-base" aria-hidden="true">{xof(subtotal)}</span>
   383→            </button>
   384→          </motion.div>
   385→        )}
   386→      </AnimatePresence>
   387→
   388→      {/* Sheet fiche produit */}
   389→      <Sheet open={!!detail} onOpenChange={(o) => !o && setDetail(null)}>
   390→        <SheetContent side="bottom" className="max-h-[88vh] rounded-t-3xl px-0 overflow-y-auto max-w-[560px] mx-auto">
   391→          {detail && (
   392→            <>
   393→              <div className="mx-auto sticky top-0 z-10 bg-card/95 backdrop-blur pt-2 pb-1">
   394→                <div className="h-1 w-10 rounded-full bg-muted mx-auto" aria-hidden="true" />
   395→              </div>
