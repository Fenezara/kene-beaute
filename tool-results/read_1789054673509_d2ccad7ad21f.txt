   124→export function ClientApp() {
   125→  const { t } = useT();
   126→  const user = useKene((s) => s.user);
   127→  // Écran courant étendu aux onglets cachés 71-c (cast typé, cf. ScreenTab).

   129→  const setClientTab = useKene((s) => s.setClientTab);
   130→  const cartCount = useKene((s) => s.cart.reduce((n, l) => n + l.qty, 0));
   131→  // Gate d'hydratation kene-store (contrat t. 63-b) : _keneHydrated passe à
   132→  // true quand la relecture localStorage est finie (onRehydrateStorage, même
   133→  // en cas d'erreur — la porte ne se verrouille jamais). Typage tolérant : si
   134→  // 63-b était absent, fallback true = comportement d'avant (zéro régression).
   135→  const hydrated = useKene((s) => (s as { _keneHydrated?: boolean })._keneHydrated ?? true);
   136→  // Badge chat honnête : faux par défaut — seul un message ENTRANT (événement
   137→  // « kene:chat:new » dispatché par ChatScreen) l'allume, voir l'effet plus bas
   138→  const [chatUnread, setChatUnread] = useState(false);
   139→  const [pendingZone, setPendingZone] = useState<BodyZone | null>(null);
   140→  const scrollRef = useRef<HTMLDivElement | null>(null);
   141→
   142→  // ─── Pull-to-refresh (Instagram / Wave) ───
   143→  const [pull, setPull] = useState(0);          // distance d'étirement (px, résistive)
   144→  const [refreshing, setRefreshing] = useState(false);
   145→  const [refreshKey, setRefreshKey] = useState(0);
   146→  const pullStart = useRef<{ y: number; x: number; atTop: boolean } | null>(null);
   147→  const wasRefreshing = useRef(false);
   148→
   149→  // ─── Swipe horizontal entre onglets (TikTok) ───
   150→  const swipeStart = useRef<{ x: number; y: number; ok: boolean } | null>(null);
   151→
   152→  // ─── Direction de transition (sens de navigation) + connectivité ───
   153→  const [navDir, setNavDir] = useState<1 | -1>(1);
   154→  const [online, setOnline] = useState(true);
   155→
   156→  // Bandeau hors-ligne — résilience réseau façon Wave
   157→  useEffect(() => {
   158→    const update = () => setOnline(isOnline());
   159→    update();
   160→    window.addEventListener("online", update);
   161→    window.addEventListener("offline", update);
   162→    return () => {
   163→      window.removeEventListener("online", update);
   164→      window.removeEventListener("offline", update);
   165→    };
   166→  }, []);
   167→
   168→  // File d'attente offline du diagnostic (t. 83-f) : les photos mises en
   169→  // attente partent TOUTES SEULES au retour du réseau — peu importe l'écran
   170→  // courant (la replay vit ici, toujours montée). Au boot : si la file n'est
   171→  // pas vide et qu'on est en ligne, elle part après 2,5 s (reprise d'app).
   172→  useEffect(() => {
   173→    let alive = true;
   174→    const flush = async () => {
   175→      const n = await replayDiagQueue();
   176→      if (!alive || n === 0) return;
   177→      haptic(HAPTIC.success);
   178→      toast.success(
   179→        n === 1 ? "Ta photo est partie — l'analyse suit son cours" : `${n} photos parties — les analyses suivent leur cours`,
   180→        { description: "Le résultat arrivera dans tes notifications." },
   181→      );
   182→    };
   183→    const onOnline = () => void flush();
   184→    window.addEventListener("online", onOnline);
   185→    const t = window.setTimeout(() => void flush(), 2500);
   186→    return () => {
   187→      alive = false;
   188→      window.removeEventListener("online", onOnline);
   189→      window.clearTimeout(t);
   190→    };
   191→  }, []);
   192→
   193→  // Relecture du store persisté (kene-store, skipHydration t. 63-b) — UNE fois
   194→  // au montage, idempotente (le store a aussi son propre filet « load ») :
   195→  // c'est ceci qui restaure l'état persisté (space/user/panier/onglet) après
   196→  // le premier rendu client — le BootSkeleton couvre exactement cette fenêtre.
   197→  useEffect(() => {
   198→    void useKene.persist.rehydrate();
   199→  }, []);
   200→
   201→  // ── Validation silencieuse de la session au boot ── DÉPLACÉE à la racine
   202→  // (t. 66-d) : le check vivait ici, or ClientApp n'est PAS monté quand

   209→  // QUE si tu n'es pas déjà sur l'onglet chat (lecture fraîche via getState,

   211→  useEffect(() => {

   213→      if (useKene.getState().clientTab !== "chat") setChatUnread(true);
   214→    };
   215→    window.addEventListener("kene:chat:new", onNewChatMessage);
   216→    return () => window.removeEventListener("kene:chat:new", onNewChatMessage);

   219→  // Remonte en haut du flux à chaque changement d'onglet (le tirage est

   221→  useEffect(() => {
   222→    scrollRef.current?.scrollTo({ top: 0 });
   223→  }, [tab]);

   225→  function goTab(t: ScreenTab) {
   226→    if (t !== tab) {
   227→      // Hors SWIPE_ORDER (écrans cachés) → pas de direction de transition,
   228→      // la motion.div du flux joue son fade par défaut.
   229→      const from = SWIPE_ORDER.indexOf(tab as ClientTab);
   230→      const to = SWIPE_ORDER.indexOf(t as ClientTab);
   231→      if (from >= 0 && to >= 0) setNavDir(to > from ? 1 : -1);
   232→      haptic(HAPTIC.tap);
   233→    }
   234→    if (t === "chat") setChatUnread(false);
   235→    setClientTab(t as ClientTab);
   236→  }
   237→
   238→  // ─── Gestes tactiles du conteneur de flux ───
   239→  const onTouchStart = (e: React.TouchEvent) => {
   240→    const el = scrollRef.current;
   241→    const t = e.touches[0];
   242→    pullStart.current = { y: t.clientY, x: t.clientX, atTop: !el || el.scrollTop <= 0 };
   243→    // le swipe est ignoré s'il démarre dans une rangée horizontale scrollable
   244→    const target = e.target as HTMLElement;
   245→    const inScrollRow = !!target.closest("[data-scroll-row]");
   246→    swipeStart.current = { x: t.clientX, y: t.clientY, ok: !inScrollRow };
   247→  };
   248→
   249→  const onTouchMove = (e: React.TouchEvent) => {
   250→    if (refreshing) return;
   251→    const s = pullStart.current;
   252→    if (!s || !s.atTop) return;
   253→    const t = e.touches[0];
   254→    const dy = t.clientY - s.y;
   255→    const dx = t.clientX - s.x;
   256→    if (dy <= 0 || Math.abs(dx) > Math.abs(dy)) return; // tirage vertical uniquement
   257→    setPull(Math.min(dy / 2.2, 96)); // résistif : l'icône s'alourdit en fin de course
   258→  };
   259→
   260→  const onTouchEnd = (e: React.TouchEvent) => {
   261→    // 1) fin de tirage → actualisation si seuil franchi
   262→    const dist = pull;
   263→    pullStart.current = null;
   264→    if (dist > 56) {
   265→      haptic(HAPTIC.medium);
   266→      setRefreshing(true);
   267→      setPull(0);
   268→      wasRefreshing.current = true;
   269→      setRefreshKey((k) => k + 1);
   270→    } else {
   271→      setPull(0);
   272→    }
   273→    // 2) swipe horizontal → onglet voisin
   274→    const s = swipeStart.current;
   275→    swipeStart.current = null;
   276→    if (!s || !s.ok || refreshing) return;
   277→    const t = e.changedTouches[0];
   278→    const dx = t.clientX - s.x;
   279→    const dy = t.clientY - s.y;
   280→    if (Math.abs(dx) < 64 || Math.abs(dx) < Math.abs(dy) * 1.8) return;
   281→    const idx = SWIPE_ORDER.indexOf(tab as ClientTab);
   282→    if (idx < 0) return;
   283→    const next = dx < 0 ? SWIPE_ORDER[idx + 1] : SWIPE_ORDER[idx - 1];
   284→    if (next) {
   285→      setNavDir(dx < 0 ? 1 : -1);
   286→      haptic(HAPTIC.tap);
   287→      setClientTab(next);
   288→    }
   289→  };
   290→
   291→  // Le fil signale la fin de son rechargement (déclenché par refreshKey)
   292→  const onRefreshed = useCallback(() => {
   293→    if (!wasRefreshing.current) return;
   294→    wasRefreshing.current = false;
   295→    setRefreshing(false);
   296→    haptic(HAPTIC.success);
   297→    toast.success("Fil actualisé", { description: `Mis à jour à ${formatTime(new Date())}` });
   298→  }, []);
   299→
   300→  const onScanZone = useCallback(
   301→    (z: BodyZone) => {
   302→      setPendingZone(z);
   303→      setClientTab("diagnostic");
   304→    },
   305→    [setClientTab],
   306→  );
   307→
   308→  const onZoneConsumed = useCallback(() => setPendingZone(null), []);
   309→
   310→  // Store persisté pas encore relu → squelette d'amorçage (évite le flash
   311→  // d'onboarding avant la restauration de la session — t. 63)
   312→  if (!hydrated) return <BootSkeleton />;
   313→
   314→  if (!user) {
   315→    // LE SEUIL (t. 74) — une seule page avant les portails : hero cinématique,

   317→    // natif puis OTP. L'ancienne chaîne (intro forcée → portes → saisie
   318→    // classique) est retirée : plus AUCUN écran obligatoire avant le choix.
   319→    return <WelcomeThreshold />;
   320→  }
   321→
   322→  const first = user.name.split(" ")[0];
   323→

   325→    <div className="h-dvh flex overflow-hidden">
   326→      {/* Atmosphère ÉCLAT 2026 — lueurs aurora derrière tout le shell (le fond

   333→        className="hidden md:flex w-[84px] xl:w-[248px] shrink-0 flex-col k-chrome"

   372→                className={cn(
   373→                  "relative flex flex-col xl:flex-row items-center justify-center xl:justify-start gap-1 xl:gap-3.5 py-2.5 xl:py-0 xl:h-12 xl:px-3 rounded-2xl transition-colors focus-visible:outline-2 focus-visible:outline-primary",
   374→                  active ? "bg-primary/12 text-primary" : "text-muted-foreground hover:bg-accent/60 hover:text-foreground",
   375→                )}

   378→                  {/* md : squircle teinté (IconBadge) ; xl : icône nue alignée au libellé */}
   379→                  <Icon className={cn("hidden xl:block size-[22px]", active && "font-bold")} />

   395→                </span>
   396→                <span className={cn("text-[10px] xl:text-[15px]", active ? "font-bold" : "font-medium")}>{t(n.labelKey)}</span>
   397→                {active && <motion.span layoutId="side-indicator" className="k-rail-line hidden xl:block absolute left-0 top-1/2 -translate-y-1/2 h-6 w-[3px] rounded-full" aria-hidden="true" />}

   403→        {/* Bas de sidebar : micro légal (l'isolation des comptes t. 69-a a
