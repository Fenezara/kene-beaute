   619→   619→function HistoryView({
   620→   620→  userId, history, compareMode, setCompareMode, compareSel, setCompareSel, onBack, onOpen,
   621→   621→}: {
   622→   622→  userId: string;
   623→   623→  history: ApiDiagnosis[] | null;
   624→   624→  compareMode: boolean;
   625→   625→  setCompareMode: (v: boolean) => void;
   626→   626→  compareSel: string[];
   627→   627→  setCompareSel: (v: string[]) => void;
   628→   628→  onBack: () => void;
   629→   629→  onOpen: (d: ApiDiagnosis) => void;
   630→   630→}) {
   631→   631→  const list = useMemo(
   632→   632→    () => [...(history ?? [])].filter((d) => d.status === "done").sort((a, b) => +new Date(b.createdAt) - +new Date(a.createdAt)),
   633→   633→    [history]
   634→   634→  );
   635→   635→  const sameZoneSel = compareSel.length === 2 && list.find((d) => d.id === compareSel[0])?.zone === list.find((d) => d.id === compareSel[1])?.zone;
   636→   636→  const a = list.find((d) => d.id === compareSel[0]);
   637→   637→  const b = list.find((d) => d.id === compareSel[1]);
   638→   638→  const ra = a ? parseDiagnosis(a.resultJson) : null;
   639→   639→  const rb = b ? parseDiagnosis(b.resultJson) : null;
   640→   640→
   641→   641→  const evolutions =
   642→   642→    ra && rb
   643→   643→      ? rb.indicateurs
   644→   644→          .map((ib) => {
   645→   645→            const ia = ra.indicateurs.find((x) => x.nom === ib.nom);
   646→   646→            return ia ? { nom: ib.nom, delta: ib.pourcentage - ia.pourcentage } : null;
   647→   647→          })
   648→   648→          .filter((x): x is { nom: string; delta: number } => x !== null)
   649→   649→          .sort((x, y) => Math.abs(y.delta) - Math.abs(x.delta))
   650→   650→          .slice(0, 3)
   651→   651→      : [];
   652→   652→
   653→   653→  return (
   654→   654→    <div className="pt-4 pb-2">
   655→   655→      <button onClick={onBack} className="inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground mb-4 focus-visible:outline-2 focus-visible:outline-primary rounded" aria-label="Retour">
   656→   656→        <ArrowLeft size={15} /> Nouveau diagnostic
   657→   657→      </button>
   658→   658→      <div className="flex items-center justify-between gap-2">
   659→   659→        <h1 className="font-heading font-black text-xl">Mon historique</h1>
   660→   660→        <button
   661→   661→          onClick={() => { setCompareMode(!compareMode); setCompareSel([]); }}
   662→   662→          aria-pressed={compareMode}
   663→   663→          className={`h-9 px-3 rounded-full text-[11px] font-bold flex items-center gap-1.5 active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${compareMode ? "bg-primary text-primary-foreground" : "border border-border bg-card"}`}
   664→   664→        >
   665→   665→          <GitCompareArrows size={14} /> Comparer
   666→   666→        </button>
   667→   667→      </div>
   668→   668→
   669→   669→      {history === null ? (
