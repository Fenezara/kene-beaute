     1→"use client";
     2→// Kènè Pro — Caisse POS : catalogue cliquable, ticket, paiement mobile money, ticket thermique imprimable
     3→import { useMemo, useState } from "react";
     4→import { motion, AnimatePresence } from "framer-motion";
     5→import { Check, Loader2, Minus, Plus, Printer, ReceiptText, Trash2, Wallet, User, UserRoundPlus, Sparkles, X } from "lucide-react";
     6→import { toast } from "sonner";
     7→import { Button } from "@/components/ui/button";
     8→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     9→import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle } from "@/components/ui/dialog";
    10→import { Input } from "@/components/ui/input";
    11→import { Label } from "@/components/ui/label";
    12→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
    13→import { Sheet, SheetContent, SheetDescription, SheetHeader, SheetTitle } from "@/components/ui/sheet";
    14→import { Skeleton } from "@/components/ui/skeleton";
    15→import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
    16→import { cn } from "@/lib/utils";
    17→import { apiGet, apiPost } from "@/lib/kene/api";
    18→import { xof, formatDate, formatTime } from "@/lib/kene/format";
    19→import { splitTVA } from "@/lib/accounting/syscohada";
    20→import { useApi } from "./useApi";
    21→import { EmptyState, ErrorState, KenteTop, SectionHeader } from "./ui-bits";
    22→import type { PaymentMethod, ProCatalog, ProClient, ProSale, SalesResponse } from "./types";
    23→
    24→interface TicketLine {
    25→  kind: "service" | "product";
    26→  id: string;
    27→  label: string;
    28→  unitPrice: number;
    29→  qty: number;
    30→}
    31→
    32→const PAY_METHODS: { code: PaymentMethod; label: string; cls: string }[] = [
    33→  { code: "wave", label: "Wave", cls: "bg-[#1DC8FF] text-[#062A33] hover:brightness-95" },
    34→  { code: "orange", label: "Orange Money", cls: "bg-[#FF7900] text-[#3A1D00] hover:brightness-95" },
    35→  { code: "cash", label: "Espèces", cls: "bg-success text-success-foreground hover:brightness-95" },
    36→  { code: "card", label: "Carte", cls: "bg-finance text-finance-foreground hover:brightness-95" },
    37→];
    38→
    39→const METHOD_LABELS: Record<string, string> = { wave: "Wave", orange: "Orange Money", cash: "Espèces", card: "Carte bancaire", wallet: "Wallet Kènè" };
    40→
    41→export function PosSection({ tenantId, tenantName, refreshKey = 0 }: { tenantId: string; tenantName: string; refreshKey?: number }) {
    42→  const [lines, setLines] = useState<TicketLine[]>([]);
    43→  const [discountInput, setDiscountInput] = useState("");
    44→  const [clientId, setClientId] = useState("");
    45→  const [busy, setBusy] = useState(false);
    46→  const [success, setSuccess] = useState(false);
    47→  const [ticket, setTicket] = useState<ProSale | null>(null);
    48→  const [sheetOpen, setSheetOpen] = useState(false);
    49→
    50→  // Cliente express — mode saisie allégé (2 champs) pour les praticiennes peu administratives
    51→  const [expressOpen, setExpressOpen] = useState(false);
    52→  const [expressName, setExpressName] = useState("");
    53→  const [expressPhone, setExpressPhone] = useState("");
    54→  const [expressBusy, setExpressBusy] = useState(false);
    55→
    56→  const catalog = useApi<ProCatalog>(() => (tenantId ? apiGet<ProCatalog>(`/api/pro/catalog?tenantId=${tenantId}`) : Promise.resolve({ services: [], products: [] })), [tenantId]);
    57→  const clients = useApi<ProClient[]>(
    58→    () => (tenantId ? apiGet<{ clients: ProClient[] }>(`/api/pro/clients?tenantId=${tenantId}`).then((r) => r.clients ?? []) : Promise.resolve([])),
    59→    [tenantId]
    60→  );
    61→  const sales = useApi<SalesResponse>(() => (tenantId ? apiGet<SalesResponse>(`/api/pro/sales?tenantId=${tenantId}`) : Promise.resolve({ sales: [] })), [tenantId, refreshKey]);
    62→
    63→  const discount = Math.max(0, Math.min(Number(discountInput) || 0, 1_000_000));
    64→  const subtotal = useMemo(() => lines.reduce((s, l) => s + l.unitPrice * l.qty, 0), [lines]);
    65→  const total = Math.max(0, subtotal - Math.min(discount, subtotal));
    66→
    67→  function addLine(kind: "service" | "product", id: string, label: string, unitPrice: number) {
    68→    setLines((ls) => {
    69→      const ex = ls.find((l) => l.kind === kind && l.id === id);
    70→      if (ex) return ls.map((l) => (l === ex ? { ...l, qty: l.qty + 1 } : l));
    71→      return [...ls, { kind, id, label, unitPrice, qty: 1 }];
    72→    });
    73→  }
    74→  function setQty(i: number, delta: number) {
    75→    setLines((ls) =>
    76→      ls.map((l, idx) => (idx === i ? { ...l, qty: l.qty + delta } : l)).filter((l) => l.qty > 0)
    77→    );
    78→  }
    79→
    80→  async function createExpressClient() {
    81→    const name = expressName.trim().replace(/\s+/g, " ");
    82→    const phone = expressPhone.trim();
    83→    if (name.length < 2) {
    84→      toast.error("Le nom est trop court");
    85→      return;
    86→    }
    87→    if (phone.replace(/\D/g, "").length < 8) {
    88→      toast.error("Téléphone invalide — au moins 8 chiffres");
    89→      return;
    90→    }
    91→    setExpressBusy(true);
    92→    try {
    93→      const r = await apiPost<{ client: ProClient; reused: boolean }>("/api/pro/clients", { tenantId, name, phone });
    94→      setClientId(r.client.id);
    95→      setExpressOpen(false);
    96→      setExpressName("");
    97→      setExpressPhone("");
    98→      void clients.refetch();
    99→      toast.success(
   100→        r.reused
   101→          ? `${r.client.name} existait déjà — fiche réutilisée`
   102→          : `${r.client.name} ajoutée au carnet clientes`
   103→      );
   104→    } catch (e) {
   105→      toast.error(e instanceof Error ? e.message : "Création impossible");
   106→    } finally {
   107→      setExpressBusy(false);
   108→    }
   109→  }
   110→
   111→  async function pay(method: PaymentMethod) {
   112→    if (lines.length === 0) {
   113→      toast.error("Le ticket est vide");
   114→      return;
   115→    }
   116→    setBusy(true);
   117→    try {
   118→      const res = await apiPost<{ sale: ProSale }>("/api/pro/sales", {
   119→        tenantId,
   120→        items: lines.map((l) => ({ kind: l.kind, id: l.id, qty: l.qty })),
   121→        paymentMethod: method,
   122→        clientProfileId: clientId || undefined,
   123→        discount: Math.min(discount, subtotal),
   124→      });
   125→      const sale: ProSale = {
   126→        ...res.sale,
   127→        items: res.sale.items ?? lines.map((l) => ({ label: l.label, qty: l.qty, unitPrice: l.unitPrice, total: l.unitPrice * l.qty, kind: l.kind })),
   128→        discount: res.sale.discount ?? Math.min(discount, subtotal),
   129→        subtotal: res.sale.subtotal ?? subtotal,
   130→      };
   131→      setSuccess(true);
   132→      window.setTimeout(() => {
   133→        setTicket(sale);
   134→        setSheetOpen(true);
   135→        setSuccess(false);
   136→        setLines([]);
   137→        setDiscountInput("");
   138→        setClientId("");
   139→        void sales.refetch();
   140→      }, 1300);
   141→    } catch (e) {
   142→      toast.error(e instanceof Error ? e.message : "Encaissement impossible");
   143→    } finally {
   144→      setBusy(false);
   145→    }
   146→  }
   147→
   148→  return (
   149→    <div className="space-y-4">
   150→      <SectionHeader title="Caisse" sub="Encaissement soins & produits — Wave, Orange Money, espèces, carte" />
   151→
   152→      <div className="grid lg:grid-cols-[1fr_400px] gap-4 items-start">
   153→        {/* ── Catalogue ── */}
   154→        <Card className="overflow-hidden pt-0">
   155→          <KenteTop />
   156→          <CardContent className="p-4">
   157→            {catalog.error && !catalog.data ? (
   158→              <ErrorState message={`Catalogue indisponible : ${catalog.error}`} onRetry={catalog.refetch} />
   159→            ) : catalog.loading && !catalog.data ? (
   160→              <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   161→                {Array.from({ length: 9 }).map((_, i) => (
   162→                  <Skeleton key={i} className="h-20" />
   163→                ))}
   164→              </div>
   165→            ) : (
   166→              <Tabs defaultValue="service">
   167→                <TabsList className="mb-3">
   168→                  <TabsTrigger value="service" className="text-xs gap-1.5">Soins</TabsTrigger>
   169→                  <TabsTrigger value="product" className="text-xs gap-1.5">Produits</TabsTrigger>
   170→                </TabsList>
   171→                <TabsContent value="service">
   172→                  {catalog.data?.services.length ? (
   173→                    <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   174→                      {catalog.data.services.filter((s) => s.active).map((s) => (
   175→                        <button
   176→                          key={s.id}
   177→                          onClick={() => addLine("service", s.id, s.name, s.price)}
   178→                          className="group rounded-xl border border-border bg-card p-3 text-left transition-all hover:border-gold hover:shadow-md hover:-translate-y-0.5 focus-visible:outline-2 focus-visible:outline-ring"
   179→                        >
   180→                          <p className="text-sm font-medium leading-tight line-clamp-2 group-hover:text-primary">{s.name}</p>
   181→                          <p className="mt-1 text-[11px] text-muted-foreground">{s.durationMin} min</p>
   182→                          <p className="mt-1 font-mono text-sm font-semibold text-gold">{xof(s.price)}</p>
   183→                        </button>
   184→                      ))}
   185→                    </div>
   186→                  ) : (
   187→                    <EmptyState label="Aucun soin actif au catalogue" />
   188→                  )}
   189→                </TabsContent>
   190→                <TabsContent value="product">
   191→                  {catalog.data?.products.length ? (
   192→                    <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   193→                      {catalog.data.products.filter((p) => p.active).map((p) => (
   194→                        <button
   195→                          key={p.id}
   196→                          onClick={() => addLine("product", p.id, p.name, p.price)}
   197→                          className="group rounded-xl border border-border bg-card p-3 text-left transition-all hover:border-gold hover:shadow-md hover:-translate-y-0.5 focus-visible:outline-2 focus-visible:outline-ring"
   198→                        >
   199→                          <p className="text-sm font-medium leading-tight line-clamp-2 group-hover:text-primary">{p.name}</p>
   200→                          <p className={cn("mt-1 text-[11px]", p.stock <= p.stockAlert ? "text-bissap font-medium" : "text-muted-foreground")}>
   201→                            Stock : {p.stock}
   202→                          </p>
   203→                          <p className="mt-1 font-mono text-sm font-semibold text-gold">{xof(p.price)}</p>
   204→                        </button>
   205→                      ))}
   206→                    </div>
   207→                  ) : (
   208→                    <EmptyState label="Aucun produit POS actif" />
   209→                  )}
   210→                </TabsContent>
   211→              </Tabs>
   212→            )}
   213→          </CardContent>
   214→        </Card>
   215→
   216→        {/* ── Ticket ── */}
   217→        <Card className="lg:sticky lg:top-24 overflow-hidden pt-0 relative">
   218→          <KenteTop />
   219→          <CardHeader className="pb-2">
   220→            <CardTitle className="font-heading text-base flex items-center gap-2">
   221→              <ReceiptText className="size-4 text-primary" aria-hidden="true" /> Ticket en cours
   222→            </CardTitle>
   223→          </CardHeader>
   224→          <CardContent className="space-y-3">
   225→            {lines.length === 0 ? (
   226→              <p className="rounded-lg border border-dashed border-border px-3 py-6 text-center text-xs text-muted-foreground">
   227→                Cliquez sur un soin ou produit pour l&apos;ajouter au ticket.
   228→              </p>
   229→            ) : (
   230→              <ul className="space-y-1.5 max-h-56 overflow-y-auto pretty-scroll pr-1">
   231→                <AnimatePresence initial={false}>
   232→                  {lines.map((l, i) => (
   233→                    <motion.li
   234→                      key={`${l.kind}-${l.id}`}
   235→                      layout
   236→                      initial={{ opacity: 0, x: 12 }}
   237→                      animate={{ opacity: 1, x: 0 }}
   238→                      exit={{ opacity: 0, x: -12 }}
   239→                      className="flex items-center gap-2 rounded-lg bg-muted/60 px-2.5 py-1.5"
   240→                    >
   241→                      <div className="min-w-0 flex-1">
   242→                        <p className="truncate text-xs font-medium">{l.label}</p>
   243→                        <p className="font-mono text-[10px] text-muted-foreground">{xof(l.unitPrice)} × {l.qty}</p>
   244→                      </div>
   245→                      <div className="flex items-center gap-1">
   246→                        <Button variant="ghost" size="icon" className="size-6" onClick={() => setQty(i, -1)} aria-label={`Retirer un ${l.label}`}>
   247→                          <Minus className="size-3" />
   248→                        </Button>
   249→                        <span className="w-5 text-center font-mono text-xs font-semibold tabular-nums">{l.qty}</span>
   250→                        <Button variant="ghost" size="icon" className="size-6" onClick={() => setQty(i, 1)} aria-label={`Ajouter un ${l.label}`}>
   251→                          <Plus className="size-3" />
   252→                        </Button>
   253→                      </div>
   254→                      <span className="w-20 shrink-0 text-right font-mono text-xs font-semibold tabular-nums">{xof(l.unitPrice * l.qty)}</span>
   255→                      <Button variant="ghost" size="icon" className="size-6 text-muted-foreground hover:text-bissap" onClick={() => setQty(i, -l.qty)} aria-label={`Supprimer ${l.label}`}>
   256→                        <Trash2 className="size-3" />
   257→                      </Button>
   258→                    </motion.li>
   259→                  ))}
   260→                </AnimatePresence>
   261→              </ul>
   262→            )}
   263→
   264→            <div className="space-y-1">
   265→              <Label htmlFor="pos-client" className="text-[11px] text-muted-foreground flex items-center gap-1">
   266→                <User className="size-3" aria-hidden="true" /> Cliente (optionnel)
   267→              </Label>
   268→              <div className="flex gap-1.5">
   269→                <Select value={clientId || undefined} onValueChange={(v) => setClientId(v === "__none" ? "" : v)}>
   270→                  <SelectTrigger id="pos-client" className="h-8 text-xs flex-1 min-w-0">
   271→                    <SelectValue placeholder="Sans cliente" />
   272→                  </SelectTrigger>
   273→                  <SelectContent>
   274→                    <SelectItem value="__none">Sans cliente</SelectItem>
   275→                    {(clients.data ?? []).map((c) => (
   276→                      <SelectItem key={c.id} value={c.id}>{c.name} — {c.phone}</SelectItem>
   277→                    ))}
   278→                  </SelectContent>
   279→                </Select>
   280→                <button
   281→                  onClick={() => setExpressOpen(true)}
   282→                  aria-label="Cliente express — créer une fiche en 2 champs"
   283→                  title="Cliente express — 2 champs"
   284→                  className="h-8 px-2.5 shrink-0 rounded-md border border-gold/50 text-gold-text text-[11px] font-bold flex items-center gap-1.5 hover:bg-gold/10 active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-ring"
   285→                >
   286→                  <UserRoundPlus size={14} aria-hidden="true" /> <span className="hidden sm:inline">Express</span>
   287→                </button>
   288→              </div>
   289→            </div>
   290→
   291→            {/* Cliente express — Dialog 2 champs (mode saisie allégé) */}
   292→            <Dialog open={expressOpen} onOpenChange={setExpressOpen}>
   293→              <DialogContent className="max-w-[360px] rounded-2xl p-5 gap-4">
   294→                <DialogHeader className="space-y-1.5 text-left">
   295→                  <DialogTitle className="font-heading font-black text-base flex items-center gap-2">
   296→                    <UserRoundPlus size={17} className="text-gold" aria-hidden="true" /> Cliente express
   297→                  </DialogTitle>
   298→                  <DialogDescription className="text-xs leading-relaxed">
   299→                    Une walk-in qui n&apos;est pas au carnet ? Deux champs suffisent — le CRM se remplit tout seul ensuite.
   300→                  </DialogDescription>
   301→                </DialogHeader>
   302→                <div className="space-y-3">
   303→                  <div className="space-y-1.5">
   304→                    <Label htmlFor="express-name" className="text-xs">Nom complet</Label>
   305→                    <Input
   306→                      id="express-name"
   307→                      value={expressName}
   308→                      onChange={(e) => setExpressName(e.target.value)}
   309→                      placeholder="Aïcha Bakayoko"
   310→                      className="h-10"
   311→                      autoFocus
   312→                      maxLength={80}
   313→                      onKeyDown={(e) => e.key === "Enter" && createExpressClient()}
   314→                    />
   315→                  </div>
   316→                  <div className="space-y-1.5">
   317→                    <Label htmlFor="express-phone" className="text-xs">Téléphone</Label>
   318→                    <Input
   319→                      id="express-phone"
   320→                      value={expressPhone}
   321→                      onChange={(e) => setExpressPhone(e.target.value)}
   322→                      inputMode="tel"
   323→                      placeholder="07 07 07 07 07"
   324→                      className="h-10"
   325→                      maxLength={20}
   326→                      onKeyDown={(e) => e.key === "Enter" && createExpressClient()}
   327→                    />
   328→                    <p className="text-[10px] text-muted-foreground">Sert aux relances WhatsApp — saisi une seule fois.</p>
   329→                  </div>
   330→                </div>
   331→                <DialogFooter className="gap-2 sm:justify-between">
   332→                  <Button
   333→                    variant="outline"
   334→                    size="sm"
   335→                    onClick={() => setExpressOpen(false)}
   336→                    disabled={expressBusy}
   337→                    className="rounded-lg"
   338→                  >
   339→                    <X size={14} aria-hidden="true" /> Annuler
   340→                  </Button>
   341→                  <Button
   342→                    size="sm"
   343→                    onClick={createExpressClient}
   344→                    disabled={expressBusy}
   345→                    className="rounded-lg bg-gold text-gold-text hover:bg-gold/90 font-bold"
   346→                  >
   347→                    {expressBusy ? <Loader2 size={14} className="animate-spin" aria-hidden="true" /> : <Check size={14} aria-hidden="true" />}
   348→                    Créer et encaisser
   349→                  </Button>
   350→                </DialogFooter>
   351→              </DialogContent>
   352→            </Dialog>
   353→
   354→            <div className="space-y-1.5 border-t border-dashed border-border pt-2 text-sm">
   355→              <div className="flex justify-between text-xs text-muted-foreground">
   356→                <span>Sous-total</span>
   357→                <span className="font-mono tabular-nums">{xof(subtotal)}</span>
   358→              </div>
   359→              <div className="flex items-center justify-between gap-2 text-xs">
   360→                <Label htmlFor="pos-remise" className="text-muted-foreground">Remise (FCFA)</Label>
   361→                <Input
   362→                  id="pos-remise"
   363→                  inputMode="numeric"
   364→                  value={discountInput}
   365→                  onChange={(e) => setDiscountInput(e.target.value.replace(/[^0-9]/g, ""))}
   366→                  placeholder="0"
   367→                  className="h-7 w-28 text-right font-mono text-xs"
   368→                />
   369→              </div>
   370→              <div className="flex items-baseline justify-between">
   371→                <span className="font-heading font-bold">TOTAL</span>
   372→                <span className="font-mono text-xl font-bold text-gold tabular-nums">{xof(total)}</span>
   373→              </div>
   374→            </div>
   375→
   376→            <div className="grid grid-cols-2 gap-2">
   377→              {PAY_METHODS.map((m) => (
   378→                <button
   379→                  key={m.code}
   380→                  disabled={busy || lines.length === 0}
   381→                  onClick={() => pay(m.code)}
   382→                  className={cn(
   383→                    "flex min-h-11 items-center justify-center gap-1.5 rounded-xl px-3 py-2.5 text-xs font-semibold shadow-sm transition-all active:scale-95 disabled:opacity-40 disabled:pointer-events-none",
   384→                    m.cls
   385→                  )}
   386→                >
   387→                  {m.code === "wave" || m.code === "orange" ? <Wallet className="size-4" aria-hidden="true" /> : m.code === "cash" ? <ReceiptText className="size-4" aria-hidden="true" /> : <Sparkles className="size-4" aria-hidden="true" />}
   388→                  {m.label}
   389→                </button>
   390→              ))}
   391→            </div>
   392→          </CardContent>
   393→
   394→          {/* Overlay succès */}
   395→          <AnimatePresence>
   396→            {success && (
   397→              <motion.div
   398→                initial={{ opacity: 0 }}
   399→                animate={{ opacity: 1 }}
   400→                exit={{ opacity: 0 }}
   401→                className="absolute inset-0 z-20 grid place-items-center rounded-xl bg-card/95"
   402→                role="status"
   403→                aria-live="polite"
   404→              >
   405→                <motion.div initial={{ scale: 0.4, rotate: -12 }} animate={{ scale: 1, rotate: 0 }} transition={{ type: "spring", stiffness: 260, damping: 16 }} className="flex flex-col items-center gap-2">
   406→                  <span className="grid size-16 place-items-center rounded-full bg-success text-success-foreground shadow-lg">
   407→                    <Check className="size-9" strokeWidth={3} aria-hidden="true" />
   408→                  </span>
   409→                  <p className="font-heading font-bold text-success">Encaissé !</p>
   410→                  <p className="font-mono text-sm font-semibold">{xof(total)}</p>
   411→                </motion.div>
   412→              </motion.div>
   413→            )}
   414→          </AnimatePresence>
   415→        </Card>
   416→      </div>
   417→
   418→      {/* ── Ventes récentes ── */}
   419→      <Card>
   420→        <CardHeader className="pb-2">
   421→          <CardTitle className="font-heading text-base">Ventes récentes</CardTitle>
   422→        </CardHeader>
   423→        <CardContent className="p-0">
   424→          {sales.loading && !sales.data ? (
   425→            <div className="space-y-2 p-4">
   426→              {Array.from({ length: 3 }).map((_, i) => (
   427→                <Skeleton key={i} className="h-10" />
   428→              ))}
   429→            </div>
   430→          ) : (sales.data?.sales ?? []).length === 0 ? (
   431→            <EmptyState label="Aucune vente enregistrée" sub="Les encaissements POS apparaîtront ici." />
   432→          ) : (
   433→            <ul className="divide-y divide-border/70">
   434→              {(sales.data?.sales ?? []).slice(0, 5).map((s) => (
   435→                <li key={s.id} className="flex items-center gap-3 px-4 py-2.5">
   436→                  <span className="font-mono text-[10px] text-muted-foreground w-24 shrink-0 tabular-nums">
   437→                    {formatDate(s.createdAt, { day: "2-digit", month: "2-digit" })} {formatTime(s.createdAt)}
