    37→export function ProfileScreen() {
    38→  const user = useKene((s) => s.user) as ClientUser;
    39→  const setUser = useKene((s) => s.setUser);
    40→  const setSpace = useKene((s) => s.setSpace);
    41→  const setClientTab = useKene((s) => s.setClientTab);
    42→  const clearCart = useKene((s) => s.clearCart);
    43→  const { t, lang, setLang } = useT();
    44→
    45→  const [edit, setEdit] = useState(false);
    46→  const [name, setName] = useState(user.name);
    47→  const [city, setCity] = useState(user.city ?? "");
    48→  const [savingId, setSavingId] = useState(false);
    49→
    50→  const [fitz, setFitz] = useState(user.fitzpatrick ?? "V");
    51→  const [skinType, setSkinType] = useState(user.skinType ?? "mixte");
    52→  const [allergies, setAllergies] = useState(user.allergies ?? "");
    53→  const [goals, setGoals] = useState<string[]>(() => {
    54→    try {
    55→      return (JSON.parse(user.goals ?? "[]") as { id: string }[]).map((g) => g.id);
    56→    } catch {
    57→      return [];
    58→    }
    59→  });
    60→  const [savingSkin, setSavingSkin] = useState(false);
    61→
    62→  const [wallet, setWallet] = useState<ApiWallet | null>(null);
    63→  const [txs, setTxs] = useState<ApiWalletTx[] | null>(null);
    64→  const [topup, setTopup] = useState(false);
    65→  const [amount, setAmount] = useState(5000);
    66→  const [method, setMethod] = useState<"wave" | "orange">("wave");
    67→  const [topupState, setTopupState] = useState<"idle" | "processing" | "done">("idle");
    68→  const [topupBusy, setTopupBusy] = useState(false);
    69→  const [exportBusy, setExportBusy] = useState(false);
    70→
    71→  // Installation PWA — même source d'événement que la bannière d'accueil.
    72→  const { canInstall, promptInstall, isStandalone, isIOS } = useInstallPrompt();
    73→
    74→  // Sécurité renforcée (2FA-lite) : code SMS exigé avant chaque paiement.
    75→  const secureEnabled = useSecurity((s) => s.enabled);
    76→  const setSecureEnabled = useSecurity((s) => s.setEnabled);
    77→
    78→  const loadWallet = useCallback(() => {
    79→    apiGet<{ wallet: ApiWallet; transactions: ApiWalletTx[] }>(`/api/wallet?userId=${user.id}`)
    80→      .then((r) => { setWallet(r.wallet); setTxs(r.transactions ?? []); })
    81→      .catch(() => setTxs([]));
    82→  }, [user.id]);
    83→
    84→  useEffect(() => {
    85→    loadWallet();
    86→  }, [loadWallet]);
    87→
    88→  /** Portabilité RGPD — télécharge « mes données » en JSON via blob */
    89→  async function downloadMyData() {
    90→    setExportBusy(true);
    91→    try {
    92→      const res = await fetch(`/api/profile/export?userId=${encodeURIComponent(user.id)}`);
    93→      if (!res.ok) throw new Error("Export impossible");
    94→      const blob = await res.blob();
    95→      const cd = res.headers.get("Content-Disposition") ?? "";
    96→      const m = /filename="([^"]+)"/.exec(cd);
    97→      const url = URL.createObjectURL(blob);
    98→      const a = document.createElement("a");
    99→      a.href = url;
   100→      a.download = m?.[1] ?? `kene-mes-donnees-${new Date().toISOString().slice(0, 10)}.json`;
   101→      document.body.appendChild(a);
   102→      a.click();
   103→      a.remove();
   104→      setTimeout(() => URL.revokeObjectURL(url), 4000);
   105→      const kb = Math.max(1, Math.round(blob.size / 1024));
   106→      toast.success(`Mes données téléchargées · ${kb} Ko`);
   107→    } catch (e) {
   108→      toast.error(e instanceof Error ? e.message : "Export impossible");
   109→    } finally {
   110→      setExportBusy(false);
   111→    }
   112→  }
   113→
   114→  /** Installer l'app (prompt natif, ou instructions iOS / navigateur). */
   115→  async function installApp() {
   116→    if (canInstall) {
   117→      const accepted = await promptInstall();
   118→      if (accepted) toast.success("Kènè installée sur ton écran d'accueil 💛");
   119→      return;
   120→    }
   121→    if (isIOS) {
   122→      toast.info("Installer sur iPhone", {
   123→        description: "Bouton Partager ⬆️ puis « Sur l'écran d'accueil ».",
   124→        duration: 8000,
   125→      });
   126→      return;
   127→    }
   128→    toast.info("Installation manuelle", {
   129→      description: "Menu du navigateur → « Installer l'application » ou « Ajouter à l'écran d'accueil ».",
   130→      duration: 8000,
   131→    });
   132→  }
   133→
   134→  async function saveIdentity() {
   135→    setSavingId(true);
   136→    try {
   137→      const r = await apiPatch<{ user: ApiUser }>("/api/auth/profile", { userId: user.id, name: name.trim() || undefined, city: city.trim() || undefined });
   138→      setUser({ ...user, ...r.user } as SessionUser);
   139→      setEdit(false);
   140→      toast.success("Profil mis à jour");
   141→    } catch (e) {
   142→      toast.error(e instanceof Error ? e.message : "Mise à jour impossible");
   143→    } finally {
   144→      setSavingId(false);
   145→    }
   146→  }
   147→
   148→  async function saveSkin() {
   149→    setSavingSkin(true);
   150→    try {
   151→      const r = await apiPatch<{ user: ApiUser }>("/api/auth/profile", {
   152→        userId: user.id,
   153→        skinType,
   154→        fitzpatrick: fitz,
   155→        allergies: allergies.trim() || undefined,
   156→        goals: goals.map((id) => ({ id, label: SKIN_GOALS.find((g) => g.id === id)?.label ?? id })),
   157→      });
   158→      setUser({ ...user, ...r.user } as SessionUser);
   159→      toast.success("Profil peau mis à jour");
   160→    } catch (e) {
   161→      toast.error(e instanceof Error ? e.message : "Mise à jour impossible");
   162→    } finally {
   163→      setSavingSkin(false);
   164→    }
   165→  }
   166→
   167→  /** Langue de l'interface (i18n) — indépendante de la langue de lecture vocale. */
   168→  function selectLang(l: Lang) {
   169→    if (l === lang) return;
   170→    setLang(l);
   171→    const label = LANGS.find((x) => x.id === l)?.label ?? "";
   172→    toast.success(`Interface en ${label.toLowerCase()}`);
   173→  }
   174→
   175→  /** Sécurité renforcée : bascule la re-vérification par code avant paiement. */
   176→  function toggleSecure() {
   177→    const next = !secureEnabled;
   178→    setSecureEnabled(next);
   179→    if (next) toast.success("Sécurité renforcée activée");
   180→    else toast("Sécurité renforcée désactivée");
   181→  }
   182→
   183→  async function runTopup() {
   184→    setTopupBusy(true);
   185→    setTopupState("processing");
   186→    try {
   187→      const r = await apiPost<{ payment: { id: string; confirmToken?: string } }>("/api/wallet/topup", { userId: user.id, amount, method });
   188→      // Contrat confirmToken (63-b/63-c) : un approvisionnement mobile money en
   189→      // attente porte son jeton — absent, on n'appelle JAMAIS confirm et on
   190→      // revient au formulaire comme après un échec (aucun crédit fantôme).
   191→      const confirmToken = r.payment.confirmToken;
   192→      if (!confirmToken) {
   193→        setTopupState("idle");
   194→        toast.error("Paiement impossible — réessaie dans quelques instants");
   195→        return;
   196→      }
   197→      await new Promise((res) => setTimeout(res, 2600));
   198→      await apiPost("/api/payments/confirm", { paymentId: r.payment.id, confirmToken });
   199→      setTopupState("done");
   200→      toast.success(`${xof(amount)} crédités sur ton wallet`);
   201→      loadWallet();
   202→    } catch (e) {
   203→      setTopupState("idle");
   204→      toast.error(e instanceof Error ? e.message : "Approvisionnement impossible");
   205→    } finally {
   206→      setTopupBusy(false);
   207→    }
   208→  }
   209→
   210→  return (
   211→    <div className="pt-4 pb-2 flex flex-col gap-6">
   212→      <button onClick={() => setClientTab("accueil")} className="self-start inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground focus-visible:outline-2 focus-visible:outline-primary rounded min-h-10 px-1" aria-label={t("profile.back.aria")}>
   213→        <ArrowLeft size={15} /> {t("tab.home")}
   214→      </button>
   215→
   216→      {/* Identité */}
   217→      <section aria-labelledby="me-t" className="rounded-3xl border border-border bg-card shadow-sm overflow-hidden">
   218→        <div className="kente-band h-1.5 w-full" aria-hidden="true" />
   219→        <div id="me-t" className="p-5">
   220→          <div className="flex items-center gap-4">
   221→            <span className="grid place-items-center h-16 w-16 rounded-full bg-gradient-to-br from-[#C8951E] to-[#A0522D] text-[#FFF9EC] font-heading font-black text-2xl shadow">
   222→              {user.name.charAt(0)}
   223→            </span>
   224→            <div className="min-w-0 flex-1">
   225→              <p className="font-heading font-black text-lg leading-tight truncate">{user.name}</p>
   226→              <p className="text-xs text-muted-foreground flex items-center gap-1.5 mt-1 font-mono"><Phone size={12} /> {user.phone}</p>
   227→              <p className="text-xs text-muted-foreground flex items-center gap-1.5 mt-0.5"><MapPin size={12} /> {user.city || "Ville non renseignée"}</p>
   228→            </div>
   229→            <button onClick={() => setEdit((v) => !v)} aria-label="Modifier mon profil" className="h-10 w-10 grid place-items-center rounded-full border border-border text-muted-foreground hover:text-primary active:scale-90 transition-all focus-visible:outline-2 focus-visible:outline-primary">
   230→              <Pencil size={16} />
   231→            </button>
   232→          </div>
   233→          {edit && (
   234→            <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: "auto" }} className="overflow-hidden">

   236→                <div>
   237→                  <label htmlFor="p-name" className="text-[11px] font-semibold text-muted-foreground">Prénom & nom</label>
   238→                  <input id="p-name" value={name} onChange={(e) => setName(e.target.value)} className="mt-1 h-11 w-full rounded-xl border border-border bg-background px-3 text-sm focus-visible:outline-2 focus-visible:outline-primary" />
   239→                </div>
   240→                <div>
   241→                  <label htmlFor="p-city" className="text-[11px] font-semibold text-muted-foreground">Ville</label>
   242→                  <input id="p-city" value={city} onChange={(e) => setCity(e.target.value)} placeholder="Abidjan" className="mt-1 h-11 w-full rounded-xl border border-border bg-background px-3 text-sm focus-visible:outline-2 focus-visible:outline-primary" />
   243→                </div>
   244→                <button onClick={saveIdentity} disabled={savingId} className="h-11 w-full rounded-xl bg-primary text-primary-foreground text-sm font-bold flex items-center justify-center gap-2 active:scale-[0.98] transition-transform disabled:opacity-60 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary">
   245→                  {savingId ? <Loader2 size={16} className="animate-spin" /> : <Check size={16} />} Enregistrer
   246→                </button>
   247→              </div>
   248→            </motion.div>
   249→          )}
   250→        </div>
   251→      </section>
   252→
   253→      {/* Profil peau */}
   254→      <section aria-labelledby="skin-t">
   255→        <SectionTitle icon={<Sparkles size={16} />}><span id="skin-t">Mon profil peau</span></SectionTitle>
   256→        <div className="rounded-2xl border border-border bg-card p-4 space-y-4">
   257→          <div>
   258→            <p className="text-[11px] font-semibold text-muted-foreground mb-2">Phototype Fitzpatrick</p>
   259→            <div className="grid grid-cols-3 gap-2">
   260→              {FITZPATRICK_CARDS.map((f) => (
   261→                <button key={f.id} onClick={() => setFitz(f.id)} aria-pressed={fitz === f.id} className={`rounded-xl p-1.5 border-2 text-left active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${fitz === f.id ? "border-primary" : "border-transparent"}`}>
   262→                  <div className="h-8 rounded-lg mb-1" style={{ background: f.gradient }} aria-hidden="true" />
   263→                  <p className="text-[11px] font-bold">{f.id}</p>

   265→              ))}
   266→            </div>
   267→          </div>
   268→          <div>
   269→            <p className="text-[11px] font-semibold text-muted-foreground mb-2">Type de peau</p>

   271→              {SKIN_TYPES.map((t) => (
   272→                <button key={t.id} onClick={() => setSkinType(t.id)} aria-pressed={skinType === t.id} className={`rounded-full px-3 min-h-10 text-xs font-medium active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${skinType === t.id ? "bg-primary text-primary-foreground" : "border border-border"}`}>{t.label}</button>
   273→              ))}
   274→            </div>

   276→          <div>

   279→              {SKIN_GOALS.map((g) => (
   280→                <button key={g.id} onClick={() => setGoals((s) => (s.includes(g.id) ? s.filter((x) => x !== g.id) : [...s, g.id]))} aria-pressed={goals.includes(g.id)} className={`rounded-full px-3 min-h-10 text-xs font-medium active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${goals.includes(g.id) ? "bg-primary text-primary-foreground" : "border border-border"}`}>{g.label}</button>
   281→              ))}
   282→            </div>
   283→          </div>
   284→          <div>
   285→            <label htmlFor="p-all" className="text-[11px] font-semibold text-muted-foreground">Allergies</label>

   288→          <button onClick={saveSkin} disabled={savingSkin} className="h-11 w-full rounded-xl border border-primary/60 text-primary text-sm font-bold flex items-center justify-center gap-2 active:scale-[0.98] transition-transform disabled:opacity-60 focus-visible:outline-2 focus-visible:outline-primary">
   289→            {savingSkin ? <Loader2 size={15} className="animate-spin" /> : <Check size={15} />} Mettre à jour mon profil peau
   290→          </button>
   291→        </div>
   292→      </section>
   293→
   294→      {/* Langue de l'interface — i18n UI, indépendante de la lecture vocale
   295→          (la langue TTS se règle dans les pilules du résumé vocal, accueil) */}
   296→      <section aria-labelledby="lang-t" className="rounded-2xl border border-border bg-card p-4">
   297→        <div className="flex items-center gap-3">
   298→          <span className="grid place-items-center h-10 w-10 rounded-xl bg-primary/15 text-primary shrink-0">
   299→            <Languages size={19} />
   300→          </span>
   301→          <div className="flex-1 min-w-0">
   302→            <p id="lang-t" className="text-xs font-bold">{t("lang.selector.label")}</p>

   304→          </div>
   305→        </div>
   306→        <div className="mt-3 grid grid-cols-2 gap-2">
   307→          {LANGS.map((l) => {

   309→            return (
   310→              <button
   311→                key={l.id}
   312→                onClick={() => selectLang(l.id)}

   337→          <div aria-hidden="true" className="absolute inset-0 bogolan-dots opacity-20" />
   338→          <div className="relative">
   339→            <p className="text-[10px] uppercase tracking-[0.18em] opacity-70">Solde disponible</p>
   340→            <p className="font-mono font-black text-3xl mt-1">{wallet ? xof(wallet.balance) : "···"}</p>

   342→              <BadgeCheck size={13} className="text-[#C8951E]" /> Cashback {Math.round((wallet?.cashbackRate ?? CASHBACK_RATE) * 100)} % sur chaque commande
   343→            </div>
   344→            <button onClick={() => { setTopup(true); setTopupState("idle"); }} className="mt-4 h-11 w-full rounded-xl bg-[#C8951E] text-[#1A1410] font-bold text-sm flex items-center justify-center gap-2 active:scale-[0.98] transition-transform focus-visible:outline-2 focus-visible:outline-[#C8951E]">
   345→              <Plus size={16} /> Approvisionner
   346→            </button>
   347→          </div>
   348→        </div>

   353→          <p className="text-xs text-muted-foreground text-center py-3">Aucune transaction pour l&apos;instant.</p>

   377→      {/* Parrainage — le fil qui relie les amies */}
   378→      <ParrainageCard userId={user.id} userName={user.name} onRedeemed={loadWallet} />
   379→
   380→      {/* Consentement */}
   381→      <section aria-labelledby="cons-t" className="rounded-2xl border border-border bg-card p-4 flex items-center gap-3">
   382→        <ShieldCheck size={20} className={user.consentHealth ? "text-success" : "text-destructive"} />
   383→        <div className="flex-1">
