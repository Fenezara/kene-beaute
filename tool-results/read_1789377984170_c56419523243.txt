    37→export function ShopScreen() {
    38→  const user = useKene((s) => s.user)!;
    39→  const cart = useKene((s) => s.cart);
    40→  const addToCart = useKene((s) => s.addToCart);
    41→  const setCartQty = useKene((s) => s.setCartQty);
    42→  const clearCart = useKene((s) => s.clearCart);
    43→
    44→  const [products, setProducts] = useState<ApiProduct[] | null>(null);
    45→  const [cat, setCat] = useState("");
    46→  const [q, setQ] = useState("");
    47→  const [favOnly, setFavOnly] = useState(false); // filtre « ♥ Favoris » (cumulable avec catégorie + recherche)
    48→  const favs = useFavorites((s) => s.favs);
    49→  const [detail, setDetail] = useState<ApiProduct | null>(null);
    50→  const [qty, setQty] = useState(1);
    51→  const [checkout, setCheckout] = useState(false);
    52→  const [wallet, setWallet] = useState<ApiWallet | null>(null);
    53→  // Fin des échecs silencieux : solde inconnu → encart discret + Réessayer
    54→  // (plus jamais un « … » éternel sur le bouton Wallet du checkout).
    55→  const [walletError, setWalletError] = useState(false);
    56→  const [payState, setPayState] = useState<{ phase: "processing" | "success"; method: PayMethod; amount: number } | null>(null);
    57→  const [paying, setPaying] = useState(false);
    58→
    59→  // Sécurité renforcée (2FA-lite) : si activée, la cliente re-vérifie son code
    60→  // AVANT que le moindre appel de paiement ne parte (voir startPay + SecureVerify).
    61→  const securityEnabled = useSecurity((s) => s.enabled);
    62→  const [pendingPay, setPendingPay] = useState<PayMethod | null>(null);
    63→
    64→  // ─── Double-tap « ajout rapide » (TikTok Shop / Instagram) ───
    65→  // 1er tap = ouvre la fiche (avec un délai court annulable) ; 2e tap < 320 ms
    66→  // = ajoute directement au panier + burst animé sur la carte.
    67→  const lastTap = useRef<{ id: string; t: number; timer: number | null }>({ id: "", t: 0, timer: null });
    68→  const [burst, setBurst] = useState<{ id: string; x: number; y: number } | null>(null);
    69→
    70→  const onCardTap = (p: ApiProduct, e: React.MouseEvent<HTMLButtonElement>) => {
    71→    const now = Date.now();
    72→    const s = lastTap.current;
    73→    if (s.id === p.id && now - s.t < 320 && s.timer) {
    74→      window.clearTimeout(s.timer);
    75→      lastTap.current = { id: "", t: 0, timer: null };
    76→      addToCart({ productId: p.id, name: p.name, price: p.price, qty: 1, image: p.image });
    77→      haptic(HAPTIC.light);
    78→      const rect = e.currentTarget.getBoundingClientRect();
    79→      setBurst({ id: p.id, x: e.clientX - rect.left, y: e.clientY - rect.top });
    80→      window.setTimeout(() => setBurst(null), 700);
    81→      toast.success(`${p.name} ajouté au panier`, { description: "Astuce : double-tape une carte pour l'ajouter en 1 geste" });
    82→    } else {
    83→      if (s.timer) window.clearTimeout(s.timer);
    84→      const timer = window.setTimeout(() => {
    85→        setDetail(p);
    86→        setQty(1);
    87→      }, 240);
    88→      lastTap.current = { id: p.id, t: now, timer };
    89→    }
    90→  };
    91→  const [promoInput, setPromoInput] = useState("");
    92→  const [promoChecking, setPromoChecking] = useState(false);
    93→  const [promo, setPromo] = useState<AppliedPromo | null>(null);
    94→  const inputRef = useRef<HTMLInputElement>(null);
    95→
    96→  /* « Mes commandes » — historique consultable (chargé au montage, rafraîchi
    97→   * après chaque commande réussie : la cliente suit ses données enregistrées). */
    98→  const [view, setView] = useState<"catalogue" | "commandes">("catalogue");
    99→  const [orders, setOrders] = useState<ApiOrder[] | null>(null);
   100→  /* La grille produits — destination du Marché vivant (tap échoppe → filtre +
   101→   * descente douce du catalogue vers la famille choisie). */
   102→  const gridRef = useRef<HTMLDivElement>(null);
   103→  const refreshOrders = useCallback(() => {
   104→    apiGet<{ orders: ApiOrder[] }>(`/api/orders?userId=${user.id}`)
   105→      .then((r) => setOrders(r.orders ?? []))
   106→      .catch(() => setOrders((prev) => prev ?? []));
   107→  }, [user.id]);
   108→
   109→  const subtotal = cart.reduce((s, l) => s + l.price * l.qty, 0);
   110→  /* Coupon invalidé si le panier a changé depuis la validation (remise recalculée
   111→   * sur un autre sous-total) — la cliente re-valide, jamais de surprise serveur. */
   112→  const livePromo = promo && promo.subtotal === subtotal ? promo : null;
   113→  const discount = livePromo?.discount ?? 0;
   114→  const total = subtotal - discount;
   115→  /* taux de cashback réellement appliqué (wallet de la cliente, sinon défaut) */
   116→  const cashbackRate = wallet?.cashbackRate ?? CASHBACK_RATE;
   117→
   118→  /* Solde wallet — échec explicite (walletError) plutôt que silence : le
   119→   * bouton Wallet reste désactivé tant que le solde est inconnu. */
   120→  const loadWallet = useCallback(() => {
   121→    apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`)
   122→      .then((r) => { setWallet(r.wallet); setWalletError(false); })
   123→      .catch(() => setWalletError(true));
   124→  }, [user.id]);
   125→
   126→  useEffect(() => {
   127→    apiGet<{ products: ApiProduct[] }>("/api/shop/products")
   128→      .then((r) => setProducts(r.products ?? []))
   129→      .catch(() => setProducts([]));
   130→    loadWallet();
   131→    refreshOrders();
   132→  }, [user.id, refreshOrders, loadWallet]);
   133→
   134→  const filtered = useMemo(() => {
   135→    const nq = q.trim().toLowerCase();
   136→    return (products ?? []).filter(
   137→      (p) =>
   138→        (!cat || p.category === cat) &&
   139→        (!favOnly || favs.includes(p.id)) &&
   140→        (!nq || `${p.name} ${p.botanicals} ${p.description}`.toLowerCase().includes(nq))
   141→    );
   142→  }, [products, cat, q, favOnly, favs]);
   143→
   144→  /* le fil de la catégorie — la navette l'illumine dans la bande tissée */
   145→  const weaveCaption =
   146→    products === null
   147→      ? "La navette monte le métier…"
   148→      : `${filtered.length} soin${filtered.length > 1 ? "s" : ""}${
   149→          cat ? ` · ${SHOP_CATEGORIES.find((c) => c.id === cat)?.label.toLowerCase() ?? cat}` : " au catalogue"
   150→        }${favOnly ? " · favoris" : ""}`;
   151→
   152→  /* Applique un code promo : aperçu de remise sans consommer le coupon
   153→   * (la consommation a lieu à la commande — toutes les gardes côté serveur). */
   154→  async function applyPromo() {
   155→    const code = promoInput.trim().toUpperCase();
   156→    if (!code || promoChecking) return;
   157→    setPromoChecking(true);
   158→    try {
   159→      const r = await apiPost<{ ok: true; coupon: { code: string; label: string | null }; discount: number; total: number }>(
   160→        "/api/coupons/validate",
   161→        { code, userId: user.id, subtotal }
   162→      );
   163→      setPromo({ code: r.coupon.code, label: r.coupon.label, discount: r.discount, subtotal });
   164→      setPromoInput("");
   165→      toast.success(`Code ${r.coupon.code} appliqué`, { description: `Remise de ${xof(r.discount)} déduite du total.` });
   166→    } catch (e) {
   167→      setPromo(null);
   168→      toast.error(e instanceof Error ? e.message : "Code promo invalide");
   169→    } finally {
   170→      setPromoChecking(false);
   171→    }
   172→  }
   173→
   174→  /** Passerelle paiement : vérification d'identité par code si la sécurité
   175→   *  renforcée est active — le paiement initialement prévu (pay) n'est lancé
   176→   *  qu'une fois le code confirmé ; annulé sinon, rien n'est engagé. */
   177→  function startPay(method: PayMethod) {
   178→    if (securityEnabled) {
   179→      haptic(HAPTIC.tap);
   180→      setPendingPay(method);
   181→      return;
   182→    }
   183→    void pay(method);
   184→  }
   185→
   186→  async function pay(method: PayMethod) {
   187→    setPaying(true);
   188→    try {
   189→      const items = cart.map((l) => ({ productId: l.productId, qty: l.qty }));
   190→      const r = await apiPost<{ order: ApiOrder; payment: ApiPayment | null; paid: boolean }>("/api/orders", {
   191→        userId: user.id,
   192→        items,
   193→        paymentMethod: method,
   194→        ...(livePromo ? { couponCode: livePromo.code } : {}),
   195→      });
   196→      const amount = r.order.total;
   197→      if (method !== "wallet" && r.payment) {
   198→        // Contrat confirmToken (63-b/63-c) : un paiement mobile money en attente
   199→        // porte son jeton — absent, on n'appelle JAMAIS confirm et on rollback
   200→        // l'état UI comme un échec de paiement (panier intact, aucun overlay).
   201→        const confirmToken = r.payment.confirmToken;
   202→        if (!confirmToken) {
   203→          setCheckout(false);
   204→          toast.error("Paiement impossible — réessaie dans quelques instants");
   205→          return;
   206→        }
   207→        setCheckout(false);
   208→        setPayState({ phase: "processing", method, amount });
   209→        await new Promise((res) => setTimeout(res, 3000));
   210→        const c = await apiPost<{ payment: ApiPayment; order?: ApiOrder; wallet?: ApiWallet }>("/api/payments/confirm", { paymentId: r.payment.id, confirmToken });
   211→        if (c.wallet) { setWallet(c.wallet); setWalletError(false); }
   212→      } else {
   213→        setCheckout(false);
   214→        setPayState({ phase: "processing", method: "wallet", amount });
   215→        await new Promise((res) => setTimeout(res, 1200));
   216→      }
   217→      setPayState({ phase: "success", method, amount });
   218→      haptic(HAPTIC.success);
   219→      clearCart();
   220→      setPromo(null); // le coupon est consommé : remise à zéro pour la prochaine commande
   221→      void refreshOrders(); // la nouvelle commande apparaît dans « Mes commandes »
   222→      const cb = r.order.cashback;
   223→      toast.success(cb > 0 ? `Commande confirmée — cashback ${xof(cb)} crédité sur ton wallet` : "Commande confirmée et payée");
   224→      if (method === "wallet") {
   225→        const w = await apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).catch(() => null);
   226→        if (w) setWallet(w.wallet);
   227→      }
   228→    } catch (e) {
   229→      setPayState(null);
   230→      toast.error(e instanceof Error ? e.message : "Paiement impossible");
   231→    } finally {
   232→      setPaying(false);
   233→    }
   234→  }
   235→
   236→  const op = (m: PayMethod) => MOMO_OPERATORS.find((o) => o.code === m);
   237→
   238→  return (
   239→    <div className="pt-4 pb-2">
   240→      <Reveal>
   241→        <RevealItem>
   242→          <header className="flex items-center justify-between">
   243→            <h1 className="flex items-center gap-2 font-heading font-black text-xl tracking-tight">
   244→              {/* Mark de marque (t. 86) — signature vectorielle de la boutique */}
   245→              <span aria-hidden="true">
   246→                <KeneMark size={26} />
   247→              </span>
   248→              Boutique Kènè
   249→            </h1>
   250→            <span className="k-chip rounded-full px-2.5 py-1 text-[10px] font-bold uppercase tracking-wide text-gold-text">Cashback {Math.round(cashbackRate * 100)} %</span>
   251→          </header>
   252→        </RevealItem>
   253→        <RevealItem>
   254→          {/* Bascule Catalogue ↔ Mes commandes — la cliente consulte ses données */}
   255→          <div className="mt-3 grid grid-cols-2 gap-1 rounded-2xl k-card p-1" role="tablist" aria-label="Vues boutique">
   256→            <button
   257→              role="tab"
   258→              aria-selected={view === "catalogue"}
   259→              onClick={() => setView("catalogue")}
   260→              className={`h-10 rounded-xl flex items-center justify-center gap-1.5 text-xs font-bold transition-colors focus-visible:outline-2 focus-visible:outline-primary ${view === "catalogue" ? "k-btn-gold text-primary-foreground" : "text-muted-foreground hover:text-foreground"}`}
   261→            >
   262→              <ShoppingBag size={14} aria-hidden="true" /> Catalogue
   263→            </button>
   264→            <button
   265→              role="tab"
   266→              aria-selected={view === "commandes"}
   267→              onClick={() => setView("commandes")}
   268→              className={`h-10 rounded-xl flex items-center justify-center gap-1.5 text-xs font-bold transition-colors focus-visible:outline-2 focus-visible:outline-primary ${view === "commandes" ? "k-btn-gold text-primary-foreground" : "text-muted-foreground hover:text-foreground"}`}
   269→            >
   270→              <History size={14} aria-hidden="true" /> Mes commandes{orders !== null && orders.length > 0 ? ` (${orders.length})` : ""}
   271→            </button>
   272→          </div>
   273→        </RevealItem>
   274→      </Reveal>
   275→
   276→      {view === "commandes" ? (
   277→        <OrdersView orders={orders} onRefresh={refreshOrders} onShop={() => setView("catalogue")} />
   278→      ) : (
   279→        <>
   280→      {/* Le Marché vivant (t. 83-c) — le marché 3D coiffe la boutique ; le tap
   281→          sur une échoppe pose le filtre réel et descend vers la grille. Clair
   282→          de Lune : bande SVG cliquable (jamais bloquant). */}
   283→      <MarcheVivant
   284→        onSelectCategory={(category) => {
   285→          setCat(category);
   286→          setQ(""); // saut exploratoire : la recherche ne masque pas l'échoppe
   287→          setFavOnly(false);
   288→          gridRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   289→        }}
   290→      />
   291→
   292→      <Reveal>
   293→        <RevealItem>
   294→          {/* Le Fil de Kente — hero tissé, le fil de la catégorie s'illumine */}
   295→          <div className="mt-4">
   296→            <KenteWeaveCard highlightIndex={cat ? categoryThread(cat) : -1} caption={weaveCaption} />
   297→          </div>
   298→        </RevealItem>
   299→
   300→        <RevealItem>
   301→          {/* Recherche */}
   302→          <div className="mt-3 relative">
   303→            <Search size={16} className="absolute left-3.5 top-1/2 -translate-y-1/2 text-muted-foreground" aria-hidden="true" />
   304→            <input
   305→              ref={inputRef}
   306→              value={q}
   307→              onChange={(e) => setQ(e.target.value)}
   308→              placeholder="Rechercher un soin, un botanique…"
   309→              aria-label="Rechercher un produit"
   310→              className="k-input h-12 w-full rounded-2xl pl-10 pr-4 text-sm"
   311→            />
   312→          </div>
   313→        </RevealItem>
   314→
   315→        <RevealItem>
   316→          {/* Filtres : catégories + favoris (bascules aria-pressed, cumulables) — Chip verre→or 2026 */}
   317→          <div className="flex gap-2 overflow-x-auto py-3 scrollbar-thin -mx-1 px-1" role="group" aria-label="Filtres de la boutique">
   318→            {SHOP_CATEGORIES.map((c) => (
   319→              <Chip key={c.id} selected={cat === c.id} onClick={() => setCat(c.id)} className="min-h-11 shrink-0">
   320→                {c.label}
   321→              </Chip>
   322→            ))}
   323→            {/* ♥ Favoris — filtre cumulable, badge count si ≥ 1 */}
   324→            <Chip
   325→              selected={favOnly}
   326→              onClick={() => { setFavOnly((v) => !v); haptic(HAPTIC.tap); }}
   327→              className="min-h-11 shrink-0"
   328→            >
   329→              <Heart size={13} fill={favOnly ? "currentColor" : "none"} aria-hidden="true" />
   330→              Favoris{favs.length > 0 ? ` · ${favs.length}` : ""}
   331→            </Chip>
   332→          </div>
   333→        </RevealItem>
   334→      </Reveal>
   335→
   336→      <div ref={gridRef} className="scroll-mt-4">
   337→      {/* Grille produits */}
   338→      {products === null ? (
   339→        <div className="grid grid-cols-2 gap-3.5 sm:grid-cols-3">
   340→          {[0, 1, 2, 3].map((i) => (
   341→            <div key={i} className="space-y-2">
   342→              <Shimmer className="aspect-square rounded-[18px]" />
   343→              <Shimmer className="h-3.5 w-4/5 rounded" />
   344→              <Shimmer className="h-3 w-2/3 rounded" />
   345→            </div>
   346→          ))}
   347→        </div>
   348→      ) : filtered.length === 0 ? (
   349→        favOnly && favs.length === 0 ? (
   350→          <EmptyBlock
   351→            icon={<Heart size={22} />}
   352→            title="Aucun favori pour l'instant"
   353→            text="Touche le cœur sur un soin pour le retrouver ici."
   354→          />
   355→        ) : favOnly ? (
   356→          <EmptyBlock
   357→            icon={<Heart size={22} />}
   358→            title="Aucun favori dans cette sélection"
   359→            text="Tes favoris ne passent pas ce filtre — essaie une autre catégorie ou efface la recherche."
   360→            cta={
   361→              <button
   362→                onClick={() => { setCat(""); setQ(""); }}
   363→                className="k-btn-gold h-11 rounded-xl px-6 text-sm font-bold text-primary-foreground focus-visible:outline-2 focus-visible:outline-primary"
   364→              >
   365→                Voir tous mes favoris
   366→              </button>
   367→            }
   368→          />
   369→        ) : (
   370→          <EmptyBlock icon={<Search size={22} />} title="Aucun produit trouvé" text="Essaie un autre mot-clé ou une autre catégorie." />
   371→        )
   372→      ) : (
   373→        <Reveal className="grid grid-cols-2 gap-3.5 sm:grid-cols-3">
   374→          {filtered.map((p) => (
   375→            /* Wrapper relatif : le cœur est un FRÈRE de la carte (jamais de <button>
   376→               imbriqué — HTML valide, focus/a11y propres), posé sur l'image en absolu. */
   377→            <RevealItem key={p.id} className="relative">
   378→              <div className="k-card k-card-hover rounded-[24px] p-2.5 pb-2">
   379→                <button
   380→                  onClick={(e) => onCardTap(p, e)}
   381→                  className="block w-full touch-manipulation rounded-[18px] text-left transition-transform focus-visible:outline-2 focus-visible:outline-primary active:scale-[0.98]"
   382→                  aria-label={`${p.name}, ${xof(p.price)} — appuie une fois pour la fiche, deux fois pour l'ajouter au panier`}
   383→                >
   384→                  <div className="relative">
   385→                    <img src={p.image} alt={p.name} loading="lazy" className="aspect-square w-full rounded-[18px] object-cover" />
   386→                    {/* Dégradé bas subtil — profondeur derrière le badge prix flottant */}
   387→                    <div aria-hidden="true" className="absolute inset-0 rounded-[18px] bg-gradient-to-t from-black/25 via-transparent to-transparent" />
   388→                    {/* Burst double-tap — panier kente qui jaillit sous le doigt */}
   389→                    {burst?.id === p.id && (
   390→                      <motion.span
   391→                        aria-hidden="true"
   392→                        initial={{ scale: 0.3, opacity: 0.95 }}
   393→                        animate={{ scale: 1.7, opacity: 0 }}
   394→                        transition={{ duration: 0.65, ease: "easeOut" }}
   395→                        className="pointer-events-none absolute z-20"
   396→                        style={{ left: burst.x, top: burst.y }}
   397→                      >
   398→                        <span className="grid place-items-center h-20 w-20 -ml-10 -mt-10 rounded-full bg-[#FFF9EC]/30 backdrop-blur-[2px] shadow-xl">
   399→                          <span className="grid place-items-center h-12 w-12 rounded-full bg-gradient-to-br from-[#A0522D] to-[#8B1A3B] shadow-lg">
   400→                            <ShoppingBag size={22} className="text-[#FFF9EC]" />
   401→                          </span>
   402→                        </span>
   403→                      </motion.span>
   404→                    )}
   405→                    {/* Badge prix — pilule de verre flottant sur l'image */}
   406→                    <span className="k-chip absolute bottom-2.5 left-2.5 rounded-full px-2.5 py-1 font-mono text-[11px] font-bold text-gold-text">{xof(p.price)}</span>
   407→                  </div>
   408→                  <div className="pt-2">
