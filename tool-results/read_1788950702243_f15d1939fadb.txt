    70→
    71→export function HomeScreen({
    72→  onScanZone,
    73→  refreshKey = 0,
    74→  onRefreshed,
    75→}: {
    76→  onScanZone: (z: BodyZone) => void;
    77→  /** Incrémenté par le pull-to-refresh du shell — déclenche un rechargement */
    78→  refreshKey?: number;
    79→  /** Appelé à la fin du chargement (le shell ferme l'indicateur de tirage) */
    80→  onRefreshed?: () => void;
    81→}) {
    82→  const user = useKene((s) => s.user)!;
    83→  const setClientTab = useKene((s) => s.setClientTab);
    84→  const { t } = useT();
    85→  const [data, setData] = useState<HomeData | null>(null);
    86→  const [err, setErr] = useState<string | null>(null);
    87→  const [ritualOpen, setRitualOpen] = useState(false);
    88→
    89→  useEffect(() => {
    90→    let alive = true;
    91→    (async () => {
    92→      try {
    93→        const [d, a, w, p, r] = await Promise.all([
    94→          apiGet<{ diagnoses: ApiDiagnosis[] }>(`/api/diagnoses?userId=${user.id}`),
    95→          apiGet<{ appointments: ApiAppointment[] }>(`/api/appointments?userId=${user.id}`),
    96→          apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).catch(() => null),
    97→          apiGet<{ products: ApiProduct[] }>("/api/shop/products"),
    98→          apiGet<ApiReminderFeed>(`/api/notifications?userId=${user.id}`).catch(() => null),
    99→        ]);
   100→        if (alive) {
   101→          setData({
   102→            diagnoses: d.diagnoses ?? [],
   103→            appointments: a.appointments ?? [],
   104→            wallet: w?.wallet ?? null,
   105→            products: p.products ?? [],
   106→            reminders: r ?? null,
   107→          });
   108→          if (alive) onRefreshed?.();
   109→        }
   110→      } catch (e) {
   111→        if (alive) {
   112→          setErr(e instanceof Error ? e.message : "Chargement impossible");
   113→          onRefreshed?.();
   114→        }
   115→      }
   116→    })();
   117→    return () => {
   118→      alive = false;
   119→    };
   120→  }, [user.id, refreshKey, onRefreshed]);
   121→
   122→  // Score multi-zones pondéré (PRD §8.8) — dernier diagnostic par zone
   123→  const multi = useMemo(() => {
   124→    const byZone = new Map<BodyZone, ApiDiagnosis>();
   125→    [...(data?.diagnoses ?? [])]
   126→      .filter((d) => d.status === "done")
   127→      .sort((x, y) => +new Date(y.createdAt) - +new Date(x.createdAt))
   128→      .forEach((d) => {
   129→        if (!byZone.has(d.zone)) byZone.set(d.zone, d);
   130→      });
   131→    const covered = BODY_ZONES.filter((z) => byZone.has(z.id));
   132→    const weightSum = covered.reduce((s, z) => s + z.weight, 0);
   133→    const score =
   134→      covered.length && weightSum > 0
   135→        ? Math.round(covered.reduce((s, z) => s + (byZone.get(z.id)?.scoreGlobal ?? 0) * z.weight, 0) / weightSum)
   136→        : null;
   137→    const missing = BODY_ZONES.filter((z) => !byZone.has(z.id));
   138→    const last = covered.length ? byZone.get(covered[0].id) : null;
   139→    return { byZone, covered, score, missing, last };
   140→  }, [data?.diagnoses]);
   141→
   142→  const nextAppt = useMemo(() => {
   143→    const now = Date.now();
   144→    return (
   145→      [...(data?.appointments ?? [])]
   146→        .filter((a) => new Date(a.startAt).getTime() >= now && a.status !== "cancelled")
   147→        .sort((x, y) => +new Date(x.startAt) - +new Date(y.startAt))[0] ?? null
   148→    );
   149→  }, [data?.appointments]);
   150→
   151→  const reco = useMemo(() => {
   152→    const all = data?.products ?? [];
   153→    const byCat = (cats: string[]) => all.filter((p) => cats.includes(p.category));
   154→    const pick =
   155→      user.skinType === "grasse"
   156→        ? byCat(["serum", "savon"])
   157→        : user.skinType === "seche"
   158→          ? byCat(["creme", "huile"])
   159→          : byCat(["serum", "creme"]);
   160→    return (pick.length >= 3 ? pick : [...pick, ...all.filter((p) => !pick.includes(p))]).slice(0, 3);
   161→  }, [data?.products, user.skinType]);
   162→
   163→  // Route de l'Or : dernier diagnostic parsable → rituel tissable depuis l'accueil
   164→  const lastResult = useMemo(() => (multi.last ? parseDiagnosis(multi.last.resultJson) : null), [multi.last]);
   165→
   166→  // Le Fil du Retour : prochaine étape dérivée de l'activité (contrôle protocole / soin de suite)
   167→  const nextStep = useMemo(
   168→    () => (data ? nextClientStep(new Date(), data.diagnoses, data.appointments) : null),
   169→    [data],
   170→  );
   171→
   172→  const first = user.name.split(" ")[0];
   173→
   174→  return (
   175→    // Reveal = entrée en cascade spring (respecte prefers-reduced-motion) :
   176→    // chaque bloc du fil est un RevealItem — le contenu/logique est inchangé.
   177→    <Reveal className="flex flex-col gap-6 pt-1">
   178→      {/* ───── Salutation ───── */}
   179→      <RevealItem>
   180→        <header className="flex items-end justify-between gap-3">
   181→          <div className="min-w-0">
   182→            {/* Filet kente — signature du bloc (décoratif) */}
   183→            <span
   184→              aria-hidden="true"
   185→              className="mb-1.5 block h-[3px] w-10 rounded-full bg-gradient-to-r from-[#C8951E] via-[#A0522D] to-[#3F7D3F] opacity-80"
   186→            />
   187→            <p className="text-[11px] uppercase tracking-[0.16em] text-primary font-semibold">{t("home.greeting")}</p>
   188→            <h2 className="font-heading font-black text-[29px] leading-tight tracking-tight truncate"><span className="kente-text-flow">{first}</span> ✨</h2>
   189→            <p className="flex items-center gap-1 text-[11px] text-muted-foreground mt-0.5">
   190→              <MapPin size={11} /> {user.city || "Abidjan"} · {user.fitzpatrick ? `Fitzpatrick ${user.fitzpatrick}` : "Phototype à définir"}
   191→            </p>
   192→          </div>
   193→          <div className="flex shrink-0 items-center gap-2">
   194→            {data?.wallet && <WalletPill balance={data.wallet.balance} onClick={() => setClientTab("profil")} />}
   195→            {/* Médaillon de marque (t. 77) — le 04b choisi par la fondatrice vit
   196→                AUSSI dans l'app (splash + Seuil ne suffisaient pas : une cliente
   197→                connectée ne les voit jamais). ≥ 56 px conformément au système
   198→                de marque 3 étages ; décoratif — le prénom est dans le h2. */}
   199→            <span aria-hidden="true" className="shrink-0 select-none">
   200→              <KeneEmblem size={56} className="drop-shadow-[0_2px_8px_rgba(200,149,30,0.22)]" />
   201→            </span>
   202→          </div>
   203→        </header>
   204→      </RevealItem>
   205→
   206→      {/* ───── Bannière d'installation PWA (auto-masquée : installée / fermée / standalone) ───── */}
   207→      <RevealItem>
   208→        <InstallBanner />
   209→      </RevealItem>
   210→
   211→      {/* ───── Stories : scan rapide + zones avec score ───── */}
   212→      <RevealItem className="-mx-3 sm:-mx-5">
   213→        <section aria-label="Scan rapide par zone" className="px-3 sm:px-5">
   214→          <ScrollFadeRow label="Stories des zones — fais défiler horizontalement" className="flex gap-3.5 overflow-x-auto no-scrollbar py-1.5 pr-2">
   215→            {/* Story Scanner — anneau signature k-cta (dégradé + halo) */}
   216→            <button
   217→              onClick={() => setClientTab("diagnostic")}
   218→              aria-label={t("home.scan.story.aria")}
   219→              className="shrink-0 w-[68px] flex flex-col items-center gap-1.5 focus-visible:outline-2 focus-visible:-outline-offset-2 focus-visible:outline-primary rounded-xl"
   220→            >
   221→              <span className="grid place-items-center h-[66px] w-[66px] rounded-full p-[3px] k-cta active:scale-95 transition-transform">
   222→                <span className="grid place-items-center h-full w-full rounded-full bg-background">
   223→                  <NeaOnnimIcon size={26} className="text-primary" />
   224→                </span>
   225→              </span>
   226→              <span className="text-[10px] font-bold text-primary">{t("tab.scan")}</span>
   227→            </button>
   228→
   229→          {/* Stories zones */}
   230→          {BODY_ZONES.map((z) => {
   231→            const d = multi.byZone.get(z.id);
   232→            const covered = !!d;
   233→            const Icon = ZONE_ICON[z.id];
   234→            const color = covered ? scoreColor(d!.scoreGlobal) : "var(--border)";
   235→            return (
   236→              <button
   237→                key={z.id}
   238→                onClick={() => onScanZone(z.id)}
   239→                aria-label={`${covered ? `Re-scanner ${z.label} — dernier score ${d!.scoreGlobal}` : `Scanner ${z.label} pour compléter ton score`} · pondération ${Math.round(z.weight * 100)} %`}
   240→                className="shrink-0 w-[68px] flex flex-col items-center gap-1.5 focus-visible:outline-2 focus-visible:-outline-offset-2 focus-visible:outline-primary rounded-xl"
   241→              >
   242→              <span
   243→                className={cn(
   244→                  "relative grid place-items-center h-[66px] w-[66px] p-[3px] active:scale-95 transition-transform",
   245→                  covered ? "rounded-full k-glow-gold" : "rounded-[22px]",
   246→                )}
   247→                style={
   248→                  covered
   249→                    ? { background: `conic-gradient(from 210deg, ${color} 0%, ${color} 45%, #C8951E 68%, ${color} 100%)` }
   250→                    : undefined
   251→                }
   252→              >
   253→                <span
   254→                  className={cn(
   255→                    "grid place-items-center h-full w-full bg-background",
   256→                    covered ? "rounded-full" : "rounded-[19px] border-[2.5px] border-dashed border-border",
   257→                  )}
   258→                >
   259→                  <Icon className={covered ? "text-foreground/80" : "text-muted-foreground"} />
   260→                </span>
   261→                  {!covered && (
   262→                    <span className="absolute -bottom-0.5 -right-0.5 grid place-items-center h-6 w-6 rounded-full bg-primary text-primary-foreground border-2 border-background">
   263→                      <Plus size={13} strokeWidth={2.5} />
   264→                    </span>
   265→                  )}
   266→                  {covered && (
   267→                    <span
   268→                      className="absolute -bottom-1 left-1/2 -translate-x-1/2 rounded-full px-1.5 py-px font-mono text-[10px] font-bold tabular-nums shadow-sm"
   269→                      style={{ backgroundColor: color, color: readableTextColor(color) }}
   270→                    >
   271→                      {d!.scoreGlobal}
   272→                    </span>
   273→                  )}
   274→                </span>
   275→                <span className={`text-[11px] w-full text-center leading-tight line-clamp-2 break-words min-h-[26px] ${covered ? "font-semibold" : "text-muted-foreground font-medium"}`}>{STORY_LABEL[z.id]}</span>
   276→              </button>
   277→            );
   278→          })}
   279→          </ScrollFadeRow>
   280→        </section>
   281→      </RevealItem>
   282→
   283→      {err && <p className="rounded-xl bg-destructive/10 text-destructive text-xs p-3">{err}</p>}
   284→
   285→      {/* ───── Carte score multi-zones (avec lecture vocale) — héro verre ───── */}
   286→      <RevealItem>
   287→        <section aria-labelledby="sc-t">
   288→          <GlassCard hero className="overflow-hidden rounded-[26px]">
   289→            <div className="kente-band h-[3px] w-full" aria-hidden="true" />
   290→            <div id="sc-t" className="p-6">
   291→              {!data ? (
   292→                <div className="flex items-center gap-4">
   293→                  <Shimmer className="h-[130px] w-[130px] rounded-full" />
   294→                  <div className="flex-1 space-y-2">
   295→                    <Shimmer className="h-4 w-3/4" />
   296→                    <Shimmer className="h-3 w-full" />
   297→                    <Shimmer className="h-3 w-2/3" />
   298→                  </div>
   299→                </div>
   300→              ) : multi.score !== null ? (
   301→            <div className="flex flex-wrap items-center gap-4">
   302→              <ScoreGauge score={multi.score} label={t("home.score.zone")} />
   303→              <div className="min-w-0 flex-1">
   304→                <h3 className="font-heading font-bold text-base">{t("home.score.title")}</h3>
   305→                <p className="text-xs text-muted-foreground mt-1 leading-relaxed">
   306→                  {multi.covered.length} zone{multi.covered.length > 1 ? "s" : ""} analysée{multi.covered.length > 1 ? "s" : ""} · pondération PRD
   307→                  {multi.last && <span className="block mt-1">Dernier scan : {formatDate(multi.last.createdAt)}</span>}
   308→                </p>
   309→                <div className="flex flex-wrap gap-1.5 mt-2">
   310→                  {multi.covered.map((z) => (
   311→                    <span key={z.id} className="k-chip rounded-full px-2 py-0.5 text-[10px] font-medium">
   312→                      {z.label} {multi.byZone.get(z.id)?.scoreGlobal}
   313→                    </span>
   314→                  ))}
   315→                  {multi.missing.map((z) => (
   316→                    <motion.button
   317→                      key={z.id}
   318→                      onClick={() => onScanZone(z.id)}
   319→                      whileTap={{ scale: 0.94 }}
   320→                      transition={{ type: "spring", stiffness: 500, damping: 24 }}
   321→                      className="rounded-full border border-primary/40 bg-primary/10 px-3 min-h-10 inline-flex items-center text-[11px] font-semibold text-primary focus-visible:outline-2 focus-visible:outline-primary"
   322→                    >
   323→                      + {z.label}
   324→                    </motion.button>
   325→                  ))}
   326→                </div>
   327→                {/* Lecture vocale du dernier diagnostic (accès non-lectrices) */}
   328→                {lastResult && (
   329→                  <div className="mt-5 pt-4 border-t border-border/60">
   330→                    <VoiceNarration result={lastResult} userName={user.name} />
   331→                  </div>
   332→                )}
   333→              </div>
   334→            </div>
   335→          ) : (
   336→            <div className="text-center py-2">
   337→              <NeaOnnimIcon size={40} className="mx-auto text-primary" />
   338→              <p className="font-heading font-bold mt-2">{t("home.first.title")}</p>
   339→              <p className="text-xs text-muted-foreground mt-1">{t("home.first.sub")}</p>
   340→            </div>
   341→          )}
   342→          {data && multi.missing.length > 0 && (
   343→            <div className="mt-4 pt-4 border-t border-dashed border-border">
   344→              <p className="text-[11px] font-semibold text-muted-foreground mb-2 flex items-center gap-1.5">
   345→                <Sparkles size={12} className="text-primary" /> {t("home.missing.title")}
   346→              </p>
   347→              <ScrollFadeRow label="Zones manquantes — fais défiler" className="flex gap-2 overflow-x-auto no-scrollbar pb-1 pr-1">
   348→                {multi.missing.map((z) => (
   349→                  <motion.button
   350→                    key={z.id}
   351→                    onClick={() => onScanZone(z.id)}
   352→                    whileTap={{ scale: 0.94 }}
   353→                    transition={{ type: "spring", stiffness: 500, damping: 24 }}
   354→                    className="shrink-0 rounded-full border border-primary/40 bg-primary/10 px-3 min-h-10 inline-flex items-center text-[11px] font-semibold text-primary focus-visible:outline-2 focus-visible:outline-primary"
   355→                  >
   356→                    + {z.label} · {Math.round(z.weight * 100)} %
   357→                  </motion.button>
   358→                ))}
   359→              </ScrollFadeRow>
   360→            </div>
   361→          )}
   362→          </div>
   363→        </GlassCard>
   364→      </section>
   365→      </RevealItem>
   366→
   367→      {/* ───── CTA Scanner ───── */}
   368→      <RevealItem>
   369→        <motion.button
   370→          whileTap={{ scale: 0.98 }}
   371→          onClick={() => setClientTab("diagnostic")}
   372→          className="relative h-24 rounded-3xl k-cta text-[#FFF9EC] overflow-hidden focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary"
   373→          aria-label={t("home.scan.aria")}
   374→        >
   375→          <div aria-hidden="true" className="absolute inset-0 bogolan-dots opacity-30" />
   376→          <div className="relative h-full flex items-center justify-center gap-3 px-4">
   377→            <span className="grid place-items-center h-14 w-14 rounded-2xl bg-[#FFF9EC]/15 border border-[#FFF9EC]/30">
   378→              <NeaOnnimIcon size={30} />
   379→            </span>
   380→            <span className="text-left">
   381→              <span className="block font-black text-lg tracking-tight">{t("home.scan.cta")}</span>
   382→              <span className="block text-[11px] opacity-90">{t("home.scan.sub")}</span>
   383→            </span>
   384→            <ChevronRight size={22} className="ml-auto opacity-80" aria-hidden="true" />
   385→          </div>
   386→        </motion.button>
   387→      </RevealItem>
   388→
   389→      {/* ───── Route de l'Or — rituel tissé depuis le dernier scan ───── */}
   390→      {data && lastResult && multi.last && (
   391→        <RevealItem>
   392→          <motion.button
   393→            whileTap={{ scale: 0.98 }}
   394→            onClick={() => setRitualOpen(true)}
   395→            className="relative w-full k-card rounded-[24px] overflow-hidden text-left focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary"
   396→            aria-label="Reprendre la Route de l'Or — tisser ma routine depuis mon dernier diagnostic"
   397→          >
   398→            <div aria-hidden="true" className="h-2 w-full" style={{ backgroundImage: "repeating-linear-gradient(90deg,#8B1A3B 0 12px,#3F7D3F 12px 20px,#C8951E 20px 28px,#E07A2B 28px 36px,#A0522D 36px 46px)" }} />
   399→            <div className="flex items-center gap-3 p-4">
   400→              <span className="grid place-items-center h-12 w-12 rounded-2xl bg-gradient-to-br from-[#C8951E] to-[#A0522D] text-[#FFF9EC] shrink-0 shadow">
   401→                <NeaOnnimIcon size={26} />
   402→              </span>
   403→              <div className="min-w-0 flex-1">
   404→                <p className="font-heading font-bold text-sm">La Route de l&apos;Or</p>
   405→                <p className="text-[11px] text-muted-foreground mt-0.5 leading-snug">
   406→                  Ta routine se tisse en 4 stations, selon ton scan du {formatDate(multi.last.createdAt, { day: "numeric", month: "short" })}
   407→                </p>
   408→              </div>
   409→              <ChevronRight size={18} className="text-primary shrink-0" />
   410→            </div>
   411→          </motion.button>
   412→        </RevealItem>
   413→      )}
   414→
   415→      {/* ───── Le Fil du Retour — ta prochaine étape ───── */}
   416→      {nextStep && (
   417→        <RevealItem>
   418→          <section
   419→            aria-labelledby="ns-t"
   420→            className="k-card k-card-hero rounded-[24px] p-4"
   421→          >
   422→            <div className="flex items-center gap-2">
   423→              <span className="grid place-items-center h-9 w-9 rounded-xl bg-gold/15 text-gold shrink-0">
   424→                <SankofaIcon size={19} />
   425→              </span>
   426→              <p id="ns-t" className="font-heading font-bold text-sm">{t("home.next.title")}</p>
   427→              <span
   428→                className={
   429→                  "ml-auto rounded-full px-2 py-0.5 text-[10px] font-semibold tabular-nums " +
