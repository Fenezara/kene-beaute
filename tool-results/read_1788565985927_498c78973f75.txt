     4→import { AnimatePresence, motion } from "framer-motion";
     5→import {

     7→  MessageSquareQuote, Star, TriangleAlert, Users, X,

    10→import { apiGet, apiPost } from "@/lib/kene/api";
    11→import { addDays, formatDate, formatTime, xof, DEPOSIT_RATE } from "@/lib/kene/format";
    12→import { cancellationRefund } from "@/lib/kene/rfm";
    13→import { SankofaIcon } from "@/components/kene/icons";
    14→import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";
    15→import { Skeleton } from "@/components/ui/skeleton";
    16→import { Tabs, TabsList, TabsTrigger } from "@/components/ui/tabs";
    17→import { useKene } from "@/store/kene";
    18→import type { ApiAppointment, ApiInstitute, ApiPayment, ApiResource, ApiReview, ApiService, ApiSlot, ApiWallet } from "./types";
    19→import { ApptBadge, EmptyBlock, MomoProcessing, SectionTitle, Stars, SuccessBurst } from "./bits";
    20→import { HAPTIC, haptic } from "@/lib/kene/ux";
    21→
    22→type PayMethod = "wave" | "wallet";
    23→
    24→function isoDate(d: Date): string {
    25→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
    26→}
    27→
    28→export function BookingScreen() {
    29→  const user = useKene((s) => s.user)!;
    30→  const [tab, setTab] = useState<"book" | "mine">("book");
    31→
    32→  // ── Réserver ──
    33→  const [institutes, setInstitutes] = useState<ApiInstitute[] | null>(null);
    34→  const [inst, setInst] = useState<ApiInstitute | null>(null);
    35→  const [services, setServices] = useState<ApiService[]>([]);
    36→  const [resources, setResources] = useState<ApiResource[]>([]);
    37→  const [reviews, setReviews] = useState<ApiReview[]>([]);
    38→  const [detailLoading, setDetailLoading] = useState(false);
    39→  const [service, setService] = useState<ApiService | null>(null);
    40→  const [day, setDay] = useState<Date>(new Date());
    41→  const [slots, setSlots] = useState<ApiSlot[] | null>(null);
    42→  const [slot, setSlot] = useState<ApiSlot | null>(null);
    43→  const [wallet, setWallet] = useState<ApiWallet | null>(null);
    44→  const [payMethod, setPayMethod] = useState<PayMethod>("wave");
    45→  const [paying, setPaying] = useState(false);
    46→  const [payOverlay, setPayOverlay] = useState<{ phase: "processing" | "done"; amount: number } | null>(null);
    47→  const [confirmed, setConfirmed] = useState<ApiAppointment | null>(null);
    48→
    49→  // ── Mes RDV ──
    50→  const [mine, setMine] = useState<ApiAppointment[] | null>(null);
    51→  const [reviewFor, setReviewFor] = useState<ApiAppointment | null>(null);
    52→  const [rating, setRating] = useState(5);
    53→  const [comment, setComment] = useState("");
    54→  const [cancelFor, setCancelFor] = useState<ApiAppointment | null>(null);
    55→
    56→  const days = useMemo(() => Array.from({ length: 7 }, (_, i) => addDays(new Date(), i)), []);
    57→  const deposit = service ? Math.round(service.price * DEPOSIT_RATE) : 0;
    58→  const practitioner = useMemo(() => (slot?.resourceIds?.length ? resources.find((r) => r.id === slot.resourceIds![0]) ?? null : null), [slot, resources]);
    59→
    60→  const loadInstitutes = useCallback(async () => {
    61→    setInstitutes(null);
    62→    try {
    63→      const r = await apiGet<{ institutes: ApiInstitute[] }>("/api/institutes");
    64→      setInstitutes(r.institutes ?? []);
    65→    } catch {
    66→      setInstitutes([]);
    67→    }
    68→  }, []);
    69→
    70→  const loadMine = useCallback(async () => {
    71→    setMine(null);
    72→    try {
    73→      const r = await apiGet<{ appointments: ApiAppointment[] }>(`/api/appointments?userId=${user.id}`);
    74→      setMine(r.appointments ?? []);
    75→    } catch {
    76→      setMine([]);
    77→    }
    78→  }, [user.id]);
    79→
    80→  useEffect(() => {
    81→    loadInstitutes();
    82→    apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).then((r) => setWallet(r.wallet)).catch(() => {});
    83→  }, [loadInstitutes, user.id]);
    84→
    85→  useEffect(() => {
    86→    if (tab === "mine") loadMine();
    87→  }, [tab, loadMine]);
    88→
    89→  async function openInstitute(i: ApiInstitute) {
    90→    setInst(i);
    91→    setService(null);
    92→    setSlot(null);
    93→    setSlots(null);
    94→    setDetailLoading(true);
    95→    try {
    96→      const r = await apiGet<{ institute: ApiInstitute; services: ApiService[]; resources: ApiResource[]; reviews: ApiReview[] }>(`/api/institutes/${i.id}`);
    97→      setServices(r.services ?? []);
    98→      setResources(r.resources ?? []);
    99→      setReviews(r.reviews ?? []);
   100→    } catch {
   101→      toast.error("Institut indisponible");
   102→      setInst(null);
   103→    } finally {
   104→      setDetailLoading(false);
   105→    }
   106→  }
   107→
   108→  async function pickDay(d: Date) {
   109→    setDay(d);
   110→    setSlot(null);
   111→    setSlots(null);
   112→    if (!service || !inst) return;
   113→    try {
   114→      const r = await apiGet<{ slots: ApiSlot[] }>(`/api/institutes/${inst.id}/availability?date=${isoDate(d)}&serviceId=${service.id}`);
   115→      setSlots(r.slots ?? []);
   116→    } catch {
   117→      setSlots([]);
   118→    }
   119→  }
   120→
   121→  async function chooseService(s: ApiService) {
   122→    setService(s);
   123→    setSlot(null);
   124→    setSlots(null);
   125→    if (!inst) return;
   126→    try {
   127→      const r = await apiGet<{ slots: ApiSlot[] }>(`/api/institutes/${inst.id}/availability?date=${isoDate(day)}&serviceId=${s.id}`);
   128→      setSlots(r.slots ?? []);
   129→    } catch {
   130→      setSlots([]);
   131→    }
   132→  }
   133→
   134→  async function book() {
   135→    if (!inst || !service || !slot) return;
   136→    setPaying(true);
   137→    try {
   138→      const [h, m] = slot.time.split(":").map(Number);
   139→      const startAt = new Date(day);
   140→      startAt.setHours(h, m, 0, 0);
   141→      const r = await apiPost<{ appointment: ApiAppointment; payment?: ApiPayment }>("/api/appointments", {
   142→        tenantId: inst.id,
   143→        serviceId: service.id,
   144→        resourceId: slot.resourceIds?.[0] ?? resources[0]?.id,
   145→        startAt: startAt.toISOString(),
   146→        clientName: user.name,
   147→        clientPhone: user.phone,
   148→        userId: user.id,
   149→        depositAmount: deposit,
   150→        paymentMethod: payMethod,
   151→      });
   152→      if (payMethod === "wave" && r.payment) {
   153→        setPayOverlay({ phase: "processing", amount: deposit });
   154→        await new Promise((res) => setTimeout(res, 3000));
   155→        await apiPost("/api/payments/confirm", { paymentId: r.payment.id });
   156→      } else {
   157→        setPayOverlay({ phase: "processing", amount: deposit });
   158→        await new Promise((res) => setTimeout(res, 1200));
   159→      }
   160→      setPayOverlay({ phase: "done", amount: deposit });
   161→      haptic(HAPTIC.success);
   162→      setTimeout(() => setPayOverlay(null), 1400);
   163→      setConfirmed(r.appointment);
   164→      const w = await apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).catch(() => null);
   165→      if (w) setWallet(w.wallet);
   166→    } catch (e) {
   167→      setPayOverlay(null);
   168→      toast.error(e instanceof Error ? e.message : "Réservation impossible");
   169→    } finally {
   170→      setPaying(false);
   171→    }
   172→  }
   173→
   174→  async function submitReview() {
   175→    if (!reviewFor) return;
   176→    try {
   177→      await apiPost(`/api/appointments/${reviewFor.id}/review`, { userId: user.id, rating, comment: comment.trim() || undefined });
   178→      toast.success("Merci pour ton avis");
   179→      setReviewFor(null);
   180→      setComment("");
   181→      setRating(5);
   182→      loadMine();
   183→    } catch (e) {
   184→      toast.error(e instanceof Error ? e.message : "Envoi impossible");
   185→    }
   186→  }
   187→
   188→  async function doCancel() {
   189→    if (!cancelFor) return;
   190→    try {
   191→      const r = await apiPost<{ appointment: ApiAppointment; refund: { amount: number; label: string } }>(`/api/appointments/${cancelFor.id}/cancel`, { userId: user.id });
   192→      toast.success(r.refund.amount > 0 ? `RDV annulé — ${xof(r.refund.amount)} remboursés sur ton wallet` : "RDV annulé");
   193→      setCancelFor(null);
   194→      loadMine();
   195→    } catch (e) {
   196→      toast.error(e instanceof Error ? e.message : "Annulation impossible");
   197→    }
   198→  }
   199→
   200→  const now = Date.now();
   201→  const upcoming = (mine ?? []).filter((a) => new Date(a.startAt).getTime() >= now && a.status !== "cancelled").sort((x, y) => +new Date(x.startAt) - +new Date(y.startAt));
   202→  const past = (mine ?? []).filter((a) => new Date(a.startAt).getTime() < now || a.status === "cancelled" || a.status === "completed").sort((x, y) => +new Date(y.startAt) - +new Date(x.startAt));
   203→
   204→  return (
   205→    <div className="pt-4 pb-2">
   206→      <Tabs value={tab} onValueChange={(v) => setTab(v as "book" | "mine")}>
   207→        <TabsList className="w-full grid grid-cols-2 bg-muted/60 rounded-2xl p-1">
   208→          <TabsTrigger value="book" className="rounded-xl data-[state=active]:bg-card data-[state=active]:text-primary font-semibold text-sm py-2.5">Réserver</TabsTrigger>
   209→          <TabsTrigger value="mine" className="rounded-xl data-[state=active]:bg-card data-[state=active]:text-primary font-semibold text-sm py-2.5">Mes RDV</TabsTrigger>
   210→        </TabsList>
   211→      </Tabs>
   212→
   213→      {/* ═══════ RÉSERVER ═══════ */}
   214→      {tab === "book" && (
   215→        <AnimatePresence mode="wait">
   216→          {confirmed ? (
   217→            /* — Confirmation — */
   218→            <motion.div key="ok" initial={{ opacity: 0, scale: 0.96 }} animate={{ opacity: 1, scale: 1 }} className="pt-8 flex flex-col items-center text-center">
   219→              <span className="grid place-items-center h-20 w-20 rounded-full bg-[#3F7D3F]/15 text-[#3F7D3D]">
   220→                <SankofaIcon size={38} />
   221→              </span>
   222→              <h2 className="font-heading font-black text-xl mt-4">Rendez-vous confirmé</h2>
   223→              <div className="mt-4 w-full rounded-2xl border border-border bg-card p-4 text-left space-y-1.5">
   224→                <p className="font-semibold text-sm">{confirmed.service?.name ?? "Soin"}</p>
   225→                <p className="text-xs text-muted-foreground">{confirmed.tenant?.name}</p>
   226→                <p className="text-xs text-muted-foreground flex items-center gap-1.5"><CalendarDays size={13} /> {formatDate(confirmed.startAt)} · <Clock size={13} /> {formatTime(confirmed.startAt)}</p>
   227→                {confirmed.resource?.name && <p className="text-xs text-muted-foreground flex items-center gap-1.5"><Users size={13} /> {confirmed.resource.name}</p>}
   228→                {confirmed.depositAmount > 0 && <p className="text-xs font-mono text-primary font-bold pt-1 border-t border-dashed border-border">Acompte réglé : {xof(confirmed.depositAmount)}</p>}
   229→              </div>
   230→              <p className="mt-4 inline-flex items-center gap-1.5 rounded-full bg-[#3F7D3F]/10 text-[#3F7D3D] px-3 py-1.5 text-[11px] font-semibold">
   231→                <MessageSquareQuote size={13} /> Rappel SMS J-1 programmé (simulé)
   232→              </p>
   233→              <div className="mt-6 grid grid-cols-2 gap-3 w-full">
   234→                <button onClick={() => { setConfirmed(null); setInst(null); setService(null); loadInstitutes(); }} className="h-12 rounded-xl border border-primary/60 text-primary text-sm font-bold active:scale-[0.98] transition-transform focus-visible:outline-2 focus-visible:outline-primary">
   235→                  Autre RDV
   236→                </button>
   237→                <button onClick={() => setTab("mine")} className="h-12 rounded-xl bg-primary text-primary-foreground text-sm font-bold active:scale-[0.98] transition-transform focus-visible:outline-2 focus-visible:outline-primary">
   238→                  Mes RDV
   239→                </button>
   240→              </div>
   241→            </motion.div>
   242→          ) : !inst ? (
   243→            /* — Liste instituts — */
   244→            <motion.div key="list" initial={{ opacity: 0, x: 30 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -30 }}>
   245→              <SectionTitle icon={<MapPin size={16} />}>Instituts partenaires</SectionTitle>
   246→              {institutes === null ? (
   247→                <div className="space-y-3">
   248→                  {[0, 1].map((i) => <Skeleton key={i} className="h-36 rounded-2xl" />)}
   249→                </div>
   250→              ) : institutes.length === 0 ? (
   251→                <EmptyBlock icon={<MapPin size={22} />} title="Aucun institut pour l'instant" text="De nouveaux instituts Kènè arrivent à Abidjan et Dakar." />
   252→              ) : (
   253→                <div className="space-y-3">
   254→                  {institutes.map((i) => (
   255→                    <button key={i.id} onClick={() => openInstitute(i)} className="w-full text-left rounded-2xl border border-border bg-card overflow-hidden shadow-sm active:scale-[0.99] transition-transform hover:border-primary/40 focus-visible:outline-2 focus-visible:outline-primary">
   256→                      <div className="relative">
   257→                        <img src={i.image} alt={i.name} loading="lazy" className="h-32 w-full object-cover" />
   258→                        <div className="absolute inset-0 bg-gradient-to-t from-[#1A1410]/70 to-transparent" />
   259→                        <div className="absolute bottom-2 left-3 right-3 flex items-end justify-between gap-2 text-[#F8F1E4]">
   260→                          <div>
   261→                            <p className="font-heading font-bold text-sm leading-tight">{i.name}</p>
   262→                            <p className="text-[10px] opacity-85 flex items-center gap-1"><MapPin size={10} /> {i.city}, {i.country}</p>
   263→                          </div>
   264→                          <span className="flex items-center gap-1 rounded-full bg-[#1A1410]/60 backdrop-blur px-2 py-0.5 text-[10px] font-bold">
   265→                            <Star size={10} className="fill-[#C8951E] text-[#C8951E]" /> {i.rating.toFixed(1)} ({i.reviewCount})
   266→                          </span>
   267→                        </div>
   268→                      </div>
   269→                      <div className="px-3 py-2.5 flex items-center justify-between text-[11px] text-muted-foreground">
   270→                        <span>{i._count?.services ?? 0} soins · {i._count?.reviews ?? 0} avis</span>
   271→                        <span className="flex items-center gap-0.5 text-primary font-semibold">Réserver <ChevronRight size={13} /></span>
   272→                      </div>
   273→                    </button>
   274→                  ))}
   275→                </div>
   276→              )}
   277→            </motion.div>
   278→          ) : (
   279→            /* — Détail institut — */
   280→            <motion.div key="detail" initial={{ opacity: 0, x: 30 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -30 }}>
   281→              <button onClick={() => setInst(null)} className="mt-4 inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground focus-visible:outline-2 focus-visible:outline-primary rounded" aria-label="Retour aux instituts">
   282→                <ArrowLeft size={15} /> Tous les instituts
   283→              </button>
   284→              <div className="relative mt-3 rounded-2xl overflow-hidden">
   285→                <img src={inst.image} alt={inst.name} className="h-44 w-full object-cover" />
   286→                <div className="absolute inset-0 bg-gradient-to-t from-[#1A1410]/80 to-transparent" />
   287→                <div className="absolute bottom-3 left-4 right-4 text-[#F8F1E4]">
   288→                  <h2 className="font-heading font-black text-lg">{inst.name}</h2>
   289→                  <p className="text-[11px] opacity-90 flex items-center gap-2">
   290→                    <span className="flex items-center gap-1"><Star size={11} className="fill-[#C8951E] text-[#C8951E]" /> {inst.rating.toFixed(1)}</span>
   291→                    <span className="flex items-center gap-1"><MapPin size={11} /> {inst.city}</span>
   292→                    <span className="flex items-center gap-1"><Clock size={11} /> {inst.openingHour}h–{inst.closingHour}h</span>
   293→                  </p>
   294→                </div>
   295→              </div>
   296→              {inst.description && <p className="text-xs text-muted-foreground leading-relaxed mt-3">{inst.description}</p>}
   297→
   298→              {detailLoading ? (
   299→                <div className="space-y-2 mt-4">{[0, 1, 2].map((i) => <Skeleton key={i} className="h-16 rounded-xl" />)}</div>
   300→              ) : (
   301→                <>
   302→                  {/* Avis */}
   303→                  {reviews.length > 0 && (
   304→                    <div className="mt-4">
   305→                      <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wide mb-2">Ce que disent les clientes</p>
   306→                      <div className="flex gap-2.5 overflow-x-auto pb-1 scrollbar-thin">
   307→                        {reviews.slice(0, 5).map((rv) => (
   308→                          <div key={rv.id} className="shrink-0 w-56 rounded-2xl border border-border bg-card p-3">
   309→                            <div className="flex items-center justify-between">
   310→                              <p className="text-xs font-semibold">{rv.user?.name ?? "Cliente"}</p>
   311→                              <Stars rating={rv.rating} size={9} />
   312→                            </div>
   313→                            <p className="text-[11px] text-muted-foreground mt-1.5 leading-snug line-clamp-3">{rv.comment ?? "Expérience agréable."}</p>
   314→                          </div>
   315→                        ))}
   316→                      </div>
   317→                    </div>
   318→                  )}
   319→
   320→                  {/* Services */}
   321→                  <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wide mt-5 mb-2">Soins & tarifs</p>
   322→                  <div className="space-y-2">
   323→                    {services.map((s) => (
   324→                      <div key={s.id} className={`rounded-2xl border p-3.5 transition-colors ${service?.id === s.id ? "border-primary bg-primary/5" : "border-border bg-card"}`}>
   325→                        <div className="flex items-start justify-between gap-2">
   326→                          <div className="min-w-0">
   327→                            <p className="text-sm font-semibold">{s.name}</p>
   328→                            <p className="text-[11px] text-muted-foreground mt-0.5 flex items-center gap-2">
   329→                              <span className="flex items-center gap-1"><Clock size={11} /> {s.durationMin} min</span>
   330→                              {s.botanicals && <span className="text-terre truncate">{s.botanicals}</span>}
   331→                            </p>
   332→                          </div>
   333→                          <p className="font-mono text-sm font-bold shrink-0">{xof(s.price)}</p>
   334→                        </div>
   335→                        <button onClick={() => chooseService(s)} className={`mt-2.5 h-10 w-full rounded-xl text-xs font-bold active:scale-[0.98] transition-all focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary ${service?.id === s.id ? "bg-primary text-primary-foreground" : "border border-primary/50 text-primary"}`}>
   336→                          {service?.id === s.id ? "Choisi" : "Choisir"}
   337→                        </button>
   338→                      </div>
   339→                    ))}
   340→                  </div>
   341→
   342→                  {/* Jours + créneaux */}
   343→                  {service && (
   344→                    <motion.div initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} className="mt-6">
   345→                      <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wide mb-2">Choisis ton jour</p>
   346→                      <div className="flex gap-2 overflow-x-auto pb-1 scrollbar-thin">
   347→                        {days.map((d) => {
   348→                          const sel = isoDate(d) === isoDate(day);
   349→                          return (
   350→                            <button key={d.toISOString()} onClick={() => pickDay(d)} aria-pressed={sel} className={`shrink-0 rounded-2xl border px-3.5 py-2 text-center transition-all active:scale-95 focus-visible:outline-2 focus-visible:outline-primary ${sel ? "border-primary bg-primary text-primary-foreground" : "border-border bg-card"}`}>
   351→                              <span className="block text-[9px] uppercase font-bold opacity-70">{formatDate(d, { weekday: "short" })}</span>
   352→                              <span className="block font-mono text-sm font-bold mt-0.5">{d.getDate()}</span>
   353→                            </button>
   354→                          );
   355→                        })}
   356→                      </div>
   357→
   358→                      <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wide mt-5 mb-2">Créneaux — {formatDate(day, { weekday: "long", day: "numeric", month: "long" })}</p>
   359→                      {slots === null ? (
   360→                        <div className="flex justify-center py-6"><Loader2 size={22} className="animate-spin text-primary" /></div>
   361→                      ) : slots.length === 0 ? (
   362→                        <p className="text-xs text-muted-foreground text-center py-4">Aucun créneau affiché — choisis un autre jour.</p>
   363→                      ) : (
   364→                        <div className="grid grid-cols-4 gap-2">
   365→                          {slots.map((s) => (
   366→                            <button
   367→                              key={s.time}
   368→                              disabled={!s.available}
   369→                              onClick={() => setSlot(s)}
   370→                              aria-pressed={slot?.time === s.time}
   371→                              className={`h-11 rounded-xl font-mono text-xs font-bold transition-all active:scale-95 focus-visible:outline-2 focus-visible:outline-primary ${
   372→                                !s.available ? "bg-muted text-muted-foreground/40 line-through" : slot?.time === s.time ? "bg-primary text-primary-foreground shadow" : "border border-border bg-card"
   373→                              }`}
   374→                            >
   375→                              {s.time}
   376→                            </button>
   377→                          ))}
   378→                        </div>
   379→                      )}
   380→                    </motion.div>
   381→                  )}
   382→
   383→                  {/* Récap + paiement */}
