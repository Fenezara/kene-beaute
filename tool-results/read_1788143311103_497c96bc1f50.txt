    39→function norm(s: string): string {
    40→  return s
    41→    .toLowerCase()
    42→    .normalize("NFD")
    43→    .replace(/[\u0300-\u036f]/g, "")
    44→    .replace(/[^a-z0-9 ]/g, " ");
    45→}

    47→/** Match une recommandation libre (« Sérum Moringa éclat ») avec un produit boutique */
    48→function matchProduct(rec: string, products: ApiProduct[]): ApiProduct | null {
    49→  const words = norm(rec).split(" ").filter((w) => w.length >= 5);
    50→  let best: { p: ApiProduct; score: number } | null = null;
    51→  for (const p of products) {
    52→    const target = `${norm(p.name)} ${norm(p.botanicals)} ${norm(p.description)}`;
    53→    let score = 0;
    54→    for (const w of words) if (target.includes(w)) score += 1;
    55→    if (score > 0 && (!best || score > best.score)) best = { p, score };
    56→  }
    57→  return best?.p ?? null;
    58→}

   586→/* ══════════════ Historique + comparaison ══════════════ */
   587→function HistoryView({
   588→  history, compareMode, setCompareMode, compareSel, setCompareSel, onBack, onOpen,
   589→}: {
   590→  history: ApiDiagnosis[] | null;
   591→  compareMode: boolean;
   592→  setCompareMode: (v: boolean) => void;
   593→  compareSel: string[];
   594→  setCompareSel: (v: string[]) => void;
   595→  onBack: () => void;
   596→  onOpen: (d: ApiDiagnosis) => void;
   597→}) {
   598→  const list = useMemo(
   599→    () => [...(history ?? [])].filter((d) => d.status === "done").sort((a, b) => +new Date(b.createdAt) - +new Date(a.createdAt)),
   600→    [history]
   601→  );
   602→  const sameZoneSel = compareSel.length === 2 && list.find((d) => d.id === compareSel[0])?.zone === list.find((d) => d.id === compareSel[1])?.zone;
   603→  const a = list.find((d) => d.id === compareSel[0]);
   604→  const b = list.find((d) => d.id === compareSel[1]);
   605→  const ra = a ? parseDiagnosis(a.resultJson) : null;
   606→  const rb = b ? parseDiagnosis(b.resultJson) : null;
   607→
   608→  const evolutions =
   609→    ra && rb
   610→      ? rb.indicateurs
   611→          .map((ib) => {
   612→            const ia = ra.indicateurs.find((x) => x.nom === ib.nom);
   613→            return ia ? { nom: ib.nom, delta: ib.pourcentage - ia.pourcentage } : null;
   614→          })
   615→          .filter((x): x is { nom: string; delta: number } => x !== null)
   616→          .sort((x, y) => Math.abs(y.delta) - Math.abs(x.delta))
   617→          .slice(0, 3)
   618→      : [];
   619→
   620→  return (
   621→    <div className="pt-4 pb-2">
   622→      <button onClick={onBack} className="inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground mb-4 focus-visible:outline-2 focus-visible:outline-primary rounded" aria-label="Retour">
   623→        <ArrowLeft size={15} /> Nouveau diagnostic
   624→      </button>
   625→      <div className="flex items-center justify-between gap-2">
   626→        <h1 className="font-heading font-black text-xl">Mon historique</h1>
   627→        <button
   628→          onClick={() => { setCompareMode(!compareMode); setCompareSel([]); }}
   629→          aria-pressed={compareMode}
   630→          className={`h-9 px-3 rounded-full text-[11px] font-bold flex items-center gap-1.5 active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${compareMode ? "bg-primary text-primary-foreground" : "border border-border bg-card"}`}
   631→        >
   632→          <GitCompareArrows size={14} /> Comparer
   633→        </button>
   634→      </div>
   635→
   636→      {history === null ? (
   637→        <div className="space-y-3 mt-5">
   638→          {[0, 1, 2].map((i) => (
   639→            <Skeleton key={i} className="h-20 rounded-2xl" />
   640→          ))}
   641→        </div>
   642→      ) : list.length === 0 ? (
   643→        <div className="mt-6 rounded-2xl border border-dashed border-border bg-card/60 p-8 text-center">
   644→          <NeaOnnimIcon size={38} className="mx-auto text-primary" />
   645→          <p className="font-heading font-bold text-sm mt-3">Aucun diagnostic pour l&apos;instant</p>
   646→          <p className="text-xs text-muted-foreground mt-1">Ton premier scan débloquera le suivi d&apos;évolution.</p>
   647→        </div>
   648→      ) : (
   649→        <>
   650→          <div className="space-y-2.5 mt-4">
   651→            {list.map((d) => {
   652→              const sel = compareSel.includes(d.id);
   653→              return (
   654→                <div key={d.id} className={`flex items-center gap-3 rounded-2xl border p-2.5 transition-colors ${sel ? "border-primary bg-primary/5" : "border-border bg-card"}`}>
   655→                  <button
   656→                    onClick={() => {
   657→                      if (compareMode) {
   658→                        setCompareSel(sel ? compareSel.filter((x) => x !== d.id) : compareSel.length < 2 ? [...compareSel, d.id] : [compareSel[1], d.id]);
   659→                      } else onOpen(d);
   660→                    }}
   661→                    className="flex items-center gap-3 flex-1 min-w-0 text-left focus-visible:outline-2 focus-visible:outline-primary rounded-xl"
   662→                    aria-label={`Diagnostic ${d.zone} du ${formatDate(d.createdAt)}, score ${d.scoreGlobal}`}
   663→                  >
   664→                    <img src={diagImgSrc(d.imageData)} alt={`Diagnostic ${d.zone}`} loading="lazy" className="h-14 w-14 rounded-xl object-cover shrink-0" />
   665→                    <div className="min-w-0">
   666→                      <p className="text-sm font-semibold">{BODY_ZONES.find((z) => z.id === d.zone)?.label ?? d.zone}</p>
   667→                      <p className="text-[11px] text-muted-foreground">{formatDate(d.createdAt)}</p>

   674→                      aria-pressed={sel}
   675→                      aria-label="Sélectionner pour comparaison"
   676→                      className={`h-8 w-8 grid place-items-center rounded-full border-2 shrink-0 transition-colors ${sel ? "bg-primary border-primary text-primary-foreground" : "border-border"}`}
