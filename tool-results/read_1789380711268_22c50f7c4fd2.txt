   188→  /* Applique un code promo : aperçu de remise sans consommer le coupon
   189→   * (la consommation a lieu à la commande — toutes les gardes côté serveur). */
   190→  async function applyPromo() {
   191→    const code = promoInput.trim().toUpperCase();
   192→    if (!code || promoChecking) return;
   193→    setPromoChecking(true);
   194→    try {
   195→      const r = await apiPost<{ ok: true; coupon: { code: string; label: string | null }; discount: number; total: number }>(
   196→        "/api/coupons/validate",
   197→        { code, userId: user.id, subtotal }

   199→      setPromo({ code: r.coupon.code, label: r.coupon.label, discount: r.discount, subtotal });
   200→      setPromoInput("");
   201→      toast.success(`Code ${r.coupon.code} appliqué`, { description: `Remise de ${xof(r.discount)} déduite du total.` });
   202→    } catch (e) {
   203→      setPromo(null);
   204→      toast.error(e instanceof Error ? e.message : "Code promo invalide");
   205→    } finally {
   206→      setPromoChecking(false);
   207→    }
   208→  }
   209→
   210→  /** Passerelle paiement : vérification d'identité par code si la sécurité
   211→   *  renforcée est active — le paiement initialement prévu (pay) n'est lancé
   212→   *  qu'une fois le code confirmé ; annulé sinon, rien n'est engagé. */
   213→  function startPay(method: PayMethod) {
   214→    if (securityEnabled) {
   215→      haptic(HAPTIC.tap);
   216→      setPendingPay(method);
   217→      return;
   218→    }
   219→    void pay(method);
   220→  }
   221→
   222→  async function pay(method: PayMethod) {
   223→    setPaying(true);
   224→    try {
   225→      const items = cart.map((l) => ({ productId: l.productId, qty: l.qty }));
   226→      const r = await apiPost<{ order: ApiOrder; payment: ApiPayment | null; paid: boolean }>("/api/orders", {
   227→        userId: user.id,
   228→        items,
   229→        paymentMethod: method,
   230→        ...(livePromo ? { couponCode: livePromo.code } : {}),
   231→      });
   232→      const amount = r.order.total;
   233→      if (method !== "wallet" && r.payment) {
   234→        // Contrat confirmToken (63-b/63-c) : un paiement mobile money en attente
   235→        // porte son jeton — absent, on n'appelle JAMAIS confirm et on rollback
   236→        // l'état UI comme un échec de paiement (panier intact, aucun overlay).
   237→        const confirmToken = r.payment.confirmToken;
   238→        if (!confirmToken) {
   239→          setCheckout(false);
   240→          toast.error("Paiement impossible — réessaie dans quelques instants");
   241→          return;
   242→        }
   243→        setCheckout(false);
   244→        setPayState({ phase: "processing", method, amount });
   245→        await new Promise((res) => setTimeout(res, 3000));
   246→        const c = await apiPost<{ payment: ApiPayment; order?: ApiOrder; wallet?: ApiWallet }>("/api/payments/confirm", { paymentId: r.payment.id, confirmToken });
   247→        if (c.wallet) { setWallet(c.wallet); setWalletError(false); }
   248→      } else {
   249→        setCheckout(false);
   250→        setPayState({ phase: "processing", method: "wallet", amount });
   251→        await new Promise((res) => setTimeout(res, 1200));
   252→      }
   253→      setPayState({ phase: "success", method, amount });
   254→      haptic(HAPTIC.success);
   255→      clearCart();
   256→      setPromo(null); // le coupon est consommé : remise à zéro pour la prochaine commande
   257→      void refreshOrders(); // la nouvelle commande apparaît dans « Mes commandes »
   258→      const cb = r.order.cashback;
   259→      toast.success(cb > 0 ? `Commande confirmée — cashback ${xof(cb)} crédité sur ton wallet` : "Commande confirmée et payée");
   260→      if (method === "wallet") {
   261→        const w = await apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).catch(() => null);
   262→        if (w) setWallet(w.wallet);
   263→      }
   264→    } catch (e) {
   265→      setPayState(null);
   266→      toast.error(e instanceof Error ? e.message : "Paiement impossible");
   267→    } finally {
   268→      setPaying(false);
   269→    }
   270→  }
   271→
   272→  const op = (m: PayMethod) => MOMO_OPERATORS.find((o) => o.code === m);
   273→
   274→  return (
   275→    <div className="pt-4 pb-2">
   276→      <Reveal>
   277→        <RevealItem>
   278→          <header className="flex items-center justify-between">
   279→            <h1 className="flex items-center gap-2 font-heading font-black text-xl tracking-tight">
   280→              {/* Mark de marque (t. 86) — signature vectorielle de la boutique */}

   282→                <KeneMark size={26} />
   283→              </span>
   284→              Boutique Kènè

   288→        </RevealItem>
   289→        <RevealItem>
   290→          {/* Bascule Catalogue ↔ Mes commandes — la cliente consulte ses données */}

   293→              role="tab"
   294→              aria-selected={view === "catalogue"}
   295→              onClick={() => setView("catalogue")}

   301→              role="tab"
   302→              aria-selected={view === "commandes"}

   307→            </button>
   308→          </div>

   315→        <>
   316→      {/* Le Marché vivant (t. 83-c) — le marché 3D coiffe la boutique ; le tap

   331→              vendeur (MAISON Kènè + instituts partenaires). La carte choisie
   332→              filtre tout le catalogue sur CET institut. */}
   333→          {products === null ? (
   334→            <div className="mt-3 flex gap-2" aria-hidden="true">
   335→              {[0, 1, 2].map((i) => (
   336→                <Shimmer key={i} className="h-[78px] w-[142px] shrink-0 rounded-[20px]" />

   344→              <div className="flex gap-2 overflow-x-auto pb-1 scrollbar-thin -mx-1 px-1">
   345→                <SellerCard
   346→                  selected={institut === ""}
   347→                  totalCount={(products ?? []).length}

   349→                />
   350→                {sellers.map((s) => (
   351→                  <SellerCard
   352→                    key={s.key}
   353→                    seller={s}
   354→                    selected={institut === (s.maison ? "maison" : s.key)}
   355→                    onClick={() => { setInstitut(s.maison ? "maison" : s.key); haptic(HAPTIC.tap); }}
   356→                  />
   357→                ))}
   358→              </div>
   359→            </div>
   360→          )}
   361→        </RevealItem>
   362→
   363→        <RevealItem>
   364→          {/* Le Fil de Kente — hero tissé, le fil de la catégorie s'illumine */}
   365→          <div className="mt-4">
   366→            <KenteWeaveCard highlightIndex={cat ? categoryThread(cat) : -1} caption={weaveCaption} />
   367→          </div>
   368→        </RevealItem>
   369→
   370→        <RevealItem>
   371→          {/* Recherche */}
   372→          <div className="mt-3 relative">
   373→            <Search size={16} className="absolute left-3.5 top-1/2 -translate-y-1/2 text-muted-foreground" aria-hidden="true" />
   374→            <input
   375→              ref={inputRef}
   376→              value={q}
   377→              onChange={(e) => setQ(e.target.value)}
   378→              placeholder="Rechercher un soin, un botanique…"
   379→              aria-label="Rechercher un produit"
   380→              className="k-input h-12 w-full rounded-2xl pl-10 pr-4 text-sm"
   381→            />
   382→          </div>
   383→        </RevealItem>
   384→
   385→        <RevealItem>
   386→          {/* Filtres : catégories + favoris (bascules aria-pressed, cumulables) — Chip verre→or 2026 */}
   387→          <div className="flex gap-2 overflow-x-auto py-3 scrollbar-thin -mx-1 px-1" role="group" aria-label="Filtres de la boutique">
   388→            {SHOP_CATEGORIES.map((c) => (
   389→              <Chip key={c.id} selected={cat === c.id} onClick={() => setCat(c.id)} className="min-h-11 shrink-0">

   402→          </div>
   403→        </RevealItem>
   404→      </Reveal>
   405→
   406→      <div ref={gridRef} className="scroll-mt-4">
   407→      {/* Grille produits */}
   408→      {products === null ? (
   409→        <div className="grid grid-cols-2 gap-3.5 sm:grid-cols-3">
   410→          {[0, 1, 2, 3].map((i) => (
   411→            <div key={i} className="space-y-2">
   412→              <Shimmer className="aspect-square rounded-[18px]" />
   413→              <Shimmer className="h-3.5 w-4/5 rounded" />
   414→              <Shimmer className="h-3 w-2/3 rounded" />
   415→            </div>
