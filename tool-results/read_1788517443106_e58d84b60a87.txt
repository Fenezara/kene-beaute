     5→import {

    10→import { apiGet, apiPost } from "@/lib/kene/api";
    11→import { addDays, formatDate, formatTime, xof, DEPOSIT_RATE } from "@/lib/kene/format";
    12→import { cancellationRefund } from "@/lib/kene/rfm";
    13→import { SankofaIcon } from "@/components/kene/icons";
    14→import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";
    15→import { Skeleton } from "@/components/ui/skeleton";
    16→import { Tabs, TabsList, TabsTrigger } from "@/components/ui/tabs";
    17→import { useKene } from "@/store/kene";
    18→import type { ApiAppointment, ApiInstitute, ApiPayment, ApiResource, ApiReview, ApiService, ApiSlot, ApiWallet } from "./types";
    19→import { ApptBadge, EmptyBlock, MomoProcessing, SectionTitle, Stars } from "./bits";
    20→
    21→type PayMethod = "wave" | "wallet";
    22→
    23→function isoDate(d: Date): string {
    24→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
    25→}
    26→
    27→export function BookingScreen() {
    28→  const user = useKene((s) => s.user)!;
    29→  const [tab, setTab] = useState<"book" | "mine">("book");
    30→
    31→  // ── Réserver ──
    32→  const [institutes, setInstitutes] = useState<ApiInstitute[] | null>(null);
    33→  const [inst, setInst] = useState<ApiInstitute | null>(null);
    34→  const [services, setServices] = useState<ApiService[]>([]);
    35→  const [resources, setResources] = useState<ApiResource[]>([]);
    36→  const [reviews, setReviews] = useState<ApiReview[]>([]);
    37→  const [detailLoading, setDetailLoading] = useState(false);
    38→  const [service, setService] = useState<ApiService | null>(null);
    39→  const [day, setDay] = useState<Date>(new Date());
    40→  const [slots, setSlots] = useState<ApiSlot[] | null>(null);
    41→  const [slot, setSlot] = useState<ApiSlot | null>(null);
    42→  const [wallet, setWallet] = useState<ApiWallet | null>(null);
    43→  const [payMethod, setPayMethod] = useState<PayMethod>("wave");
    44→  const [paying, setPaying] = useState(false);
    45→  const [payOverlay, setPayOverlay] = useState<{ phase: "processing" | "done"; amount: number } | null>(null);
    46→  const [confirmed, setConfirmed] = useState<ApiAppointment | null>(null);
    47→
    48→  // ── Mes RDV ──
    49→  const [mine, setMine] = useState<ApiAppointment[] | null>(null);
    50→  const [reviewFor, setReviewFor] = useState<ApiAppointment | null>(null);
    51→  const [rating, setRating] = useState(5);
    52→  const [comment, setComment] = useState("");
    53→  const [cancelFor, setCancelFor] = useState<ApiAppointment | null>(null);
    54→
    55→  const days = useMemo(() => Array.from({ length: 7 }, (_, i) => addDays(new Date(), i)), []);
    56→  const deposit = service ? Math.round(service.price * DEPOSIT_RATE) : 0;
    57→  const practitioner = useMemo(() => (slot?.resourceIds?.length ? resources.find((r) => r.id === slot.resourceIds![0]) ?? null : null), [slot, resources]);
    58→
    59→  const loadInstitutes = useCallback(async () => {
    60→    setInstitutes(null);
    61→    try {
    62→      const r = await apiGet<{ institutes: ApiInstitute[] }>("/api/institutes");
    63→      setInstitutes(r.institutes ?? []);
    64→    } catch {
    65→      setInstitutes([]);
    66→    }
    67→  }, []);
    68→
    69→  const loadMine = useCallback(async () => {
    70→    setMine(null);
    71→    try {
    72→      const r = await apiGet<{ appointments: ApiAppointment[] }>(`/api/appointments?userId=${user.id}`);
    73→      setMine(r.appointments ?? []);
    74→    } catch {
    75→      setMine([]);
    76→    }
    77→  }, [user.id]);
    78→
    79→  useEffect(() => {
    80→    loadInstitutes();
    81→    apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).then((r) => setWallet(r.wallet)).catch(() => {});
    82→  }, [loadInstitutes, user.id]);
    83→
    84→  useEffect(() => {
    85→    if (tab === "mine") loadMine();
    86→  }, [tab, loadMine]);
    87→
    88→  async function openInstitute(i: ApiInstitute) {
    89→    setInst(i);
    90→    setService(null);
    91→    setSlot(null);
    92→    setSlots(null);
    93→    setDetailLoading(true);
    94→    try {
    95→      const r = await apiGet<{ institute: ApiInstitute; services: ApiService[]; resources: ApiResource[]; reviews: ApiReview[] }>(`/api/institutes/${i.id}`);
    96→      setServices(r.services ?? []);
    97→      setResources(r.resources ?? []);
    98→      setReviews(r.reviews ?? []);
    99→    } catch {
   100→      toast.error("Institut indisponible");
   101→      setInst(null);
   102→    } finally {
   103→      setDetailLoading(false);
   104→    }
   105→  }
   106→
   107→  async function pickDay(d: Date) {
   108→    setDay(d);
   109→    setSlot(null);
   110→    setSlots(null);
   111→    if (!service || !inst) return;
   112→    try {
   113→      const r = await apiGet<{ slots: ApiSlot[] }>(`/api/institutes/${inst.id}/availability?date=${isoDate(d)}&serviceId=${service.id}`);
   114→      setSlots(r.slots ?? []);
   115→    } catch {
   116→      setSlots([]);
   117→    }
   118→  }
   119→
   120→  async function chooseService(s: ApiService) {
   121→    setService(s);
   122→    setSlot(null);
   123→    setSlots(null);
   124→    if (!inst) return;
   125→    try {
   126→      const r = await apiGet<{ slots: ApiSlot[] }>(`/api/institutes/${inst.id}/availability?date=${isoDate(day)}&serviceId=${s.id}`);
   127→      setSlots(r.slots ?? []);
   128→    } catch {
   129→      setSlots([]);
   130→    }
   131→  }
   132→
   133→  async function book() {
   134→    if (!inst || !service || !slot) return;
   135→    setPaying(true);
   136→    try {
   137→      const [h, m] = slot.time.split(":").map(Number);
   138→      const startAt = new Date(day);
   139→      startAt.setHours(h, m, 0, 0);
   140→      const r = await apiPost<{ appointment: ApiAppointment; payment?: ApiPayment }>("/api/appointments", {
   141→        tenantId: inst.id,
   142→        serviceId: service.id,
   143→        resourceId: slot.resourceIds?.[0] ?? resources[0]?.id,
   144→        startAt: startAt.toISOString(),
   145→        clientName: user.name,
   146→        clientPhone: user.phone,
   147→        userId: user.id,
   148→        depositAmount: deposit,
   149→        paymentMethod: payMethod,
   150→      });
   151→      if (payMethod === "wave" && r.payment) {
   152→        setPayOverlay({ phase: "processing", amount: deposit });
   153→        await new Promise((res) => setTimeout(res, 3000));
   154→        await apiPost("/api/payments/confirm", { paymentId: r.payment.id });
   155→      } else {
   156→        setPayOverlay({ phase: "processing", amount: deposit });
   157→        await new Promise((res) => setTimeout(res, 1200));
   158→      }
   159→      setPayOverlay({ phase: "done", amount: deposit });
   160→      setTimeout(() => setPayOverlay(null), 1400);
   161→      setConfirmed(r.appointment);
   162→      const w = await apiGet<{ wallet: ApiWallet }>(`/api/wallet?userId=${user.id}`).catch(() => null);
   163→      if (w) setWallet(w.wallet);
   164→    } catch (e) {
   165→      setPayOverlay(null);
   166→      toast.error(e instanceof Error ? e.message : "Réservation impossible");
   167→    } finally {
   168→      setPaying(false);
   169→    }
   170→  }
   171→
   172→  async function submitReview() {
   173→    if (!reviewFor) return;
   174→    try {
   175→      await apiPost(`/api/appointments/${reviewFor.id}/review`, { userId: user.id, rating, comment: comment.trim() || undefined });
   176→      toast.success("Merci pour ton avis");
   177→      setReviewFor(null);
   178→      setComment("");
   179→      setRating(5);
   180→      loadMine();
   181→    } catch (e) {
   182→      toast.error(e instanceof Error ? e.message : "Envoi impossible");
   183→    }
   184→  }
   185→
   186→  async function doCancel() {
   187→    if (!cancelFor) return;
   188→    try {
   189→      const r = await apiPost<{ appointment: ApiAppointment; refund: { amount: number; label: string } }>(`/api/appointments/${cancelFor.id}/cancel`, { userId: user.id });
   190→      toast.success(r.refund.amount > 0 ? `RDV annulé — ${xof(r.refund.amount)} remboursés sur ton wallet` : "RDV annulé");
   191→      setCancelFor(null);
   192→      loadMine();
   193→    } catch (e) {
   194→      toast.error(e instanceof Error ? e.message : "Annulation impossible");
   195→    }
   196→  }
   197→
   198→  const now = Date.now();
   199→  const upcoming = (mine ?? []).filter((a) => new Date(a.startAt).getTime() >= now && a.status !== "cancelled").sort((x, y) => +new Date(x.startAt) - +new Date(y.startAt));
   200→  const past = (mine ?? []).filter((a) => new Date(a.startAt).getTime() < now || a.status === "cancelled" || a.status === "completed").sort((x, y) => +new Date(y.startAt) - +new Date(x.startAt));
   201→
   202→  return (
   203→    <div className="pt-4 pb-2">
   204→      <Tabs value={tab} onValueChange={(v) => setTab(v as "book" | "mine")}>
   205→        <TabsList className="w-full grid grid-cols-2 bg-muted/60 rounded-2xl p-1">
   206→          <TabsTrigger value="book" className="rounded-xl data-[state=active]:bg-card data-[state=active]:text-primary font-semibold text-sm py-2.5">Réserver</TabsTrigger>
   207→          <TabsTrigger value="mine" className="rounded-xl data-[state=active]:bg-card data-[state=active]:text-primary font-semibold text-sm py-2.5">Mes RDV</TabsTrigger>
   208→        </TabsList>
   209→      </Tabs>
   210→
   211→      {/* ═══════ RÉSERVER ═══════ */}
   212→      {tab === "book" && (
   213→        <AnimatePresence mode="wait">
   214→          {confirmed ? (
   215→            /* — Confirmation — */
   216→            <motion.div key="ok" initial={{ opacity: 0, scale: 0.96 }} animate={{ opacity: 1, scale: 1 }} className="pt-8 flex flex-col items-center text-center">
   217→              <span className="grid place-items-center h-20 w-20 rounded-full bg-[#3F7D3F]/15 text-[#3F7D3D]">
   218→                <SankofaIcon size={38} />
   219→              </span>
   220→              <h2 className="font-heading font-black text-xl mt-4">Rendez-vous confirmé</h2>
   221→              <div className="mt-4 w-full rounded-2xl border border-border bg-card p-4 text-left space-y-1.5">
   222→                <p className="font-semibold text-sm">{confirmed.service?.name ?? "Soin"}</p>
   223→                <p className="text-xs text-muted-foreground">{confirmed.tenant?.name}</p>
   224→                <p className="text-xs text-muted-foreground flex items-center gap-1.5"><CalendarDays size={13} /> {formatDate(confirmed.startAt)} · <Clock size={13} /> {formatTime(confirmed.startAt)}</p>
   225→                {confirmed.resource?.name && <p className="text-xs text-muted-foreground flex items-center gap-1.5"><Users size={13} /> {confirmed.resource.name}</p>}
   226→                {confirmed.depositAmount > 0 && <p className="text-xs font-mono text-primary font-bold pt-1 border-t border-dashed border-border">Acompte réglé : {xof(confirmed.depositAmount)}</p>}
   227→              </div>
   228→              <p className="mt-4 inline-flex items-center gap-1.5 rounded-full bg-[#3F7D3F]/10 text-[#3F7D3D] px-3 py-1.5 text-[11px] font-semibold">
   229→                <MessageSquareQuote size={13} /> Rappel SMS J-1 programmé (simulé)
   230→              </p>
   231→              <div className="mt-6 grid grid-cols-2 gap-3 w-full">
   232→                <button onClick={() => { setConfirmed(null); setInst(null); setService(null); loadInstitutes(); }} className="h-12 rounded-xl border border-primary/60 text-primary text-sm font-bold active:scale-[0.98] transition-transform focus-visible:outline-2 focus-visible:outline-primary">
   233→                  Autre RDV
   234→                </button>
   235→                <button onClick={() => setTab("mine")} className="h-12 rounded-xl bg-primary text-primary-foreground text-sm font-bold active:scale-[0.98] transition-transform focus-visible:outline-2 focus-visible:outline-primary">
   236→                  Mes RDV
   237→                </button>
   238→              </div>
   239→            </motion.div>
   240→          ) : !inst ? (
   241→            /* — Liste instituts — */
   242→            <motion.div key="list" initial={{ opacity: 0, x: 30 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -30 }}>
   243→              <SectionTitle icon={<MapPin size={16} />}>Instituts partenaires</SectionTitle>
   244→              {institutes === null ? (
   245→                <div className="space-y-3">
   246→                  {[0, 1].map((i) => <Skeleton key={i} className="h-36 rounded-2xl" />)}
   247→                </div>
   248→              ) : institutes.length === 0 ? (
   249→                <EmptyBlock icon={<MapPin size={22} />} title="Aucun institut pour l'instant" text="De nouveaux instituts Kènè arrivent à Abidjan et Dakar." />
   250→              ) : (
   251→                <div className="space-y-3">
   252→                  {institutes.map((i) => (
   253→                    <button key={i.id} onClick={() => openInstitute(i)} className="w-full text-left rounded-2xl border border-border bg-card overflow-hidden shadow-sm active:scale-[0.99] transition-transform hover:border-primary/40 focus-visible:outline-2 focus-visible:outline-primary">
   254→                      <div className="relative">
   255→                        <img src={i.image} alt={i.name} loading="lazy" className="h-32 w-full object-cover" />
   256→                        <div className="absolute inset-0 bg-gradient-to-t from-[#1A1410]/70 to-transparent" />
   257→                        <div className="absolute bottom-2 left-3 right-3 flex items-end justify-between gap-2 text-[#F8F1E4]">
   258→                          <div>
   259→                            <p className="font-heading font-bold text-sm leading-tight">{i.name}</p>
   260→                            <p className="text-[10px] opacity-85 flex items-center gap-1"><MapPin size={10} /> {i.city}, {i.country}</p>
   261→                          </div>
   262→                          <span className="flex items-center gap-1 rounded-full bg-[#1A1410]/60 backdrop-blur px-2 py-0.5 text-[10px] font-bold">
   263→                            <Star size={10} className="fill-[#C8951E] text-[#C8951E]" /> {i.rating.toFixed(1)} ({i.reviewCount})
   264→                          </span>
   265→                        </div>
   266→                      </div>
   267→                      <div className="px-3 py-2.5 flex items-center justify-between text-[11px] text-muted-foreground">
   268→                        <span>{i._count?.services ?? 0} soins · {i._count?.reviews ?? 0} avis</span>
   269→                        <span className="flex items-center gap-0.5 text-primary font-semibold">Réserver <ChevronRight size={13} /></span>
   270→                      </div>
   271→                    </button>
   272→                  ))}
   273→                </div>
   274→              )}
   275→            </motion.div>
   276→          ) : (
   277→            /* — Détail institut — */
   278→            <motion.div key="detail" initial={{ opacity: 0, x: 30 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -30 }}>
   279→              <button onClick={() => setInst(null)} className="mt-4 inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground focus-visible:outline-2 focus-visible:outline-primary rounded" aria-label="Retour aux instituts">
   280→                <ArrowLeft size={15} /> Tous les instituts
   281→              </button>
   282→              <div className="relative mt-3 rounded-2xl overflow-hidden">
   283→                <img src={inst.image} alt={inst.name} className="h-44 w-full object-cover" />
   284→                <div className="absolute inset-0 bg-gradient-to-t from-[#1A1410]/80 to-transparent" />
   285→                <div className="absolute bottom-3 left-4 right-4 text-[#F8F1E4]">
   286→                  <h2 className="font-heading font-black text-lg">{inst.name}</h2>
   287→                  <p className="text-[11px] opacity-90 flex items-center gap-2">
   288→                    <span className="flex items-center gap-1"><Star size={11} className="fill-[#C8951E] text-[#C8951E]" /> {inst.rating.toFixed(1)}</span>
   289→                    <span className="flex items-center gap-1"><MapPin size={11} /> {inst.city}</span>
   290→                    <span className="flex items-center gap-1"><Clock size={11} /> {inst.openingHour}h–{inst.closingHour}h</span>
   291→                  </p>
   292→                </div>
   293→              </div>
   294→              {inst.description && <p className="text-xs text-muted-foreground leading-relaxed mt-3">{inst.description}</p>}
   295→
   296→              {detailLoading ? (
   297→                <div className="space-y-2 mt-4">{[0, 1, 2].map((i) => <Skeleton key={i} className="h-16 rounded-xl" />)}</div>
   298→              ) : (
   299→                <>
   300→                  {/* Avis */}
   301→                  {reviews.length > 0 && (
   302→                    <div className="mt-4">
   303→                      <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wide mb-2">Ce que disent les clientes</p>
   304→                      <div className="flex gap-2.5 overflow-x-auto pb-1 scrollbar-thin">
   305→                        {reviews.slice(0, 5).map((rv) => (
   306→                          <div key={rv.id} className="shrink-0 w-56 rounded-2xl border border-border bg-card p-3">
   307→                            <div className="flex items-center justify-between">
   308→                              <p className="text-xs font-semibold">{rv.user?.name ?? "Cliente"}</p>
   309→                              <Stars rating={rv.rating} size={9} />
   310→                            </div>
   311→                            <p className="text-[11px] text-muted-foreground mt-1.5 leading-snug line-clamp-3">{rv.comment ?? "Expérience agréable."}</p>
   312→                          </div>
   313→                        ))}
   314→                      </div>
   315→                    </div>
   316→                  )}
   317→
   318→                  {/* Services */}
   319→                  <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wide mt-5 mb-2">Soins & tarifs</p>
   320→                  <div className="space-y-2">
   321→                    {services.map((s) => (
   322→                      <div key={s.id} className={`rounded-2xl border p-3.5 transition-colors ${service?.id === s.id ? "border-primary bg-primary/5" : "border-border bg-card"}`}>
   323→                        <div className="flex items-start justify-between gap-2">
   324→                          <div className="min-w-0">
   325→                            <p className="text-sm font-semibold">{s.name}</p>
   326→                            <p className="text-[11px] text-muted-foreground mt-0.5 flex items-center gap-2">
   327→                              <span className="flex items-center gap-1"><Clock size={11} /> {s.durationMin} min</span>
   328→                              {s.botanicals && <span className="text-terre truncate">{s.botanicals}</span>}
   329→                            </p>
   330→                          </div>
   331→                          <p className="font-mono text-sm font-bold shrink-0">{xof(s.price)}</p>
   332→                        </div>
   333→                        <button onClick={() => chooseService(s)} className={`mt-2.5 h-10 w-full rounded-xl text-xs font-bold active:scale-[0.98] transition-all focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary ${service?.id === s.id ? "bg-primary text-primary-foreground" : "border border-primary/50 text-primary"}`}>
   334→                          {service?.id === s.id ? "Choisi" : "Choisir"}
   335→                        </button>
   336→                      </div>
   337→                    ))}
   338→                  </div>
   339→
   340→                  {/* Jours + créneaux */}
   341→                  {service && (
   342→                    <motion.div initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} className="mt-6">
   343→                      <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wide mb-2">Choisis ton jour</p>
   344→                      <div className="flex gap-2 overflow-x-auto pb-1 scrollbar-thin">
   345→                        {days.map((d) => {
   346→                          const sel = isoDate(d) === isoDate(day);
   347→                          return (
   348→                            <button key={d.toISOString()} onClick={() => pickDay(d)} aria-pressed={sel} className={`shrink-0 rounded-2xl border px-3.5 py-2 text-center transition-all active:scale-95 focus-visible:outline-2 focus-visible:outline-primary ${sel ? "border-primary bg-primary text-primary-foreground" : "border-border bg-card"}`}>
   349→                              <span className="block text-[9px] uppercase font-bold opacity-70">{formatDate(d, { weekday: "short" })}</span>
   350→                              <span className="block font-mono text-sm font-bold mt-0.5">{d.getDate()}</span>
   351→                            </button>
   352→                          );
   353→                        })}
   354→                      </div>
   355→
   356→                      <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wide mt-5 mb-2">Créneaux — {formatDate(day, { weekday: "long", day: "numeric", month: "long" })}</p>
   357→                      {slots === null ? (
   358→                        <div className="flex justify-center py-6"><Loader2 size={22} className="animate-spin text-primary" /></div>
   359→                      ) : slots.length === 0 ? (
   360→                        <p className="text-xs text-muted-foreground text-center py-4">Aucun créneau affiché — choisis un autre jour.</p>
   361→                      ) : (

   363→                          {slots.map((s) => (

   366→                              disabled={!s.available}

   371→                              }`}
   372→                            >
   373→                              {s.time}
   374→                            </button>
   375→                          ))}
   376→                        </div>
   377→                      )}
   378→                    </motion.div>
   379→                  )}
   380→
   381→                  {/* Récap + paiement */}
   382→                  {service && slot && (
