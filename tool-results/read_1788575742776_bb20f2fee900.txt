   131→  async function onFile(f: File | undefined) {

   159→  async function launch() {
   160→    if (!image) return;
   161→    setStep(2);
   162→    setAnalyzing(true);
   163→    setProgress(0);
   164→    setCheckedSteps(0);
   165→    setNetNotice(null);

   168→      setProgress((p) => Math.min(96, p + Math.random() * 7 + 2));

   170→    const steps = setInterval(() => {

   175→
   176→    // Résultat acquis (POST direct ou récupéré après un 502) : même animation,
   177→    // durée minimale ~3,4 s, minuteurs suivis et nettoyés dans tous les cas.
   178→    function showResult(d: { id: string; result: DiagnosisResult; imageData: string; createdAt: string }, recovered = false) {
   179→      const wait = Math.max(0, 3400 - (Date.now() - t0));
   180→      const to1 = setTimeout(() => {
   181→        setProgress(100);
   182→        setCheckedSteps(ANALYSIS_STEPS.length);
   183→        const to2 = setTimeout(() => {
   184→          setDiag(d);
   185→          setAnalyzing(false);
   186→          setNetNotice(null);
   187→          setStep(3);
   188→          clearTimers();
   189→          if (recovered) {
   190→            toast.success("Analyse retrouvée — voici tes résultats", {
   191→              description: "Le réseau a été instable une seconde, mais ton diagnostic était bien enregistré.",
   192→            });
   193→          }
   194→        }, 450);
   195→        timersRef.current.timeouts.push(to2);
   196→      }, wait);
   197→      timersRef.current.timeouts.push(to1);
   198→    }
   199→
   200→    // Récupération après échec réseau : le diagnostic a très bien pu être
   201→    // ENREGISTRÉ avant le redémarrage du serveur (le back crée la ligne
   202→    // « pending » puis la passe à « done » avant de répondre). On va le
   203→    // chercher : done récent (< 4 min) → résultats ; pending → re-poll 3 s
   204→    // × 10 max. Retourne true si les résultats sont affichés.
   205→    async function tryRecover(): Promise<boolean> {
   206→      const RECENT_MS = 4 * 60 * 1000;
   207→      const findRecent = (list: ApiDiagnosis[] | undefined): ApiDiagnosis | undefined =>
   208→        (list ?? [])
   209→          .filter(
   210→            (d) =>
   211→              d.zone === zone &&
   212→              (d.status === "done" || d.status === "pending") &&
   213→              Date.now() - new Date(d.createdAt).getTime() < RECENT_MS,
   214→          )
   215→          .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())[0];
   216→      try {
   217→        const first = findRecent((await apiGet<{ diagnoses: ApiDiagnosis[] }>(`/api/diagnoses?userId=${user.id}`)).diagnoses);
   218→        let found = first;
   219→        let polls = 0;
   220→        while (found?.status === "pending" && polls < 10) {
   221→          polls += 1;
   222→          setNetNotice("Récupération de ton analyse…");
   223→          await new Promise((res) => setTimeout(res, 3000));
   224→          const again = await apiGet<{ diagnoses: ApiDiagnosis[] }>(`/api/diagnoses?userId=${user.id}`);
   225→          found = findRecent(again.diagnoses);
   226→        }
   227→        if (!found || found.status !== "done") return false;
   228→        const result = parseDiagnosis(found.resultJson);
   229→        if (!result) return false;
   230→        showResult(
   231→          { id: found.id, result, imageData: diagImgSrc(found.imageData), createdAt: found.createdAt },
   232→          true,
   233→        );
   234→        return true;
   235→      } catch {
   236→        // Le GET lui-même est injoignable (serveur toujours au redémarrage) :
   237→        // échec final honnête, sans crash.
   238→        return false;
   239→      }
   240→    }
   241→
   242→    try {
   243→      // POST /api/diagnoses : 3 tentatives quand la gateway renvoie 502/503/504
   244→      // (fenêtre transitoire de redémarrage du serveur Next) — backoff 1,5 s
   245→      // puis 4 s. Les autres erreurs (400/404/429…) ne sont JAMAIS rejouées.
   246→      let r: { diagnosis: ApiDiagnosis } | undefined;
   247→      let fatal: unknown = new Error("Analyse impossible");
   248→      const backoffs = [1500, 4000];
   249→      for (let attempt = 0; attempt < 3; attempt += 1) {
   250→        try {
   251→          r = await apiPost<{ diagnosis: ApiDiagnosis }>("/api/diagnoses", {
   252→            userId: user.id,
   253→            zone,
   254→            image,
   255→            fitzpatrick: user.fitzpatrick ?? undefined,
   256→            allergies: user.allergies ?? undefined,
   257→          });
   258→          break;
   259→        } catch (e) {
   260→          fatal = e;
   261→          const retryable = e instanceof ApiError && [502, 503, 504].includes(e.status);
   262→          if (!retryable || attempt === 2) break;
   263→          setNetNotice("Rétablissement de la connexion…");
   264→          await new Promise((res) => setTimeout(res, backoffs[attempt]));
   265→          setNetNotice(null);
   266→        }
   267→      }
   268→      if (!r) throw fatal;
   269→      const result = parseDiagnosis(r.diagnosis.resultJson);
   270→      if (!result) throw new Error("Résultat IA illisible");
   271→      showResult({ id: r.diagnosis.id, result, imageData: image, createdAt: r.diagnosis.createdAt });
   272→    } catch (e) {
   273→      // 502/503/504 après 3 tentatives : tenter la récupération AVANT l'erreur.
   274→      const gatewayish = e instanceof ApiError && [502, 503, 504].includes(e.status);
   275→      if (gatewayish && (await tryRecover())) return;
   276→      clearTimers();
   277→      setAnalyzing(false);
   278→      setNetNotice(null);
   279→      toast.error(
   280→        gatewayish
   281→          ? "Moteur d'analyse momentanément injoignable — tes photos restent prêtes, retente dans un instant."
   282→          : e instanceof Error
   283→            ? e.message
   284→            : "Analyse impossible",
   285→      );
   286→      setStep(1); // retour capture : la photo est conservée
   287→    }
   288→  }
   289→

   304→                key={z.id}

   308→              >
   309→                <div className="flex items-start justify-between">
   310→                  <span className="grid place-items-center h-10 w-10 rounded-xl bg-primary/10 text-primary">
   311→                    <Icon size={20} />
   312→                  </span>
   313→                  <span className="rounded-full bg-muted px-2 py-0.5 font-mono text-[10px] font-bold">{Math.round(z.weight * 100)} %</span>

   317→              </motion.button>

   319→          })}
   320→        </div>

   322→          <History size={16} className="text-primary" /> Voir mon historique

   334→          <ArrowLeft size={15} /> Changer de zone
   335→        </button>
   336→        <span className="rounded-full bg-primary/10 text-primary px-3 py-1 text-[11px] font-bold uppercase tracking-wide">{zoneDef.label} · {Math.round(zoneDef.weight * 100)} %</span>
   337→        <h2 className="font-heading font-black text-xl mt-3">Prends ta photo</h2>
   338→
   339→        <ul className="mt-4 space-y-2">
   340→          {[
   341→            "Lumière naturelle de préférence, sans ombre directe",
   342→            "Distance ~30 cm, cadre serré sur la zone",
   343→            "Cheveux dégagés, visage net (pas de maquillage)",
   344→          ].map((t, i) => (
   345→            <li key={i} className="flex items-start gap-2 text-xs text-muted-foreground">

   350→        </ul>
   351→
   352→        <div className="mt-5">
   353→          {image ? (

   356→              <button onClick={() => setImage("")} className="absolute top-2 right-2 h-10 w-10 grid place-items-center rounded-full bg-[#1A1410]/70 text-white backdrop-blur active:scale-90 transition-transform" aria-label="Retirer la photo">
   357→                <X size={16} />
   358→              </button>
   359→            </div>
   360→          ) : (
   361→            <button
   362→              onClick={() => fileRef.current?.click()}
   363→              className="w-full aspect-square rounded-3xl border-2 border-dashed border-primary/40 bg-primary/5 grid place-items-center gap-3 active:scale-[0.99] transition-transform focus-visible:outline-2 focus-visible:outline-primary"
   364→            >
   365→              <span className="flex flex-col items-center gap-3 text-primary">
   366→                <Camera size={44} />
   367→                <span className="text-sm font-semibold">Ouvrir l&apos;appareil photo</span>

   372→          <input ref={fileRef} type="file" accept="image/*" capture="user" className="sr-only" onChange={(e) => onFile(e.target.files?.[0])} aria-label="Photo de la zone" />

   377→            <ImagePlus size={15} className="text-primary" /> Photo démo

   379→          {image && (
   380→            <button onClick={() => fileRef.current?.click()} className="h-11 px-4 rounded-xl border border-border bg-card text-xs font-semibold flex items-center gap-1.5 active:scale-[0.98] transition-transform focus-visible:outline-2 focus-visible:outline-primary">

   390→        >
   391→          <NeaOnnimIcon size={22} /> Lancer l&apos;analyse IA
   392→        </button>
   393→      </div>
   394→    );
   395→  }
   396→
   397→  /* ─────────── Étape 2 — Analyse en cours ─────────── */
   398→  if (step === 2 && analyzing) {
   399→    return (
   400→      <div className="pt-6 flex flex-col items-center">
   401→        <div className="relative w-full max-w-[320px] rounded-3xl overflow-hidden border border-border shadow-xl">
   402→          <img src={image} alt="Photo en cours d'analyse" className="aspect-square w-full object-cover" />
   403→          <motion.div
   404→            className="absolute left-0 right-0 h-16 bg-gradient-to-b from-transparent via-[#C8951E]/50 to-transparent border-y-2 border-[#C8951E]"
   405→            animate={{ top: ["8%", "78%", "8%"] }}
   406→            transition={{ duration: 2.6, repeat: Infinity, ease: "easeInOut" }}
   407→            aria-hidden="true"
   408→          />
   409→        </div>
   410→        <div className="w-full max-w-[320px] mt-6 space-y-3">
   411→          {ANALYSIS_STEPS.map((label, i) => {
   412→            const done = i < checkedSteps;
   413→            const active = i === checkedSteps;
   414→            return (
   415→              <div key={i} className={`flex items-center gap-3 rounded-xl border p-3 transition-colors ${done ? "border-[#3F7D3F]/40 bg-[#3F7D3F]/5" : active ? "border-primary/50 bg-primary/5" : "border-border bg-card opacity-50"}`}>
   416→                <span className={`grid place-items-center h-6 w-6 rounded-full shrink-0 ${done ? "bg-[#3F7D3F] text-white" : active ? "bg-primary/20 text-primary" : "bg-muted text-muted-foreground"}`}>
   417→                  {done ? <Check size={13} /> : active ? <Loader2 size={13} className="animate-spin" /> : <span className="font-mono text-[10px] font-bold">{i + 1}</span>}
   418→                </span>
   419→                <span className={`text-xs ${done ? "text-[#3F7D3F] font-semibold" : "text-muted-foreground"}`}>{label}</span>
   420→              </div>
   421→            );
   422→          })}
   423→        </div>
   424→        <div className="w-full max-w-[320px] mt-5">

   773→            <p className="text-[11px] font-semibold leading-tight line-clamp-2">{ind.nom}</p>
   774→            <CircleHelp size={13} className="text-primary shrink-0 mt-px" aria-hidden="true" />
   775→          </button>
   776→        ) : (
   777→          <p className="text-[11px] font-semibold leading-tight line-clamp-2">{ind.nom}</p>
   778→        )}
   779→        <span className={`font-mono text-[11px] font-bold shrink-0 ${sev.text}`}>
   780→          {ind.pourcentage}%
   781→        </span>
   782→      </div>
   783→      <div className="mt-1.5 h-1.5 rounded-full bg-muted overflow-hidden">
   784→        <motion.div initial={{ width: 0 }} animate={{ width: `${Math.min(100, Math.max(0, ind.pourcentage))}%` }} transition={{ duration: 0.8 }} className={`h-full rounded-full ${sev.dot}`} />
   785→      </div>
   786→    </div>
   787→  );

   790→/* ══════════════ Historique + comparaison ══════════════ */
   791→function HistoryView({
   792→  userId, history, compareMode, setCompareMode, compareSel, setCompareSel, onBack, onOpen,
   793→}: {
   794→  userId: string;
   795→  history: ApiDiagnosis[] | null;
   796→  compareMode: boolean;
   797→  setCompareMode: (v: boolean) => void;
   798→  compareSel: string[];
   799→  setCompareSel: (v: string[]) => void;
   800→  onBack: () => void;
   801→  onOpen: (d: ApiDiagnosis) => void;
   802→}) {
   803→  const list = useMemo(
   804→    () => [...(history ?? [])].filter((d) => d.status === "done").sort((a, b) => +new Date(b.createdAt) - +new Date(a.createdAt)),
   805→    [history]
   806→  );
   807→  const sameZoneSel = compareSel.length === 2 && list.find((d) => d.id === compareSel[0])?.zone === list.find((d) => d.id === compareSel[1])?.zone;
   808→  const a = list.find((d) => d.id === compareSel[0]);
   809→  const b = list.find((d) => d.id === compareSel[1]);
   810→  const ra = a ? parseDiagnosis(a.resultJson) : null;
   811→  const rb = b ? parseDiagnosis(b.resultJson) : null;
   812→
   813→  const evolutions =
   814→    ra && rb
   815→      ? rb.indicateurs
   816→          .map((ib) => {
   817→            const ia = ra.indicateurs.find((x) => x.nom === ib.nom);
   818→            return ia ? { nom: ib.nom, delta: ib.pourcentage - ia.pourcentage } : null;
   819→          })
   820→          .filter((x): x is { nom: string; delta: number } => x !== null)
   821→          .sort((x, y) => Math.abs(y.delta) - Math.abs(x.delta))
   822→          .slice(0, 3)
   823→      : [];
   824→
   825→  return (
   826→    <div className="pt-4 pb-2">
   827→      <button onClick={onBack} className="inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-foreground mb-4 focus-visible:outline-2 focus-visible:outline-primary rounded" aria-label="Retour">
   828→        <ArrowLeft size={15} /> Nouveau diagnostic
   829→      </button>
   830→      <div className="flex items-center justify-between gap-2">
   831→        <h2 className="font-heading font-black text-xl">Mon historique</h2>
   832→        <button
   833→          onClick={() => { setCompareMode(!compareMode); setCompareSel([]); }}
   834→          aria-pressed={compareMode}
   835→          className={`h-10 px-3 rounded-full text-[11px] font-bold flex items-center gap-1.5 active:scale-95 transition-all focus-visible:outline-2 focus-visible:outline-primary ${compareMode ? "bg-primary text-primary-foreground" : "border border-border bg-card"}`}
   836→        >
   837→          <GitCompareArrows size={14} /> Comparer
   838→        </button>
   839→      </div>
   840→
   841→      {history === null ? (
   842→        <div className="space-y-3 mt-5">
   843→          {[0, 1, 2].map((i) => (
   844→            <Skeleton key={i} className="h-20 rounded-2xl" />
   845→          ))}
   846→        </div>
   847→      ) : list.length === 0 ? (
   848→        <div className="mt-6 rounded-2xl border border-dashed border-border bg-card/60 p-8 text-center">
   849→          <NeaOnnimIcon size={38} className="mx-auto text-primary" />
   850→          <p className="font-heading font-bold text-sm mt-3">Aucun diagnostic pour l&apos;instant</p>
   851→          <p className="text-xs text-muted-foreground mt-1">Ton premier scan débloquera le suivi d&apos;évolution.</p>
   852→        </div>
   853→      ) : (
   854→        <>
   855→          {/* Le Fil du Temps — courbes d'évolution */}
   856→          <EvolutionCard userId={userId} className="mt-4" />
   857→
   858→          <div className="space-y-2.5 mt-4">
   859→            {list.map((d) => {
   860→              const sel = compareSel.includes(d.id);
   861→              return (
   862→                <div key={d.id} className={`flex items-center gap-3 rounded-2xl border p-2.5 transition-colors ${sel ? "border-primary bg-primary/5" : "border-border bg-card"}`}>
   863→                  <button
   864→                    onClick={() => {
   865→                      if (compareMode) {
   866→                        setCompareSel(sel ? compareSel.filter((x) => x !== d.id) : compareSel.length < 2 ? [...compareSel, d.id] : [compareSel[1], d.id]);
   867→                      } else onOpen(d);
   868→                    }}
   869→                    className="flex items-center gap-3 flex-1 min-w-0 text-left focus-visible:outline-2 focus-visible:outline-primary rounded-xl"
   870→                    aria-label={`Diagnostic ${d.zone} du ${formatDate(d.createdAt)}, score ${d.scoreGlobal}`}
   871→                  >
   872→                    <img src={diagImgSrc(d.imageData)} alt={`Diagnostic ${d.zone}`} loading="lazy" className="h-14 w-14 rounded-xl object-cover shrink-0" />
   873→                    <div className="min-w-0">
   874→                      <p className="text-sm font-semibold">{BODY_ZONES.find((z) => z.id === d.zone)?.label ?? d.zone}</p>
   875→                      <p className="text-[11px] text-muted-foreground">{formatDate(d.createdAt)}</p>
   876→                    </div>
   877→                    <div className="ml-auto"><ScoreChip score={d.scoreGlobal} /></div>
   878→                  </button>
   879→                  {compareMode && (
   880→                    <button
   881→                      onClick={() => setCompareSel(sel ? compareSel.filter((x) => x !== d.id) : compareSel.length < 2 ? [...compareSel, d.id] : [compareSel[1], d.id])}
   882→                      aria-pressed={sel}
   883→                      aria-label="Sélectionner pour comparaison"
   884→                      className={`h-9 w-9 grid place-items-center rounded-full border-2 shrink-0 transition-colors ${sel ? "bg-primary border-primary text-primary-foreground" : "border-border"}`}
   885→                    >
   886→                      {sel && <Check size={14} />}
   887→                    </button>
   888→                  )}
   889→                </div>
   890→              );
   891→            })}
   892→          </div>
   893→
   894→          {compareMode && (
   895→            <AnimatePresence>
   896→              {compareSel.length === 2 && sameZoneSel && ra && rb && a && b ? (
   897→                <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: "auto" }} exit={{ opacity: 0, height: 0 }} className="mt-5 overflow-hidden">
   898→                  <div className="rounded-3xl border border-primary/40 bg-card p-4 shadow-md">
   899→                    <p className="font-heading font-bold text-sm mb-3 flex items-center gap-2">
   900→                      <GitCompareArrows size={15} className="text-primary" /> Avant / Après — {BODY_ZONES.find((z) => z.id === a.zone)?.label}
   901→                    </p>
   902→                    <div className="grid grid-cols-2 gap-3">
   903→                      {[{ d: a, r: ra, tag: "Avant" }, { d: b, r: rb, tag: "Après" }].map(({ d, r, tag }) => (
   904→                        <div key={tag}>
   905→                          <img src={diagImgSrc(d.imageData)} alt={`${tag} — ${d.zone}`} loading="lazy" className="aspect-square w-full rounded-2xl object-cover border border-border" />
   906→                          <div className="flex items-center justify-between mt-1.5">
   907→                            <span className="text-[10px] uppercase tracking-wide font-bold text-muted-foreground">{tag} · {formatDate(d.createdAt, { day: "numeric", month: "short" })}</span>
   908→                            <ScoreChip score={r.score_global} />
   909→                          </div>
   910→                        </div>
   911→                      ))}
   912→                    </div>
   913→                    <div className="mt-3 rounded-xl bg-muted/60 p-3 flex items-center justify-between">
   914→                      <span className="text-xs font-semibold">Évolution du score</span>
   915→                      <span className="font-mono text-lg font-black" style={{ color: rb.score_global >= ra.score_global ? "#3F7D3F" : "#8B1A3B" }}>
   916→                        {rb.score_global >= ra.score_global ? "+" : ""}{rb.score_global - ra.score_global} pts
   917→                      </span>
   918→                    </div>
   919→                    {evolutions.length > 0 && (
   920→                      <div className="mt-3 space-y-1.5">
   921→                        {evolutions.map((e) => (
   922→                          <div key={e.nom} className="flex items-center justify-between text-[11px]">
   923→                            <span className="text-muted-foreground">{e.nom}</span>

   925→                              {e.delta >= 0 ? "+" : ""}{e.delta} %
   926→                            </span>
   927→                          </div>
   928→                        ))}
   929→                      </div>
   930→                    )}
   931→                  </div>

   933→              ) : (
   934→                <p className="mt-4 text-center text-[11px] text-muted-foreground">Sélectionne 2 diagnostics de la même zone pour comparer l&apos;évolution.</p>
   935→              )}
   936→            </AnimatePresence>
   937→          )}
   938→        </>
   939→      )}
   940→      <div className="h-2" />
   941→    </div>
   942→  );
