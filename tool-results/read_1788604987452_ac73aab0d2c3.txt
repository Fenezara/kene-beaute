    81→  const loadWallet = useCallback(() => {
    82→    apiGet<{ wallet: ApiWallet; transactions: ApiWalletTx[] }>(`/api/wallet?userId=${user.id}`)

    84→      .catch(() => setTxs([]));
    85→  }, [user.id]);
    86→
    87→  useEffect(() => {
    88→    loadWallet();
    89→  }, [loadWallet]);

    91→  /** Portabilité RGPD — télécharge « mes données » en JSON via blob */
    92→  async function downloadMyData() {
    93→    setExportBusy(true);
    94→    try {
    95→      const res = await fetch(`/api/profile/export?userId=${encodeURIComponent(user.id)}`);
    96→      if (!res.ok) throw new Error("Export impossible");
    97→      const blob = await res.blob();
    98→      const cd = res.headers.get("Content-Disposition") ?? "";
    99→      const m = /filename="([^"]+)"/.exec(cd);
   100→      const url = URL.createObjectURL(blob);
   101→      const a = document.createElement("a");
   102→      a.href = url;
   103→      a.download = m?.[1] ?? `kene-mes-donnees-${new Date().toISOString().slice(0, 10)}.json`;
   104→      document.body.appendChild(a);
   105→      a.click();
   106→      a.remove();
   107→      setTimeout(() => URL.revokeObjectURL(url), 4000);
   108→      const kb = Math.max(1, Math.round(blob.size / 1024));
   109→      toast.success(`Mes données téléchargées · ${kb} Ko`);
   110→    } catch (e) {
   111→      toast.error(e instanceof Error ? e.message : "Export impossible");
   112→    } finally {
   113→      setExportBusy(false);
   114→    }
   115→  }
   116→
   117→  /** « Créer un compte entreprise » (t. 69-a) : l'espace Pro vit sur un compte
   118→   *  DÉDIÉ, séparé du compte cliente — on pose le pont sessionStorage que
   119→   *  l'onboarding consomme à son montage (mode entreprise), on vide le panier
   120→   *  (règle de déconnexion) puis on ferme la session cliente. */
   121→  function startProSignup() {
   122→    try {
   123→      sessionStorage.setItem("kene-pro-signup", "1");
   124→    } catch {
   125→      /* stockage indisponible : l'onboarding démarrera simplement en mode cliente */
   126→    }
   127→    clearCart();
   128→    setUser(null);
   129→    toast.info("À très vite — bienvenue dans l'aventure entreprise");
   130→  }

   132→  /** Installer l'app (prompt natif, ou instructions iOS / navigateur). */
   133→  async function installApp() {

   135→      const accepted = await promptInstall();

   141→        description: "Bouton Partager ⬆️ puis « Sur l'écran d'accueil ».",
   142→        duration: 8000,
   143→      });
   144→      return;
   145→    }
   146→    toast.info("Installation manuelle", {
   147→      description: "Menu du navigateur → « Installer l'application » ou « Ajouter à l'écran d'accueil ».",
   148→      duration: 8000,
   149→    });
   150→  }

   152→  async function saveIdentity() {
   153→    setSavingId(true);
   154→    try {
   155→      const r = await apiPatch<{ user: ApiUser }>("/api/auth/profile", { userId: user.id, name: name.trim() || undefined, city: city.trim() || undefined });

   157→      setEdit(false);
   158→      toast.success("Profil mis à jour");
   159→    } catch (e) {
   160→      toast.error(e instanceof Error ? e.message : "Mise à jour impossible");

   162→      setSavingId(false);
   163→    }
   164→  }

   166→  async function saveSkin() {
   167→    setSavingSkin(true);
   168→    try {
   169→      const r = await apiPatch<{ user: ApiUser }>("/api/auth/profile", {
   170→        userId: user.id,
   171→        skinType,
   172→        fitzpatrick: fitz,
   173→        allergies: allergies.trim() || undefined,
   174→        goals: goals.map((id) => ({ id, label: SKIN_GOALS.find((g) => g.id === id)?.label ?? id })),
   175→      });
   176→      setUser({ ...user, ...r.user } as SessionUser);
   177→      toast.success("Profil peau mis à jour");
   178→    } catch (e) {
   179→      toast.error(e instanceof Error ? e.message : "Mise à jour impossible");
   180→    } finally {
   181→      setSavingSkin(false);
   182→    }
   183→  }

   193→  /** Sécurité renforcée : bascule la re-vérification par code avant paiement. */

   201→  async function runTopup() {
   202→    setTopupBusy(true);
   203→    setTopupState("processing");
   204→    try {
   205→      const r = await apiPost<{ payment: { id: string; confirmToken?: string } }>("/api/wallet/topup", { userId: user.id, amount, method });

   207→      // attente porte son jeton — absent, on n'appelle JAMAIS confirm et on
   208→      // revient au formulaire comme après un échec (aucun crédit fantôme).
   209→      const confirmToken = r.payment.confirmToken;
   210→      if (!confirmToken) {
   211→        setTopupState("idle");
   212→        toast.error("Paiement impossible — réessaie dans quelques instants");
   213→        return;
   214→      }
   215→      await new Promise((res) => setTimeout(res, 2600));
   216→      await apiPost("/api/payments/confirm", { paymentId: r.payment.id, confirmToken });
   217→      setTopupState("done");
   218→      toast.success(`${xof(amount)} crédités sur ton wallet`);
   219→      loadWallet();
   220→    } catch (e) {
   221→      setTopupState("idle");
   222→      toast.error(e instanceof Error ? e.message : "Approvisionnement impossible");
   223→    } finally {
   224→      setTopupBusy(false);
   225→    }
   226→  }
